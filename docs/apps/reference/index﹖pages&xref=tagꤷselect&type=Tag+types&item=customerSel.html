App: regression/errors - Page: syntax/a_nested_select_names_the_select_it_stands_in
PHP
syntax/a_nested_select_names_the_select_it_stands_in.php 		HTML
syntax/a_nested_select_names_the_select_it_stands_in.pad 		Result 
<?php

  // A select related to none of the selects around it is named with the one it stands in,
  // the nearest: the outermost was named, a select the inner one was never written in.

  $padSelect ['customerSel'] = [ 'db' => 'customers', 'key' => 'customerNumber' ];
  $padSelect ['orderSel']    = [ 'db' => 'orders',    'key' => 'orderNumber' ];
  $padSelect ['staffSel']    = [ 'db' => 'staff',     'key' => 'name' ];

  $padRelations ['orderSel'] ['customerSel'] = 'customerNumber';

?>
 
{customerSel $customerNumber=103}{orderSel rows=1}{staffSel}x{/staffSel}{/orderSel}{/customerSel}

App: regression/pages - Page: select/demo/a_select_in_the_else_of_an_empty_select_stands_alone
PHP
select/demo/a_select_in_the_else_of_an_empty_select_stands_alone.php 		HTML
select/demo/a_select_in_the_else_of_an_empty_select_stands_alone.pad 		Result 
<?php

  // A select in the @else@ of a related select that found no row has no row to follow, so it
  // stands alone. It ended the page on "the relation to 'customerSel' is followed by its
  // field 'customerNumber', which the row of 'customerSel' does not hold - select it in
  // fields=". Over the Classic Models tables of the demo database.

  $padSelect ['customerSel'] = [ 'db' => 'customers', 'key' => 'customerNumber' ];
  $padSelect ['orderSel']    = [ 'db' => 'orders',    'key' => 'orderNumber' ];

  $padRelations ['orderSel'] ['customerSel'] = 'customerNumber';

?>
 
[{customerSel $customerNumber=1}{$customerName}@else@none, the latest order: {orderSel rows=1, order="orderDate desc, orderNumber desc"}{$orderNumber}{/orderSel}{/customerSel}]
[{customerSel $customerNumber=103}{$customerName}, the latest order: {orderSel rows=1, order="orderDate desc, orderNumber desc"}{$orderNumber}{/orderSel}@else@none{/customerSel}]

App: regression/pages - Page: select/demo/an_empty_page_is_the_first_page_of_the_limit
PHP
select/demo/an_empty_page_is_the_first_page_of_the_limit.php 		HTML
select/demo/an_empty_page_is_the_first_page_of_the_limit.pad 		Result 
<?php

  // page= on a select is the SQL limit whatever its value: an empty page - ?pg= with
  // page=$pg ?? 1 - or a page 0 is the first page, ten rows read and then sorted, the same
  // rows as page 1, a sort= before it or not. Over the 122 customers of the demo database.

  $padSelect ['customerSel'] = [ 'db' => 'customers', 'key' => 'customerNumber' ];

  $none = '';
  $zero = 0;

?>
 
one: [{customerSel fields="customerNumber", sort="customerNumber desc", page=1}{$customerNumber} {/customerSel}]
empty: [{customerSel fields="customerNumber", sort="customerNumber desc", page=$none}{$customerNumber} {/customerSel}]
zero: [{customerSel fields="customerNumber", sort="customerNumber desc", page=$zero}{$customerNumber} {/customerSel}]

App: regression/pages - Page: select/demo/json_rows_are_a_list_in_the_order_of_the_select
PHP
select/demo/json_rows_are_a_list_in_the_order_of_the_select.php 		HTML
select/demo/json_rows_are_a_list_in_the_order_of_the_select.pad 		Result 
<?php

  // htmlAttrJson hands a component the rows as a JSON list in the order of the select, also
  // when they hold an id column - by which db() keys the rows of an array. Over the
  // customers of the demo database.

  $padSelect ['customerSel'] = [ 'db' => 'customers', 'key' => 'customerNumber' ];

?>
 
list: [{customerSel htmlAttrJson, fields="customerNumber as id, customerName", where="country = 'Norway'", order="customerName desc"/}]
record: [{customerSel htmlAttrJson='record', fields="customerNumber as id, customerName", $customerNumber=121/}]
