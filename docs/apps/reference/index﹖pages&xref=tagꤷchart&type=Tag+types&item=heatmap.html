App: charts - Page: heatmap/traffic
PHP
heatmap/traffic.php 		HTML
heatmap/traffic.pad 		Result 
<?php

  // Visits per weekday and hour: a busy lunch, a busier evening, a quiet weekend morning,
  // and a week that gets busier towards Thursday and quieter on Friday night.

  $traffic = [];
  $week    = [ 'Mon' => 0.85, 'Tue' => 0.95, 'Wed' => 1.0, 'Thu' => 1.1, 'Fri' => 0.9, 'Sat' => 1.2, 'Sun' => 1.3 ];

  foreach ( $week as $day => $busy )
    foreach ( [ 8, 10, 12, 14, 16, 18, 20, 22 ] as $hour ) {

      $visits = 40 + 60 * exp ( - ( ( $hour - 12.5 ) ** 2 ) / 3 )
                   + 90 * exp ( - ( ( $hour - 20 )   ** 2 ) / 4 );

      $visits = $visits * $busy - ( $busy > 1.15 and $hour < 12 ? 35 : 0 );

      $traffic [] = [ 'day' => $day, 'hour' => "{$hour}h", 'visits' => (int) round ( $visits ) ];

    }

?>
 
{chart:heatmap data='traffic',
       x='hour',
       y='day',
       value='visits',
       title='Visits per weekday and hour',
       width=640,
       height=360/}

App: regression/framework - Page: tags/chart_heatmap_takes_time_by_its_cells
PHP
tags/chart_heatmap_takes_time_by_its_cells.php 		HTML
tags/chart_heatmap_takes_time_by_its_cells.pad 		Result 
<?php

  // A heatmap's time goes by the cells it has, not by every pair of a row and a column:
  // 20,000 rows of a cell each - each its own row and column - took ten seconds.

  $cells = [];

  for ( $i = 0; $i < 20000; $i++ )
    $cells [] = [ 'r' => "r$i", 'c' => "c$i", 'v' => $i % 7 ];

  $start = hrtime ( TRUE );

?>
 
{chart:heatmap data='cells', x='c', y='r', value='v' | strlen/} {pad:echo hrtime(TRUE) - $start lt 3e9 ? 'in time' : 'too slow'/}

App: widgets - Page: palette/batlow
PHP
[group]/[kind].php 		HTML
[group]/[kind].pad 		Result 
<?php

  // A kind of one of the four groups, by the name in the address - ?section/hero,
  // /sections/effect/typewriter, ?palette/viridis: one the engine has, else a page not found.
  // Its examples are shown beside the templates that make them; a theme's page lists the
  // contrast of its colours and has a button that puts every page in it, a palette's the
  // colours padPalette answers.

  $one = sectionsKind ( $group, $kind );

  if ( ! $one )
    padAbort ( 404 );

  $name     = $one ['name'];
  $title    = $name . ' ' . strtolower ( rtrim ( sectionsGroups () [$group] [0], 's' ) );
  $text     = $one ['text'];
  $wide     = in_array ( $group, [ 'section', 'palette' ] );
  $examples = sectionsExamples ( $group, $kind );

  $contrasts = ( $group == 'theme' ) ? sectionsContrasts ( $kind ) : [];
  $aaa       = ( $group == 'theme' and padThemeNeeds ( $kind ) >= 7 );
  $colors    = ( $group == 'palette' ) ? implode ( ' ', padPalette ( $kind ) ) : '';

?>
 
<p class="sx-lead">{$text}</p>

{include:exampleList/}

{pad:if $group eq 'theme'}
  <h2>Contrast</h2>
  <p>Every text of the theme on each of its backgrounds, as WCAG 2 measures it -
     {pad:if $aaa}7 to 1 or more, AAA{pad:else}4.5 to 1 or more, AA{/pad:if}.</p>
  <table class="sx-contrast">
    <tr><th>Text</th><th>On</th><th>Ratio</th></tr>
    {array:contrasts}<tr><td>{$fore}</td><td>{$back}</td><td>{$ratio} : 1</td></tr>{/array:contrasts}
  </table>
  <form method="post" class="sx-use">
    <input type="hidden" name="padForm" value="padThemes">
    <input type="hidden" name="padTheme" value="{$kind}">
    <button type="submit">Put every page in {$name}</button>
  </form>
{/pad:if}

{pad:if $group eq 'palette'}
  <h2>In PHP</h2>
  <p><code>padPalette ( '{$kind}' )</code> answers <code>{$colors}</code>.</p>
{/pad:if}
