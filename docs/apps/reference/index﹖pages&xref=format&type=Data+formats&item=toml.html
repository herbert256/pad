<?php

  // A fetched body is read in the format of the formats subsystem its media type names - the
  // NDJSON of a stream API (application/x-ndjson), a calendar (text/calendar), TOML
  // (application/toml), TSV (text/tab-separated-values) - where application/x-ndjson was
  // taken for one JSON document by the json in its name and failed, and TOML read as CSV;
  // a feed (application/rss+xml) is the XML it is, as before, and JSON is JSON.

  padCurlFake ( [
    'https://api.example.com/ndjson' => [ 'data' => "{\"name\":\"rex\"}\n{\"name\":\"tom\"}", 'headers' => [ 'Content-Type' => 'application/x-ndjson; charset=utf-8' ] ],
    'https://api.example.com/ics'    => [ 'data' => "BEGIN:VCALENDAR\nBEGIN:VEVENT\nSUMMARY:walk\nDTSTART:20261101\nEND:VEVENT\nEND:VCALENDAR", 'headers' => [ 'Content-Type' => 'text/calendar' ] ],
    'https://api.example.com/toml'   => [ 'data' => "[[pets]]\nname = \"rex\"\n[[pets]]\nname = \"tom\"", 'headers' => [ 'Content-Type' => 'application/toml' ] ],
    'https://api.example.com/tsv'    => [ 'data' => "name\nrex\ntom", 'headers' => [ 'Content-Type' => 'text/tab-separated-values' ] ],
    'https://api.example.com/rss'    => [ 'data' => '<rss><channel><item><title>a</title></item></channel></rss>', 'headers' => [ 'Content-Type' => 'application/rss+xml' ] ],
    'https://api.example.com/json'   => [ 'data' => '[{"name":"rex"}]', 'headers' => [ 'Content-Type' => 'application/json' ] ]
  ] );

  $fetchShow = [];

  foreach ( [ 'ndjson', 'ics', 'toml', 'tsv', 'rss', 'json' ] as $fetchKind ) {
    $fetchCurl   = padCurl ( "https://api.example.com/$fetchKind" );
    $fetchShow [] = "$fetchKind: {$fetchCurl ['type']} " . json_encode ( padCurlData ( $fetchCurl, 'x' ) );
  }

  padCurlFakeStop ();

  $fetchShow = implode ( ' | ', $fetchShow );

?>

<?php

  // TOML written: plain values first, then a [table] per table - left out for one that holds
  // tables only - and an [[array]] per table of a list of tables, an empty one among them; a
  // text that is a date written bare, a float with its .0, inf and nan, a key that is no bare
  // key quoted. Read again it is the value it was written from.

  $tomlValue = [
    'title'   => 'Weekend "away"',
    'start'   => '2026-11-06',
    'at'      => '2026-11-06T09:30:00Z',
    'ratio'   => 2.0,
    'limit'   => INF,
    'tags'    => [ 'a', 'b' ],
    'point'   => [ 'x' => 1, 'y' => [ 2, 3 ] ],
    'a b'     => 'spaced key',
    'servers' => [ 'alpha' => [ 'ip' => '10.0.0.1' ] ],
    'stops'   => [ [ 'name' => 'Gravensteen', 'hours' => 2 ], [], [ 'name' => 'Graslei', 'hours' => 1.5 ] ]
  ];

  $tomlText = padFormatWrite ( 'toml', $tomlValue );
  $tomlBack = padFormatRead  ( 'toml', $tomlText );

  $tomlSame = ( $tomlBack == $tomlValue ) ? 'the same' : 'another value: ' . json_encode ( $tomlBack );
  $tomlShow = str_replace ( "\n", ' | ', trim ( $tomlText ) );

?>

App: regression/framework - Page: formats/an_ini_toml_or_geojson_fault_reads_as_no_rows_on_the_lenient_walk

<?php

  // other fault of theirs is: an INI key named like a later section, arrays and inline

    [ 'geojson', '{"type":"Feature","geometry":{"type":["x"]},"properties":{}}' ],

App: regression/framework - Page: formats/the_toml_spec_strings_numbers_and_dates_read_as_written
PHP
formats/the_toml_spec_strings_numbers_and_dates_read_as_written.php 		HTML
formats/the_toml_spec_strings_numbers_and_dates_read_as_written.pad 		Result 
<?php

  // The examples of the TOML 1.0 specification (toml.io/en/v1.0.0) for strings, integers,
  // floats, booleans and dates, each read and shown as JSON: the three ways to write the
  // quick brown fox are one text, the escapes and the multi-line forms keep what the
  // specification says they keep, 0xDEADBEEF is 3735928559, and a date is the text it is -
  // the space between date and time read as a T.

  $tomlStrings = <<<'TOML'
str1 = "The quick brown fox jumps over the lazy dog."
str2 = """
The quick brown \


  fox jumps over \
    the lazy dog."""
str3 = """\
       The quick brown \
       fox jumps over \
       the lazy dog.\
       """
str4 = "I'm a string. \"You can quote me\". Name\tJos\u00E9\nLocation\tSF."
winpath2 = '\\ServerX\admin$\system32\'
regex2 = '''I [dw]on't need \d{2} apples'''
lines  = '''
The first newline is
trimmed in raw strings.
'''
quot15 = '''Here are fifteen quotation marks: """""""""""""""'''
str = ''''That,' she said, 'is still pointless.''''
str7 = """"This," she said, "is just a pointless statement.""""
TOML;

  $tomlNumbers = <<<'TOML'
int1 = +99
int4 = -17
int6 = 5_349_221
hex1 = 0xDEADBEEF
oct1 = 0o01234567
bin1 = 0b11010110
max = 9223372036854775807
flt1 = +1.0
flt4 = 5e+22
flt6 = -2E-2
flt8 = 224_617.445_991_228
bool1 = true
odt1 = 1979-05-27T07:32:00Z
odt3 = 1979-05-27T00:32:00.999999-07:00
odt4 = 1979-05-27 07:32:00Z
ldt1 = 1979-05-27T07:32:00
ld1 = 1979-05-27
lt2 = 00:32:00.999999
TOML;

  $tomlRead = padFormatRead ( 'toml', $tomlStrings );

  $tomlFox  = ( $tomlRead ['str1'] === $tomlRead ['str2'] and $tomlRead ['str1'] === $tomlRead ['str3'] ) ? 'one fox' : 'three foxes';
  $tomlText = json_encode ( array_diff_key ( $tomlRead, [ 'str2' => 1, 'str3' => 1 ] ), JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES );

  $tomlKinds = [];

  foreach ( padFormatRead ( 'toml', $tomlNumbers ) as $key => $value )
    $tomlKinds [] = $key . '=' . gettype ( $value ) . ':' . var_export ( $value, TRUE );

  $tomlNumbers = implode ( ' ', $tomlKinds );

?>
