App: notes - Page: index
PHP
index.php 		HTML
index.pad 		Result 
<?php

  // The notes, pinned first and then the last changed: all of them, those of one tag, or
  // those a search finds in a title, a text or a tag - the tag and the search read by
  // _inits.php, whose menu has the tags. The list is a live region - the star pins a note,
  // and this PHP answers the click with the region alone - and a fragment, which notes.js
  // asks for while a search is typed.

  if ( padLiveEvent () == 'pin' )
    notesPin ( (int) padLiveValue () );

  $notes = notesList ( $search, $only );
  $total = (int) db ( "FIELD count(*) FROM notes" );
  $found = count ( $notes );
  $title = $only !== '' ? "#$only" : ( $search !== '' ? "Found for “{$search}”" : 'All notes' );

?>
 
<section class="list">
  <div class="list-head">
    <span class="count">{$found} of {$total} notes</span>
  </div>
  <div id="results" data-tag="{$only}">
    {pad:fragment 'list'}
    {pad:live 'notes'}
    <div class="grid">
      {array:notes}
      <article class="note color-{$color}{pad:if $pinned} pinned{/pad:if}">
        <button type="button" class="pin" pad-click="pin" pad-value="{$id}" aria-pressed="{pad:if $pinned}true{pad:else}false{/pad:if}" title="{pad:if $pinned}Unpin{pad:else}Pin to the top{/pad:if}">★</button>
        <a class="open" href="{$padGo}note&id={$id}">
          <h2>{$title}</h2>
          <p>{$excerpt}</p>
        </a>
        <footer>
          <time datetime="{$updated}">{$day}</time>
          {pad:if $files gt 0}<span class="clip" title="Attached files">📎 {$files}</span>{/pad:if}
          {level:tags}<a class="tag" href="{$padGo}index{?tag}">#{$tag}</a>{/level:tags}
        </footer>
      </article>
      @else@
      <p class="empty">{pad:if $search}Nothing found for “{$search}”.{pad:else}No notes here yet.{/pad:if} <a href="{$padGo}edit">Write one</a>.</p>
      {/array:notes}
    </div>
    {/pad:live}
    {/pad:fragment}
  </div>
</section>

App: regression/framework - Page: formulas/an_editable_spreadsheet_in_a_live_region_grows_with_its_cells_not_their_square
PHP
formulas/an_editable_spreadsheet_in_a_live_region_grows_with_its_cells_not_their_square.php 		HTML
formulas/an_editable_spreadsheet_in_a_live_region_grows_with_its_cells_not_their_square.pad 		Result 
<?php

  // An editable spreadsheet of 30 x 30 numbers in a {pad:live} region is one form the live script
  // posts, which a change of a cell submits: every cell's text stands once, in its field, and
  // the page grows with the cells - some 115 KB, as outside a region. Each of the 900 fields
  // carried the texts of all the others in its pad-value, and the page was 10.7 MB.

  $grid = [];

  for ( $row = 0; $row < 30; $row++ )
    for ( $column = 0; $column < 30; $column++ )
      $grid [$row] [$column] = ( $row * 30 + $column ) * 1.5;

?>
 
{pad:live 'grid'}{pad:spreadsheet 'grid', data='grid', editable, headers=FALSE/}{/pad:live | php:substr_count(@, '<input name="padSheet[grid]') . ' cells, ' . php:substr_count(@, ' pad-value=') . ' values, ' . php:substr_count(@, 'pad-submit="sheet-grid"') . ' form, ' . ( php:strlen(@) lt 150000 ? 'under' : 'over' ) . ' 150 KB'}

App: regression/framework - Page: tags/live_answers_a_change_of_a_multiselect_with_every_choice
PHP
tags/live_answers_a_change_of_a_multiselect_with_every_choice.php 		HTML
tags/live_answers_a_change_of_a_multiselect_with_every_choice.pad 		Result 
<?php

  // The post the live script sends for a change of a multiselect of which a and c are chosen:
  // every choice as the form posts it, padLists[] naming the list, padValue the choices
  // joined by ', '. The region comes back with both chosen, and padLiveValue () is a, c.

  $_SERVER ['REQUEST_METHOD'] = 'POST';
  $_POST = [ 'tags' => [ 'a', 'c' ], 'padLists' => [ 'tags[]' ], 'padEvent' => 'pick', 'padValue' => 'a, c', 'padLive' => 'box' ];

  $picked = padLiveEvent () . ': ' . padLiveValue ();

?>
 
{pad:live 'box'}{input:multiselect 'tags', options='a, b, c', value='a', pad-change='pick'/}<p>{$picked}</p>{/pad:live}
