<?php

  // A single word spliced into the expression of a {pad:bool} named on its closing tag is no
  // string when the evaluator knows it as a reference: the request value connection_status
  // called that PHP function with no arguments, and its 0 made the comparison TRUE. A word
  // that names nothing, {$s} eq 'paid', still compares.

  $w = 'connection_status';

?>
 
{pad:bool}{$w} eq 0{/pad:bool 'b'}{b}T@else@F{/b}

App: regression/framework - Page: escaping/a_bool_value_is_a_truth_test_not_run_as_pad
PHP
escaping/a_bool_value_is_a_truth_test_not_run_as_pad.php 		HTML
escaping/a_bool_value_is_a_truth_test_not_run_as_pad.pad 		Result 
<?php

  // {pad:bool} from a value is that value's own truth, not the value run as PAD: $v names a
  // sequence tag and '1 eq 2' is an expression - run, the first would make a sequence and
  // the second would be FALSE - so a value kept its text and both are TRUE; '0' is FALSE.
  // A written-out literal second parameter is still the expression it is: 'false' is FALSE.

  $v = 'sequence:fibonacci(6)';
  $x = '1 eq 2';
  $w = '0';

?>
 
{pad:bool 'a', $v/}{pad:bool 'b', $x/}{pad:bool 'c', $w/}{pad:bool 'd', 'false'/}[{bool:a}T@else@F{/bool:a}{bool:b}T@else@F{/bool:b}{bool:c}T@else@F{/bool:c}{bool:d}T@else@F{/bool:d}]

App: regression/framework - Page: escaping/a_value_rendered_into_a_bool_named_at_its_close_is_text
PHP
escaping/a_value_rendered_into_a_bool_named_at_its_close_is_text.php 		HTML
escaping/a_value_rendered_into_a_bool_named_at_its_close_is_text.pad 		Result 
<?php

  // A {pad:bool} named on its closing tag reduces its content after the walk, its fields' values
  // in it. Content that is values alone is their text's own truth, as {pad:bool 'name', $v}
  // takes it: $a ran as the expression it reads like and stored FALSE, $f called the PHP
  // function it names; 'Hello world' stays TRUE, '0' FALSE. An expression of the author's
  // own still evaluates with a number in it, a word, or a value inside its quotes.

  $a = 'php:strlen(12345) eq 6';
  $f = 'connection_status';
  $t = 'Hello world';
  $z = '0';
  $n = 3;
  $m = 1;
  $u = 'Ann';
  $s = 'paid';

?>
 
{pad:bool}{$a}{/pad:bool 'b1'}{pad:bool}{$f}{/pad:bool 'b2'}{pad:bool}{$t}{/pad:bool 'b3'}{pad:bool}{$z}{/pad:bool 'b4'}{pad:bool}{$n} gt 2{/pad:bool 'b5'}{pad:bool}{$m} gt 2{/pad:bool 'b6'}{pad:bool}'{$u}' eq 'Ann'{/pad:bool 'b7'}{pad:bool}'{$u}' eq 'Bob'{/pad:bool 'b8'}{pad:bool}$m gt 0{/pad:bool 'b9'}{pad:bool}{$s} eq 'paid'{/pad:bool 'b10'}{pad:bool}{$s} eq 'due'{/pad:bool 'b11'}[{bool:b1}T@else@F{/bool:b1}{bool:b2}T@else@F{/bool:b2}{bool:b3}T@else@F{/bool:b3}{bool:b4}T@else@F{/bool:b4}{bool:b5}T@else@F{/bool:b5}{bool:b6}T@else@F{/bool:b6}{bool:b7}T@else@F{/bool:b7}{bool:b8}T@else@F{/bool:b8}{bool:b9}T@else@F{/bool:b9}{bool:b10}T@else@F{/bool:b10}{bool:b11}T@else@F{/bool:b11}]
