Reference - PAD functions - abbreviate

App: manual - Page: fragments/numbers_4
PHP
fragments/numbers_4.php 		HTML
fragments/numbers_4.pad 		Result 
<?php

  // The PHP hands over plain numbers; the template writes them
  // for people.

  $runners = [
    [ 'name' => 'Ann', 'place' => 1, 'views' => 1534200 ],
    [ 'name' => 'Bob', 'place' => 2, 'views' => 48210   ],
    [ 'name' => 'Cee', 'place' => 3, 'views' => 950     ]
  ];

?>
 
{runners}
  <p>{$place | ordinal}: {$name},
     {$views | abbreviate(1)} views</p>
{/runners}
