App: alpine - Page: examples/form
PHP
examples/form.php 		HTML
examples/form.pad 		Result 
<?php

  // One list of rules for both sides. padValidate checks a post against it; for the browser
  // padValidateClient exports the same list, every rule with the message padValidate would
  // give, and {input:validator} in the template brings the checker that reads it. A good post is
  // kept in the session; the answer is JSON either way - the messages, or the new list.

  $rules = [ 'name'  => 'required|max:40',
             'email' => 'required|email',
             'team'  => 'required|in:web,platform,data',
             'seats' => 'required|integer|min:1|max:8' ];

  $people = padSession ( 'alpinePeople', [] );
  $errors = [];
  $saved  = FALSE;

  if ( padRequestIs ( 'POST' ) ) {

    $errors = padValidate ( $rules );

    if ( ! $errors ) {
      array_unshift ( $people, [ 'name' => trim ( padRequest ( 'name' ) ), 'team' => padRequest ( 'team' ), 'seats' => (int) padRequest ( 'seats' ) ] );
      $saved = padSessionPut ( 'alpinePeople', array_slice ( $people, 0, 8 ) );
      $people = array_slice ( $people, 0, 8 );
    }

  }

  $start     = [ 'rules' => padValidateClient ( $rules ), 'people' => $people ];
  $padExpose = [ 'saved', 'errors', 'people' ];

?>
 
<div class="cols-2" x-data="signup({^start})">

  <form novalidate @submit.prevent="send()" class="stack-fields">
    <label class="label">Name <input class="input" name="name" x-model="form.name" @blur="leave('name')" :class="errors.name && 'is-invalid'"></label>
    <span class="error" x-text="errors.name" x-show="errors.name" x-cloak></span>

    <label class="label">E-mail <input class="input" name="email" type="email" x-model="form.email" @blur="leave('email')" :class="errors.email && 'is-invalid'"></label>
    <span class="error" x-text="errors.email" x-show="errors.email" x-cloak></span>

    <label class="label">Team
      <select class="input" name="team" x-model="form.team" @change="leave('team')" :class="errors.team && 'is-invalid'">
        <option value="">Choose ...</option>
        <option value="web">Web</option>
        <option value="platform">Platform</option>
        <option value="data">Data</option>
        <option value="sales">Sales - not one of the rules</option>
      </select>
    </label>
    <span class="error" x-text="errors.team" x-show="errors.team" x-cloak></span>

    <label class="label">Seats <input class="input" name="seats" x-model="form.seats" @blur="leave('seats')" :class="errors.seats && 'is-invalid'"></label>
    <span class="error" x-text="errors.seats" x-show="errors.seats" x-cloak></span>

    <label class="small check-line"><input type="checkbox" x-model="browser"> Check in the browser first</label>
    <button :disabled="busy">Book</button>
    <p class="small" x-text="note" x-show="note" x-cloak></p>
  </form>

  <div>
    <div class="panel-title">Booked in your session</div>
    <template x-for="(one, at) in people" :key="at">
      <div class="card booked"><strong x-text="one.name"></strong> <span class="badge" x-text="one.team"></span> <span class="muted small" x-text="one.seats + ' seats'"></span></div>
    </template>
    <p class="muted small" x-show="people.length === 0">Nobody yet.</p>
  </div>

</div>

App: react - Page: examples/feedback
PHP
examples/feedback.php 		HTML
examples/feedback.pad 		Result 
<?php

  // The wall is this visitor's own, kept in the session: padSession reads it, padSessionPut
  // writes it. A post comes from the component's fetch - with $padCsrf on in the application's
  // configuration it carries the session's token in an X-CSRF-Token header, or PAD answers 403
  // before this file runs. padValidate answers a message per field that broke its rules;
  // with none the message goes on the wall. The answer to the post is the JSON of $padExpose.
  //
  // One list of rules for both sides: padValidate checks the post with it, and the component
  // gets it as padValidateClient exports it, for the browser's checker that {input:validator} in
  // the template brings - the same rules, the same messages, before the post leaves.

  $rules = [ 'name'    => 'required|max:40',
             'email'   => 'required|email',
             'mood'    => 'required|in:happy,curious,puzzled',
             'message' => 'required|min:10|max:280' ];

  $wall   = padSession ( 'reactWall', [] );
  $errors = [];
  $saved  = FALSE;

  if ( padRequestIs ( 'POST' ) ) {

    $errors = padValidate ( $rules );

    if ( ! $errors ) {

      array_unshift ( $wall, [ 'name'    => trim ( padRequest ( 'name' ) ),
                               'mood'    => padRequest ( 'mood' ),
                               'message' => trim ( padRequest ( 'message' ) ),
                               'at'      => date ( 'H:i' ) ] );

      $wall  = array_slice ( $wall, 0, 12 );
      $saved = padSessionPut ( 'reactWall', $wall );

    }

  }

  $start     = [ 'wall' => $wall, 'rules' => padValidateClient ( $rules ) ];
  $padExpose = [ 'saved', 'errors', 'wall' ];

?>
 
<div data-island="FeedbackWall" data-props="{^start}">
  <p class="island-fallback">The form needs JavaScript - it posts with fetch().</p>
</div>

{input:validator/}
<script type="text/babel" src="{$padRoot}react/examples/feedback.js"></script>
