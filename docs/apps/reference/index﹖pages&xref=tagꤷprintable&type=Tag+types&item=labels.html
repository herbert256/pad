App: regression/framework - Page: printables/a_barcode_in_a_box_of_a_printable_is_drawn_in_the_pdf
PHP
printables/a_barcode_in_a_box_of_a_printable_is_drawn_in_the_pdf.php 		HTML
printables/a_barcode_in_a_box_of_a_printable_is_drawn_in_the_pdf.pad 		Result 
<?php

  // The SVG of a barcode or a QR code in a box of a printable is drawn in the PDF: its white
  // ground and its bars as filled rectangles - the bars a path of M, h, v and z, turned into
  // the PDF's own path - its digits as text, in Courier for the monospaced font of the SVG.
  // Outside a printable an SVG is left out of a PDF as before.

  $label = padUnprotect ( padUnescape ( padCode ( "{printable:labels copies=1}{barcode:ean13 '400638133393'/}{/printable:labels}" ) ) );
  $plain = padUnprotect ( padUnescape ( padCode ( "<p>Plain</p>{barcode:ean13 '400638133393'/}" ) ) );

  $count = function ( $html ) {
    $pdf = padPdf ( $html, [ 'compress' => FALSE ] );
    preg_match_all ( '/\(([0-9]+)\) Tj/', $pdf, $digits );
    return substr_count ( $pdf, ' re f Q' ) . ' rectangles, ' . substr_count ( $pdf, ' f Q' ) - substr_count ( $pdf, ' re f Q' ) . ' paths, digits '
         . implode ( ' ', $digits [1] ) . ( str_contains ( $pdf, '/BaseFont /Courier' ) ? ' in Courier' : '' );
  };

  $answer = 'in a label: ' . $count ( $label ) . ' // in a page: ' . $count ( $plain );

?>
 
{!answer}

App: regression/framework - Page: printables/a_printable_in_a_pdf_is_a_page_of_its_own_size
PHP
printables/a_printable_in_a_pdf_is_a_page_of_its_own_size.php 		HTML
printables/a_printable_in_a_pdf_is_a_page_of_its_own_size.pad 		Result 
<?php

  // In a PDF a printable's sheet is a page of its own size - an envelope DL, 623.62 by 311.81
  // points, a sheet of labels A4 - the first one the document's first page while nothing
  // stands on it, and what the flow sets after the sheets goes on a page of the document's own
  // size, Letter here. The sheets carry no page number, and the one page of the flow none
  // either: a number stands under the pages of a flow of more than one.

  $html = padUnprotect ( padUnescape ( padCode ( <<<'PAD'
{printable:envelope sender='PAD, Utrecht'}Ann Smith{/printable:envelope}
{printable:labels sheet='a4-2x7', copies=3}Fragile{/printable:labels}
<p>After the sheets.</p>
PAD ) ) );

  $pdf = padPdf ( $html, [ 'compress' => FALSE, 'size' => 'Letter' ] );

  preg_match_all ( '#/MediaBox \[0 0 ([0-9.]+) ([0-9.]+)\]#', $pdf, $boxes, PREG_SET_ORDER );
  preg_match_all ( '#stream\n(.*?)\nendstream#s', $pdf, $streams );

  $pages = [];

  foreach ( $boxes as $i => $box ) {
    preg_match_all ( '/\(((?:[^()\\\\]|\\\\.)*)\) Tj/', $streams [1] [$i] ?? '', $texts );
    $pages [] = $box [1] . ' by ' . $box [2] . ': ' . implode ( ', ', $texts [1] );
  }

  $answer = count ( $boxes ) . ' pages - ' . implode ( ' // ', $pages );

?>
 
{!answer}
