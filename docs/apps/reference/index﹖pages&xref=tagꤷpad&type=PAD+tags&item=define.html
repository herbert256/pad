App: regression/framework - Page: sequence/an_eval_expression_is_refused_only_for_a_value_that_could_run

sequence/an_eval_expression_is_refused_only_for_a_value_that_could_run.php 		HTML
sequence/an_eval_expression_is_refused_only_for_a_value_that_could_run.pad 		Result 
<?php

  // The eval sequence refuses an expression only for a spliced value that could call code or
  // read a variable - a (, a $ or a : in it: a - or a * 2 printed in a tag's parameters
  // elsewhere on the page, and a {define} written into the expression, made eval='@ - 1',
  // eval='@ * 2' and eval='@ {twice}' refusals.

  $op   = '* 2';
  $dash = '-';

?>
 
op=[{echo '{$op}'}] result=[{sequence '1..3', eval='@ * 2'}{$sequence} {/sequence}] a=[{echo '{$dash}'}] b=[{sequence '5..3', eval='@ - 1'}{$sequence} {/sequence}] {define 'twice'}* 2{/define}c=[{sequence '1..3', eval='@ {twice}'}{$sequence} {/sequence}]
