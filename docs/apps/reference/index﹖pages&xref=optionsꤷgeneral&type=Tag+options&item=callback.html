App: regression/pages - Page: callback
PHP
callback.php 		HTML
callback.pad 		Result 
<?php

  // The row phase of a streaming callback reads the occurrence's fields as plain PHP variables.
  // A case rendered inside another request cannot show that: those fields are the nested
  // pass's locals rather than globals, so the callback never sees them. In a real request it
  // does, which is what the manual's staff example relies on and what this page pins.

  $rows = [
    [ 'n' => 1 ],
    [ 'n' => 2 ],
    [ 'n' => 3 ]
  ];

?>
 
{rows callback='stream.php'}{$n},{/rows}total={$sum}

catalog/properties.pad 		Result 
<?php

  // What the at-type lines read: a global for @all, and one for the saved group to shadow.
  // The function line runs the catsum callback from _callbacks/, whose row phase totals the
  // rows into $t.

  $catAll   = 'found';
  $catSaved = 'outer';

?>

nameProp: {cXs}{name@cXs},{/cXs}

notFirst: {cXs}{notFirst@cXs}-{/notFirst@cXs}{$cXs}{/cXs}
notLast:  {cXs}{$cXs}{notLast@cXs},{/notLast@cXs}{/cXs}
even:     {cXs}{even@cXs}E{/even@cXs}{/cXs}
odd:      {cXs}{odd@cXs}O{/odd@cXs}{/cXs}
border:   {cXs}{border@cXs}|{/border@cXs}{$cXs}{/cXs}
middle:   {data 'cYs'}[1,2,3]{/data}{cYs}{middle@cYs}M{/middle@cYs}{/cYs}
fields:   {data 'cU'}[{"a":1,"b":2}]{/data}{cU}{fields@cU}{$name}={$value},{/fields@cU}{/cU}
firstFieldName:  {cU}{firstFieldName@cU},{/cU}
firstFieldValue: {cU}{firstFieldValue@cU},{/cU}
keys:     {data 'cKv' ignore}{"a":1,"b":2}{/data}{cKv}{first@cKv}{keys@cKv}[{$name}]{/keys@cKv}{/first@cKv}{/cKv}
parameter: {true 'p1', 'p2'}{parameter.1@true}/{echo $parameter.2@true}{/true}
variable:  {true $t='9'}{variable.t@true}/{echo $variable.t@true}{/true}
option:    {true opt='v'}{echo $option.opt@true}{/true}
any:       {true $cAny='9'}{echo $cAny@any}{/true}
options:   {true opt='w'}{echo $opt@options}{/true}
parameters: {true 'q1', 'q2'}{echo $2@parameters}{/true}
saved:     {data 'cSv'}["x"]{/data}{cSv %catSaved='inner'}{$catSaved}/{echo $catSaved@saved}{/cSv}
variables: {true $cV='n'}{echo $cV@variables}{/true}
all:       {echo $catAll@all}
sequences: {sequence '1..5', push='cSeq'/}{echo $cSeq.3@sequences}
tags:      {true $cT='7'}{echo $cT@tags}{/true}
level:     {data 'cLv'}["x","y"]{/data}{cLv}{echo $0.cLv@level}{echo $1.cLv@level},{/cLv}
dataAt:    {data 'cAd'}{"cKey":9}{/data}{echo $cKey.cAd@data}
function:  {data 'cFn'}[1,2]{/data}{cFn callback='catsum.php', before}{last@cFn}{echo $t@function}{/last@cFn}{/cFn}
dataProp:  {cLv}{first@cLv}[{data@cLv}{$cLv},{/data@cLv}]{/first@cLv}{/cLv}
condition: {cLv}{if first@cLv}F{/if}{$cLv}{if notLast@cLv}-{/if}{/cLv}
