App: regression/pages - Page: db/a_database_tag_binds_a_value_put_in_as_name
PHP
db/a_database_tag_binds_a_value_put_in_as_name.php 		HTML
db/a_database_tag_binds_a_value_put_in_as_name.pad 		Result 
<?php

  // The SQL of a database tag is a quoted string, and a value goes into it as $name, bound
  // as an escaped literal of that variable - an array as a list, a row's field inside a
  // loop too. A value concatenated in was SQL, and one put in as {$name} reached the
  // database as PAD's inert stand-ins, so no tag had a way to take a value that worked.

  $who   = 'jim';
  $names = [ 'bob', 'joe' ];
  $odd   = "O'Br\\ien=a{b}";

?>
 
field:  [{field "phone from staff where name = $who"}]
array:  [{array "name from staff where name in ($names) order by name"}{$name} {/array}]
check:  [{check "staff where name = $who"}yes@else@no{/check}]
record: [{record "name, phone from staff where name = $who"}{$name} {$phone}{/record}]
odd:    [{field "concat($odd, '|')"}]
quoted: [{field "count(*) from staff where name = '$who'"}]
loop:   [{array "name from staff where salary > 2000 order by name"}{$name} {field "phone from staff where name = $name"} {/array}]

App: regression/pages - Page: db/a_float_read_from_the_database_is_bound_back_as_the_same_number
PHP
db/a_float_read_from_the_database_is_bound_back_as_the_same_number.php 		HTML
db/a_float_read_from_the_database_is_bound_back_as_the_same_number.pad 		Result 
<?php

  // A REAL or DOUBLE comes back from the database as a PHP float, and bound back - a db()
  // placeholder, a $name of a database tag - it was written with PHP's 14 significant
  // digits, 1234567.8912346, and no longer found its own row; 0.1 + 0.2 went in as 0.3.
  // A whole float stays a count after LIMIT.

  $k     = db ( "field 1234567.891234567e0 as k" );
  $sum   = db ( "field 0.1e0 + 0.2e0 as k" );
  $three = db ( "field 3.0e0 as k" );

  $same  = db ( "field {0} = 1234567.891234567e0", [ $k ] );
  $added = db ( "field {0} = 0.1e0 + 0.2e0", [ $sum ] );
  $limit = count ( db ( "array name from staff order by name limit {0}", [ $three ] ) );

?>
 
same:  [{$same}]
added: [{$added}]
bound: [{field "$k = 1234567.891234567e0"}]
limit: [{$limit}]

App: regression/pages - Page: db/a_one_row_limit_sees_past_a_trailing_comment
PHP
db/a_one_row_limit_sees_past_a_trailing_comment.php 		HTML
db/a_one_row_limit_sees_past_a_trailing_comment.pad 		Result 
<?php

  // A record, field or check whose statement ends in a line comment gets its one-row limit
  // judged and placed as if the comment were not there: its own limit before the comment is
  // seen, a semicolon or a locking clause before it is too. The tests ran on the text with
  // the comment, so "limit 1 -- the one row" got a second limit on a line after the comment,
  // as did "'bob'; -- the one": MySQL syntax errors near 'limit 0,1'.

  $count  = db ( "field count(*) from staff where name = 'bob' limit 1 -- the one row" );
  $first  = db ( "record name from staff order by name limit 1 # first" );
  $check  = db ( "check staff where salary > 0 limit 1 -- any" );
  $semi   = db ( "field name from staff where name = 'bob'; -- the one" );
  $locked = db ( "field phone from staff where name = 'jim' for update -- lock it" );
  $plain  = db ( "check staff where salary > 0 -- any" );

?>
 
count:  {$count}
first:  {$first.name}
check:  {$check}
semi:   {$semi}
locked: {$locked}
plain:  {$plain}
tag:    {field "count(*) from staff where name = 'bob' limit 1 -- the count"}

App: regression/pages - Page: db/sqlite_binds_a_name_after_a_literal_ending_in_a_backslash
PHP
db/sqlite_binds_a_name_after_a_literal_ending_in_a_backslash.php 		HTML
db/sqlite_binds_a_name_after_a_literal_ending_in_a_backslash.pad 		Result 
<?php

  // On SQLite a backslash in a quoted literal is a character like any other, so 'C:\' ends
  // where its quote does and a $name after it is bound. The bind took the backslash for an
  // escape on every driver: the quote that ends the literal counted as escaped, the rest of
  // the statement as quoted, and $n went to SQLite as written - a parameter of its own, NULL.

  $GLOBALS ['padSqlConnect'] = padDbSqlite ( ':memory:' );

  db ( "create table staff (name text, folder text)" );
  db ( "insert into staff values ('bob', 'C:\\'), ('jim', 'D:\\')" );

  $n = 'bob';

?>
 
A[{field "count(*) from staff where name = $n"}] B[{field "count(*) from staff where folder <> 'D:\\' and name = $n"}] C[{record "folder from staff where folder = 'C:\\' and name = $n"}{$folder}{/record}]
