App: regression/pages - Page: db/a_negative_number_after_a_hyphen_stays_as_written_outside_sql_code
PHP
db/a_negative_number_after_a_hyphen_stays_as_written_outside_sql_code.php 		HTML
db/a_negative_number_after_a_hyphen_stays_as_written_outside_sql_code.pad 		Result 
<?php

  // A negative number spliced right after a minus gets a space before it in the SQL a tag
  // writes, outside its quoted literals, where 2000--1000 is a line comment on SQLite - and
  // nowhere else: the page's text, an attribute, an echo and a literal keep the value as it is,
  // a literal in the fields= or join= of a select too, where no other value may stand.

  $lo  = -12;
  $hi  = -4;
  $id  = -1;
  $adj = -1000;

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name' ];

?>
 
text:    [{$lo}-{$hi}] [row-{$id}] <i id="item-{$id}"></i>
echo:    [{echo 'UTC-{$id}'/}]
literal: [{field "concat('X-{$id}')"/}]
code:    [{field "2000-{$adj}"/}]
fields:  [{staffSel fields="name, concat('X-{$id}') as code", rows=1}{$code}{/staffSel}]
join:    [{staffSel join="staff as b on b.name = staff.name and concat('X-{$id}') = 'X--1'", fields="staff.name", rows=1}{$name}{/staffSel}]

<?php

  // A select with a join meets its key and its relations with its own table's columns: a
  // field both tables hold - orderNumber in orders and orderdetails, customerNumber in orders

  $padSelect ['orderLines'] = [ 'db' => 'orders', 'key' => 'orderNumber', 'fields' => 'orders.orderNumber',
                                'join' => [ 'inner' => 'lineDetail', 'key' => 'orders.orderNumber' ],
                                'order' => 'orderdetails.orderLineNumber' ];

  $padRelations ['orders'] ['customers'] = 'customerNumber';

?>
 
key: [{orderLines $orderNumber=10100}{$product} {/orderLines}]
nested: [{customers $customerNumber=103}{orders join="payments on payments.customerNumber = orders.customerNumber", fields="count(*) as n"}{$n}{/orders}{/customers}]

  // The {pager} of a joined select whose select list decides what a row is - distinct, a

<?php

  // Paging a declared select table: the page is cut once, by the SQL limit, so page 2 holds
  // the third and fourth rows; a row count in the declaration is a row count, not the
  // query's command word; and a joined table declared without fields joins, adding none.
  // Over the four-row staff table and the join tables of the demo database.

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name' ];
  $padSelect ['staffTwo'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name', 'rows' => 2 ];

  $padSelect ['bare']     = [ 'db' => 'table2', 'key' => 'key' ];
  $padSelect ['joined']   = [ 'db' => 'table1', 'fields' => 'table1.text', 'order' => 'table1.key',
                              'join' => [ 'inner' => 'bare', 'key' => 'table1.key' ] ];

?>
 
page 1: [{staffSel page=1, rows=2}{$name} {/staffSel}]
page 2: [{staffSel page=2, rows=2}{$name} {/staffSel}]
declared rows: [{staffTwo}{$name} {/staffTwo}]
join without fields: [{joined}{$text} {/joined}]
