App: regression/pages - Page: db/sqlite_numeric_text_keeps_its_zeros
PHP
db/sqlite_numeric_text_keeps_its_zeros.php 		HTML
db/sqlite_numeric_text_keeps_its_zeros.pad 		Result 
<?php

  // On SQLite numeric text goes into a bare placeholder, or a select's $name, as a number
  // only when SQLite gives the same text back: '007' became the integer 7, so the code
  // '007' found the row of code '7', the zip '01234' was stored as '1234' and '1.10' as
  // '1.1'. A numeral like '3' stays a number, so a count still compares with it.

  $GLOBALS ['padSqlConnect'] = padDbSqlite ( ':memory:' );

  db ( "create table codes (code text primary key, zip varchar(10), label text)" );
  db ( "insert into codes values ('007', '01234', 'bond'), ('7', '1234', 'seven')" );

  $byCode = db ( "field label from codes where code = {0}",                   [ '007' ] );
  $byZip  = db ( "field zip from codes where zip = {0}",                      [ '01234' ] );
  $inList = db ( "array code from codes where code in ({0}) order by code",   [ [ '007', '7' ] ] );
  $inList = implode ( ' ', array_column ( $inList, 'code' ) );

  db ( "insert into codes values ({0}, {1}, 'kept')", [ '0042', '1.10' ] );

  $stored = db ( "record code, zip from codes where label = 'kept'" );
  $count  = db ( "field 'three' where (select count(*) from codes) = {0}",    [ '3' ] );

  $padSelect ['codes'] = [ 'db' => 'codes', 'key' => 'code' ];

  $c = '007';

?>
 
code:   {$byCode}
zip:    {$byZip}
list:   {$inList}
stored: {$stored.code} {$stored.zip}
count:  {$count}
select: {codes where="code = $c"}{$label}{/codes}

App: regression/pages - Page: select/demo/sqlite_key_and_relation_match_an_untyped_column
PHP
select/demo/sqlite_key_and_relation_match_an_untyped_column.php 		HTML
select/demo/sqlite_key_and_relation_match_an_untyped_column.pad 		Result 
<?php

  // On SQLite a key bound on a select tag and the relation to the enclosing select compare
  // a number as a number: the value went in as quoted text, and a column declared without
  // a type has no affinity, so '2' never equalled the integer 2 - {boards $id=2} found no
  // board, and every board's topics came out empty. A key that only looks like a number,
  // '007', stays text.

  $GLOBALS ['padSqlConnect'] = padDbSqlite ( ':memory:' );

  db ( "create table boards (id, name)" );
  db ( "create table topics (id integer primary key, board_id, title text)" );
  db ( "create table codes (code, label)" );
  db ( "insert into boards values (1, 'one'), (2, 'two')" );
  db ( "insert into topics (board_id, title) values (1, 'a'), (1, 'b'), (2, 'c')" );
  db ( "insert into codes values ('007', 'bond'), (7, 'seven')" );

  $padSelect ['boards'] = [ 'db' => 'boards', 'key' => 'id' ];
  $padSelect ['topics'] = [ 'db' => 'topics', 'key' => 'id' ];
  $padSelect ['codes']  = [ 'db' => 'codes',  'key' => 'code' ];

  $padRelations ['topics'] ['boards'] = [ 'board_id' => 'id' ];

  $wanted = '2';
  $code   = '007';

?>
 
key:      [{boards $id=2}{$name}@else@none{/boards}]
request:  [{boards $id=$wanted}{$name}@else@none{/boards}]
relation: [{boards}{$name}={topics}{$title}{/topics} {/boards}]
text:     [{codes $code=$code}{$label}@else@none{/codes}]
