================================================================================
A pair is a block
================================================================================

{array:orders}
  <li>{$id}</li>
{/array:orders}

--------------------------------------------------------------------------------

(template
  (block
    (open_tag
      (tag_name
        (prefix)
        (identifier)))
    (body
      (text)
      (field_tag
        (variable))
      (text))
    (close_tag
      (tag_name
        (prefix)
        (identifier)))))
