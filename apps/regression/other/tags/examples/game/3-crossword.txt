A game's
   script registers itself - PadGames.register('memory', function (k, cfg) { ... }) - and is
   started on every board of its kind: the ones on the page, and the ones a fragment, a live
   region or a script brings later. k is the game's kit: its board, its status line and its
   bar, a random number seeded by seed= (the same games on every load) or by the browser,
   timers that stop with the game, and the end of a game, which the board carries as
   data-result="won" or "lost" and announces with a pad-game-end event. */

(function () {
  'use strict';

  if (window.PadGames) return;

  var kinds = {};

  // mulberry32: a small, fast generator whose numbers follow from the seed alone.

  function random(seed) {
    var a = seed >>> 0;
    return function () {
      a = (a + 0x6D2B79F5) >>> 0;
      var t = a;
      t = Math.imul(t ^ (t >>> 15), t | 1);
      t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  function el(tag, cls, text) {
    var e = document.createElement(tag);
    if (cls) e.className = cls;
    if (text !== undefined && text !== null) e.textContent = text;
    return e;
  }

  function Kit(root, cfg) {
    this.root = root;
    this.cfg = cfg;
    this.board = root.querySelector('.pg-board');
    this.bar = root.querySelector('.pg-bar');
    this.statusEl = root.querySelector('.pg-status');
    this.seed = (typeof cfg.seed === 'number') ? cfg.seed : Math.floor(Math.random() * 2147483647);
    this.rand = random(this.seed);
    this.timers = [];
  }

  Kit.prototype = {
    el: el,
    int: function (n) { return Math.floor(this.rand() * n); },
    pick: function (list) { return list[this.int(list.length)]; },
    shuffle: function (list) {
      for (var i = list.length - 1; i > 0; i--) {
        var j = this.int(i + 1), t = list[i];
        list[i] = list[j];
        list[j] = t;
      }
      return list;
    },
    status: function (text) { this.statusEl.textContent = text; },
    button: function (label, fn, cls) {
      var b = el('button', 'pg-button' + (cls ? ' ' + cls : ''), label);
      b.type = 'button';
      b.addEventListener('click', fn);
      this.bar.appendChild(b);
      return b;
    },
    // A new game: the board emptied, the result and the timers of the last one gone.
    fresh: function () {
      this.stop();
      this.board.textContent = '';
      this.root.classList.remove('is-won', 'is-lost');
      this.root.removeAttribute('data-result');
    },
    won: function (text) { this.finish('won', text); },
    lost: function (text) { this.finish('lost', text); },
    finish: function (result, text) {
      this.stop();
      this.root.classList.add(result === 'won' ? 'is-won' : 'is-lost');
      this.root.setAttribute('data-result', result);
      this.status(text);
      this.root.dispatchEvent(new CustomEvent('pad-game-end', { bubbles: true, detail: { result: result } }));
    },
    over: function () { return this.root.hasAttribute('data-result'); },
    later: function (fn, ms) { var t = setTimeout(fn, ms); this.timers.push(t); return t; },
    every: function (fn, ms) { var t = setInterval(fn, ms); this.timers.push(t); return t; },
    stop: function () {
      this.timers.forEach(function (t) { clearTimeout(t); clearInterval(t); });
      this.timers = [];
    },
    // Keys the game reads while it has the focus; the arrows then do not scroll the page.
    keys: function (map) {
      this.root.addEventListener('keydown', function (e) {
        if (e.altKey || e.ctrlKey || e.metaKey) return;
        var fn = map[e.key];
        if (fn && fn(e) !== false) e.preventDefault();
      });
    },
    // A swipe over the board, as the arrow it points: 'ArrowUp', 'ArrowLeft' ...
    swipe: function (fn) {
      var x0 = null, y0 = null, board = this.board;
      board.addEventListener('touchstart', function (e) { x0 = e.touches[0].clientX; y0 = e.touches[0].clientY; }, { passive: true });
      board.addEventListener('touchend', function (e) {
        if (x0 === null) return;
        var dx = e.changedTouches[0].clientX - x0, dy = e.changedTouches[0].clientY - y0;
        x0 = null;
        if (Math.max(Math.abs(dx), Math.abs(dy)) < 24) return;
        fn(Math.abs(dx) > Math.abs(dy) ? (dx > 0 ? 'ArrowRight' : 'ArrowLeft') : (dy > 0 ? 'ArrowDown' : 'ArrowUp'));
      });
    }
  };

  function start(root) {
    if (root.padGameKit) return;
    var init = kinds[root.getAttribute('data-pad-game')];
    if (!init) return;
    var cfg = {};
    try { cfg = JSON.parse(root.getAttribute('data-config') || '{}') || {}; } catch (e) { cfg = {}; }
    var kit = new Kit(root, cfg);
    root.padGameKit = kit;
    root.classList.add('is-ready');
    init(kit, cfg);
  }

  // The games of the page, or of a part of it - and of the open shadow roots in it: a game in
  // a web component's {shadow}, which the document's own querySelectorAll does not see into.
  // A shadow root found is watched for games that come, as the page is.
  function scan(where) {
    where = where || document;
    var list = where.querySelectorAll('.pad-game[data-pad-game]');
    for (var i = 0; i < list.length; i++) start(list[i]);
    var all = where.querySelectorAll('*');
    for (var j = 0; j < all.length; j++)
      if (all[j].shadowRoot) { watch(all[j].shadowRoot); scan(all[j].shadowRoot); }
  }

  window.PadGames = {
    register: function (name, init) { kinds[name] = init; scan(); },
    scan: scan,
    random: random,
    el: el
  };

  document.addEventListener('DOMContentLoaded', function () { scan(); });

  var watcher = window.MutationObserver ? new MutationObserver(function (records) {
    for (var i = 0; i < records.length; i++)
      for (var j = 0; j < records[i].addedNodes.length; j++) {
        var n = records[i].addedNodes[j];
        if (n.nodeType === 1 && (n.matches('.pad-game') || n.querySelector('.pad-game') || n.shadowRoot)) { scan(); return; }
      }
  }) : null;

  if (watcher) watcher.observe(document.documentElement, { childList: true, subtree: true });

  var watched = [];
  function watch(root) {
    if (!watcher || watched.indexOf(root) >= 0) return;
    watched.push(root);
    watcher.observe(root, { childList: true, subtree: true });
  }
})();

  var R = cfg.rows, C = cfg.cols, letters, inputs = [], across = true, current = null, clueItems = [];

  letters = new TextDecoder().decode(Uint8Array.from(atob(cfg.letters), function (c) { return c.charCodeAt(0); })).split('\n');

  // The clue of a square in a direction, and the squares of that clue.
  function clueOf(i, way) {
    for (var n = 0; n < cfg.clues.length; n++) if (cfg.clues[n].across === way && cfg.clues[n].squares.indexOf(i) >= 0) return cfg.clues[n];
    return null;
  }

  function focus(i, way) {
    if (way !== undefined) across = way;
    if (!clueOf(i, across)) across = !across;
    current = clueOf(i, across);
    inputs[i].focus();
    mark();
  }

  function mark() {
    inputs.forEach(function (inp) { if (inp) inp.parentNode.classList.remove('is-word'); });
    clueItems.forEach(function (li) { li.classList.remove('is-current'); });
    if (!current) return;
    current.squares.forEach(function (j) { inputs[j].parentNode.classList.add('is-word'); });
    var li = clueItems[cfg.clues.indexOf(current)];
    li.classList.add('is-current');
    clueBar.textContent = current.n + (current.across ? ' across: ' : ' down: ') + current.clue;
  }

  function step(i, forward) {
    if (!current) return;
    var at = current.squares.indexOf(i), next = current.squares[at + (forward ? 1 : -1)];
    if (next !== undefined) inputs[next].focus();
  }

  function check(marking) {
    var right = 0, total = 0, wrong = 0;
    inputs.forEach(function (inp, i) {
      if (!inp) return;
      total++;
      var v = inp.value.toUpperCase(), ok = v === letters[i].toUpperCase();
      if (ok) right++;
      if (marking) inp.parentNode.classList.toggle('is-wrong', !!v && !ok);
      if (v && !ok) wrong++;
    });
    if (right === total) {
      inputs.forEach(function (inp) { if (inp) { inp.readOnly = true; inp.parentNode.classList.remove('is-wrong'); } });
      return k.won('Solved - every word is right!');
    }
    if (marking) k.status(wrong ? wrong + (wrong === 1 ? ' letter is wrong' : ' letters are wrong') + ' - marked in red' : 'Nothing wrong so far - ' + (total - right) + ' squares to go.');
  }

  function reveal(squares) {
    squares.forEach(function (j) { inputs[j].value = letters[j]; inputs[j].parentNode.classList.add('is-revealed'); inputs[j].parentNode.classList.remove('is-wrong'); });
    check(false);
  }

  var grid = k.el('div', 'cwd-grid'), clueBar = k.el('p', 'cwd-bar');
  grid.style.gridTemplateColumns = 'repeat(' + C + ', var(--cwd-size))';
  var numbers = {};
  cfg.clues.forEach(function (c) { numbers[c.squares[0]] = c.n; });

  for (var i = 0; i < R * C; i++) (function (i) {
    if (!letters[i]) { grid.appendChild(k.el('span', 'cwd-block')); inputs.push(null); return; }
    var sq = k.el('span', 'cwd-square'), inp = k.el('input', 'cwd-input');
    if (numbers[i]) sq.appendChild(k.el('span', 'cwd-number', numbers[i]));
    inp.maxLength = 1;
    inp.autocomplete = 'off';
    inp.setAttribute('autocapitalize', 'characters');
    inp.setAttribute('aria-label', 'row ' + (Math.floor(i / C) + 1) + ', column ' + (i % C + 1) + (numbers[i] ? ', ' + numbers[i] : ''));
    inp.addEventListener('focus', function () { if (!current || current.squares.indexOf(i) < 0) { current = clueOf(i, across) || clueOf(i, !across); if (current) across = current.across; mark(); } inp.select(); });
    inp.addEventListener('mousedown', function (e) {
      if (document.activeElement === inp && clueOf(i, !across)) { e.preventDefault(); focus(i, !across); }
    });
    inp.addEventListener('input', function () {
      inp.value = inp.value.slice(-1).toUpperCase();
      sq.classList.remove('is-wrong', 'is-revealed');
      if (inp.value) step(i, true);
      check(false);
    });
    inp.addEventListener('keydown', function (e) {
      var r = Math.floor(i / C), c = i % C, to = null;
      if (e.key === 'Backspace' && !inp.value) { step(i, false); e.preventDefault(); return; }
      if (e.key === 'ArrowRight') to = [r, c + 1, true];
      if (e.key === 'ArrowLeft') to = [r, c - 1, true];
      if (e.key === 'ArrowDown') to = [r + 1, c, false];
      if (e.key === 'ArrowUp') to = [r - 1, c, false];
      if (!to) return;
      e.preventDefault();
      for (var rr = to[0], cc = to[1]; rr >= 0 && cc >= 0 && rr < R && cc < C; rr += to[0] - r, cc += to[1] - c) {
        if (inputs[rr * C + cc]) { focus(rr * C + cc, to[2]); return; }
        if (to[0] === r && to[1] === c) break;
      }
    });
    sq.appendChild(inp);
    grid.appendChild(sq);
    inputs.push(inp);
  })(i);

  var lists = k.el('div', 'cwd-lists');
  [true, false].forEach(function (way) {
    var mine = cfg.clues.filter(function (c) { return c.across === way; });
    if (!mine.length) return;
    var sec = k.el('section', 'cwd-clues');
    sec.appendChild(k.el('h4', null, way ? 'Across' : 'Down'));
    var ol = k.el('ol');
    mine.forEach(function (c) {
      var li = k.el('li', null);
      li.value = c.n;
      var b = k.el('button', 'cwd-clue', c.clue + ' (' + c.squares.length + ')');
      b.type = 'button';
      b.addEventListener('click', function () { focus(c.squares[0], c.across); });
      li.appendChild(b);
      ol.appendChild(li);
      clueItems[cfg.clues.indexOf(c)] = li;
    });
    sec.appendChild(ol);
    lists.appendChild(sec);
  });
  if (cfg.skipped && cfg.skipped.length) lists.appendChild(k.el('p', 'pg-note', 'Not placed: ' + cfg.skipped.join(', ')));

  var wrap = k.el('div', 'cwd-wrap'), left = k.el('div', 'cwd-left');
  left.appendChild(clueBar);
  left.appendChild(grid);
  wrap.appendChild(left);
  wrap.appendChild(lists);
  k.board.appendChild(wrap);

  k.button('Check', function () { if (!k.over()) check(true); });
  k.button('Reveal word', function () { if (!k.over() && current) reveal(current.squares); });
  k.button('Reveal all', function () { if (!k.over()) { var all = []; inputs.forEach(function (inp, j) { if (inp) all.push(j); }); reveal(all); } });
  k.button('Clear', function () {
    inputs.forEach(function (inp) { if (inp) { inp.value = ''; inp.readOnly = false; inp.parentNode.classList.remove('is-wrong', 'is-revealed'); } });
    k.root.classList.remove('is-won');
    k.root.removeAttribute('data-result');
    k.status(cfg.clues.length + ' words - click a square or a clue and type');
  });
  k.status(cfg.clues.length + ' words - click a square or a clue and type');
});
