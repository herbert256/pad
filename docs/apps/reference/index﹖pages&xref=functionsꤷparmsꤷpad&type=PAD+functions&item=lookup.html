<?php

  $customers = [
    [ 'id' => 1, 'name' => 'Ann',  'country' => 'NL' ],
    [ 'id' => 2, 'name' => 'Bob',  'country' => 'BE' ]
  ];

  $orders = [
    [ 'number' => 101, 'customer_id' => 2 ],
    [ 'number' => 102, 'customer_id' => 1 ],
    [ 'number' => 103, 'customer_id' => 7 ]
  ];

?>
 
{orders}
  <p>Order {$number}: {echo $customer_id | lookup('customers', 'id', 'name') | ?? 'unknown customer'}</p>
{/orders}
