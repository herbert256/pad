App: showcase - Page: progress/uploads
PHP
progress/uploads.php 		HTML
progress/uploads.pad 		Result 
<?php

  $uploads = [
    [ 'file' => 'holiday-photos.zip',  'done' => 100, 'color' => 'success' ],
    [ 'file' => 'quarterly-report.pdf','done' => 72,  'color' => 'accent'  ],
    [ 'file' => 'site-backup.tar.gz',  'done' => 38,  'color' => 'accent'  ],
    [ 'file' => 'video-raw.mov',       'done' => 91,  'color' => 'warning' ],
    [ 'file' => 'database.sql',        'done' => 12,  'color' => 'danger'  ] ];

?>
 
{meta title='Uploads in progress', about='A native progress element per file, styled, with its label and percentage - the colour slot says how it goes.'}
<div style="display:grid; gap:18px; width:min(100%, 460px); margin:0 auto">
  {uploads}{progress $done, label=$file, color=$color}{/uploads}
</div>
