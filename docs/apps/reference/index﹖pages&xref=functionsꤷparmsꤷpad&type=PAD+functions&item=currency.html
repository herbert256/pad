fragments/locale_1.pad 		Result 
<?php

  // The page picks Dutch; an application listing $padLocales lets the visitor's
  // Accept-Language or ?lang= choose instead.

  $padLocale = 'nl';

  $count = 3;
  $user  = 'Ann';
  $total = 1234.5;
  $when  = '2026-10-05';

?>
 
<h4>{trans 'cart.title'}</h4>
<p>{trans 'cart.items', count=$count} - {trans 'cart.owner', name=$user}</p>
<p>{$total | currency('EUR', 'nl_NL')} - {$when | localDate('d MMMM y')}</p>
