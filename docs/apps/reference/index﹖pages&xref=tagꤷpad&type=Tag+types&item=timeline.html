App: regression/framework - Page: tags/a_date_without_a_zone_keeps_its_clock_in_the_hour_summer_time_skips
PHP
tags/a_date_without_a_zone_keeps_its_clock_in_the_hour_summer_time_skips.php 		HTML
tags/a_date_without_a_zone_keeps_its_clock_in_the_hour_summer_time_skips.pad 		Result 
<?php

  // A date written without a zone is read as it stands, also when its clock time falls in
  // the hour the application's timezone skips: read in that zone, 2026-03-29 02:30 in
  // Amsterdam became 03:30, and in Santiago, where the summer starts at midnight, the plain
  // date 2026-09-06 became 01:00 of that day - a moment, no longer a day. A date with its
  // offset is still put in the application's timezone.

  include_once PAD . 'tags/_lib/chart.php';

  $read = [];

  foreach ( [ 'Europe/Amsterdam' => [ '2026-03-29 02:30', '2026-03-29T00:30:00+00:00' ],
              'America/Santiago' => [ '2026-09-06', '2026-09-06 12:00' ] ] as $zone => $dates ) {
    $padTimezone = $zone;
    foreach ( $dates as $one ) {
      $stamp   = padChartStamp ( $one );
      $read [] = $one . '=' . ( $stamp === FALSE ? 'no date' : gmdate ( 'Y-m-d H:i', $stamp ) );
    }
  }

  $padTimezone = 'America/Santiago';

  $agenda = [ [ 'when' => '2026-09-06', 'what' => 'Spring' ] ];

  $read = implode ( '; ', $read );

?>
 
{$read} | {timeline data='agenda', date='when', label='what', vertical | php:substr_count(@, 'Sep 6, 2026: Spring')}

<?php

  // A format= that names the zone or the epoch writes a timestamp or a date with its offset
  // as the moment it is in the application's timezone - its zone's name, offset and the
  // real timestamp - where the wall clock was written as if in UTC: midnight in Amsterdam
  // said 00:00 GMT (+00:00), and U was the timestamp plus two hours. A date written without
  // a zone has none to name and is written in UTC, as before.

  $padTimezone = 'Europe/Amsterdam';

  $agenda = [ [ 'when' => '1760306400',                'what' => 'stamp'  ],
              [ 'when' => '2025-10-13T00:30:00+02:00', 'what' => 'offset' ],
              [ 'when' => '2025-10-13 00:30:00',       'what' => 'plain'  ] ];

?>

  // lays them: across the page the place of each card took time with the cube of their
