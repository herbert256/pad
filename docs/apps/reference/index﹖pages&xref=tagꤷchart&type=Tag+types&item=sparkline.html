<?php

  // Five shares and their closing price over the last fifteen days.

  $stocks = [
    'ASML'    => [ 640, 652, 648, 661, 670, 668, 684, 679, 692, 701, 698, 712, 709, 725, 731 ],
    'Philips' => [ 21.4, 21.1, 20.8, 21.0, 20.6, 20.2, 20.5, 19.9, 19.6, 19.8, 19.3, 19.5, 19.1, 18.8, 18.9 ],
    'Shell'   => [ 30.1, 30.4, 30.2, 30.9, 31.3, 30.8, 30.6, 31.1, 31.4, 31.0, 31.6, 31.9, 31.5, 31.8, 32.2 ],
    'Heineken'=> [ 88, 87, 89, 86, 88, 90, 87, 85, 88, 89, 86, 87, 90, 88, 89 ],
    'Adyen'   => [ 1210, 1185, 1240, 1302, 1276, 1330, 1295, 1361, 1408, 1372, 1440, 1415, 1478, 1452, 1499 ]
  ];

  $shares = [];

  foreach ( $stocks as $name => $closes )
    $shares [] = [ 'name'   => $name,
                   'closes' => $closes,
                   'last'   => end ( $closes ),
                   'change' => round ( ( end ( $closes ) / $closes [0] - 1 ) * 100, 1 ) ];

?>
 
<table class="quotes">
  <tr><th>Share</th><th>15 days</th><th>Last</th><th>Change</th></tr>
  {array:shares}
  <tr>
    <td>{$name}</td>
    <td>{chart:sparkline data='closes', width=140, height=30, title=$name/}</td>
    <td>{$last}</td>
    <td class="{pad:echo $change ge 0 ? 'up' : 'down'/}">{$change} %</td>
  </tr>
  {/array:shares}
</table>

App: sequence - Page: gallery
PHP
gallery.php 		HTML
gallery.pad 		Result 
<?php

  // The gallery: every sequence type that needs no parameter, its first 24 terms drawn as a
  // {chart:sparkline}, the first eight written out, and the OEIS entry the terms come from - a
  // link to oeis.org, found in the table the oeis type reads (seqFunOeis in _lib/fun.php).

  $drawn = [];

  foreach ( seqFunTypes () as $type ) {

    $terms = seqFunTerms ( $type, 24 );
    $oeis  = seqFunOeis ( $terms, $type );

    $drawn [] = [
      'type'  => $type,
      'first' => implode ( ', ', array_slice ( $terms, 0, 8 ) ),
      'oeis'  => $oeis ? sprintf ( 'A%06d', $oeis ) : ''
    ];

  }

?>
 
<p>Every sequence type that needs no parameter, its first 24 terms drawn by
<code>{chart:sparkline sequence=$type, rows=24}</code>, beside the
<a href="https://oeis.org">OEIS</a> entry the terms come from. Follow a name to hear it.</p>

<table cellpadding="4">
  {array:drawn}
    <tr>
      <td align="right"><a href="?listen&type={$type}">{$type}</a></td>
      <td>{chart:sparkline sequence=$type, rows=24/}</td>
      <td align="left"><small>{$first}, ...</small></td>
      <td>{pad:if $oeis ne ''}<a href="https://oeis.org/{$oeis}">{$oeis}</a>{/pad:if}</td>
    </tr>
  {/array:drawn}
</table>
