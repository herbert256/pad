<?php

  // Examples page: picks the group of examples to show and the one item out of it.
  //
  // example= names a directory of the application, so it is only used when it is one of the
  // groups the page offers - the same list examples.pad walks. Interpolating it unchecked
  // meant a ../ read the file names of any directory the server could reach, and an unknown
  // group ended the request with a 500. item= is checked against the group's own contents,
  // which it already was.

  $examples = [ 'sequences', 'specials', 'random', 'play/single', 'play/double', 'keepRemoveFlag' ];

  if ( ! isset ( $example ) or ! in_array ( $example, $examples ) )
    $example = 'specials';

  if ( $example == 'specials' )
    if ( ! isset ( $item ) or ! $item )
      $item = 'basic';

  if ( $example == 'sequences')
    $exam = sequenceDir ( APP . "basic" ) ;
  else
    $exam = sequenceDir ( APP . "$example" ) ;

  if ( ! count ( $exam ) )
    $exam = [ 'basic' ];

  if ( isset ( $item ) and ! in_array ( $item, $exam) )
    $item = $exam [0];

  if ( ! isset ( $item ) or ! $item )
    $item = $exam [0];

?>
 
{data 'list'} 
  (
    'sequences',
    'specials',
    'random',
    'play/single',
    'play/double',
    'keepRemoveFlag'
  )
{/data}

{block}

<p>Examples ! Many are generated and do not always make sence, <br>the examples from 'specials' are hand crafted and gives a good view of what the Sequences subsystem of PAD can do.</p>

{/block}

<br>
{lines}
  {list}
    {if $example eq $list}
      <b>{$list}</b>
    @else@
      <a href="?examples&example={$list}{?item}">{$list}</a>
    {/if}
  {notLast} | {/notLast}
  {/list}
{/lines}

<br>

{block}<center>

  {exam} 
    {if $item eq $exam}
      <b>{$exam}</b>
    @else@
      <a href="?examples&example={$example}&item={$exam}">{$exam}</a>
    {/if}
    &nbsp;
  {/exam}

  <br><br><hr>

  {if $example eq 'sequences' }
    <h3>{$item}</h3>
      {exists APP . 'basic/{$item}.pad'}
        <h5>{echo 'basic/{$item}.pad'}</h5>
        {example 'basic/{$item}', onlyResult}
      {/exists}
    {list}

gallery.php 		HTML
gallery.pad 		Result 
<?php

  // The gallery: every sequence type that needs no parameter, its first 24 terms drawn as a
  // {sparkline}, the first eight written out, and the OEIS entry the terms come from - a
  // link to oeis.org, found in the table the oeis type reads (seqFunOeis in _lib/fun.php).

  $drawn = [];

  foreach ( seqFunTypes () as $type ) {

    $terms = seqFunTerms ( $type, 24 );
    $oeis  = seqFunOeis ( $terms, $type );

    $drawn [] = [
      'type'  => $type,
      'first' => implode ( ', ', array_slice ( $terms, 0, 8 ) ),
      'oeis'  => $oeis ? sprintf ( 'A%06d', $oeis ) : ''
    ];

  }

?>
 
<p>Every sequence type that needs no parameter, its first 24 terms drawn by
<code>{sparkline sequence=$type, rows=24}</code>, beside the
<a href="https://oeis.org">OEIS</a> entry the terms come from. Follow a name to hear it.</p>

<table cellpadding="4">
  {drawn}
    <tr>
      <td align="right"><a href="?listen&type={$type}">{$type}</a></td>
      <td>{sparkline sequence=$type, rows=24}</td>
      <td align="left"><small>{$first}, ...</small></td>
      <td>{if $oeis ne ''}<a href="https://oeis.org/{$oeis}">{$oeis}</a>{/if}</td>
    </tr>
  {/drawn}
</table>
App: sequence Page: listen
PHP
listen.php 		HTML
listen.pad 		Result 
<?php

  // Listen: a sequence played as notes. Each of the first 32 terms becomes a note - the
  // term modulo 60, five octaves up from C2, the way the OEIS 'listen' button folds a term
  // into the keyboard - and the notes are written as a WAV file into the page itself
  // (seqFunWav in _lib/fun.php), so an <audio> element plays them without JavaScript.
  // Recamán's sequence, which jumps back and forth, is the famous one to listen to.
  //
  // type arrives from the query string and is used only when it names a type the gallery
  // draws.

  $choices = seqFunTypes ();

  if ( ! isset ( $type ) or ! is_string ( $type ) or ! in_array ( $type, $choices ) )
    $type = 'recaman';

  $terms = seqFunTerms ( $type, 32 );
  $oeis  = seqFunOeis ( $terms, $type );
  $oeis  = $oeis ? sprintf ( 'A%06d', $oeis ) : '';
  $list  = implode ( ', ', $terms );
  $notes = implode ( ' ', array_map ( 'seqFunNoteName', seqFunNotes ( $terms ) ) );
  $audio = seqFunWav ( $terms );

?>
 
<p>{choices}{notFirst} | {/notFirst}{if $choices eq $type}<b>{$choices}</b>@else@<a href="?listen&type={$choices}">{$choices}</a>{/if}{/choices}</p>

<h3>{$type | ucfirst}{if $oeis ne ''} - <a href="https://oeis.org/{$oeis}">{$oeis}</a>{/if}</h3>

{chart 'line', data='terms', title='{$type | ucfirst}, the first 32 terms', width=640, height=220}

<p><audio controls preload="none" src="{$audio}"></audio></p>

<p>Each term is a note: the term modulo 60, counted in semitones up from C2.</p>

<p><small>{$list}</small></p>

<p><small>{$notes}</small></p>
