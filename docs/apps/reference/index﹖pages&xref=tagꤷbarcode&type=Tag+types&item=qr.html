App: demo - Page: qr

qr.php 		HTML
qr.pad 		Result 
<?php

  $title = 'QR Codes';

  // The text to encode: what the form posted - its rules, on the field in qr.pad, are kept
  // before this runs - else the address of PAD itself.

  if ( ! padPosted ( 'qr' ) or ! is_string ( $text ?? NULL ) )
    $text = 'https://github.com/herbert256/pad';

  $levels = [
    [ 'level' => 'L', 'restores' => '7%'  ],
    [ 'level' => 'M', 'restores' => '15%' ],
    [ 'level' => 'Q', 'restores' => '25%' ],
    [ 'level' => 'H', 'restores' => '30%' ],
  ];

  $wifi = 'WIFI:T:WPA;S:PAD Demo;P:inversion-of-control;;';

?>
 
<p>The <code>{pad:ignore}{barcode:qr}{/pad:ignore}</code> tag encodes a text as a QR code and writes it as inline SVG -
no image library, no remote service, no JavaScript. Type a text and scan the result with your phone.</p>

{pad:form 'qr', error='Please correct the errors below.'}
  <p>{pad:input 'text', label='Text or address:', value=$text, rules='required|max:1000'/}</p>
  <p><input type="submit" value="Make the QR code"></p>
{/pad:form}

<div class="qr-big">{barcode:qr $text, size=220/}</div>

<p><code>{pad:ignore}{barcode:qr $text, size=220/}{/pad:ignore}</code></p>

<h2>Error correction</h2>

<p>The higher the level, the more of the symbol may be damaged or covered - a logo in the
middle - and the more modules it takes.</p>

<table>
  <tr>
    {array:levels}<th>level='{$level}' - restores {$restores}</th>{/array:levels}
  </tr>
  <tr>
    {array:levels}<td>{barcode:qr $text, level=$level, size=150/}</td>{/array:levels}
  </tr>
</table>

<h2>Not only addresses</h2>

<p>A phone joins a Wi-Fi network from this one:</p>

<p>{barcode:qr $wifi, size=150, title='Join the PAD Demo network', color='#2e7d32'/}</p>

<p><code>{pad:ignore}{barcode:qr 'WIFI:T:WPA;S:PAD Demo;P:inversion-of-control;;', color='#2e7d32'/}{/pad:ignore}</code></p>

App: regression/framework - Page: tags/a_field_inside_the_quoted_text_of_a_qr_code_or_barcode_is_encoded_as_itself
PHP
tags/a_field_inside_the_quoted_text_of_a_qr_code_or_barcode_is_encoded_as_itself.php 		HTML
tags/a_field_inside_the_quoted_text_of_a_qr_code_or_barcode_is_encoded_as_itself.pad 		Result 
<?php

  // A field inside the quoted text of {barcode:qr} or {barcode:auto} - the usual way to make a mailto:
  // or a code from a field - is encoded as the characters it holds, the same symbol as the
  // whole text in a variable gives, where the stand-ins a value travels with went into the
  // code: the @ of the address scanned as U+E040, and a , refused a Code 128.

  $email = 'herbert@example.com';
  $sku   = 'A,B';

?>
 
{barcode:qr 'mailto:{$email}', size=21/}
{barcode:code128 'SKU-{$sku}', plain/}

App: regression/framework - Page: tags/a_field_inside_the_quoted_text_of_a_tag_that_computes_from_it_counts_as_its_characters

<?php

  // gives what the field alone gives: the characters it holds - not the stand-ins it
  // travels with, nor the HTML its sanitize step made of it, which went into the initials,
  // the hash and the code, and into a label escaped once more.

  $name  = 'Åsa Ørsted & Co';
  $email = "o'brien@example.com";
  $query = 'https://example.com/?a=1&b=2';

?>
 
{pad:avatar '{$name}'/}
{pad:gravatar '{$email}'/}
{pad:identicon '{$name}'/}
{barcode:qr '{$query}', size=21/}

App: regression/framework - Page: tags/a_quote_stand_in_in_a_value_stays_inside_the_attribute_of_a_drawing_tag

tags/a_quote_stand_in_in_a_value_stays_inside_the_attribute_of_a_drawing_tag.php 		HTML
tags/a_quote_stand_in_in_a_value_stays_inside_the_attribute_of_a_drawing_tag.pad 		Result 
<?php

  // A value holding the stand-in of a quote - what %EE%80%A2 in a request is - written by
  // the tags that draw into an attribute: escaped as the quote it becomes when the page is
  // written out, where their escapers left it and it closed the attribute and opened one
  // of the visitor's own, an event handler.

  $hostile = "red\u{E022} onclick=\u{E022}alert(1)";

?>
 
{barcode:qr 'https://example.com', size=60, color=$hostile/}
{pad:countdown '2030-01-01', now='2026-01-01', past=$hostile, live/}
{pad:rating 3, label=$hostile/}
{pad:progress 40, label=$hostile/}
{pad:placeholder '60x30', text=$hostile/}
