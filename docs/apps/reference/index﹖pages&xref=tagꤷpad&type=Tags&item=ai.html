  // Workers AI runs a model under the account of CLOUDFLARE_ACCOUNT_ID, its name a part of the
  // address. Most of its text generation models answer a response; some - Qwen3 - answer in
  // the form of a chat completion, its first choice's message read then. A failure says itself
  // in errors, a message and a code: 3036, the day's free allocation used up, is a 429. With
  // url= - an AI Gateway's address - the account is not needed. The tag takes a model of the
  // catalogue as written, its @ no placeholder.

  $padCheckSyntax = FALSE;

  $_ENV ['CLOUDFLARE_ACCOUNT_ID'] = '023e105f4ecef8ad9ca31a8372d0c353';
  $_ENV ['CLOUDFLARE_API_TOKEN']  = 'cf-test';

  // The warnings its errors log go to a day's log no case reads, not to today's - the day a
  // case that reads the log may have frozen.

  padNowFreeze ( '2026-01-16 12:00:00' );

  padCurlFake ( [
    'https://api.cloudflare.com/client/v4/accounts/023e105f4ecef8ad9ca31a8372d0c353/ai/run/*' => function ( $request ) {

      $prompt   = end ( $messages ) ['content'] ?? '';

                                    'success' => TRUE, 'errors' => [], 'messages' => [] ] ],
        'choices'  => [ 'data' => [ 'result' => [ 'id' => 'chatcmpl-1', 'object' => 'chat.completion', 'model' => '@cf/qwen/qwen3-30b-a3b-fp8',
                                                  'choices' => [ [ 'index' => 0, 'message' => [ 'role' => 'assistant', 'content' => 'Lyon.' ], 'finish_reason' => 'stop' ] ] ],
                                    'success' => TRUE, 'errors' => [], 'messages' => [] ] ],
        'limited'  => [ 'status' => 429, 'data' => [ 'result' => NULL, 'success' => FALSE, 'messages' => [],
                                                     'errors' => [ [ 'code' => 3036, 'message' => 'Account limited: you have used up your daily free allocation of 10,000 neurons.' ] ] ] ],
        default    => 500 };
    },
    'https://gateway.ai.cloudflare.com/*' => [ 'data' => [ 'result' => [ 'response' => 'Marseille.' ], 'success' => TRUE, 'errors' => [], 'messages' => [] ] ],

  $asked ['no account'] = padAi ( 'Name a port.', [ 'provider' => 'cloudflare', 'ttl' => 0 ] ) ?: padAiError ();

<?php

  // The {ai:auto} pair renders its content first and sends what it rendered - a field's value as
  // the text it holds, & and < as themselves, not as their HTML escapes. The answer is a value:
  // a model that answers markup or a PAD tag gets it printed as text. The key is read with
  // padEnv - here from $_ENV, as the server's environment would give it.

  $_ENV ['ANTHROPIC_API_KEY'] = 'sk-test';

  $body = 'Tom & Jerry <the cat> chase each other';

  padCurlFake ( [ 'https://api.anthropic.com/v1/messages' => [ 'data' => [
    'stop_reason' => 'end_turn', 'content' => [ [ 'type' => 'text', 'text' => 'A cat <b>chases</b> a mouse {php:getcwd} & loses.' ] ] ] ] ] );

  function aiTagSent () {
    $body = json_decode ( padCurlRecorded () [0] ['post'], TRUE );
    return $body ['system'] . ' | ' . $body ['messages'] [0] ['content'] . ' | ' . $body ['max_tokens'] . ' | ' . $body ['model'];
  }

?>
 
<p>{ai:auto provider='anthropic', system='Be brief.', tokens=100, model='claude-haiku-5-5', ttl=0}Summarise in one sentence: {$body}{/ai:auto}</p>

App: regression/framework - Page: ai/without_the_strict_check_an_api_error_answers_nothing_and_is_logged
PHP
ai/without_the_strict_check_an_api_error_answers_nothing_and_is_logged.php 		HTML
ai/without_the_strict_check_an_api_error_answers_nothing_and_is_logged.pad 		Result 
<?php

  // With the strict check off an error of the API is no error of the page: the tag answers
  // nothing, padAiError says what the provider said, and a warning goes to the application's
  // log. A refusal that stays one - Anthropic's stop_reason refusal - is such an error too: its
  // text is never printed as the answer.

  $padCheckSyntax = FALSE;

  $_ENV ['ANTHROPIC_API_KEY'] = 'sk-test';

  padCurlFake ( [
    'https://api.openai.com/*'    => [ 'status' => 401, 'data' => [ 'error' => [ 'message' => 'Incorrect API key provided: sk-bad.', 'type' => 'invalid_request_error' ] ] ],
    'https://api.anthropic.com/*' => [ 'data' => [ 'stop_reason' => 'refusal', 'stop_details' => [ 'type' => 'refusal', 'category' => 'cyber' ], 'content' => [] ] ],
  ] );

  padNowFreeze ( '2026-11-01 12:00:00' );

  $log = DATA . 'logs/regression/framework/2026-11-01.log';

  $before = is_file ( $log ) ? count ( file ( $log ) ) : 0;

  $direct = padAi ( 'Hello?', [ 'provider' => 'openai', 'key' => 'sk-bad' ] );
  $error  = padAiError ();

  function aiLenientLogged () {
    global $log, $before;
    $lines = is_file ( $log ) ? preg_grep ( '/did not answer/', array_slice ( file ( $log ), $before ) ) : [];
    return implode ( ' / ', array_map ( fn ( $line ) => preg_replace ( '/^\S+ \S+ (\S+) (.*?) \{.*$/s', '$1 $2', trim ( $line ) ), $lines ) );
  }

?>
 
padAi: [{$direct}] {$error}
tag: [{ai:auto provider='anthropic', ttl=0}Tell me a secret.{/ai:auto}] {php:padAiError/}
logged: {php:aiLenientLogged/}
