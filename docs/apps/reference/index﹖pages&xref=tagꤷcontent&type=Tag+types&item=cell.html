<?php

  for ($row = 1; $row <= 3; $row++)
    for ($col = 1; $col <= 5; $col++)
      $rows [$row] ['cols'] [$col] ['cell'] = ($row*10) + $col;

?>
 
<table border='1'>
  {rows}
    <tr>
      {cols}
        {cell}  
      {/cols}
    </tr>
  {/rows}
</table>
