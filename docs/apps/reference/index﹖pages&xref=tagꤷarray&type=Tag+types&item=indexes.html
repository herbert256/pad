Reference - Tag types - indexes

App: tags - Page: index
PHP
index.php 		HTML
index.pad 		Result 
<?php

  $groups   = tagsByGroup ();
  $indexes  = [
    [ 'page' => 'names',      'label' => 'By name',     'text' => 'Every tag from A to Z, with its one line.' ],
    [ 'page' => 'groups',     'label' => 'By group',    'text' => 'The tags by what they are for - conditions, loops, forms, charts ...' ],
    [ 'page' => 'forms',      'label' => 'By form',     'text' => 'Single tags, tag pairs, and the tags that can be either.' ],
    [ 'page' => 'options',    'label' => 'By option',   'text' => 'Every option a tag takes of its own, and the tags that take it.' ],
    [ 'page' => 'cheatsheet', 'label' => 'Cheat sheet', 'text' => 'The syntax of every tag on one page.' ] ];

  $exampleCount = 0;
  foreach ( tagsCatalog () as $name => $tag )
    $exampleCount += count ( tagsExamples ( $name ) );

?>
 
{meta title='Every PAD tag'}
<section class="hero">
  <h1>Every PAD tag</h1>
  <p class="lead">A page for each built-in tag of PAD: what it does in one line, its syntax and
  the options that belong to it, a longer description and live examples - every result
  rendered by the tag it documents. Find a tag by name, by group, by form or by option, or
  search for it.</p>
  <ul class="stats">
    <li><b>{$tagsCount}</b> tags</li>
    <li><b>{$tagsGroupCnt}</b> groups</li>
    <li><b>{$exampleCount}</b> examples</li>
  </ul>
</section>

<div class="cards">

  <a class="card" href="?{$page}">
    <h2>{$label}</h2>
    <p>{$text}</p>
  </a>
  {/indexes}
</div>

    <div class="tagcloud">{tags}<a href="?tag/{$name}" title="{$about}">{$name}</a>{/tags}</div>
  </section>
  {/groups}
</div>
