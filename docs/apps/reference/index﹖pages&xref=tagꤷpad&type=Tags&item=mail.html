App: regression/pages - Page: mail/a_header_written_as_a_quoted_text_with_a_field_is_its_plain_text
PHP
mail/a_header_written_as_a_quoted_text_with_a_field_is_its_plain_text.php 		HTML
mail/a_header_written_as_a_quoted_text_with_a_field_is_its_plain_text.pad 		Result 
<?php

  // A header of {mail} written as a quoted text with a field in it - subject='Welcome
  // {$name}', to='{$email}' - is the text the field holds: the field's escaping is for the
  // HTML of a page and its stand-ins are for the template, neither for a mail header. The
  // subject went out as "Welcome Zo&euml; O&#039;Neil", and to='{$email}' was refused as
  // 'annexample.com', its @ a stand-in.

  function mailQuotedKeep ( $message ) {

    $GLOBALS ['mails'] [] = json_encode ( [ $message ['to'], $message ['subject'], $message ['replyTo'] ],
                                          JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES );

    return TRUE;

  }

  $padMailTransport = 'mailQuotedKeep';
  $padMailFrom      = 'shop@example.com';

  $name  = "Zoë O'Neil";
  $email = 'ann@example.com';
  $s     = 'Hello, world = "x" | y';
  $mails = [];

?>
 
{mail to='{$email}', subject='Welcome {$name}', replyTo='{$name} <{$email}>'}<p>Hi</p>{/mail}
{mail to=$email, subject='Re: {$s}'}<p>x</p>{/mail}
{mails}{!mails}
{/mails}

App: regression/pages - Page: mail/tags
PHP
mail/tags.php 		HTML
mail/tags.pad 		Result 
<?php

  // The {mail} tag, with a transport of the application's own: a function that gets each
  // message - here it keeps them for the template to show. template= looks _mail/ up from
  // this page's directory to the application's, which has welcome and the layout around
  // it; the pair form's content is the HTML part, rendered where it stands - in the loop,
  // with the fields of the row.

  function mailTagsKeep ( $message ) {

    $GLOBALS ['mails'] [] = [ 'to' => $message ['to'], 'subject' => $message ['subject'],
                              'text' => str_replace ( "\n", ' / ', $message ['text'] ), 'html' => $message ['html'] ];

    return TRUE;

  }

  $padMailTransport = 'mailTagsKeep';
  $padMailFrom      = 'shop@example.com';

  $name    = 'Ann';
  $people  = [ [ 'email' => 'bob@example.com', 'first' => 'Bob' ], [ 'email' => 'cy@example.com', 'first' => 'Cy' ] ];
  $mails   = [];

?>
 
before{mail to='ann@example.com', template='welcome', subject='Welcome, ' . $name}{people}{mail to=$email, subject='Hi ' . $first}<p>Hi <b>{$first}</b></p>{/mail}{/people}after
{mails}{$to} | {$subject} | {$text} | {!html}
{/mails}
