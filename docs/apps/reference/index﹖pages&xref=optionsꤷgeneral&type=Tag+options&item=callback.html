App: regression/framework - Page: tags/a_callback_runs_no_row_phase_for_the_else_branch
PHP
tags/a_callback_runs_no_row_phase_for_the_else_branch.php 		HTML
tags/a_callback_runs_no_row_phase_for_the_else_branch.pad 		Result 
<?php

  // An empty level and one its where= emptied show their @else@: that branch is no row, so
  // the callback's row phase does not run for it - init and exit do. The row phase read the
  // row's fields and ended the request ("Undefined variable $n").

  $xs = [ [ 'n' => 1 ], [ 'n' => 2 ], [ 'n' => 3 ] ];

?>
 
{data 'e'}[]{/data}{e callback='rowfield'}{$n}@else@none{/e}{$sumRows} {xs callback='before', before, where='$n gt 9'}{$n}@else@none{/xs}{$sum}

App: regression/pages - Page: callback
PHP
callback.php 		HTML
callback.pad 		Result 
<?php

  // The row phase of a streaming callback reads the occurrence's fields as plain PHP variables.
  // A case rendered inside another request cannot show that: those fields are the nested
  // pass's locals rather than globals, so the callback never sees them. In a real request it
  // does, which is what the manual's staff example relies on and what this page pins.

  $rows = [
    [ 'n' => 1 ],
    [ 'n' => 2 ],
    [ 'n' => 3 ]
  ];

?>
 
{rows callback='stream.php'}{$n},{/rows}total={$sum}
