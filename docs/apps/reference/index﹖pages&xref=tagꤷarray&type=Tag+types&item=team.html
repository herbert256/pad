<?php

  $team = [
    [ 'name' => 'Herbert Jebbink',    'role' => 'Founder and engine'     ],
    [ 'name' => 'Ada Lovelace',       'role' => 'Sequences'              ],
    [ 'name' => 'Grace Hopper',       'role' => 'Compilers and tooling'  ],
    [ 'name' => 'Alan Turing',        'role' => 'Expression evaluator'   ],
    [ 'name' => 'Émile Zola',         'role' => 'Translations'           ],
    [ 'name' => 'Margaret Hamilton',  'role' => 'Error handling'         ] ];

?>
 
{meta title='A team list', about='Every member gets a circle with their initials - the colour comes from a hash of the name, the text colour from its contrast.'}

  {team}
  <li style="display:flex; align-items:center; gap:12px">
    {avatar $name, size=44}
    <span><strong>{$name}</strong><br><small style="opacity:.7">{$role}</small></span>
  </li>
  {/team}
</ul>
