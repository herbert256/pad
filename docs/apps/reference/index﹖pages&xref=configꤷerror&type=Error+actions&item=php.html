App: regression/error_php - Page: answered_then_failed

answered_then_failed.php 		HTML
answered_then_failed.pad 		Result 
<?php

  // Fails once its answer has gone out: a shutdown function of the page raises an error
  // after the engine has sent the body and its Content-Length.

  register_shutdown_function ( function () { padError ( 'raised after the answer went out' ); } );

?>
 
<p>answered</p>

App: regression/error_php - Page: answered_then_threw
PHP
answered_then_threw.php 		HTML
answered_then_threw.pad 		Result 
<?php

  // Fails once its answer has gone out with an error of its own, not padError: a shutdown
  // function throws, and another warns.

  register_shutdown_function ( function () { trigger_error ( 'warned after the answer went out', E_USER_WARNING ); } );
  register_shutdown_function ( function () { throw new Exception ( 'thrown after the answer went out' ); } );

?>
 
<p>answered</p>

App: regression/error_php - Page: index
PHP
index.php 		HTML
index.pad 		Result 
<?php

  // Test fetches the boom page - an undefined variable its .php reads - and asserts what
  // the 'php' error action is supposed to do with it: PHP reports the warning itself and
  // the page finishes. A plain load only offers the link, so a crawl raises nothing.

  $tested = isset ( $test ) ? 1 : 0;

  if ( $tested ) {

    $r    = padCurl ( $padHost . 'regression/error_php/?boom&padInclude' );
    $code = $r ['result'];
    $body = $r ['data'];

    $verdict = ( $code == 200 and str_contains ( $body, "Warning" )
               and str_contains ( $body, "after" ) ) ? "yes" : "NO";

  }

  $action = $padErrorAction;

?>
 
<p>action: {$action}</p>

{pad:if $tested eq 1}
  <p>the boom page behaved as this action promises: <b>{$verdict}</b></p>
{pad:else}
  <p><a href="?index&test">Test</a> the boom page</p>
{/pad:if}

App: regression/error_php - Page: late
PHP
late.php 		HTML
late.pad 		Result 
<?php

  // Under this action an error is PHP's to show: once the answer had gone out, PHP's own
  // "Fatal error: Uncaught Exception" landed behind the body, past the Content-Length the
  // headers declared. The page that fails after answering is read off the socket.

  $url  = parse_url ( $padHost . 'regression/error_php/?answered_then_failed&padInclude' );
  $sock = fsockopen ( $url ['host'], $url ['port'] ?? 80, $errno, $errstr, 10 );

  fwrite ( $sock, "GET {$url ['path']}?{$url ['query']} HTTP/1.1\r\nHost: {$url ['host']}\r\nConnection: close\r\n\r\n" );

  [ $head, $body ] = explode ( "\r\n\r\n", stream_get_contents ( $sock ), 2 ) + [ '', '' ];

  fclose ( $sock );

  preg_match ( '/^Content-Length:\s*(\d+)/mi', $head, $length );

  $endsThere = ( isset ( $length [1] ) and strlen ( $body ) == $length [1] ) ? 'yes' : 'NO';

?>
 
<p>the answer ends where its Content-Length says: {$endsThere}</p>
