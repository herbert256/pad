Reference - Tag types - accountList

App: ledger - Page: add
PHP
add.php 		HTML
add.pad 		Result 
<?php

  // A transaction written down: a day, an amount, a category, an account, a note. The rules
  // stand on the fields; a post that kept them is booked - its day as Y-m-d, its sign from
  // its category - and the page goes to the overview of its month.

  if ( padPosted ( 'entry' ) ) {

    $day = ledgerEntryDay ( padRequest ( 'day' ) );

    if ( $day and ledgerBook ( $day, padRequest ( 'category' ), padRequest ( 'account' ), str_replace ( ',', '.', padRequest ( 'amount' ) ), padRequest ( 'note', '' ) ) ) {
      padFlash ( 'Booked.' );
      padRedirect ( 'index', [ 'month' => substr ( $day, 0, 7 ) ] );
    }

    padFlash ( 'That category or account is not there.', 'error' );

  }

  $incomeList  = array_values ( array_filter ( ledgerCategories (), fn ( $c ) => $c ['kind'] == 'income' ) );
  $expenseList = array_values ( array_filter ( ledgerCategories (), fn ( $c ) => $c ['kind'] == 'expense' ) );
  $accountList = ledgerAccounts ();
  $dayDefault  = date ( 'Y-m-d' );
  $title       = 'Add a transaction';

?>
 
<div class="narrow">

  {input:form 'entry', error, class='entry'}
    {input:field 'day', type='date', label='Day', value=$dayDefault, rules='required|date'/}
    {input:field 'amount', label='Amount', inputmode='decimal', placeholder='42.50', rules='required|regex:/^[0-9]+([.,][0-9][0-9]?)?$/'/}
    <label for="category">Category</label>
    <select id="category" name="category">
      <optgroup label="Spending">{array:expenseList}<option value="{$id}">{$name}</option>{/array:expenseList}</optgroup>
      <optgroup label="Income">{array:incomeList}<option value="{$id}">{$name}</option>{/array:incomeList}</optgroup>
    </select>
    <label for="account">Account</label>
    <select id="account" name="account">{array:accountList}<option value="{$id}">{$name}</option>{/array:accountList}</select>
    {input:field 'note', label='Note', placeholder='What it was', rules='max:120'/}
    <div class="bar"><button class="button primary">Book it</button><a class="button" href="{$padGo}index{?month}">Cancel</a></div>
  {/input:form}

  <p class="dim">Spending goes out and income comes in, whichever sign you type.</p>

</div>

App: ledger - Page: transactions
PHP
transactions.php 		HTML
transactions.pad 		Result 
<?php

  // A month's transactions as one table - sorted and paged on the server by {table:standard} -
  // narrowed to a category, an account or a word in the note. The same rows answer as CSV
  // (?transactions&padFormat=csv) and JSON: $padExpose names them.

  $only     = (int) padRequest ( 'category', 0 );
  $from     = (int) padRequest ( 'account', 0 );
  $word     = trim ( ledgerText ( 'q' ) );

  $rows     = array_map ( fn ( $r ) => array_diff_key ( $r, [ 'id' => 1, 'kind' => 1 ] ), ledgerRows ( $ledgerMonth, $only, $from, $word ) );
  $shown    = count ( $rows );
  $sum      = round ( array_sum ( array_column ( $rows, 'amount' ) ), 2 );

  $categoryList = array_map ( fn ( $c ) => $c + [ 'chosen' => $c ['id'] == $only ], ledgerCategories () );
  $accountList  = array_map ( fn ( $a ) => $a + [ 'chosen' => $a ['id'] == $from ], ledgerAccounts () );
  $monthName    = ledgerMonthName ( $ledgerMonth );
  $csvName      = "ledger-$ledgerMonth.csv";
  $csvQuery     = http_build_query ( [ 'month' => $ledgerMonth, 'category' => $only, 'account' => $from, 'q' => $word, 'padFormat' => 'csv' ] );

  $padExpose  = [ 'rows' ];
  $title      = "Transactions, $monthName";

?>
 
<form class="filters" method="get">
  <input type="hidden" name="transactions" value="">
  <input type="hidden" name="month" value="{$ledgerMonth}">
  <label>Category
    <select name="category"><option value="0">All</option>{array:categoryList}<option value="{$id}"{pad:if $chosen} selected{/pad:if}>{$name}</option>{/array:categoryList}</select>
  </label>
  <label>Account
    <select name="account"><option value="0">All</option>{array:accountList}<option value="{$id}"{pad:if $chosen} selected{/pad:if}>{$name}</option>{/array:accountList}</select>
  </label>
  <label>Note <input type="search" name="q" value="{$word}" placeholder="a word"></label>
  <button class="button">Show</button>
  <a class="button csv" href="{$padGo}transactions&{$csvQuery}" download="{$csvName}">⬇ CSV</a>
</form>

<section class="card table">
  {pad:if $shown gt 0}
  {table:standard data='rows', columns='day, category, account, note, amount', format="amount:money, day:date('D j M')", totals='amount', sortable, rows=15, striped/}
  {pad:else}
  <p class="dim">No transactions here.</p>
  {/pad:if}
  <p class="dim sum">{$shown} transactions, together {$sum | money}.</p>
</section>
