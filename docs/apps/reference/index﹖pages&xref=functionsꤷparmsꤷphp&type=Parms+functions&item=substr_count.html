  $read = [];

  foreach ( [ 'Europe/Amsterdam' => [ '2026-03-29 02:30', '2026-03-29T00:30:00+00:00' ],

      $read [] = $one . '=' . ( $stamp === FALSE ? 'no date' : gmdate ( 'Y-m-d H:i', $stamp ) );
    }
  }

  $padTimezone = 'America/Santiago';
