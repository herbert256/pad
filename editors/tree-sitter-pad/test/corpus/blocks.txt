================================================================================
A pair is a block
================================================================================

{orders}
  <li>{$id}</li>
{/orders}

--------------------------------------------------------------------------------

(template
  (block
    (open_tag
      (tag_name
        (identifier)))
    (body
      (text)
      (field_tag
        (variable))
      (text))
    (close_tag
      (tag_name
        (identifier)))))

================================================================================
Same-name pairs nest
================================================================================

{if $a eq 1}{if $b eq 2}x{/if}y{/if}

--------------------------------------------------------------------------------

(template
  (block
    (open_tag
      (tag_name
        (identifier))
      (arguments
        (variable)
        (identifier)
        (number)))
    (body
      (block
        (open_tag
          (tag_name
            (identifier))
          (arguments
            (variable)
            (identifier)
            (number)))
        (body
          (text))
        (close_tag
          (tag_name
            (identifier))))
      (text))
    (close_tag
      (tag_name
        (identifier)))))
