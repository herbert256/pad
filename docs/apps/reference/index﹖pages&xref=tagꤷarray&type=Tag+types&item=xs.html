<?php

  // A field piped into a fallback - default(), ??, optional - may be missing under the strict
  // check in {echo} and inside an expression too, not only in {$x | ...}: they named it missing.

  $user = [ 'name' => 'Ann' ];
  $xs   = [ [ 'n' => 1 ], [ 'n' => 2, 'm' => 5 ] ];

?>
 
[{echo $user.middle | default("-")}][{echo $user.middle | ?? "-"}][{echo $user.middle | optional}][{echo $user.name | default("-")}] {xs where='$m | ?? 0 | @ gt 1'}{$n}{/xs}

App: regression/framework - Page: expressions/a_value_spliced_after_a_string_of_the_other_quote_is_text
PHP
expressions/a_value_spliced_after_a_string_of_the_other_quote_is_text.php 		HTML
expressions/a_value_spliced_after_a_string_of_the_other_quote_is_text.pad 		Result 
<?php

  // A value spliced into the quoted expression of where= or eval= is a string of it also when
  // a string of the other quote stands before it - '"': the quotes were counted, the odd one
  // made the value seem to stand in a string already, and it went in bare - the request value
  // 1 or php:gettype() eq 3 became part of the condition, true for every row, and of the
  // eval expression.

  $q  = '1 or php:gettype() eq 3';
  $g  = 'php:gettype()';
  $xs = [ [ 'n' => 1, 's' => 'a' ], [ 'n' => 3, 's' => 'b' ] ];

?>
 
W[{xs where='$s ne \'"\' and $n eq {$q}'}{$n} {/xs}] E[{sequence '1..2', eval='@ . \'-\' . {$q}'}{$sequence} {/sequence}] F[{sequence '1..2', eval='@ . \'"\' . {$g}'}{$sequence}{/sequence}]

  // toBool and toContent - or toData - on one tag, in either order: each reads the tag's

<?php

  // A sort before first= and one after it, a sort before a group and one after it: each
  // sorts on its own value - both sorted on the last one written.

  $xs     = [ [ 'n' => 3 ], [ 'n' => 1 ], [ 'n' => 2 ], [ 'n' => 5 ], [ 'n' => 4 ] ];
  $orders = [ [ 'c' => 'a', 't' => 1, 'd' => '2026-03' ], [ 'c' => 'b', 't' => 9, 'd' => '2026-01' ], [ 'c' => 'a', 't' => 5, 'd' => '2026-02' ],
              [ 'c' => 'a', 't' => 2, 'd' => '2026-01' ], [ 'c' => 'b', 't' => 1, 'd' => '2026-02' ] ];

?>
 
[{xs sort='n', first=2, sort='n DESC'}{$n}{/xs}] {orders sort='d', group='c', sum='t', sort='t DESC'}[{$c}:{$t}:{rows}{$d}/{$t},{/rows}]{/orders}

  // An empty level and one its where= emptied show their @else@: that branch is no row, so
