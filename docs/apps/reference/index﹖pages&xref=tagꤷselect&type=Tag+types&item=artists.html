App: padSelect - Page: aggregates/having
PHP
aggregates/having.php 		HTML
aggregates/having.pad 		Result 
<?php

  $least = 1;

?>
 
<table>
  <tr><th>Artist</th><th class="n">Albums</th><th>Years</th></tr>
  {albums fields="artist_id, count(*) as n, min(released) as first, max(released) as last",
          group="artist_id", having="count(*) > $least", order="n desc, first"}
  <tr><td>{artists}{$name}{/artists}</td><td class="n">{$n}</td><td>{$first} - {$last}</td></tr>
  {/albums}
</table>
