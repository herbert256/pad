Reference - Tag types - incomeList

App: ledger - Page: add
PHP
add.php 		HTML
add.pad 		Result 
<?php

  // A transaction written down: a day, an amount, a category, an account, a note. The rules
  // stand on the fields; a post that kept them is booked - its day as Y-m-d, its sign from
  // its category - and the page goes to the overview of its month.

  if ( padPosted ( 'entry' ) ) {

    $day = ledgerEntryDay ( padRequest ( 'day' ) );

    if ( $day and ledgerBook ( $day, padRequest ( 'category' ), padRequest ( 'account' ), str_replace ( ',', '.', padRequest ( 'amount' ) ), padRequest ( 'note', '' ) ) ) {
      padFlash ( 'Booked.' );
      padRedirect ( 'index', [ 'month' => substr ( $day, 0, 7 ) ] );
    }

    padFlash ( 'That category or account is not there.', 'error' );

  }

  $incomeList  = array_values ( array_filter ( ledgerCategories (), fn ( $c ) => $c ['kind'] == 'income' ) );
  $expenseList = array_values ( array_filter ( ledgerCategories (), fn ( $c ) => $c ['kind'] == 'expense' ) );
  $accountList = ledgerAccounts ();
  $dayDefault  = date ( 'Y-m-d' );
  $ledgerPage  = 'Add a transaction';

?>
 
<div class="narrow">

  <h1>Add a transaction</h1>

  {form 'entry', error, class='entry'}
    {input 'day', type='date', label='Day', value=$dayDefault, rules='required|date'}
    {input 'amount', label='Amount', inputmode='decimal', placeholder='42.50', rules='required|regex:/^[0-9]+([.,][0-9][0-9]?)?$/'}
    <label for="category">Category</label>
    <select id="category" name="category">
      <optgroup label="Spending">{expenseList}<option value="{$id}">{$name}</option>{/expenseList}</optgroup>
      <optgroup label="Income">{incomeList}<option value="{$id}">{$name}</option>{/incomeList}</optgroup>
    </select>
    <label for="account">Account</label>
    <select id="account" name="account">{accountList}<option value="{$id}">{$name}</option>{/accountList}</select>
    {input 'note', label='Note', placeholder='What it was', rules='max:120'}
    <div class="bar"><button class="button primary">Book it</button><a class="button" href="{$padGo}index{?month}">Cancel</a></div>
  {/form}

  <p class="dim">Spending goes out and income comes in, whichever sign you type.</p>

</div>
