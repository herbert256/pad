App: regression/framework - Page: tags/a_field_inside_the_label_of_a_drawing_tag_is_escaped_once
PHP
tags/a_field_inside_the_label_of_a_drawing_tag_is_escaped_once.php 		HTML
tags/a_field_inside_the_label_of_a_drawing_tag_is_escaped_once.pad 		Result 
<?php

  // A field written inside the quoted label of {progress}, {rating}, {icon} or the text of
  // {placeholder} is escaped once, as the field alone is: it arrived escaped by its
  // sanitize step and the tag escaped it again - Q&amp;A was shown as Q&amp;amp;A.

  $topic = 'Q&A';

?>
 
{progress 40, label='{$topic}'}
{rating 3, label='{$topic}'}
{icon 'check', label='{$topic}'}
{placeholder '60x30', text='{$topic}'}

App: showcase - Page: icon/gallery
PHP
icon/gallery.php 		HTML
icon/gallery.pad 		Result 
<?php

  $icons = [];

  foreach ( padIconNames () as $name )
    $icons [] = [ 'name' => $name ];

?>
 
{meta title='Every icon', about='The whole set, listed by padIconNames () - lines on a 24 by 24 grid in the colour of the text.'}
<ul style="list-style:none; margin:0; padding:0; display:grid; grid-template-columns:repeat(auto-fill, minmax(92px, 1fr)); gap:6px">
  {icons}
  <li style="display:grid; justify-items:center; gap:6px; padding:12px 4px; border:1px solid rgba(127,127,127,.2); border-radius:10px">
    {icon $name, size=24}
    <code style="font-size:11px">{$name}</code>
  </li>
  {/icons}
</ul>
