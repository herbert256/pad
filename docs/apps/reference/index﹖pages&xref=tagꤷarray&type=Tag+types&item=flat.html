App: manual - Page: fragments/arrays_6

fragments/arrays_6.php 		HTML
fragments/arrays_6.pad 		Result 
<?php

  $settings = [
    'mail'  => [
      'from' => 'shop@example.com',
      'smtp' => [ 'host' => 'localhost', 'port' => 25 ]
    ],
    'theme' => 'dark'
  ];

  $flat = padArrDot ( $settings );

  $changes = padArrUndot ( [ 'mail.smtp.port' => 587,
                             'theme'          => 'light' ] );
  $port    = $changes ['mail'] ['smtp'] ['port'];

  $nested  = [ 'php', [ 'pad', [ 'templates', 'helpers' ] ] ];
  $tags    = implode ( ', ', padArrFlatten ( $nested ) );

  $wrapped = count ( padArrWrap ( 'one recipient' ) ) . ' / '
           . count ( padArrWrap ( NULL ) );

?>
 
{flat}
  <p>{key@flat} = {$flat}</p>
{/flat}
<p>Undot: the port is {$port}</p>
<p>Flatten: {$tags}</p>
<p>Wrap: {$wrapped}</p>
