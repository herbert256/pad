App: regression/pages - Page: live/counter
PHP
live/counter.php 		HTML
live/counter.pad 		Result 
<?php

  // A live counter on a page that has $padCsrf on (the application's _config/config.php
  // switches it on for this page): live/csrf.php fetches it and posts its events.

  $count = (int) padLiveValue ();

  if ( padLiveEvent () == 'add' )
    $count++;

?>
 
{live 'counter'}<p>{$count}</p><button pad-click="add" pad-value="{$count}">+1</button>{/live}
