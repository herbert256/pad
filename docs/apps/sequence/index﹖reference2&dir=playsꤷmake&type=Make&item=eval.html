<?php

  // A value spliced into the quoted expression of where= or eval= is a string of it also when
  // a string of the other quote stands before it - '"': the quotes were counted, the odd one
  // made the value seem to stand in a string already, and it went in bare - the request value
  // 1 or php:gettype() eq 3 became part of the condition, true for every row, and of the
  // eval expression.

  $q  = '1 or php:gettype() eq 3';
  $g  = 'php:gettype()';
  $xs = [ [ 'n' => 1, 's' => 'a' ], [ 'n' => 3, 's' => 'b' ] ];

?>
 
W[{xs where='$s ne \'"\' and $n eq {$q}'}{$n} {/xs}] E[{sequence '1..2', eval='@ . \'-\' . {$q}'}{$sequence} {/sequence}] F[{sequence '1..2', eval='@ . \'"\' . {$g}'}{$sequence}{/sequence}]

<?php

  // A value spliced into the quoted expression of the eval sequence is a string of it in every
  // spelling - {eval '...'}, {sequence:eval '...'}, make='eval|...', sequence:eval(2, '...') -
  // as in eval='...': only eval= quoted it, and in the others the value became part of the
  // expression and called the function it named.

  $q = 'php:gettype()';

?>
 
e[{eval '@ . {$q}', rows=2}{$eval} {/eval}] s[{sequence:eval '@ . {$q}', rows=2}{$eval} {/sequence:eval}] m[{sequence '1..2', make='eval|@ . {$q}'}{$sequence} {/sequence}] c[{echo sequence:eval(2, '@ . {$q}')}]
App: regression/framework Page: sequence/an_eval_expression_is_refused_only_for_a_value_that_could_run

sequence/an_eval_expression_is_refused_only_for_a_value_that_could_run.pad 		Result 
<?php

  // The eval sequence refuses an expression only for a spliced value that could call code or
  // read a variable - a (, a $ or a : in it: a - or a * 2 printed in a tag's parameters
  // elsewhere on the page, and a {define} written into the expression, made eval='@ - 1',
  // eval='@ * 2' and eval='@ {twice}' refusals.

  $op   = '* 2';
  $dash = '-';

?>
 
op=[{echo '{$op}'}] result=[{sequence '1..3', eval='@ * 2'}{$sequence} {/sequence}] a=[{echo '{$dash}'}] b=[{sequence '5..3', eval='@ - 1'}{$sequence} {/sequence}] {define 'twice'}* 2{/define}c=[{sequence '1..3', eval='@ {twice}'}{$sequence} {/sequence}]
