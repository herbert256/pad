App: sequence - Page: examples
PHP
examples.php 		HTML
examples.pad 		Result 
<?php

  // Examples page: picks the group of examples to show and the one item out of it.
  //
  // example= names a directory of the application, so it is only used when it is one of the
  // groups the page offers - the same list examples.pad walks. Interpolating it unchecked
  // meant a ../ read the file names of any directory the server could reach, and an unknown
  // group ended the request with a 500. item= is checked against the group's own contents,
  // which it already was.

  $examples = [ 'sequences', 'specials', 'random', 'play/single', 'play/double', 'keepRemoveFlag' ];

  if ( ! isset ( $example ) or ! in_array ( $example, $examples ) )
    $example = 'specials';

  if ( $example == 'specials' )
    if ( ! isset ( $item ) or ! $item )
      $item = 'basic';

  if ( $example == 'sequences')
    $exam = sequenceDir ( APP . "basic" ) ;
  else
    $exam = sequenceDir ( APP . "$example" ) ;

  if ( ! count ( $exam ) )
    $exam = [ 'basic' ];

  if ( isset ( $item ) and ! in_array ( $item, $exam) )
    $item = $exam [0];

  if ( ! isset ( $item ) or ! $item )
    $item = $exam [0];

?>
 
{data 'list'} 
  (
    'sequences',
    'specials',
    'random',
    'play/single',
    'play/double',
    'keepRemoveFlag'
  )
{/data}

{block}

<p>Examples ! Many are generated and do not always make sence, <br>the examples from 'specials' are hand crafted and gives a good view of what the Sequences subsystem of PAD can do.</p>

{/block}

<br>
{lines}
  {list}
    {if $example eq $list}
      <b>{$list}</b>
    @else@
      <a href="?examples&example={$list}{?item}">{$list}</a>
    {/if}
  {notLast} | {/notLast}
  {/list}
{/lines}

<br>

{block}<center>

  {exam} 
    {if $item eq $exam}
      <b>{$exam}</b>
    @else@
      <a href="?examples&example={$example}&item={$exam}">{$exam}</a>
    {/if}
    &nbsp;
  {/exam}

  <br><br><hr>

  {if $example eq 'sequences' }
    <h3>{$item}</h3>
      {exists APP . 'basic/{$item}.pad'}
        <h5>{echo 'basic/{$item}.pad'}</h5>
        {example 'basic/{$item}', onlyResult}
      {/exists}
    {list}
      {exists APP . '{$list}/{$item}.pad'}
        <h5>{echo '{$list}/{$item}.pad'}</h5>
        {example '{$list}/{$item}', onlyResult}
      {/exists}
    {/list}
  @else@
    {example '{$example}/{$item}', onlyResult, include}
  {/if}

</center>{/block}

App: sequence - Page: index
PHP
index.php 		HTML
index.pad 		Result 
<?php

  // Front page: names the five ideas the subsystem is built from, and describes the one the
  // concept= parameter asks for.
  //
  // concept arrives from the query string, so it is only used when it names one of them -
  // reading $concepts with an unknown key ended the request with a 500 instead of a page,
  // and so did concept[]=, which arrives as an array.

  $concepts ['sequences']  = 'Something that defines a Sequence list';
  $concepts ['stores']     = 'A stored sequence list';
  $concepts ['actions']    = 'An action executed on a Sequence list';
  $concepts ['plays']      = 'Execute a Sequence on a Sequence';
  $concepts ['resume']     = 'Resume on a stored Sequence';

  if ( ! isset ( $concept ) or ! is_string ( $concept ) or ! isset ( $concepts [$concept] ) )
    $concept = 'sequences';

  $title = 'Sequences';

  $conceptTitle = ucfirst ($concept ) . ' - ' . $concepts [$concept];

?>
 
{lines}
  {concepts}
    {if &concepts:key NE $concept}
      <a href="?index&go=index&concept={&concepts:key}">{&concepts:key | ucfirst}</a>
    @else@
      <b> {&concepts:key | ucfirst} </b>
    {/if}
    {notLast} | {/notLast}
  {/concepts}
{/lines}

<h2>{$conceptTitle}</h2>

{block}
  {page 'concepts/{$concept}', include}
{/block}
