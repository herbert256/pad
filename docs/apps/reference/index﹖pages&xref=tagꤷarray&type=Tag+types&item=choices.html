App: manual - Page: fragments/arrays_3
PHP
fragments/arrays_3.php 		HTML
fragments/arrays_3.pad 		Result 
<?php

  $customers = [
    [ 'id' => 3, 'name' => 'Cleo', 'country' => 'NL' ],
    [ 'id' => 1, 'name' => 'Ann',  'country' => 'BE' ],
    [ 'id' => 2, 'name' => 'Bob',  'country' => 'NL' ]
  ];

  $names   = implode ( ', ', padArrPluck ( $customers, 'name' ) );

  $sorted  = padArrSortBy ( $customers, 'name' );
  $choices = padArrPluck  ( $sorted, 'name', 'id' );

  $byId    = padArrKeyBy ( $customers, 'id' );
  $two     = $byId [2] ['name'];

?>
 
<p>{$names}</p>
<select name="customer">
  {choices~}
    <option value="{key@choices}">{$choices}</option>
  {~/choices}
</select>
<p>Customer 2 is {$two}</p>

App: notes - Page: edit
PHP
edit.php 		HTML
edit.pad 		Result 
<?php

  // A note written or changed: its title, its Markdown, its tags, its colour, a file to
  // attach. The rules stand on the fields (edit.pad) and are checked before this runs - a
  // post that broke them comes back refilled with the messages; one that kept them is
  // stored here, and the page goes to the note. A file refused by its type or size keeps
  // the note from being stored, its reason beside the field.

  $editId = (int) padRequest ( 'id', 0 );
  $draft  = $editId ? notesNote ( $editId ) : NULL;

  if ( $editId and ! $draft )
    padAbort ( 404, 'There is no such note' );

  if ( padPosted ( 'note' ) ) {

    $attach = padUpload ( 'file', types: notesTypes (), max: '8M', dir: 'notes/files' );

    if ( $attach !== FALSE ) {

      $now   = notesNow ();
      $color = notesText ( 'color' );
      $color = isset ( notesColors () [$color] ) ? $color : 'plain';

      if ( $draft )
        db ( "UPDATE notes SET title='{0}', body='{1}', color='{2}', updated='{3}' WHERE id={4}",
             [ padRequest ( 'title' ), padRequest ( 'body', '' ), $color, $now, $editId ] );
      else
        $editId = db ( "INSERT INTO notes (title, body, color, created, updated) VALUES ('{0}', '{1}', '{2}', '{3}', '{3}')",
                       [ padRequest ( 'title' ), padRequest ( 'body', '' ), $color, $now ] );

      notesSaveTags ( $editId, padRequest ( 'tags', '' ) );

      if ( $attach )
        db ( "INSERT INTO attachments (note_id, name, file, type, size, created) VALUES ({0}, '{1}', '{2}', '{3}', {4}, '{5}')",
             [ $editId, $attach ['name'], $attach ['file'], $attach ['type'], $attach ['size'], $now ] );

      padFlash ( $draft ? 'Saved.' : 'Your note is written.' );
      padRedirect ( 'note', [ 'id' => $editId ] );

    }

  }

  $draft     = $draft ?: [ 'id' => 0, 'title' => '', 'body' => '', 'color' => 'plain', 'tags' => [], 'files' => [] ];
  $draftTags = implode ( ', ', array_column ( $draft ['tags'], 'tag' ) );
  $current   = notesText ( 'color', $draft ['color'] );
  $choices   = [];

  foreach ( notesColors () as $key => $label )
    $choices [] = [ 'key' => $key, 'label' => $label, 'chosen' => $key == $current ];

  $leave     = $draft ['id'] ? "note&id={$draft ['id']}" : 'index';
  $notesPage = $draft ['id'] ? "Edit: {$draft ['title']}" : 'New note';

?>
 
<div class="editor">

  <h1>{if $draft.id}Edit note{else}New note{/if}</h1>

  {form 'note', error}

    {input 'title', label='Title', value=$draft.title, rules='required|max:120', placeholder='What is it about?'}

    <div class="tabs" role="tablist" aria-label="Text">
      <button type="button" role="tab" data-tab="write" aria-selected="true">Write</button>
      <button type="button" role="tab" data-tab="preview" aria-selected="false">Preview</button>
    </div>
    {textarea 'body', label='Text, in Markdown', value=$draft.body, rows=16, rules='max:20000'}
    <div id="preview" class="preview body" hidden></div>

    {input 'tags', label='Tags', value=$draftTags, placeholder='work, ideas - separated by commas', rules='max:200'}

    <fieldset class="colors">
      <legend>Colour</legend>
      {choices}<label class="color-{$key}"><input type="radio" name="color" value="{$key}"{if $chosen} checked{/if}> {$label}</label>{/choices}
    </fieldset>

    <div class="drop">
      <span>Drop a picture, a PDF or a text file here - or choose one</span>
      {input 'file', type='file', label='Attach a file', accept='image/*,application/pdf,text/plain,text/markdown,text/csv'}
      <span class="chosen"></span>
    </div>

    <div class="bar">
      <button class="button primary">Save</button>
      <a class="button" href="{$padGo}{$leave}">Cancel</a>
      <small><kbd>Ctrl</kbd> + <kbd>S</kbd> saves</small>
    </div>

  {/form}

</div>

App: sequence - Page: listen
PHP
listen.php 		HTML
listen.pad 		Result 
<?php

  // Listen: a sequence played as notes. Each of the first 32 terms becomes a note - the
  // term modulo 60, five octaves up from C2, the way the OEIS 'listen' button folds a term
  // into the keyboard - and the notes are written as a WAV file into the page itself
  // (seqFunWav in _lib/fun.php), so an <audio> element plays them without JavaScript.
  // Recamán's sequence, which jumps back and forth, is the famous one to listen to.
  //
  // type arrives from the query string and is used only when it names a type the gallery
  // draws.

  $choices = seqFunTypes ();

  if ( ! isset ( $type ) or ! is_string ( $type ) or ! in_array ( $type, $choices ) )
    $type = 'recaman';

  $terms = seqFunTerms ( $type, 32 );
  $oeis  = seqFunOeis ( $terms, $type );
  $oeis  = $oeis ? sprintf ( 'A%06d', $oeis ) : '';
  $list  = implode ( ', ', $terms );
  $notes = implode ( ' ', array_map ( 'seqFunNoteName', seqFunNotes ( $terms ) ) );
  $audio = seqFunWav ( $terms );

?>
 
<p>{choices}{notFirst} | {/notFirst}{if $choices eq $type}<b>{$choices}</b>@else@<a href="?listen&type={$choices}">{$choices}</a>{/if}{/choices}</p>

<h3>{$type | ucfirst}{if $oeis ne ''} - <a href="https://oeis.org/{$oeis}">{$oeis}</a>{/if}</h3>

{chart 'line', data='terms', title='{$type | ucfirst}, the first 32 terms', width=640, height=220}

<p><audio controls preload="none" src="{$audio}"></audio></p>

<p>Each term is a note: the term modulo 60, counted in semitones up from C2.</p>

<p><small>{$list}</small></p>

<p><small>{$notes}</small></p>
