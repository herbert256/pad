<?php

  // The login - and, while DATA/admin/users.json holds nobody, the making of the first user,
  // which only this machine may do: there is no default password to guess. Five tries a
  // minute per address. The password is read as it was posted: a space at its end is part
  // of it.

  $title      = 'Log in';
  $loginSetup = adminUsers () ? 0 : 1;
  $loginError = '';
  $loginHint  = ( $loginSetup and PHP_SAPI !== 'cli' and ! padLoopback () ) ? 1 : 0;

  if ( adminUserValid ( $adminUser, $adminStamp ) )
    padRedirect ( 'index' );

  if ( padPosted ( 'login' ) ) {

    $loginKey = 'admin-login:' . ( $_SERVER ['REMOTE_ADDR'] ?? 'cli' );

    if ( ! padRateLimit ( $loginKey, 5, 60 ) ) {

      $loginError = 'Too many tries - wait ' . padRateLimitAvailableIn ( $loginKey ) . ' seconds.';

    } else {

      $loginName     = trim ( adminField ( 'name' ) );
      $loginPassword = adminField ( 'password' );
      $loginStamp    = '';

      try {

        if ( ! $loginSetup )
          $loginStamp = adminUserCheck ( $loginName, $loginPassword );
        elseif ( $loginHint )
          $loginError = 'The first user can only be made from this machine.';
        elseif ( $loginPassword !== adminField ( 'again' ) )
          $loginError = 'The two passwords are not the same.';
        else
          $loginStamp = adminUserAdd ( $loginName, $loginPassword );

      } catch ( RuntimeException $e ) {

        $loginError = ucfirst ( $e->getMessage () ) . '.';

      }

      if ( $loginStamp !== '' ) {

        padRateLimitClear ( $loginKey );
        padSessionRegenerate ();

        $adminUser  = $loginName;
        $adminStamp = $loginStamp;

        padRedirect ( 'index' );

      }

      if ( $loginError === '' )
        $loginError = 'The name or the password is not right.';

    }

  }

<?php

  $title    = 'Contact Form';
  $dataFile = 'demo/messages.json';   // under DATA/, for padFileGet and padFilePut

  // The rules of a valid message stand on the fields in contact.pad - rules='required|email' -
  // and are checked before this file runs: padPosted ( 'contact' ) is TRUE only for a post
  // that keeps them. A post that breaks one never gets here; the form comes back refilled
  // with what was typed, each message beside its field.

  // The rules let a field posted as a list - name[]=x - through, so a message whose fields
  // are not all text is not stored: htmlspecialchars() was handed the array and the post
  // ended on a 500. The form comes back as it was sent.

  $sent = padRequestOnly ( [ 'name', 'email', 'subject', 'message' ] );

  if ( padPosted ( 'contact' ) and count ( array_filter ( $sent, 'is_string' ) ) == count ( $sent ) ) {

    $messages = json_decode ( padFileGet ( $dataFile ), TRUE ) ?: [];

    $entry          = array_map ( 'htmlspecialchars', $sent );
    $entry ['date'] = padNow ( 'Y-m-d H:i:s' );

    $messages [] = $entry;

    padFilePut ( $dataFile, json_encode ( $messages, JSON_PRETTY_PRINT ) );

    // Post, redirect, get: the browser is sent on to a plain GET of this page, so a refresh
    // shows the empty form instead of sending the message a second time - the thanks go
    // along as a flash message.

    padFlash ( 'Thank you for your message! We will get back to you soon.' );

    padRedirect ( 'contact' );

  }

?>

<?php

  // One card: changed, moved to another column, or deleted - or a new one, without an id.
  // A plain form, the rules on its fields; the board's dialog adds cards the same way.

  $cardId = (int) padRequest ( 'id', 0 );
  $card   = $cardId ? db ( "RECORD * FROM cards WHERE id={0}", [ $cardId ] ) : NULL;

  if ( $cardId and ! $card )
    padAbort ( 404, 'There is no such card' );

  if ( padPosted ( 'remove' ) and $card ) {
    db ( "DELETE FROM cards WHERE id={0}", [ $cardId ] );
    kanbanLog ( "{$card ['title']} was deleted" );
    padFlash ( "“{$card ['title']}” is deleted." );
    padRedirect ( 'index' );
  }

  if ( padPosted ( 'edit' ) ) {

    $label  = kanbanText ( 'label' );
    $label  = isset ( kanbanLabels () [$label] ) ? $label : '';
    $due    = kanbanText ( 'due' );
    $due    = preg_match ( '/^\d{4}-\d\d-\d\d$/', $due ) ? $due : '';
    $column = (int) padRequest ( 'column', 0 );

    if ( $card ) {
      db ( "UPDATE cards SET title='{0}', body='{1}', label='{2}', due='{3}', updated='{4}' WHERE id={5}",
           [ padRequest ( 'title' ), padRequest ( 'body', '' ), $label, $due, kanbanNow (), $cardId ] );
      if ( $column and $column != $card ['column_id'] )
        kanbanMove ( $cardId, $column );
      else
        kanbanLog ( padRequest ( 'title' ) . ' was changed', $cardId );
    } else {
      $column = $column ?: (int) db ( "FIELD id FROM columns ORDER BY position LIMIT 1" );
      $cardId = db ( "INSERT INTO cards (column_id, position, title, body, label, due, created, updated) VALUES ({0}, {1}, '{2}', '{3}', '{4}', '{5}', '{6}', '{6}')",
                     [ $column, (float) db ( "FIELD coalesce(max(position), 0) FROM cards WHERE column_id={0}", [ $column ] ) + 1,
                       padRequest ( 'title' ), padRequest ( 'body', '' ), $label, $due, kanbanNow () ] );
      kanbanLog ( padRequest ( 'title' ) . ' was added', $cardId );
    }

    padFlash ( 'Saved.' );
    padRedirect ( 'index' );

  }

  $draft      = $card ?: [ 'id' => 0, 'title' => '', 'body' => '', 'label' => '', 'due' => '', 'column_id' => 0 ];

  $cardTitle  = $draft ['title'];
  $isCard     = $card ? 1 : 0;
  $kanbanPage = $card ? $card ['title'] : 'New card';

<?php

  // What each category of spending may cost a month. A budget of 0 takes the category's
  // budget away; the overview shows how far each month got.

  if ( padPosted ( 'budgets' ) ) {

    foreach ( (array) padRequest ( 'budget', [] ) as $id => $amount ) {
      if ( ! is_scalar ( $amount ) )
        continue;
      $amount = (float) str_replace ( ',', '.', (string) $amount );
      if ( $amount > 0 )
        db ( "INSERT OR REPLACE INTO budgets (category_id, monthly) SELECT id, {1} FROM categories WHERE id={0} AND kind = 'expense'", [ (int) $id, round ( $amount, 2 ) ] );
      else
        db ( "DELETE FROM budgets WHERE category_id={0}", [ (int) $id ] );
    }

    padFlash ( 'The budgets are saved.' );
    padRedirect ( 'index', [ 'month' => $ledgerMonth ] );

<?php

  // Transactions from a CSV file - a bank's export, or this application's own: a header row
  // naming day, amount, category (and account and note when there are any). A category not
  // known yet is made, as spending when its amounts are below zero; a day that is no date -
  // digits are read year first or day first, 02/09/2026 is 2 September (ledgerImportDay) -
  // an amount that is no number, or more cells than the first row names (ledgerCsv), skips
  // the row and says which.

  $report = [];

  if ( padPosted ( 'import' ) ) {

    $upload = padUpload ( 'csv', types: [ 'text/csv', 'text/plain', 'application/csv' ], max: '5M', dir: 'ledger/imports' );

    if ( $upload ) {

      $handle  = fopen ( $upload ['path'], 'r' );
      [ $header, $rows, $refused ] = ledgerCsv ( $handle );
      $lines   = $rows + $refused;
      $added   = 0;
      $account = (int) db ( "FIELD id FROM accounts ORDER BY id LIMIT 1" );

      ksort ( $lines );

      if ( ! in_array ( 'day', $header ) or ! in_array ( 'amount', $header ) or ! in_array ( 'category', $header ) )
        $report [] = [ 'line' => 1, 'said' => 'The first row must name the columns day, amount and category' ];

      else foreach ( $lines as $line => $row ) {

        // A line refused as it was read - more cells than the first row names - is reported
        // in its place among the rows.

        if ( is_string ( $row ) ) {
          $report [] = [ 'line' => $line, 'said' => $row ];
          continue;
        }

        $amount = str_replace ( [ ',', '€', ' ', '−' ], [ '.', '', '', '-' ], (string) $row ['amount'] );
        $day    = ledgerImportDay ( $row ['day'] );

        if ( $day === NULL or ! is_numeric ( $amount ) ) {
          $report [] = [ 'line' => $line, 'said' => $day === NULL ? 'skipped - no day like 2026-09-02 or 02/09/2026' : 'skipped - no amount' ];
          continue;
        }

        $name = trim ( (string) $row ['category'] ) ?: 'Other';
        $kind = (float) $amount < 0 ? 'expense' : 'income';

        db ( "INSERT OR IGNORE INTO categories (name, kind) VALUES ('{0}', '{1}')", [ mb_substr ( $name, 0, 40 ), $kind ] );

        $category = (int) db ( "FIELD id FROM categories WHERE name = '{0}'", [ mb_substr ( $name, 0, 40 ) ] );
        $from     = isset ( $row ['account'] ) ? (int) db ( "FIELD id FROM accounts WHERE name = '{0}'", [ trim ( $row ['account'] ) ] ) : 0;

        if ( ledgerBook ( $day, $category, $from ?: $account, $amount, (string) ( $row ['note'] ?? '' ) ) )
          $added++;
        else
          $report [] = [ 'line' => $line, 'said' => "skipped - '$name' is a transfer" ];

      }

      fclose ( $handle );
      @unlink ( $upload ['path'] );

      padFlash ( ( $added == 1 ? 'One transaction' : "$added transactions" ) . ' imported' . ( $report ? ', ' . count ( $report ) . ' rows skipped.' : '.' ) );

      if ( ! $report )
        padRedirect ( 'transactions', [ 'month' => ledgerMonth ( '' ) ] );

    }

  }

  $ledgerPage = 'Import';

?>
 
<div class="narrow">

  <h1>Import</h1>
  <p class="lead">Transactions from a CSV file - a bank's export, or one this application wrote on the Transactions page.</p>

  {form 'import', class='entry'}
    {input 'csv', type='file', label='The CSV file', accept='.csv,text/csv,text/plain'}
    <div class="bar"><button class="button primary">Import</button></div>
  {/form}

  {report}
    {first@report}<h2>Rows not imported</h2><ul class="report">{/first@report}
    <li>Line {$line}: {$said}</li>
    {last@report}</ul>{/last@report}
  {/report}

  <h2>The columns</h2>
  <p>The first row names them: <code>day</code>, <code>amount</code> and <code>category</code>, and
  <code>account</code> and <code>note</code> when there are any. Money going out is below zero;
  a category not known yet is made.</p>
  day,amount,category,account,note
2026-10-02,-23.40,Groceries,Current account,Market
2026-10-25,3200,Salary,Current account,Salary
