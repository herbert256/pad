tags/fake_rows_of_a_seed_are_the_rows_padfakerows_gives_and_go_to_a_data_store.php 		HTML
tags/fake_rows_of_a_seed_are_the_rows_padfakerows_gives_and_go_to_a_data_store.pad 		Result 
<?php

  // The rows of {fake ..., rows=, seed=} are those padFakeRows gives for the seed, and an
  // email of a row with a name is that name's.

  $fromPhp = padFakeRows ( [ 'customer' => 'name', 'mail' => 'email', 'total' => [ 'price', [ 'from' => 10, 'to' => 99 ] ] ], 3, 11 );

?>
 
{fake 'customer:name, mail:email, total:price', from=10, to=99, rows=3, seed=11, toData='people'/}{people}[{$customer}|{$mail}|{$total}]{/people} = {fromPhp}[{$customer}|{$mail}|{$total}]{/fromPhp}

App: regression/framework - Page: tags/the_fake_tag_leaves_the_pages_own_fake_sequence_where_it_was

tags/the_fake_tag_leaves_the_pages_own_fake_sequence_where_it_was.pad 		Result 
<?php

  // The page seeds the fake sequence for values of its own; the {fake} tags between - with
  // a seed of their own and without - take theirs from elsewhere and put the page's back, so
  // the next padFakeName is the one the seed gives first.

  padFakeSeed ( 42 );
  $expected = padFakeName ();
  padFakeSeed ( 42 );

?>
 
{fake 'name', seed=3/}{fake 'name, email', rows=2/}{fake 'name'}{/fake}[{echo php:padFakeName()/}] [{$expected}]
