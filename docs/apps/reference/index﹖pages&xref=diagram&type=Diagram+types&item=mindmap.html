App: regression/framework - Page: tags/diagram_mindmap_of_a_deep_chain_draws_the_same_picture_in_good_time

tags/diagram_mindmap_of_a_deep_chain_draws_the_same_picture_in_good_time.pad 		Result 
<?php

  // A mind map that is one chain of 2000 nested lines, each indented one more than the last.

  $chain = "{diagram:mindmap}\n";

  for ( $i = 0; $i < 2000; $i++ )
    $chain .= str_repeat ( ' ', $i ) . "n$i\n";

  $chain .= "{/diagram:mindmap}";

?>
 
{pad:echo $chain | code | php:md5(@)/}
