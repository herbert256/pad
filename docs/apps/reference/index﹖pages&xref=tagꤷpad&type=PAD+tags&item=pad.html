App: reference - Page: index
PHP
 HTML
index.pad 		Result 
<blockquote>

{block}

  {pad data='reference'}

    {sequence '1..{$lvl}'}
      <blockquote>
    {/sequence}

    <h3 id="{$type | slug}">{$type}</h3>

    <blockquote>

      {getReference '{$dir}', '{$xref}'}

        {if $dir}
          <a href="?dir{?xref}{?type}{?item}">{$item}</a>
        {elseif $pages}
          <a href="?pages{?xref}{?type}{?item}">{$item}</a>
        @else@
          {$item}
        {/if}

        &nbsp; 
        
      {/getReference}
            
    </blockquote>

    {sequence '1..{$lvl}'}
      </blockquote>
    {/sequence}

  {/pad}

{/block}

</blockquote>

syntax/a_content_value_does_not_leave_the_application.pad 		Result 
<?php

  // A value naming content is looked up as a content block, a snippet or a page of this
  // application and nowhere else: with a .. in it, it walked out of the application and
  // included and ran any .php it reached - here the hello application's page.

  $walk = '../../hello/index';

?>
 
[{pad content=$walk}{/pad}][{$message | optional}]
