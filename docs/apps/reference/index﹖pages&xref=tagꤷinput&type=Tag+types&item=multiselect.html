App: regression/framework - Page: tags/a_disabled_multiselect_keeps_its_value_after_a_failed_post
PHP
tags/a_disabled_multiselect_keeps_its_value_after_a_failed_post.php 		HTML
tags/a_disabled_multiselect_keeps_its_value_after_a_failed_post.pad 		Result 
<?php

  // The browser leaves a disabled select out of the post, and its padLists[], written disabled
  // with it, too: form f came back - the other field to be corrected - with neither, so the
  // locked choices show value= again, as the disabled text field beside it does. The
  // padLists[] written enabled went along alone, and the choices came back emptied.

  $_SERVER ['REQUEST_METHOD'] = 'POST';
  $_POST = [ 'padForm' => 'f', 'other' => '' ];

?>
 
{input:form 'f'}{input:multiselect 'locked', options='a, b, c', value='a, b', disabled/}{input:field 'name', value='x', disabled/}{input:field 'other', rules='required'/}{/input:form}

  // A form that asks by GET with its one field a multiselect left empty sends padLists[]

App: regression/framework - Page: tags/a_multiselect_in_a_get_form_keeps_its_value_on_a_link_with_a_value
PHP
tags/a_multiselect_in_a_get_form_keeps_its_value_on_a_link_with_a_value.php 		HTML
tags/a_multiselect_in_a_get_form_keeps_its_value_on_a_link_with_a_value.pad 		Result 
<?php

  // The page of a form that asks by GET, asked with a value of its own - a sort link,
  // ?page&sort=name - and padLists[] naming two of its lists, as the form sends them: tags,
  // not named, shows value=, where the value in the query emptied it; emptied, named and
  // not in the query, has nothing chosen; chosen shows what the query holds.

  $_GET ['sort']     = 'name';
  $_GET ['padLists'] = [ 'emptied[]', 'chosen[]' ];
  $_GET ['chosen']   = [ 'c' ];

?>
 
{input:form 'f', method='get'}{input:multiselect 'tags', options='a, b, c', value='a, b'/}{input:multiselect 'emptied', options='a, b, c', value='a'/}{input:multiselect 'chosen', options='a, b, c', value='a'/}{/input:form}

App: regression/framework - Page: tags/a_multiselect_its_form_posted_without_has_nothing_chosen
PHP
tags/a_multiselect_its_form_posted_without_has_nothing_chosen.php 		HTML
tags/a_multiselect_its_form_posted_without_has_nothing_chosen.pad 		Result 
<?php

  // The browser leaves a select with nothing chosen out of the post: form f came back without
  // the field, its padLists[] naming it as the browser posts it, so nothing is chosen in it -
  // value= is what it shows before a post, and shown again it undid the emptying.
