App: regression/pages - Page: select/demo/a_bound_array_is_a_list
PHP
select/demo/a_bound_array_is_a_list.php 		HTML
select/demo/a_bound_array_is_a_list.pad 		Result 
<?php

  // An array bound into where= as $name is a list, as an array in a bare db() placeholder
  // is: name in ($names) is name in ('jim','bob'), where the array became the one literal
  // 'jim,bob' and matched nothing - and on SQLite an "Array to string conversion".

  $padSelect ['staffList'] = [ 'db' => 'staff' ];

  $names = [ 'jim', 'bob' ];

?>
 
[{staffList where="name in ($names)", order="name"}{$name} {/staffList}]

App: regression/pages - Page: select/demo/a_nested_view_follows_its_base_relation
PHP
select/demo/a_nested_view_follows_its_base_relation.php 		HTML
select/demo/a_nested_view_follows_its_base_relation.pad 		Result 
<?php

  // A virtual table nested in a related table is related to it through its base: a view of
  // the employees inside {offices} lists the employees of that office. Only the outer
  // tag's base= chain was walked, so every office listed every sales rep of the view, and
  // a view without a where of its own ended the page on a missing relation.

  $padSelect ['offices']   = [ 'db' => 'offices',   'key' => 'officeCode' ];
  $padSelect ['employees'] = [ 'db' => 'employees', 'key' => 'employeeNumber' ];

  $padSelect ['salesReps'] = [ 'base' => 'employees', 'where' => "jobTitle like 'Sales%'", 'order' => 'lastName' ];
  $padSelect ['staffOf']   = [ 'base' => 'employees', 'order' => 'lastName' ];

  $padRelations ['employees'] ['offices'] = 'officeCode';

?>
 
{offices order='officeCode', rows=3}
{$officeCode} reps:  {salesReps}{$lastName} {/salesReps}
{$officeCode} staff: {staffOf}{$lastName} {/staffOf}
{/offices}

App: regression/pages - Page: select/demo/a_pager_counts_a_select_with_a_join
PHP
select/demo/a_pager_counts_a_select_with_a_join.php 		HTML
select/demo/a_pager_counts_a_select_with_a_join.pad 		Result 
<?php

  // A pager after a select that joins two tables sharing a column name - orders and
  // customers, both with customerNumber - counts its rows: the count ran the select list
  // as a derived table, which MySQL refuses with two columns of one name ("1060: Duplicate
  // column name 'customerNumber'"), and the page ended at the pager.

  $padSelect ['ordersPaged'] = [ 'db' => 'orders', 'key' => 'orderNumber' ];

?>
 
[{ordersPaged join='customers on customers.customerNumber = orders.customerNumber', where="customers.country = 'Norway'", order='orderNumber', page=1, rows=3}{$orderNumber} {$customerName}; {/ordersPaged}]
{pager 'ordersPaged', window=0}{$kind}:{$page} {/pager}

App: regression/pages - Page: select/demo/a_pager_counts_the_rows_its_fields_make
PHP
select/demo/a_pager_counts_the_rows_its_fields_make.php 		HTML
select/demo/a_pager_counts_the_rows_its_fields_make.pad 		Result 
<?php

  // A pager counts the rows the select answers also when its fields= decides what a row is:
  // distinct countries, or one row of totals. The count over the table alone was taken
  // whenever distinct=, group=, having= and union= were not given, so 27 countries had the
  // 13 pages of the 122 customers, and the one row of totals a pager of 28 pages.

  $padSelect ['countriesPaged'] = [ 'db' => 'customers', 'key' => 'customerNumber' ];
  $padSelect ['paymentTotals']  = [ 'db' => 'payments',  'key' => 'checkNumber' ];

?>
 
[{countriesPaged fields='distinct country', order='country', page=1, rows=10}{$country}; {/countriesPaged}]
({pager 'countriesPaged', window=0}{$kind}:{$page} {/pager})
[{paymentTotals fields='count(*) as n', page=1, rows=10}{$n}{/paymentTotals}]
({pager 'paymentTotals', window=0}{$kind}:{$page} {/pager})

App: regression/pages - Page: select/demo/bound
PHP
select/demo/bound.php 		HTML
select/demo/bound.pad 		Result 
<?php

  // A value written into where= as $name is bound by the select as a quoted literal: a
  // quote in it stays inside the string and matches nothing, a comma is still a comma. In
  // order= a $name is spliced as column names. Over the customers table of the demo database.

  $padSelect ['bound'] = [ 'db' => 'customers' ];

  $who   = "x' or customerNumber > '0";
  $comma = 'Australian Collectors, Co.';
  $sort  = 'customerNumber desc';

?>
 
a quote in a bound value: [{bound where="customerName = $who"}{$customerNumber},{/bound}]
a comma in a bound value: [{bound where="customerName = $comma"}{$customerNumber}{/bound}]
a bound order: [{bound where="customerNumber < 120", order="$sort"}{$customerNumber},{/bound}]

App: regression/sqlite - Page: select
PHP
select.php 		HTML
select.pad 		Result 
<?php

  // The Select subsystem over the same staff table: the statement it builds - backquoted
  // names, a quoted key, limit offset,count - is one SQLite reads as MySQL does.

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name' ];

  // A value bound into where= is escaped as SQLite wants it: the quote stays in the string.

  $who = "x' or name > '";

  // A negative number bound after a minus keeps its distance, so "salary > 2000-$adj" is
  // 2000 - (-1000) = 3000 and not a -- comment that leaves SQLite with half a clause.

  $adj = -1000;

?>
 
where: [{staffSel where="salary >= 2000", order="salary"}{$name} {/staffSel}]
order: [{staffSel order="salary desc", fields="name"}{$name} {/staffSel}]
first: [{staffSel order="name", rows=2}{$name} {/staffSel}]
bound: [{staffSel where="name = $who"}{$name} @else@none{/staffSel}]
minus: [{staffSel where="salary > 2000-$adj", order="name"}{$name} @else@none{/staffSel}]
