Reference - Tag types - icons

App: showcase - Page: icon/gallery
PHP
icon/gallery.php 		HTML
icon/gallery.pad 		Result 
<?php

  $icons = [];

  foreach ( padIconNames () as $name )
    $icons [] = [ 'name' => $name ];

?>
 
{meta title='Every icon', about='The whole set, listed by padIconNames () - lines on a 24 by 24 grid in the colour of the text.'}
<ul style="list-style:none; margin:0; padding:0; display:grid; grid-template-columns:repeat(auto-fill, minmax(92px, 1fr)); gap:6px">
  {icons}
  <li style="display:grid; justify-items:center; gap:6px; padding:12px 4px; border:1px solid rgba(127,127,127,.2); border-radius:10px">
    {icon $name, size=24}
    <code style="font-size:11px">{$name}</code>
  </li>
  {/icons}
</ul>
