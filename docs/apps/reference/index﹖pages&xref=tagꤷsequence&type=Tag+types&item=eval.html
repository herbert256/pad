<?php

  // A tag's answer spliced into the expression of the eval sequence - {echo $q} inside its
  // quotes - is refused as a field's value is: it went in bare and ran, 3 3.

  $q = '0 + php:abs(-3)';

?>
 
{eval '@ * {echo $q}', rows=2}{$eval} {/eval}

<?php

  // A value spliced into the quoted expression of the eval sequence is a string of it in every
  // spelling - {eval '...'}, {sequence:eval '...'}, make='eval|...', sequence:eval(2, '...') -
  // as in eval='...': only eval= quoted it, and in the others the value became part of the
  // expression and called the function it named.

  $q = 'php:gettype()';

?>
 
e[{eval '@ . {$q}', rows=2}{$eval} {/eval}] s[{sequence:eval '@ . {$q}', rows=2}{$eval} {/sequence:eval}] m[{sequence '1..2', make='eval|@ . {$q}'}{$sequence} {/sequence}] c[{echo sequence:eval(2, '@ . {$q}')}]
