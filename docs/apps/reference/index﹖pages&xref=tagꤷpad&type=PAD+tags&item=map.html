tags/map_region_views_show_their_own_islands_and_name_what_they_cannot.pad 		Result 
<?php

  // A region's view shows the countries of its region - Cabo Verde, Mauritius and the
  // Seychelles in africa, Palau and the Marshall Islands in oceania - and a country with a
  // value that the view cannot show, Samoa across the 180th meridian, is named in the
  // title: they were dropped from the picture without a word, while their value still set
  // the colour scale.

  $africa  = [ [ 'c' => 'Cape Verde', 'n' => 5 ], [ 'c' => 'Mauritius', 'n' => 6 ], [ 'c' => 'Seychelles', 'n' => 7 ], [ 'c' => 'Nigeria', 'n' => 8 ] ];
  $oceania = [ [ 'c' => 'Samoa', 'n' => 1 ], [ 'c' => 'Palau', 'n' => 2 ], [ 'c' => 'Marshall Islands', 'n' => 3 ], [ 'c' => 'Australia', 'n' => 4 ] ];

?>
 
{map 'africa', data='africa', key='c', value='n'/}
{map 'oceania', data='oceania', key='c', value='n'/}
