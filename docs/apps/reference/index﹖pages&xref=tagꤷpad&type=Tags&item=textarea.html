App: demo - Page: contact

contact.pad 		Result 
<?php

  $title    = 'Contact Form';
  $dataFile = 'demo/messages.json';   // under DATA/, for padFileGet and padFilePut

  // The rules of a valid message stand on the fields in contact.pad - rules='required|email' -
  // and are checked before this file runs: padPosted ( 'contact' ) is TRUE only for a post
  // that keeps them. A post that breaks one never gets here; the form comes back refilled
  // with what was typed, each message beside its field.

  // The rules let a field posted as a list - name[]=x - through, so a message whose fields
  // are not all text is not stored: htmlspecialchars() was handed the array and the post
  // ended on a 500. The form comes back as it was sent.

  $sent = padRequestOnly ( [ 'name', 'email', 'subject', 'message' ] );

  if ( padPosted ( 'contact' ) and count ( array_filter ( $sent, 'is_string' ) ) == count ( $sent ) ) {

    $messages = json_decode ( padFileGet ( $dataFile ), TRUE ) ?: [];

    $entry          = array_map ( 'htmlspecialchars', $sent );
    $entry ['date'] = padNow ( 'Y-m-d H:i:s' );

    $messages [] = $entry;

    padFilePut ( $dataFile, json_encode ( $messages, JSON_PRETTY_PRINT ) );

    // Post, redirect, get: the browser is sent on to a plain GET of this page, so a refresh
    // shows the empty form instead of sending the message a second time - the thanks go
    // along as a flash message.

    padFlash ( 'Thank you for your message! We will get back to you soon.' );

    padRedirect ( 'contact' );

  }

?>
 
<p>Have a question or feedback? Send us a message!</p>

{pad:flash}
  <div class="message">{$message}</div>
{/pad:flash}

{input:form 'contact', error='Please correct the errors below.'}

  <p>{input:field 'name', label='Your Name: *', rules='required|max:100'/}</p>

  <p>{input:field 'email', type='email', label='Your Email: *', rules='required|email'/}</p>

  <p>{input:field 'subject', label='Subject: *', rules='required|max:200'/}</p>

  <p>{input:textarea 'message', label='Message: *', rows=6, rules='required|max:2000'/}</p>

  <p>
    <input type="submit" value="Send Message">
  </p>

{/input:form}

<?php

  // One card: changed, moved to another column, or deleted - or a new one, without an id.
  // A plain form, the rules on its fields; the board's dialog adds cards the same way.

  $cardId = (int) padRequest ( 'id', 0 );
  $card   = $cardId ? db ( "RECORD * FROM cards WHERE id={0}", [ $cardId ] ) : NULL;

  if ( $cardId and ! $card )
    padAbort ( 404, 'There is no such card' );

  if ( padPosted ( 'remove' ) and $card ) {
    db ( "DELETE FROM cards WHERE id={0}", [ $cardId ] );
    kanbanLog ( "{$card ['title']} was deleted" );
    padFlash ( "“{$card ['title']}” is deleted." );
    padRedirect ( 'index' );
  }

  if ( padPosted ( 'edit' ) ) {

    $label  = kanbanText ( 'label' );
    $label  = isset ( kanbanLabels () [$label] ) ? $label : '';
    $due    = kanbanText ( 'due' );
    $due    = preg_match ( '/^\d{4}-\d\d-\d\d$/', $due ) ? $due : '';
    $column = (int) padRequest ( 'column', 0 );

    if ( $card ) {
      db ( "UPDATE cards SET title='{0}', body='{1}', label='{2}', due='{3}', updated='{4}' WHERE id={5}",
           [ padRequest ( 'title' ), padRequest ( 'body', '' ), $label, $due, kanbanNow (), $cardId ] );
      if ( $column and $column != $card ['column_id'] )
        kanbanMove ( $cardId, $column );
      else
        kanbanLog ( padRequest ( 'title' ) . ' was changed', $cardId );
    } else {
      $column = $column ?: (int) db ( "FIELD id FROM columns ORDER BY position LIMIT 1" );
      $cardId = db ( "INSERT INTO cards (column_id, position, title, body, label, due, created, updated) VALUES ({0}, {1}, '{2}', '{3}', '{4}', '{5}', '{6}', '{6}')",
                     [ $column, (float) db ( "FIELD coalesce(max(position), 0) FROM cards WHERE column_id={0}", [ $column ] ) + 1,
                       padRequest ( 'title' ), padRequest ( 'body', '' ), $label, $due, kanbanNow () ] );
      kanbanLog ( padRequest ( 'title' ) . ' was added', $cardId );
    }

    padFlash ( 'Saved.' );
    padRedirect ( 'index' );

  }

  $draft      = $card ?: [ 'id' => 0, 'title' => '', 'body' => '', 'label' => '', 'due' => '', 'column_id' => 0 ];
  $columnList = array_map ( fn ( $c ) => $c + [ 'chosen' => $c ['id'] == $draft ['column_id'] ], array_values ( db ( "ARRAY id, name FROM columns ORDER BY position" ) ?: [] ) );
  $labelList  = array_map ( fn ( $key, $name ) => [ 'key' => $key, 'name' => $name, 'chosen' => $key == $draft ['label'] ], array_keys ( kanbanLabels () ), kanbanLabels () );
  $cardTitle  = $draft ['title'];
  $isCard     = $card ? 1 : 0;
  $title      = $card ? $card ['title'] : 'New card';

?>
 
<div class="narrow">

  <p class="crumbs"><a href="{$padGo}index">← The board</a></p>

  {input:form 'edit', error, class='card-form sheet'}
    {input:field 'title', label='Title', value=$draft.title, rules='required|max:100'/}
    {input:textarea 'body', label='Details', value=$draft.body, rows=6, rules='max:2000'/}
    <div class="row">
      <div><label for="label">Label</label>
        <select id="label" name="label"><option value="">None</option>{array:labelList}<option value="{$key}"{pad:if $chosen} selected{/pad:if}>{$name}</option>{/array:labelList}</select></div>
      <div><label for="column">Column</label>
        <select id="column" name="column">{array:columnList}<option value="{$id}"{pad:if $chosen} selected{/pad:if}>{$name}</option>{/array:columnList}</select></div>
      <div>{input:field 'due', type='date', label='Due', value=$draft.due/}</div>
    </div>
    <div class="bar"><button class="button primary">Save</button><a class="button" href="{$padGo}index">Cancel</a></div>
  {/input:form}

  {pad:if $isCard}
  <details class="danger-zone">
    <summary>Delete this card…</summary>
    {input:form 'remove'}<button class="button danger">Delete “{$cardTitle}”</button>{/input:form}
  </details>
  {/pad:if}

</div>

App: notes - Page: edit

edit.php 		HTML
edit.pad 		Result 
<?php

  // A note written or changed: its title, its Markdown, its tags, its colour, a file to
  // attach. The rules stand on the fields (edit.pad) and are checked before this runs - a
  // post that broke them comes back refilled with the messages; one that kept them is
  // stored here, and the page goes to the note. A file refused by its type or size keeps
  // the note from being stored, its reason beside the field.

  $editId = (int) padRequest ( 'id', 0 );
  $draft  = $editId ? notesNote ( $editId ) : NULL;

  if ( $editId and ! $draft )
    padAbort ( 404, 'There is no such note' );

  if ( padPosted ( 'note' ) ) {

    $attach = padUpload ( 'file', types: notesTypes (), max: '8M', dir: 'notes/files' );

    if ( $attach !== FALSE ) {

      $now   = notesNow ();
      $color = notesText ( 'color' );
      $color = isset ( notesColors () [$color] ) ? $color : 'plain';

      if ( $draft )
        db ( "UPDATE notes SET title='{0}', body='{1}', color='{2}', updated='{3}' WHERE id={4}",
             [ padRequest ( 'title' ), padRequest ( 'body', '' ), $color, $now, $editId ] );
      else
        $editId = db ( "INSERT INTO notes (title, body, color, created, updated) VALUES ('{0}', '{1}', '{2}', '{3}', '{3}')",
                       [ padRequest ( 'title' ), padRequest ( 'body', '' ), $color, $now ] );

      notesSaveTags ( $editId, padRequest ( 'tags', '' ) );

      if ( $attach )
        db ( "INSERT INTO attachments (note_id, name, file, type, size, created) VALUES ({0}, '{1}', '{2}', '{3}', {4}, '{5}')",
             [ $editId, $attach ['name'], $attach ['file'], $attach ['type'], $attach ['size'], $now ] );

      padFlash ( $draft ? 'Saved.' : 'Your note is written.' );
      padRedirect ( 'note', [ 'id' => $editId ] );

    }

  }

  $draft     = $draft ?: [ 'id' => 0, 'title' => '', 'body' => '', 'color' => 'plain', 'tags' => [], 'files' => [] ];
  $draftTags = implode ( ', ', array_column ( $draft ['tags'], 'tag' ) );
  $current   = notesText ( 'color', $draft ['color'] );
  $choices   = [];

  foreach ( notesColors () as $key => $label )
    $choices [] = [ 'key' => $key, 'label' => $label, 'chosen' => $key == $current ];

  $leave     = $draft ['id'] ? "note&id={$draft ['id']}" : 'index';
  $title     = $draft ['id'] ? "Edit: {$draft ['title']}" : 'New note';

?>
 
<div class="editor">

  {input:form 'note', error}

    {input:field 'title', label='Title', value=$draft.title, rules='required|max:120', placeholder='What is it about?'/}

    <div class="tabs" role="tablist" aria-label="Text">
      <button type="button" role="tab" data-tab="write" aria-selected="true">Write</button>
      <button type="button" role="tab" data-tab="preview" aria-selected="false">Preview</button>
    </div>
    {input:textarea 'body', label='Text, in Markdown', value=$draft.body, rows=16, rules='max:20000'/}
    <div id="preview" class="preview body" hidden></div>

    {input:field 'tags', label='Tags', value=$draftTags, placeholder='work, ideas - separated by commas', rules='max:200'/}

    <fieldset class="colors">
      <legend>Colour</legend>
      {array:choices}<label class="color-{$key}"><input type="radio" name="color" value="{$key}"{pad:if $chosen} checked{/pad:if}> {$label}</label>{/array:choices}
    </fieldset>

    <div class="drop">
      <span>Drop a picture, a PDF or a text file here - or choose one</span>
      {input:field 'file', type='file', label='Attach a file', accept='image/*,application/pdf,text/plain,text/markdown,text/csv'/}
      <span class="chosen"></span>
    </div>

    <div class="bar">
      <button class="button primary">Save</button>
      <a class="button" href="{$padGo}{$leave}">Cancel</a>
      <small><kbd>Ctrl</kbd> + <kbd>S</kbd> saves</small>
    </div>

  {/input:form}

</div>
