App: regression/framework - Page: printables/a_barcode_in_a_box_of_a_printable_is_drawn_in_the_pdf
PHP
printables/a_barcode_in_a_box_of_a_printable_is_drawn_in_the_pdf.php 		HTML
printables/a_barcode_in_a_box_of_a_printable_is_drawn_in_the_pdf.pad 		Result 
<?php

  // The SVG of a barcode or a QR code in a box of a printable is drawn in the PDF: its white
  // ground and its bars as filled rectangles - the bars a path of M, h, v and z, turned into
  // the PDF's own path - its digits as text, in Courier for the monospaced font of the SVG.
  // Outside a printable an SVG is left out of a PDF as before.

  $label = padUnprotect ( padUnescape ( padCode ( "{printable:labels copies=1}{barcode:ean13 '400638133393'/}{/printable:labels}" ) ) );
  $plain = padUnprotect ( padUnescape ( padCode ( "<p>Plain</p>{barcode:ean13 '400638133393'/}" ) ) );

  $count = function ( $html ) {
    $pdf = padPdf ( $html, [ 'compress' => FALSE ] );
    preg_match_all ( '/\(([0-9]+)\) Tj/', $pdf, $digits );
    return substr_count ( $pdf, ' re f Q' ) . ' rectangles, ' . substr_count ( $pdf, ' f Q' ) - substr_count ( $pdf, ' re f Q' ) . ' paths, digits '
         . implode ( ' ', $digits [1] ) . ( str_contains ( $pdf, '/BaseFont /Courier' ) ? ' in Courier' : '' );
  };

  $answer = 'in a label: ' . $count ( $label ) . ' // in a page: ' . $count ( $plain );

?>
 
{!answer}

App: regression/framework - Page: tags/subsystem_selectors_keep_dynamic_and_automatic_behavior
PHP
tags/subsystem_selectors_keep_dynamic_and_automatic_behavior.php 		HTML
tags/subsystem_selectors_keep_dynamic_and_automatic_behavior.pad 		Result 
<?php

  $selectedIcon = 'check';
  $selectedWidget = 'toast';
  $absentKind = 'missing';
  $checks = [];
  $checks ['optional dynamic kind'] = padCode ( '{icon:$absentKind optional/}' ) === '';
  $checks ['dynamic icon'] = str_contains ( padCode ( '{icon:$selectedIcon/}' ), 'pad-icon-check' );
  $checks ['dynamic pair'] = str_contains ( padCode ( "{widget:\$selectedWidget 'Saved'}Your changes{/widget:\$selectedWidget}" ), 'Your changes' );
  $checks ['automatic barcode'] = padCode ( "{barcode:auto '400638133393'/}" ) === padCode ( "{barcode:ean13 '400638133393'/}" );
  $checks ['automatic diagram'] = str_contains ( padCode ( "{diagram:auto}sequenceDiagram\nA ->> B: Hello{/diagram:auto}" ), 'Hello' );
  $checks ['auto node is not a header'] = padDiagramHeader ( "auto\nA --> B", FALSE ) === '';
  $checks ['automatic embed'] = str_contains ( padCode ( "{embed:auto 'https://youtu.be/aqz-KE-bpKQ', title='Video'/}" ), 'youtube-nocookie.com' );
  $expectedRows = padFakeRows ( 'customer:name, email', 2, 7 );
  $checks ['fake rows'] = padCode ( "{fake:rows 'customer:name, email', rows=2, seed=7}{\$customer}|{\$email};{/fake:rows}" )
                       === padCode ( '{array:expectedRows}{$customer}|{$email};{/array:expectedRows}' );
  $answer = implode ( '; ', array_map ( fn ( $name, $ok ) => $name . ': ' . ( $ok ? 'yes' : 'NO' ), array_keys ( $checks ), $checks ) );

?>
 
{$answer}

App: printables - Page: labels
PHP
[kind].php 		HTML
[kind].pad 		Result 
<?php

  // A printable, by the name in the address - ?labels, /printables/labels: one of the engine's
  // kinds, else a page not found. Its template - kinds/<kind>.pad, sample data from {fake:rows} with
  // fixed seeds - is shown beside what it makes, coloured, and run; its pdf option links the
  // same sheets as a PDF.

  $one = array_values ( array_filter ( $printables, fn ( $item ) => $item ['kind'] === $kind ) ) [0] ?? NULL;

  if ( ! $one )
    padAbort ( 404 );

  $title    = $one ['name'];
  $text     = $one ['text'];
  $template = printablesTemplate ( $kind );
  $source   = padHighlightTokens ( $template, 'pad' );

  // The icons at the top right of the template that take it to the WASM editor and the live
  // application (_common/_lib/playground.php).

  include_once COMMON . '_lib/playground.php';

  $play     = padPlayLink ( "$template\n" );

?>
 
<p class="pr-lead">{$text}</p>

<div class="pr-stage">
  <section class="pr-template" aria-labelledby="pr-template">
    <h2 id="pr-template">The template</h2>
    <div class="pad-play-box">{!play}<pre class="pr-source"><code>{!source}</code></div>
  </section>
  <section class="pr-preview" aria-labelledby="pr-preview">
    <h2 id="pr-preview">What it prints</h2>
    {pad:echo $template | code/}
  </section>
</div>
