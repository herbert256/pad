Reference - Tag types - servers

App: charts - Page: gauge/servers
PHP
gauge/servers.php 		HTML
gauge/servers.pad 		Result 
<?php

  // Three servers and how loaded they are, as a monitoring page would read them.

  $servers = [
    [ 'name' => 'web-1', 'cpu' => 34, 'memory' => 61 ],
    [ 'name' => 'web-2', 'cpu' => 71, 'memory' => 78 ],
    [ 'name' => 'db-1',  'cpu' => 93, 'memory' => 88 ],
  ];

?>
 
<div class="gauges">
  {servers}
  {chart 'gauge',
         value=$cpu,
         label=$name,
         unit='%',
         bands='70, 90',
         width=200,
         height=130}
  {/servers}
</div>
