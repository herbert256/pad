holidays/a_calendar_without_holidays_leaves_the_holiday_field_to_the_page.pad 		Result 
<?php

  // A page's own $holiday is read inside the days of a {pad:calendar} pair without holidays=:
  // the days get a field holiday only when holidays= asks for the holidays, and then every
  // day has it - the name, or '' - so the page's value never stands in for a day without.

  $holiday = 'PAGEVAR';

?>
 
{pad:calendar '2026-01', first=1}{array:days first=4}{$date}:{$holiday};{/array:days}{/pad:calendar}|
{pad:calendar '2026-01', holidays='nl', first=1}{array:days first=4}{$date}:{$holiday};{/array:days}{/pad:calendar}
