<?php

  // The groups of the catalog with their examples, for the cards of the overview.

  $groups = [];

  foreach ( psGroups () as $group => $tagline ) {

    $examples = [];

    foreach ( $psCatalog [$group] as $page => [ $label, $about ] )
      $examples [] = [ 'page' => $page, 'label' => $label, 'about' => $about ];

    $groups [] = [ 'group' => $group, 'tagline' => $tagline, 'examples' => $examples ];

  }

?>

  <p class="ps-lead">The Select subsystem makes the tables of a database tags. Declare a table
  once - its key, its order - and the relations between the tables, and a template walks them
  without writing a join: a table nested in another reads only the rows related to the row
  around it. Every example here shows the declarations it uses, its template, the result and
  the SQL PAD sent for it, over a small music catalogue - <a href="?model">the model</a>.</p>

  <div class="ps-groups">
    {array:groups}
    <section class="ps-group">
      <h2>{$group}</h2>
