App: react - Page: examples/search
PHP
examples/search.php 		HTML
examples/search.pad 		Result 
<?php

  // One page, two answers. The browser asking ?examples/search gets search.pad with the
  // first result in it; the component asks the same page with padFormat=json and gets
  // $result as JSON - the templates never run for that request. $padExpose is the list of
  // what may leave the server: nothing else of this page does.

  $query    = trim ( reactText ( 'q' ) );
  $category = reactText ( 'category' );
  $order    = reactText ( 'sort', 'name' );
  $perPage  = 6;

  $products   = reactData ( 'products' );
  $categories = array_values ( array_unique ( array_column ( $products, 'category' ) ) );

  $found = array_values ( array_filter ( $products, fn ( $product ) =>
             ( $category === '' or $product ['category'] === $category ) and
             ( $query    === '' or stripos ( $product ['name'] . ' ' . $product ['blurb'], $query ) !== FALSE ) ) );

  usort ( $found, match ( $order ) {
    'cheap'  => fn ( $a, $b ) => $a ['price']  <=> $b ['price'],
    'dear'   => fn ( $a, $b ) => $b ['price']  <=> $a ['price'],
    'rating' => fn ( $a, $b ) => $b ['rating'] <=> $a ['rating'],
    default  => fn ( $a, $b ) => strcasecmp ( $a ['name'], $b ['name'] )
  } );

  $pages  = max ( 1, (int) ceil ( count ( $found ) / $perPage ) );
  $pageNo = min ( $pages, max ( 1, (int) padRequest ( 'p', 1 ) ) );

  $result = [ 'items' => array_slice ( $found, ( $pageNo - 1 ) * $perPage, $perPage ),
              'total' => count ( $found ),
              'page'  => $pageNo,
              'pages' => $pages ];

  $start = [ 'result' => $result, 'categories' => $categories ];
  $items = $result ['items'];

  $padExpose = [ 'result' ];

?>
 
<div data-island="CatalogSearch" data-props="{^start}">
  <ul>
    {items}<li>{$emoji} {$name} - {$price}</li>{/items}
  </ul>
</div>

<script type="text/babel" src="{$padRoot}react/examples/search.js"></script>

App: regression/framework - Page: variables/a_dotted_path_a_row_lacks_reads_the_enclosing_levels_not_a_sibling_row
PHP
variables/a_dotted_path_a_row_lacks_reads_the_enclosing_levels_not_a_sibling_row.php 		HTML
variables/a_dotted_path_a_row_lacks_reads_the_enclosing_levels_not_a_sibling_row.pad 		Result 
<?php

  // Rows without the dotted path read it from the enclosing levels and the page's variables,
  // never from another row: every row printed Bob, the row's customer.address.city stood in
  // for the page's $address.city, a row without purchases got the first row's.

  $user    = [ 'name' => 'Ann' ];
  $rows    = [ [ 'a' => 1 ], [ 'a' => 3, 'user' => [ 'name' => 'Bob' ] ], [ 'a' => 5 ] ];
  $address = [ 'city' => 'Amsterdam' ];
  $orders  = [ [ 'id' => 1, 'customer' => [ 'name' => 'X', 'address' => [ 'city' => 'Paris' ] ] ], [ 'id' => 2, 'customer' => [ 'name' => 'Y' ] ] ];
  $users   = [ [ 'name' => 'A', 'bought' => [ [ 'id' => 1 ] ] ], [ 'name' => 'B' ], [ 'name' => 'C', 'bought' => [ [ 'id' => 3 ] ] ] ];
  $a       = [ 'n' => 2 ];
  $items   = [ [ 'name' => 'x', 'a' => [ 'n' => 1 ] ], [ 'name' => 'y' ] ];

?>
 
{rows where='$user.name eq "Bob"'}{$a} {/rows}|{rows}{$a}:{$user.name} {/rows}
{orders}{$id}:{$address.city} {/orders}
{users}{$name}:{$bought.0.id | optional} {/users}
{items}{if $a.n eq 1}one:{$a.n}{else}other:{$a.n}{/if} {/items}

App: regression/framework - Page: variables/a_dotted_path_named_like_a_property_reads_the_variable
PHP
variables/a_dotted_path_named_like_a_property_reads_the_variable.php 		HTML
variables/a_dotted_path_named_like_a_property_reads_the_variable.pad 		Result 
<?php

  // Page arrays named like iteration properties - data, count, name, key, options. A dotted
  // path into one is the array's key: $data.status answered the level's data and ended the
  // page, $count.total the loop's row count, $name.first the loop's name, $key.id the row's key.

  $data    = [ 'title' => 'T', 'status' => 'open' ];
  $count   = [ 'total' => 5 ];
  $name    = [ 'first' => 'Ann' ];
  $key     = [ 'id' => 'k1' ];
  $options = [ 'color' => 'red' ];
  $items   = [ [ 'n' => 1 ], [ 'n' => 2 ] ];

?>
 
{if $data.status eq "open"}open{else}closed{/if}|{echo $count.total + 1}|{items}{$name.first}/{$count.total}/{$key.id}/{$options.color};{/items}

<?php

  // The thing an in-request case cannot be: a page whose .php half leaves variables for its
  // .pad half. A nested pass has a scope of its own, so a case rendered inside another
