<?php

  // A field holding a date object of the page's PHP is guessed as the date field, as a
  // field holding a date as text is: read only when date= named it, the timeline without
  // date= failed - "the timeline has no events with a date" - and the calendar showed no
  // event at all, the text date of the other row neither.

  $agenda = [ [ 'when' => new DateTimeImmutable ( '2026-10-05 10:00', new DateTimeZone ( 'UTC' ) ), 'what' => 'Review' ],
              [ 'when' => '2026-10-07', 'what' => 'Text date' ] ];

?>
 
{calendar '2026-10', data='agenda'}{days}{if $count gt 0}{$date}: {events}{$what}{/events}; {/if}{/days}{/calendar}
{timeline data='agenda', label='what'}

App: regression/framework - Page: tags/calendar_and_timeline_put_a_moment_on_its_day_in_the_application_timezone
PHP
tags/calendar_and_timeline_put_a_moment_on_its_day_in_the_application_timezone.php 		HTML
tags/calendar_and_timeline_put_a_moment_on_its_day_in_the_application_timezone.pad 		Result 
<?php

  // A moment - a Unix timestamp, a date with its offset - falls on the day it is in the
  // application's timezone, the one today is: an event at local midnight in Amsterdam is
  // on the 13th, where the calendar put a timestamp and an offset date on their UTC day,
  // the 12th, and the timeline wrote them as 22:00 and 22:30 of the 12th.

  include_once PAD . 'tags/_lib/timeline.php';

  $padTimezone = 'Europe/Amsterdam';

  padNowFreeze ( '2025-10-13 00:30:00' );

  $agenda = [ [ 'when' => '1760306400',                'what' => 'stamp'  ],
              [ 'when' => '2025-10-13T00:30:00+02:00', 'what' => 'offset' ],
              [ 'when' => '2025-10-13 00:30:00',       'what' => 'plain'  ] ];

  $shown = padTimelineShow ( ...padTimelineDate ( '1760306400' ) );

?>
 
{calendar '2025-10', data='agenda', date='when', title='what'}{days}{events}{$date}={$what} {/events}{/days}{/calendar}
timeline: {$shown}

App: showcase - Page: countdown/events
PHP
countdown/events.php 		HTML
countdown/events.pad 		Result 
<?php

  // Events at moments counted from now - from half a minute ahead, so the words come out
  // round - and the page always has some to come.

  $events = [];
  $from   = padNow ()->modify ( '+30 seconds' );

  foreach ( [ 'Jazz in the park'   => '-2 days',
              'Pottery workshop'   => '+3 hours 20 minutes',
              'City half marathon' => '+4 days 6 hours',
              'Film night'         => '+26 days 2 hours',
              'New year concert'   => '+84 days 3 hours' ] as $name => $when )
    $events [] = [ 'name' => $name, 'starts' => $from->modify ( $when )->format ( 'Y-m-d H:i:s' ) ];

?>
 
{meta title='Upcoming events', about='The two largest units by default, a past= text for an event that has started, and the full moment in the title on hover.'}
<ul style="list-style:none;margin:0;padding:0;min-width:24em">
  {events}
  <li style="display:flex;justify-content:space-between;gap:24px;padding:10px 0;border-bottom:1px solid rgba(128,128,128,.25)">
    <strong>{$name}</strong>
    <span style="opacity:.8">{countdown $starts, past='Sold out'}</span>
  </li>
  {/events}
</ul>
