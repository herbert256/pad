Reference - Chart types - calendar

App: charts - Page: heatmap/commits
PHP
heatmap/commits.php 		HTML
heatmap/commits.pad 		Result 
<?php

  // A year of commits: none on most weekends, more mid-week, a quiet August and a busy
  // run before the release in November - the same picture every time, from a fixed seed.

  mt_srand ( 2026 );

  $commits = [];

  for ( $day = new DateTimeImmutable ( '2026-01-01' ); $day->format ( 'Y' ) == '2026'; $day = $day->modify ( '+1 day' ) ) {

    $weekday = (int) $day->format ( 'N' );
    $month   = (int) $day->format ( 'n' );
    $chance  = ( $weekday >= 6 ) ? 0.12 : ( $month == 8 ? 0.25 : 0.8 );
    $busy    = ( $month == 11 ) ? 2.2 : ( in_array ( $weekday, [ 2, 3, 4 ] ) ? 1.3 : 1 );

    if ( mt_rand () / mt_getrandmax () < $chance )
      $commits [] = [ 'day' => $day->format ( 'Y-m-d' ), 'count' => (int) round ( mt_rand ( 1, 6 ) * $busy ) ];

  }

?>
 
{chart 'calendar',
       data='commits',
       date='day',
       value='count',
       year=2026,
       title='Commits in 2026',
       width=720,
       height=150/}

App: regression/framework - Page: tags/a_millisecond_timestamp_is_no_date
PHP
tags/a_millisecond_timestamp_is_no_date.php 		HTML
tags/a_millisecond_timestamp_is_no_date.pad 		Result 
<?php

  // A timestamp past the year 9999 - JavaScript's Date.now (), in milliseconds - is no
  // date: written back as 57742-03-07 it was read as 2002, and a calendar of one such row
  // beside 2026 showed ten empty years and left the real day out.

  include_once PAD . 'charts/_lib/chart.php';

  $read = [];

  foreach ( [ '1760000000', '1760000000000' ] as $one ) {
    $stamp   = padChartStamp ( $one );
    $read [] = $one . '=' . ( $stamp === FALSE ? 'no date' : gmdate ( 'Y-m-d', $stamp ) );
  }

  $read = implode ( '; ', $read );

?>
 
{$read} - {chart 'calendar', date='day', value='n'}
day,n
2026-10-01,3
1760000000000,4
{/chart}
