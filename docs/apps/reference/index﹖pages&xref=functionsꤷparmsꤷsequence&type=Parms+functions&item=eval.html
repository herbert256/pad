<?php

  // Only a literal parameter of the eval sequence is an expression the template wrote: the
  // value spliced into sequence:repeat(2, '...') was taken for one, and sequence:eval(2, $q),
  // refused on its own, then ran the same value from the request.

  $q = '0 + php:abs(-3)';

?>
 
{echo sequence:repeat(2, '{$q}')}{echo sequence:eval(2, $q)}

<?php

  // eval(2, '...') is sequence:eval(2, '...') written without its prefix, and a value spliced
  // into its quoted expression is a string of it there too: it went in bare, and the request
  // value 0 + php:abs(-3) called PHP - the call answered 3.

  $q = '0 + php:abs(-3)';

?>
 
{echo eval(2, '@ * {$q}')}

App: regression/framework - Page: sequence/a_value_spliced_into_any_eval_expression_is_text
PHP
sequence/a_value_spliced_into_any_eval_expression_is_text.php 		HTML
sequence/a_value_spliced_into_any_eval_expression_is_text.pad 		Result 
<?php

  // A value spliced into the quoted expression of the eval sequence is a string of it in every
  // spelling - {eval '...'}, {sequence:eval '...'}, make='eval|...', sequence:eval(2, '...') -
  // as in eval='...': only eval= quoted it, and in the others the value became part of the
  // expression and called the function it named.

  $q = 'php:gettype()';

?>
 
e[{eval '@ . {$q}', rows=2}{$eval} {/eval}] s[{sequence:eval '@ . {$q}', rows=2}{$eval} {/sequence:eval}] m[{sequence '1..2', make='eval|@ . {$q}'}{$sequence} {/sequence}] c[{echo sequence:eval(2, '@ . {$q}')}]
