App: regression/framework - Page: expressions/the_expression_of_echo_if_case_and_while_runs_once
PHP
expressions/the_expression_of_echo_if_case_and_while_runs_once.php 		HTML
expressions/the_expression_of_echo_if_case_and_while_runs_once.pad 		Result 
<?php

  // The expression of {pad:echo}, {pad:if}, {pad:case} and {pad:while} runs once: it was evaluated when the
  // parameters were read and again by the tag, so a PHP call in it ran twice.

  $calls = 0;

  function onceCall () { return ++ $GLOBALS ['calls']; }

?>
 
{pad:echo php:onceCall()/}{pad:if php:onceCall() gt 0}{/pad:if}{pad:case php:onceCall()}{pad:when 3}{/pad:case}{pad:while php:onceCall() lt 0}{/pad:while} {$calls}
