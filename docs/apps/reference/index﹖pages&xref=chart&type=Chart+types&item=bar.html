App: react - Page: examples/chart
PHP
examples/chart.php 		HTML
examples/chart.pad 		Result 
<?php

  // The chart the request asks for: its kind, how many months, which channels. The browser
  // gets the whole page; the component asks for &padFragment=chart with its choices and gets
  // the {fragment 'chart'} of chart.pad alone - the SVG PAD drew, nothing around it.

  $kind     = padRequest ( 'kind', 'bar' );
  $kind     = in_array ( $kind, [ 'bar', 'stacked', 'line', 'hbar' ] ) ? $kind : 'bar';
  $months   = max ( 3, min ( 12, (int) padRequest ( 'months', 12 ) ) );
  $channels = array_values ( array_intersect ( [ 'online', 'shop', 'wholesale' ], explode ( ',', reactText ( 'channels', 'online,shop,wholesale' ) ) ) );

  if ( ! $channels )
    $channels = [ 'online' ];

  $sales     = array_slice ( reactData ( 'sales' ), -$months );
  $value     = implode ( ', ', $channels );
  $stacked   = ( $kind == 'stacked' );
  $chartKind = $stacked ? 'bar' : $kind;
  $total     = array_sum ( array_map ( fn ( $row ) => array_sum ( array_intersect_key ( $row, array_flip ( $channels ) ) ), $sales ) );

  $choice = [ 'kind' => $kind, 'months' => $months, 'channels' => $channels ];

?>
 
<div data-island="ChartStudio" data-props="{^choice}">
{fragment 'chart'}
  <div class="chart-frame" data-total="{$total}">
    {if $stacked}
      {chart 'bar', data='sales', label='month', value=$value, stacked, title='Sales per month, in thousands'/}
    {else}
      {chart $chartKind, data='sales', label='month', value=$value, title='Sales per month, in thousands'/}
    {/if}
  </div>
{/fragment}
</div>

<script type="text/babel" src="{$padRoot}react/examples/chart.js"></script>

App: regression/framework - Page: tags/a_kept_chart_and_the_same_chart_drawn_now_have_their_own_ids
PHP
tags/a_kept_chart_and_the_same_chart_drawn_now_have_their_own_ids.php 		HTML
tags/a_kept_chart_and_the_same_chart_drawn_now_have_their_own_ids.pad 		Result 
<?php

  // An earlier request stores the named section with the bar chart; this request serves it
  // from the cache and draws the same chart beside it. The kept one was numbered by the
  // request that stored it - the first of its kind - and so was the one drawn now: one id
  // twice in the page. Kept, a chart's id is made with the section's key as well.

  if ( ! isset ( $warm ) ) {
    padFragmentForget ( 'fwChartSameIds' );
    padCurl ( [ 'url' => $padHost . "regression/framework/?tags/a_kept_chart_and_the_same_chart_drawn_now_have_their_own_ids&padInclude&warm=1" ] );
  }

?>
 
{data 'sales'}[{"month":"Jan","amount":120},{"month":"Feb","amount":90}]{/data}
{cache 'fwChartSameIds'}{chart 'bar', data='sales', value='amount', width=300, height=150/}{/cache}
{chart 'bar', data='sales', value='amount', width=300, height=150/}
