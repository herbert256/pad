<?php

  // A quote's stand-in - U+E022, which a request can carry in a value - is escaped as the
  // quote it will become at output (exits.php restores it after all escaping), so it cannot
  // close the attribute {pad:attrs} or {pad:classes} builds and add an event handler.

  $t = "hi\u{E022} onmouseover=1";
  $c = "p\u{E022} onx=1";

?>
 
<a {pad:attrs title=$t/}>x</a>|<i class="{pad:classes p=1, $c/}">y</i>

<?php

  $tasks = [
    [ 'title' => 'Write the release notes', 'done' => TRUE,  'days' => 0 ],
    [ 'title' => 'Update the screenshots',  'done' => FALSE, 'days' => 1 ],
    [ 'title' => 'Plan the next version',   'done' => FALSE, 'days' => 14 ] ];

?>
 
{pad:meta title='Per row of a loop', about='Each row\'s classes from its own fields and from the loop\'s properties.'/}
<ul>
  {array:tasks}
    <li class="{pad:classes 'task', done=$done, first=first@tasks, urgent=$days lt 2/}">{$title}: <code>{pad:classes 'task', done=$done, first=first@tasks, urgent=$days lt 2/}</code></li>
  {/array:tasks}
</ul>
