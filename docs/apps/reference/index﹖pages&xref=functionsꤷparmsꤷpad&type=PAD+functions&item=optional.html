App: regression/pages - Page: tags/code
PHP
tags/code.php 		HTML
tags/code.pad 		Result 
<?php

  // A truthy page variable, on purpose: the unset that makes sandbox and clean leave no
  // trace once compared names against snapshot values, loosely, and any TRUE in the page
  // matched every name - so {code sandbox} on a page like this one unset nothing. The
  // sandbox suite could not see it, because a case's snapshot is empty.

  $codePageFlag = TRUE;

?>
 
{# The four isolation forms of a nested pass, run from a real page: sandbox and clean unset
   what the pass created, plain and reset leave it standing. #}

sandbox: [{code sandbox}{set $zq = 'q'/}{/code}{$zq | optional}]
clean: [{code clean}{set $zc = 'q'/}{/code}{$zc | optional}]
plain: [{code}{set $zp = 'q'/}{/code}{$zp | optional}]
reset: [{code reset}{set $zr = 'q'/}{/code}{$zr | optional}]
