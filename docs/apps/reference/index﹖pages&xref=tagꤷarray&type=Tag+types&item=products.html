App: htmx - Page: examples/cart
PHP
examples/cart.php 		HTML
examples/cart.pad 		Result 
<?php

  // The cart is the session's: product id => quantity. A button posts its product - op=add
  // or op=remove - and the answer is the {fragment 'item'} of that product: the card with its
  // new count, then the parts elsewhere on the page that changed too, each marked
  // hx-swap-oob - htmx puts the card back where the button was and the rest where their ids
  // are: the count in the header, the summary beside the shelf. Emptying the cart is an
  // ordinary {form} post that ends in a redirect, the page coming back whole.

  $counts   = padSession ( 'htmxCart', [] );
  $products = array_slice ( htmxData ( 'products' ), 0, 8 );
  $posted   = padRequestIs ( 'POST' );

  if ( $posted ) {

    $id    = (int) padRequest ( 'id', 0 );
    $op    = htmxText ( 'op' );
    $stock = array_column ( $products, 'stock', 'id' );

    if ( $op == 'clear' ) {
      padSessionPut ( 'htmxCart', [] );
      padRedirect ();
    }

    if ( $op == 'add' and ( $stock [$id] ?? 0 ) > ( $counts [$id] ?? 0 ) )
      $counts [$id] = ( $counts [$id] ?? 0 ) + 1;
    elseif ( $op == 'remove' and isset ( $counts [$id] ) and --$counts [$id] <= 0 )
      unset ( $counts [$id] );

    padSessionPut ( 'htmxCart', $counts );

    $products = array_values ( array_filter ( $products, fn ( $product ) => $product ['id'] == $id ) );

    // A product the shelf does not have - a stale or made-up request - has no card to
    // answer: a 404, which htmx swaps nothing for, where the empty loop rendered no
    // {fragment 'item'} and the answer was a 500, "there is no fragment named 'item'".

    if ( ! $products )
      padAbort ( 404, 'There is no such product' );

  }

  foreach ( $products as $at => $product )
    $products [$at] ['inCart'] = $counts [ $product ['id'] ] ?? 0;

  $cart      = htmxCart ( $counts );
  $cartCount = $cart ['count'];
  $lines     = $cart ['lines'];
  $total     = $cart ['total'];

?>
 
<div class="shop">
  <div id="shelf" class="grid shelf">
  {products}
  {fragment 'item'}
    <div class="card item" id="item-{$id}">
      <div class="item-top"><span class="item-emoji">{$emoji}</span>{if $inCart}<span class="badge is-accent">{$inCart} in cart</span>{/if}</div>
      <strong>{$name}</strong>
      <div class="muted small">{$price | currency('EUR')} &middot; {if $stock}{$stock} in stock{else}sold out{/if}</div>
      <div class="item-buttons">
        <button class="btn-sm" hx-post="?examples/cart&id={$id}&op=add&padFragment=item" hx-target="#item-{$id}" hx-swap="outerHTML">+ Add</button>
        {if $inCart}<button class="btn-sm btn-ghost" hx-post="?examples/cart&id={$id}&op=remove&padFragment=item" hx-target="#item-{$id}" hx-swap="outerHTML">−</button>{/if}
      </div>
    </div>
    {if $posted}
      <span id="cart-count" hx-swap-oob="true">{$cartCount}</span>
      {cartSummary}
    {/if}
  {/fragment}
  {/products}
  </div>
  {cartSummary}
</div>

App: webcomponents - Page: examples/rating
PHP
examples/rating.php 		HTML
examples/rating.pad 		Result 
<?php

  // The votes are the session's: product id => [ the sum, the count ], started from a few
  // votes of others. The rating of a product is the visitor's own last vote; a click posts
  // the product and the stars, and the answer is the new average ($padExpose).

  $products = [ [ 'id' => 1, 'name' => 'Aurora Headphones', 'emoji' => '🎧' ],
                [ 'id' => 2, 'name' => 'Tactile Keyboard',  'emoji' => '⌨️' ],
                [ 'id' => 3, 'name' => 'Ember Desk Lamp',   'emoji' => '💡' ] ];

  $votes = padSession ( 'wcVotes', [ 1 => [ 'sum' => 18, 'count' => 4, 'mine' => 0 ],
                                     2 => [ 'sum' => 9,  'count' => 2, 'mine' => 0 ],
                                     3 => [ 'sum' => 7,  'count' => 2, 'mine' => 0 ] ] );

  // Asked for its JSON without a vote, the page answers no rating; a vote for a product it
  // does not have is a 404 - where both were a 500, the one for a $rating never set, the
  // other for the votes of product 99.

  $rating = NULL;

  if ( padRequestIs ( 'POST' ) ) {

    $id    = (int) padRequest ( 'id', 0 );
    $value = (int) padRequest ( 'value', 0 );

    if ( ! isset ( $votes [$id] ) )
      padAbort ( 404, 'There is no such product' );

    if ( $value >= 1 and $value <= 5 ) {
      $votes [$id] ['sum']  += $value - $votes [$id] ['mine'];
      $votes [$id] ['count'] += $votes [$id] ['mine'] ? 0 : 1;
      $votes [$id] ['mine']  = $value;
      padSessionPut ( 'wcVotes', $votes );
    }

    $rating = [ 'average' => round ( $votes [$id] ['sum'] / max ( 1, $votes [$id] ['count'] ), 1 ),
                'votes'   => $votes [$id] ['count'] ];

  }

  foreach ( $products as $at => $product )
    $products [$at] += [ 'mine'    => $votes [ $product ['id'] ] ['mine'],
                         'average' => round ( $votes [ $product ['id'] ] ['sum'] / $votes [ $product ['id'] ] ['count'], 1 ),
                         'votes'   => $votes [ $product ['id'] ] ['count'] ];

  $padExpose = [ 'rating' ];

?>
 
<div class="grid">
{products}
  <div class="card product">
    <span class="product-emoji">{$emoji}</span>
    <strong>{$name}</strong>
    {rating name=$id, value=$mine, average=$average, votes=$votes}
  </div>
{/products}
</div>

<script type="module" src="{$padRoot}webcomponents/elements/rating.js"></script>
