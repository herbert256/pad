examples/stepper.php 		HTML
examples/stepper.pad 		Result 
<?php

  // An order: the quantity comes from <pad-stepper>, a form-associated custom element whose
  // value the {form} posts like any field; the size from an {input}. padValidate checks both
  // on the server - the stepper lets the visitor go to 12, the rule stops at 10.

  $errors  = [];
  $ordered = FALSE;

  if ( padPosted ( 'order' ) ) {
    $errors  = padValidate ( [ 'qty' => 'required|integer|min:1|max:10', 'size' => 'required|in:s,m,l' ],
                             NULL, [ 'qty.max' => 'At most :n at a time', 'qty' => 'The quantity is a number from 1 to 10' ] );
    $ordered = ! $errors;
  }

  $qty      = (int) padRequest ( 'qty', 2 );
  $size     = wcText ( 'size' );
  $qtyError = $errors ['qty'] ?? '';

?>
 
{if $ordered}
  <div class="callout is-good"><p><strong>Ordered:</strong> {$qty} × size {$size} - the stepper's value came with the post like any field.</p></div>
{/if}

{form 'order', error, class='stack'}
  <label for="qty">Quantity</label>
  {stepper name='qty', value=$qty, min=1, max=12}
  {if $qtyError}<span class="error">{$qtyError}</span>{/if}

  {input 'size', label='Size - s, m or l', class='input', value=$size}

  <button>Order</button>
{/form}

<script type="module" src="{$padRoot}webcomponents/elements/stepper.js"></script>
