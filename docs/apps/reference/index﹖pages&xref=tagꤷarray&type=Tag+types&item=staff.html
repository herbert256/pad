Reference - Tag types - staff

App: regression/pages - Page: sample/orders
PHP
sample/orders.php 		HTML
sample/orders.pad 		Result 
<?php

  // The real page: its rows come from the database.

  $title = 'Staff';
  $staff = db ( 'array name from staff order by name limit 2' );

?>
 
<h1>{$title}</h1>
<p>{$who}</p>
<ul>{staff}<li>{$name}</li>{/staff}</ul>
<p>{field "count(*) from staff", name='staffCount'} on the staff</p>

App: regression/sqlite - Page: verbs
PHP
verbs.php 		HTML
verbs.pad 		Result 
<?php

  // db()'s reading verbs on SQLite, answering in the shapes they have on MySQL.

  $field  = db ( "field phone from staff where name = 'jim'" );
  $joe    = db ( "record name, salary from staff where name = 'joe'" );
  $staff  = db ( "array name, phone from staff order by name" );
  $found  = db ( "check staff where name = 'bob'" );
  $none   = db ( "check staff where name = 'nobody'" );
  $count  = db ( "field count(*) from staff" );

  $types  = gettype ( $count ) . ' ' . gettype ( $joe ['salary'] );

?>
 
field:  {$field}
record: {joe}{$name} {$salary}{/joe}
array:  {staff}{$name}:{$phone} {/staff}
check:  {$found} {$none}
count:  {$count}
types:  {$types}
