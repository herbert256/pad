Reference - Tag types - temperatures

App: tags - Page: examples/if/2-elseif
PHP
examples/if/2-elseif.php 		HTML
examples/if/2-elseif.pad 		Result 
<?php

  $temperatures = [
    [ 'city' => 'Oslo',    'degrees' => -4 ],
    [ 'city' => 'Utrecht', 'degrees' => 12 ],
    [ 'city' => 'Seville', 'degrees' => 27 ] ];

?>
 
{meta title='elseif and else', about='The branches are tried in order - the first that holds wins.'}
{temperatures}
  {if $degrees lt 0}
    <p>{$city}: freezing</p>
  {elseif $degrees lt 20}
    <p>{$city}: mild</p>
  {else}
    <p>{$city}: warm</p>
  {/if}
{/temperatures}
