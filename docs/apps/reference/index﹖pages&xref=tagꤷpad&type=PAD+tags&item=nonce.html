<?php

  // A section holding the CSP nonce - a widget's or a chart's <style> under style-src 'nonce'
  // - is kept, the nonce written in on a hit as the request's: it was never kept at all.

  $padCsp = "default-src 'self'; style-src 'nonce'";

  padFragmentForget ( 'nonce-section' );

?>

App: regression/framework - Page: tags/a_cached_section_holding_the_csrf_token_or_the_nonce_is_not_kept
PHP
tags/a_cached_section_holding_the_csrf_token_or_the_nonce_is_not_kept.php 		HTML
tags/a_cached_section_holding_the_csrf_token_or_the_nonce_is_not_kept.pad 		Result 
<?php

  // A section whose rendering holds the visitor's CSRF token - a {form}, {csrf}, a live
  // region - is rendered every time and never kept: the copy went to every later visitor,
  // who got the first one's token to post with on that visitor's behalf. The second section
  // of that name finds no stored copy and renders its own content. One holding the CSP
  // nonce is kept with a marker in its place, which a hit fills with the nonce of the
  // request it answers - the second section of that name is the first one's copy.

  padFragmentForget ( 'fwTokenSection' );
  padFragmentForget ( 'fwNonceSection' );

?>
 
{cache 'fwTokenSection', ttl=60}<i>{csrf token}</i>{/cache}{cache 'fwTokenSection', ttl=60}<b>kept</b>{/cache}{cache 'fwNonceSection', ttl=60}<i>{nonce}</i>{/cache}{cache 'fwNonceSection', ttl=60}<b>kept</b>{/cache}

App: regression/framework - Page: tags/a_value_in_a_cached_section_never_takes_the_nonce

tags/a_value_in_a_cached_section_never_takes_the_nonce.php 		HTML
tags/a_value_in_a_cached_section_never_takes_the_nonce.pad 		Result 
<?php

  // A value in a {cache} section that holds the text of a nonce marker stays that text on a
  // hit - the marker is sealed: the fixed one, brought in by a value, took each later
  // request's real nonce, and a script injected with it passed the policy.

  $markerText = "\u{E0A0}padNonce\u{E0A0}";

  padFragmentForget ( 'nonce-mark' );

?>
 
[{cache 'nonce-mark', ttl=60}{!markerText}|{nonce}{/cache}][{cache 'nonce-mark', ttl=60}x{/cache}]
