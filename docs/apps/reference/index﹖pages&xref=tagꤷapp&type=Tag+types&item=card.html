App: webcomponents - Page: examples/cards
PHP
examples/cards.php 		HTML
examples/cards.pad 		Result 
<?php

  // Three cards from one custom tag: {card} writes a <pad-card> with its shadow root, the
  // stylesheet inlined by {shadow css=} and named slots; the page hands it the title and the
  // tone as parameters and the rest as content - the light DOM the browser slots in.

  $cards = [ [ 'title' => 'Revenue',  'tone' => 'good', 'value' => '€ 48,210', 'note' => '12% more than last month' ],
             [ 'title' => 'Visitors', 'tone' => 'info', 'value' => '12,804',   'note' => 'Most of them on a phone' ],
             [ 'title' => 'Returns',  'tone' => 'bad',  'value' => '38',       'note' => 'Three more than usual' ] ];

?>
 
<div class="grid">
{cards}
  {card title=$title, tone=$tone}
    <p class="stat">{$value}</p>
    <p class="muted">{$note}</p>
    <a slot="footer" href="?examples/cards">Details →</a>
  {/card}
{/cards}
</div>

<p class="muted small" style="margin-top: 18px">
  Open the developer tools: each card is <code><pad-card></code> with a <code>#shadow-root</code> PAD wrote. The page
  loads no script for it - the browser builds the root from the <code><template shadowrootmode></code>, and the
  styles inside it neither leak out nor let the page's in.
</p>
