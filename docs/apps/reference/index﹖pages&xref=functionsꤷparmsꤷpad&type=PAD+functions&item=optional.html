App: regression/framework - Page: expressions/a_field_piped_into_a_fallback_may_be_missing_in_echo_too

expressions/a_field_piped_into_a_fallback_may_be_missing_in_echo_too.php 		HTML
expressions/a_field_piped_into_a_fallback_may_be_missing_in_echo_too.pad 		Result 
<?php

  // A field piped into a fallback - default(), ??, optional - may be missing under the strict
  // check in {echo} and inside an expression too, not only in {$x | ...}: they named it missing.

  $user = [ 'name' => 'Ann' ];
  $xs   = [ [ 'n' => 1 ], [ 'n' => 2, 'm' => 5 ] ];

?>
 
[{echo $user.middle | default("-")}][{echo $user.middle | ?? "-"}][{echo $user.middle | optional}][{echo $user.name | default("-")}] {xs where='$m | ?? 0 | @ gt 1'}{$n}{/xs}
