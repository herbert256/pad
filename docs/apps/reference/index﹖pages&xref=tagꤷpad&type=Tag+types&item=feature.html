App: regression/framework - Page: tags/auth_owns_its_else_inside_an_if_and_an_if_inside_it_owns_its_own

tags/auth_owns_its_else_inside_an_if_and_an_if_inside_it_owns_its_own.php 		HTML
tags/auth_owns_its_else_inside_an_if_and_an_if_inside_it_owns_its_own.pad 		Result 
<?php

  // An {else} belongs to the innermost owner it stands in - {if}, {case}, {auth}, {guest},
  // {can}, {cannot}, {feature} - whichever of them encloses which.

  padLogin ( [ 'id' => 1, 'name' => 'Bo' ] );

  padGate ( 'x', fn ( $user ) => FALSE );

  $padFeatures = [ 'off' => FALSE ];

?>
 
{if 1 eq 2}no{else}{guest}no{else}in{/guest}{/if} | {auth}{if 1 eq 2}no{else}{$name}{/if}{else}no{/auth} | {case 'a'}{when 'a'}{feature 'off'}no{else}off{/feature}{else}no{/case} | {can 'x'}{if 1 eq 1}x{else}no{/if}{else}not x{/can}

App: regression/framework - Page: tags/feature_shows_its_content_when_the_flag_is_on
PHP
tags/feature_shows_its_content_when_the_flag_is_on.php 		HTML
tags/feature_shows_its_content_when_the_flag_is_on.pad 		Result 
<?php

  // TRUE, FALSE, a share - decided by the id, fixed here - and a function given the user.

  function featureCaseBeta ( $user ) { return ( $user ['role'] ?? '' ) == 'beta'; }

  $padFeatures = [ 'on' => TRUE, 'off' => FALSE, 'all' => 1, 'none' => 0.0, 'half' => 0.5, 'beta' => 'featureCaseBeta' ];

  padLogin ( [ 'id' => 3, 'role' => 'beta' ] );
  padFeatureId ( 'visitor:case' );

?>
 
{feature 'on'}on{else}off{/feature} {feature 'off'}on{else}off{/feature} {feature 'all'}on{/feature} {feature 'none'}on@else@off{/feature} {feature 'half'}on{else}off{/feature} {feature 'beta'}beta{/feature}

examples/feature/1-on-off.php 		HTML
examples/feature/1-on-off.pad 		Result 
<?php

  $padFeatures = [ 'newSearch' => TRUE, 'darkMode' => FALSE ];

?>
 
{meta title='A flag on and a flag off', about='TRUE and FALSE in $padFeatures switch a part for everybody - normally set in _config/config.php.'}
{feature 'newSearch'}
  <p>The new search, with suggestions.</p>
{else}
  <p>The old search.</p>
{/feature}
{feature 'darkMode'}
  <p>Dark mode is on.</p>
@else@
  <p>Dark mode is not released yet.</p>
{/feature}
