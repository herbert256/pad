App: regression/pages - Page: db/a_null_of_the_enclosing_row_is_bound_as_null
PHP
db/a_null_of_the_enclosing_row_is_bound_as_null.php 		HTML
db/a_null_of_the_enclosing_row_is_bound_as_null.pad 		Result 
<?php

  // A NULL column of the enclosing row is NULL in the SQL it goes into - a $name of a
  // database tag, the relation a nested select follows - where it was bound as the empty
  // text '': a boss of NULL found the row whose key is '', on MySQL an integer key 0.

  $padSelect ['staffBoss']   = [ 'db' => 'staff' ];
  $padSelect ['staffByBoss'] = [ 'db' => 'staff' ];

  $padRelations ['staffByBoss'] ['staffBoss'] = [ 'name' => 'boss' ];

?>
 
bound:    [{array "name, NULL as boss from staff where name = 'jim'"}{field "$boss is null"/}{/array}]
relation: [{staffBoss fields="name, NULL as boss", where="name = 'jim'"}{$name}:{staffByBoss}{$name}{/staffByBoss}{/staffBoss}]

App: regression/pages - Page: db/sqlite_numeric_text_keeps_its_zeros
PHP
db/sqlite_numeric_text_keeps_its_zeros.php 		HTML
db/sqlite_numeric_text_keeps_its_zeros.pad 		Result 
<?php

  // On SQLite numeric text goes into a bare placeholder, or a select's $name, as a number
  // only when SQLite gives the same text back: '007' became the integer 7, so the code
  // '007' found the row of code '7', the zip '01234' was stored as '1234' and '1.10' as
  // '1.1'. A numeral like '3' stays a number, so a count still compares with it.

  $GLOBALS ['padSqlConnect'] = padDbSqlite ( ':memory:' );

  db ( "create table codes (code text primary key, zip varchar(10), label text)" );
  db ( "insert into codes values ('007', '01234', 'bond'), ('7', '1234', 'seven')" );

  $byCode = db ( "field label from codes where code = {0}",                   [ '007' ] );
  $byZip  = db ( "field zip from codes where zip = {0}",                      [ '01234' ] );
  $inList = db ( "array code from codes where code in ({0}) order by code",   [ [ '007', '7' ] ] );
  $inList = implode ( ' ', array_column ( $inList, 'code' ) );

  db ( "insert into codes values ({0}, {1}, 'kept')", [ '0042', '1.10' ] );

  $stored = db ( "record code, zip from codes where label = 'kept'" );
  $count  = db ( "field 'three' where (select count(*) from codes) = {0}",    [ '3' ] );

  $padSelect ['codes'] = [ 'db' => 'codes', 'key' => 'code' ];

  $c = '007';

?>
 
code:   {$byCode}
zip:    {$byZip}
list:   {$inList}
stored: {$stored.code} {$stored.zip}
count:  {$count}
select: {codes where="code = $c"}{$label}{/codes}

App: regression/pages - Page: select/demo/a_bound_array_is_a_list
PHP
select/demo/a_bound_array_is_a_list.php 		HTML
select/demo/a_bound_array_is_a_list.pad 		Result 
<?php

  // An array bound into where= as $name is a list, as an array in a bare db() placeholder
  // is: name in ($names) is name in ('jim','bob'), where the array became the one literal
  // 'jim,bob' and matched nothing - and on SQLite an "Array to string conversion".

  $padSelect ['staffList'] = [ 'db' => 'staff' ];

  $names = [ 'jim', 'bob' ];

?>
 
[{staffList where="name in ($names)", order="name"}{$name} {/staffList}]

App: regression/pages - Page: select/demo/a_comment_in_a_view_s_where_ends_with_its_part
PHP
select/demo/a_comment_in_a_view_s_where_ends_with_its_part.php 		HTML
select/demo/a_comment_in_a_view_s_where_ends_with_its_part.pad 		Result 
<?php

  // A -- comment in a where ends with its own part of the where a view composes - with the
  // tag's where=, or with the where of the view it stands on - as it does in a where alone:
  // the parenthesis that closed the part stood after it, and the select ended on an SQL
  // syntax error. Over the four-row staff table of the demo database.

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name' ];
  $padSelect ['paid']     = [ 'base' => 'staffSel', 'where' => 'salary > 1000 -- the paid ones' ];
  $padSelect ['generous'] = [ 'base' => 'staffSel', 'where' => 'bonus > 100' ];
  $padSelect ['midPaid']  = [ 'base' => 'paid',     'where' => 'salary < 4000' ];

?>
 
declared: [{paid where="bonus > 200"}{$name} {/paid}]
tag: [{generous where="salary > 1000 -- the paid ones"}{$name} {/generous}]
chain: [{midPaid}{$name} {/midPaid}]
