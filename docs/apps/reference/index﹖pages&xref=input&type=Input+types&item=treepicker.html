App: regression/framework - Page: tags/a_treepicker_refills_the_item_its_form_posted
PHP
tags/a_treepicker_refills_the_item_its_form_posted.php 		HTML
tags/a_treepicker_refills_the_item_its_form_posted.pad 		Result 
<?php

  // A post of form f: its city shows the item posted in place of value=, the city of form g,
  // which did not come back, shows value=; town came back without a choice - a radio group
  // with none is left out of the post, the padLists[] beside it says it was there - so none
  // is chosen, not value= again; and place was
  // posted a value that is no item of it, which is not shown. The page's padValidate left a
  // message for the town, tied to its radios.

  $_SERVER ['REQUEST_METHOD'] = 'POST';
  $_POST = [ 'padForm' => 'f', 'city' => 'Paris', 'place' => 'Tokyo', 'padLists' => [ 'city', 'town', 'place' ] ];

  padValidate ( [ 'town' => 'required' ] );

?>
 
{pad:form 'g'}{input:treepicker 'city', options='Europe > Paris, Europe > Rome', value='Rome'/}{/pad:form}{pad:form 'f'}{input:treepicker 'city', options='Europe > Paris, Europe > Rome', value='Rome'/}{input:treepicker 'town', options='Lyon, Nice', value='Nice'/}{input:treepicker 'place', options='Kyoto, Osaka'/}{/pad:form}

<?php

  // Rows of a table whose parent_id is NOT NULL DEFAULT 0: a parent 0 that is no row's value

  ];

?>
