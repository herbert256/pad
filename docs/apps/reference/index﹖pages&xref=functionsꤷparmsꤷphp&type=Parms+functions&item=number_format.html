App: manual - Page: fragments/pipes_1
PHP
fragments/pipes_1.php 		HTML
fragments/pipes_1.pad 		Result 
<?php

  $name = 'john doe';
  $email = 'ALICE@EXAMPLE.COM';
  $price = 1234.56;

?>
 
<p>Original: {$name}</p>
<p>Uppercase: {echo $name | upper}</p>
<p>Capitalize: {echo $name | ucwords}</p>

<p>Original: {$email}</p>
<p>Lowercase: {echo $email | lower}</p>

<p>Original: {$price}</p>
<p>Formatted: {echo $price | number_format(@, 2)}</p>

App: showcase - Page: country/visitors
PHP
country/visitors.php 		HTML
country/visitors.pad 		Result 
<?php

  // Last month's visits by country, as an analytics export gives them.

  $visitors = [
    [ 'code' => 'NL', 'visits' => 18230 ],
    [ 'code' => 'US', 'visits' => 12904 ],
    [ 'code' => 'DE', 'visits' => 8711  ],
    [ 'code' => 'GB', 'visits' => 6120  ],
    [ 'code' => 'BE', 'visits' => 4385  ],
    [ 'code' => 'JP', 'visits' => 2017  ],
    [ 'code' => 'BR', 'visits' => 1460  ],
    [ 'code' => 'CI', 'visits' => 312   ],
  ];

?>
 
{meta title='Visitors by country', about='The flag and the name from a two-letter code, with name - the flag is then decoration and the name is what a screen reader says.'}
<ol style="margin:0;padding:0;list-style:none;min-width:20em">
  {visitors}
  <li style="display:flex;justify-content:space-between;gap:24px;padding:8px 0;border-bottom:1px solid rgba(128,128,128,.25);font-size:15px">
    {country $code, name}
    <span style="font-variant-numeric:tabular-nums;opacity:.8">{$visits | number_format}</span>
  </li>
  {/visitors}
</ol>

App: tags - Page: examples/macro/2-loop
PHP
examples/macro/2-loop.php 		HTML
examples/macro/2-loop.pad 		Result 
<?php

  $macroProducts = [
    [ 'product' => 'Kettle',  'price' => 39.5 ],
    [ 'product' => 'Toaster', 'price' => 24   ],
    [ 'product' => 'Blender', 'price' => 79.9 ] ];

?>
 
{meta title='Used inside a loop', about='The macro\'s own parameters are its fields, and the row round the use stays visible behind them.'}
{macro 'macroPrice', amount}
  <li>{$product}: € {$amount | number_format(@, 2)}</li>
{/macro}

<ul>
{macroProducts}
  {macroPrice $price}
{/macroProducts}
</ul>
