App: regression/framework - Page: ai/an_api_error_whose_message_is_no_text_is_an_error_like_any_other
PHP
ai/an_api_error_whose_message_is_no_text_is_an_error_like_any_other.php 		HTML
ai/an_api_error_whose_message_is_no_text_is_an_error_like_any_other.pad 		Result 
<?php

  // url= points a provider at any server that speaks its API, and their error answers take
  // many shapes: a message that is a list or an object - a validation error's location and
  // text - is an error like any other, said as its JSON, where it ended the page with
  // "Array to string conversion". With the strict check off the tag answers nothing.

  $padCheckSyntax = FALSE;

  $_ENV ['OPENAI_API_KEY'] = 'sk-test';

  padCurlFake ( [ '*' => [ 'status' => 422, 'data' => [ 'error' => [ 'message' => [ 'loc' => 'body', 'msg' => 'field required' ] ] ] ] ] );

  // The warning it logs goes to a day's log of its own: the cases that read the log, run at
  // the same time, find only their own lines in theirs.

  padNowFreeze ( '2026-01-13 12:00:00' );

?>
 
before [{pad:ai provider="openai", ttl=0}hello{/pad:ai}] after - {php:padAiError/}

App: regression/framework - Page: ai/without_the_strict_check_an_api_error_answers_nothing_and_is_logged
PHP
ai/without_the_strict_check_an_api_error_answers_nothing_and_is_logged.php 		HTML
ai/without_the_strict_check_an_api_error_answers_nothing_and_is_logged.pad 		Result 
<?php

  // With the strict check off an error of the API is no error of the page: the tag answers
  // nothing, padAiError says what the provider said, and a warning goes to the application's
  // log. A refusal that stays one - Anthropic's stop_reason refusal - is such an error too: its
  // text is never printed as the answer.

  $padCheckSyntax = FALSE;

  $_ENV ['ANTHROPIC_API_KEY'] = 'sk-test';

  padCurlFake ( [
    'https://api.openai.com/*'    => [ 'status' => 401, 'data' => [ 'error' => [ 'message' => 'Incorrect API key provided: sk-bad.', 'type' => 'invalid_request_error' ] ] ],
    'https://api.anthropic.com/*' => [ 'data' => [ 'stop_reason' => 'refusal', 'stop_details' => [ 'type' => 'refusal', 'category' => 'cyber' ], 'content' => [] ] ],
  ] );

  padNowFreeze ( '2026-11-01 12:00:00' );

  $log = DATA . 'logs/regression/framework/2026-11-01.log';

  $before = is_file ( $log ) ? count ( file ( $log ) ) : 0;

  $direct = padAi ( 'Hello?', [ 'provider' => 'openai', 'key' => 'sk-bad' ] );
  $error  = padAiError ();

  function aiLenientLogged () {
    global $log, $before;
    $lines = is_file ( $log ) ? preg_grep ( '/did not answer/', array_slice ( file ( $log ), $before ) ) : [];
    return implode ( ' / ', array_map ( fn ( $line ) => preg_replace ( '/^\S+ \S+ (\S+) (.*?) \{.*$/s', '$1 $2', trim ( $line ) ), $lines ) );
  }

?>
 
padAi: [{$direct}] {$error}
tag: [{pad:ai provider='anthropic', ttl=0}Tell me a secret.{/pad:ai}] {php:padAiError/}
logged: {php:aiLenientLogged/}
