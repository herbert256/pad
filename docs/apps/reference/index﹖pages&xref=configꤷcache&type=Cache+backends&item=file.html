App: regression/cache_file - Page: a_304_carries_the_etag_its_200_carries
PHP
a_304_carries_the_etag_its_200_carries.php 		HTML
a_304_carries_the_etag_its_200_carries.pad 		Result 
<?php

  // A 304 carries the ETag the 200 to the same request carries. The stable page, fetched at
  // a fresh address: the miss goes out as it was made, the hit gzipped as the cache keeps it,
  // with the tag of the gzip body ("...-gzip"), and asked again with that tag the answer is
  // a 304 with the same one - it said the identity tag, which a cache holding the gzip copy
  // may not take for the one it holds.

  function tagOf ( $answer ) {

    foreach ( $answer ['headers'] ?? [] as $name => $value )
      if ( strtolower ( $name ) == 'etag' )
        return trim ( $value );

    return '';

  }

  $tagUrl  = $padHost . 'regression/cache_file/?stable&fresh=' . bin2hex ( random_bytes ( 6 ) );
  $tagMiss = padCurl ( $tagUrl );
  $tagHit  = padCurl ( $tagUrl );
  $tagAsk  = padCurl ( [ 'url' => $tagUrl, 'headers' => [ 'If-None-Match' => tagOf ( $tagHit ) ] ] );

  $tagResult = 'miss: '  . ( str_ends_with ( tagOf ( $tagMiss ), '-gzip"' ) ? 'gzip tag' : 'plain tag' )
             . ', hit: ' . ( str_ends_with ( tagOf ( $tagHit ),  '-gzip"' ) ? 'gzip tag' : 'plain tag' )
             . ', asked with it: ' . $tagAsk ['result'] . ' ' . ( tagOf ( $tagAsk ) === tagOf ( $tagHit ) ? 'the same tag' : 'another tag' );

?>
 
{pad:meta cache=0/}{$tagResult}

App: regression/cache_file - Page: a_public_answer_among_several_locales_varies_on_the_cookie
PHP
a_public_answer_among_several_locales_varies_on_the_cookie.php 		HTML
a_public_answer_among_several_locales_varies_on_the_cookie.pad 		Result 
<?php

  // A page a shared cache may keep, of an application with several $padLocales, says Vary:
  // Cookie on its public answer as well as Accept-Language: a padLang cookie chooses the
  // locale (lib/locale.php), and the answer to that visitor is private - but a shared cache
  // that kept the guest's English answer, stored under the address and Accept-Language
  // alone, handed it to the next visitor whose padLang=nl cookie chose Dutch, the server
  // never asked (RFC 9111 4.1: only the headers Vary names are matched). Apache joins the
  // Vary names with a comma, php -S with a comma and a blank: the blank is taken out before
  // they are compared.

  $varyAnswer = padCurl ( [ 'url' => $padHost . 'regression/cache_file/?proxied&padInclude&fresh=' . bin2hex ( random_bytes ( 6 ) ),
                            'headers' => [ 'Accept-Language' => 'en' ] ] );

  $varyControl = '';
  $varyList    = '';

  foreach ( $varyAnswer ['headers'] ?? [] as $varyName => $varyValue )
    if ( strtolower ( $varyName ) == 'cache-control' )
      $varyControl = $varyValue;
    elseif ( strtolower ( $varyName ) == 'vary' )
      $varyList = preg_replace ( '/\s*,\s*/', ',', $varyValue );

  $varyResult = trim ( (string) ( $varyAnswer ['data'] ?? '' ) ) . ' '
              . ( preg_match ( '/^(public|private)\b/', $varyControl, $varyMatch ) ? $varyMatch [1] : "'$varyControl'" )
              . ', Vary: ' . $varyList;

?>
 
{pad:meta cache=0/}{$varyResult}

App: regression/cache_file - Page: a_visitor_with_an_identity_is_never_answered_public
PHP
a_visitor_with_an_identity_is_never_answered_public.php 		HTML
a_visitor_with_an_identity_is_never_answered_public.pad 		Result 
<?php

  // A page a shared cache may keep ($padCacheProxyAge) is public to a guest only. Asked with
  // a cookie of the visitor's own, or with an Authorization header, it is that visitor's -
  // private, as the page cache already treats such a request: it went out public with an
  // s-maxage, and a proxy kept "Hello Alice" for everyone after her (RFC 9111 3.5 lets a
  // shared cache reuse an answer to an Authorization request that says public). So is the
  // answer whose locale a padLang cookie chose: a shared cache keys it on the address and the
  // Accept-Language alone, and handed the Dutch page to the next English visitor.

  function proxyType ( $extra ) {

    global $padHost;

    $answer = padCurl ( [ 'url' => $padHost . 'regression/cache_file/?proxied&padInclude&fresh=' . bin2hex ( random_bytes ( 6 ) ) ] + $extra );

    $control = '';

    foreach ( $answer ['headers'] ?? [] as $name => $value )
      if ( strtolower ( $name ) == 'cache-control' )
        $control = $value;

    return trim ( $answer ['data'] ) . ' ' . ( preg_match ( '/^(public|private)\b/', $control, $match ) ? $match [1] : "'$control'" );

  }

  $proxyResult = 'guest: '           . proxyType ( [] )
               . ', cookie: '        . proxyType ( [ 'cookies' => [ 'viewer' => 'Alice' ] ] )
               . ', authorization: ' . proxyType ( [ 'headers' => [ 'Authorization' => 'Basic ' . base64_encode ( 'Alice:secret' ) ] ] )
               . ', language cookie: ' . proxyType ( [ 'cookies' => [ 'padLang' => 'nl' ], 'headers' => [ 'Accept-Language' => 'en' ] ] );

?>
 
{pad:meta cache=0/}{$proxyResult}

<?php

  // A page that shows this machine something it shows nobody else - a {pad:debug} - is not
  // stored: built for a local request it went into the cache, and every visitor after it
  // was answered with the value {pad:debug} keeps from them. The second fetch says it was

App: regression/cache_file - Page: etaglist
PHP
etaglist.php 		HTML
etaglist.pad 		Result 
<?php

  // On a cache miss the 304 decision reads the whole If-None-Match list: the stable page,
  // fetched at a fresh address each time so the cache never holds it, answers 304 to a list
  // holding its tag second, and to *, and 200 to a tag it does not have. Only the first tag
  // of the list was compared there - the cache's own path already read them all.

  function etagAsk ( $tags ) {

    global $padHost;

    $input = [ 'url' => $padHost . 'regression/cache_file/?stable&fresh=' . bin2hex ( random_bytes ( 6 ) ) ];

    if ( $tags !== '' )
      $input ['headers'] = [ 'If-None-Match' => $tags ];

    return padCurl ( $input );

  }

  $etagFirst = etagAsk ( '' );
  $etagTag   = '';

  foreach ( $etagFirst ['headers'] ?? [] as $etagName => $etagValue )
    if ( strtolower ( $etagName ) == 'etag' )
      $etagTag = trim ( $etagValue );

  $etagResult = 'tag: ' . ( $etagTag !== '' ? 'sent' : 'none' )
              . ' | list: '  . etagAsk ( '"AAAAAAAAAAAAAAAAAAAAAA", ' . $etagTag ) ['result']
              . ' | star: '  . etagAsk ( '*' ) ['result']
              . ' | other: ' . etagAsk ( '"AAAAAAAAAAAAAAAAAAAAAA"' ) ['result'];

?>
 
{$etagResult}

App: regression/cache_file - Page: fragment

fragment.pad 		Result 
<?php

  // A page whose PHP picks a response fragment - from an HX-Request header, which the page
  // cache's key does not hold - is answered fresh for such a request and its fragment is
  // never stored: the HTMX swap went into the cache under the page's address, and every
  // browser after it got the bare fragment for the whole page; or the stored whole page

<?php

  // A page under a directory guard stays out of the page cache: a hit is answered before
  // any guard runs, so a cached copy would go to whoever asked, whatever the guard says. The
  // guard here wants a header, and a request with it gets a fresh build each time while one
  // without it is refused - it was given the copy the first request left in the cache.

  $keyTwo = padCurl ( [ 'url' => $url, 'headers' => [ 'X-Key' => 'open' ] ] );
  $noKey  = padCurl ( $url );

  $guardedResult = 'built fresh: ' . ( $keyOne ['data'] !== $keyTwo ['data'] ? 'yes' : 'NO' )
                 . ', without the key: ' . $noKey ['result'];

?>
