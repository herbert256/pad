<?php

  $user = [
    'id'       => 7,
    'name'     => 'Ann',
    'email'    => 'ann@example.com',
    'password'=> '*REDACTED*',
    'login'    => [ 'ip'      => '10.0.0.1',
                    'browser' => 'Firefox' ]
  ];

  $contact = padArrOnly   ( $user, 'email, name' );
  $public  = padArrExcept ( $user, 'password, login.ip' );

  {~notLast@contact}, {/notLast@contact}
{~/contact}</p>

App: manual - Page: fragments/arrays_4
PHP
fragments/arrays_4.php 		HTML
fragments/arrays_4.pad 		Result 
<?php

  $orders = [
    [ 'number' => 101, 'customer' => 'Ann',
      'status' => 'paid', 'total' => '120.00' ],
    [ 'number' => 102, 'customer' => 'Bob',
      'status' => 'open', 'total' => '35.50' ],
    [ 'number' => 103, 'customer' => 'Cleo',
      'status' => 'paid', 'total' => '89.90' ],
    [ 'number' => 104, 'customer' => 'Carl',
      'status' => 'open', 'total' => '15.00' ]
  ];

  $large  = padArrWhere  ( $orders, 'total', '>=', 50 );
  $large  = padArrSortBy ( $large, 'total', 'desc' );
  $withC  = padArrWhere  ( $orders, 'customer', 'like', 'c%' );
  $unpaid = padArrWhere  ( $orders, 'status', 'open' );

  $isOpen = fn ( $order ) => $order ['status'] == 'open';
  $isPaid = fn ( $order ) => $order ['status'] == 'paid';

  $first = padArrFirst ( $orders, $isOpen ) ['number'];
  $last  = padArrLast  ( $orders, $isPaid ) ['number'];

?>
 
<p>50 or more, largest first:
  {large~}
    {$number} ({$total}){notLast@large}, {/notLast@large}
  {~/large}</p>
<p>A customer starting with C: {withC~}
  {$customer}{notLast@withC}, {/notLast@withC}
{~/withC}</p>
<p>Not paid yet: {unpaid~}
  {$number}{notLast@unpaid}, {/notLast@unpaid}
{~/unpaid}</p>
<p>First open order: {$first}, last paid order: {$last}</p>

App: sequence - Page: examples

examples.php 		HTML
examples.pad 		Result 
<?php

  // Examples page: picks the group of examples to show and the one item out of it.
  //
  // example= names a directory of the application, so it is only used when it is one of the
  // groups the page offers - the same list examples.pad walks. Interpolating it unchecked
  // meant a ../ read the file names of any directory the server could reach, and an unknown
  // group ended the request with a 500. item= is checked against the group's own contents,
  // which it already was.

  $examples = [ 'sequences', 'specials', 'random', 'play/single', 'play/double', 'keepRemoveFlag' ];

  if ( ! isset ( $example ) or ! in_array ( $example, $examples ) )
    $example = 'specials';

  if ( $example == 'specials' )
    if ( ! isset ( $item ) or ! $item )
      $item = 'basic';

  if ( $example == 'sequences')
    $exam = sequenceDir ( APP . "basic" ) ;
  else
    $exam = sequenceDir ( APP . "$example" ) ;

  if ( ! count ( $exam ) )
    $exam = [ 'basic' ];

  if ( isset ( $item ) and ! in_array ( $item, $exam) )
    $item = $exam [0];

  if ( ! isset ( $item ) or ! $item )
    $item = $exam [0];

?>
 
{data 'list'} 
  (
    'sequences',
    'specials',
    'random',
    'play/single',
    'play/double',
    'keepRemoveFlag'
  )
{/data}

{block}

<p>Examples ! Many are generated and do not always make sence, <br>the examples from 'specials' are hand crafted and gives a good view of what the Sequences subsystem of PAD can do.</p>

{/block}

<br>
{lines}
  {list}
    {if $example eq $list}
      <b>{$list}</b>
    @else@
      <a href="?examples&example={$list}{?item}">{$list}</a>
    {/if}
  {notLast} | {/notLast}
  {/list}
{/lines}

<br>

{block}<center>

  {exam} 
    {if $item eq $exam}
      <b>{$exam}</b>
    @else@
      <a href="?examples&example={$example}&item={$exam}">{$exam}</a>
    {/if}
    &nbsp;
  {/exam}

  <br><br><hr>

  {if $example eq 'sequences' }
    <h3>{$item}</h3>
      {exists APP . 'basic/{$item}.pad'}
        <h5>{echo 'basic/{$item}.pad'}</h5>
        {example 'basic/{$item}', onlyResult}
      {/exists}
    {list}
      {exists APP . '{$list}/{$item}.pad'}
        <h5>{echo '{$list}/{$item}.pad'}</h5>
        {example '{$list}/{$item}', onlyResult}
      {/exists}
    {/list}
  @else@
    {example '{$example}/{$item}', onlyResult, include}
  {/if}

</center>{/block}
