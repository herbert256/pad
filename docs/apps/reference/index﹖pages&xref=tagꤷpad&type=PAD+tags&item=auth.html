<?php

  // padLogin keeps the row - less its password and token fields - and {auth} hands it to
  // its content as the data of one occurrence.

  padLogin ( [ 'id' => 7, 'name' => 'Ann', 'role' => 'editor', 'password' => padHash ( 'secret' ), 'remember_token' => 'x' ] );

  $kept = json_encode ( array_keys ( padUser () ) );

?>
 
{auth}{$id} {$name} {$role}{else}guest{/auth} | {guest}guest{else}not a guest{/guest} | {!kept}

<?php

  // An {else} belongs to the innermost owner it stands in - {if}, {case}, {auth}, {guest},
  // {can}, {cannot}, {feature} - whichever of them encloses which.

  padLogin ( [ 'id' => 1, 'name' => 'Bo' ] );

  padGate ( 'x', fn ( $user ) => FALSE );

  $padFeatures = [ 'off' => FALSE ];

?>
