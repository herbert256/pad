The template

{pad:live 'shop'}
  {sheet:grid 'shop', editable='B2:C6, C8, B9', caption='Change a quantity or a price',
    formats='C2:C6:€ #,##0.00, D:€ #,##0.00'}
    Item,Qty,Price,Total
    Bread,2,2.45,=B2*C2
    Cheese,1,8.90,=B3*C3
    Coffee,2,6.75,=B4*C4
    Apples,6,0.45,=B5*C5
    Flowers,1,12.50,=B6*C6
    Subtotal,,,=SUM(D2:D6)
    Discount over € 50,,10%,=IF(D7>50,-D7*C8,0)
    VAT,21%,,=(D7+D8)*B9
    Total,,,=D7+D8+D9
  {/sheet:grid}
{/pad:live}
