tags/can_asks_a_gate_with_a_value_and_with_the_row_of_an_enclosing_loop.pad 		Result 
<?php

  // A gate gets the user and the values {can} names: $post a page array, $posts inside
  // {posts} the row of that occurrence.

  padGate ( 'edit-post', fn ( $user, $post ) => $user ['id'] == $post ['user_id'] );

  padLogin ( [ 'id' => 7, 'name' => 'Ann' ] );

  $posts = [ [ 'title' => 'mine', 'user_id' => 7 ], [ 'title' => 'theirs', 'user_id' => 8 ] ];
  $post  = $posts [1];

?>
 
{can 'edit-post', $post}edit{else}read{/can} | {posts}{$title}: {can 'edit-post', $posts}edit{else}read{/can}; {/posts}| {posts}{cannot 'edit-post', $posts}{$title} is not mine{/cannot}{/posts}

App: regression/framework - Page: tags/can_lets_a_guest_reach_only_a_gate_that_takes_null
PHP
tags/can_lets_a_guest_reach_only_a_gate_that_takes_null.php 		HTML
tags/can_lets_a_guest_reach_only_a_gate_that_takes_null.pad 		Result 
<?php

  // Nobody is logged in: a gate whose first parameter allows NULL is asked, any other is
  // not - and a before hook is skipped the same way.

  padGateBefore ( fn ( $user, $ability ) => TRUE );

  padGate ( 'view-post',   fn ( ?array $user, $post ) => $post ['public'] );
  padGate ( 'delete-post', fn ( $user, $post ) => TRUE );

  $post = [ 'public' => TRUE ];

?>
 
{can 'view-post', $post}view{else}no view{/can} | {can 'delete-post', $post}delete{else}no delete{/can} | {cannot 'delete-post', $post}cannot delete@else@can delete{/cannot}

{set $note = [ 'title' => 'Minutes', 'published' => TRUE, 'author' => 7 ]}
{cannot 'edit-post', $note}
  <p>You cannot change these minutes.</p>
{else}
  <p><a href="#edit">Change the minutes</a></p>
{/cannot}
