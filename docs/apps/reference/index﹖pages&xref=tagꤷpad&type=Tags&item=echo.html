<?php

  // Five shares and their closing price over the last fifteen days.

  $stocks = [
    'ASML'    => [ 640, 652, 648, 661, 670, 668, 684, 679, 692, 701, 698, 712, 709, 725, 731 ],
    'Philips' => [ 21.4, 21.1, 20.8, 21.0, 20.6, 20.2, 20.5, 19.9, 19.6, 19.8, 19.3, 19.5, 19.1, 18.8, 18.9 ],
    'Shell'   => [ 30.1, 30.4, 30.2, 30.9, 31.3, 30.8, 30.6, 31.1, 31.4, 31.0, 31.6, 31.9, 31.5, 31.8, 32.2 ],
    'Heineken'=> [ 88, 87, 89, 86, 88, 90, 87, 85, 88, 89, 86, 87, 90, 88, 89 ],
    'Adyen'   => [ 1210, 1185, 1240, 1302, 1276, 1330, 1295, 1361, 1408, 1372, 1440, 1415, 1478, 1452, 1499 ]
  ];

  $shares = [];

  foreach ( $stocks as $name => $closes )
    $shares [] = [ 'name'   => $name,
                   'closes' => $closes,
                   'last'   => end ( $closes ),
                   'change' => round ( ( end ( $closes ) / $closes [0] - 1 ) * 100, 1 ) ];

?>
 
<table class="quotes">
  <tr><th>Share</th><th>15 days</th><th>Last</th><th>Change</th></tr>
  {array:shares}
  <tr>
    <td>{$name}</td>
    <td>{chart:sparkline data='closes', width=140, height=30, title=$name/}</td>
    <td>{$last}</td>
    <td class="{pad:echo $change ge 0 ? 'up' : 'down'/}">{$change} %</td>
  </tr>
  {/array:shares}
</table>

App: islands - Page: examples/build
PHP
examples/build.php 		HTML
examples/build.pad 		Result 
<?php

  // What {pad:vite 'src/main.js'} wrote into this page's <head>, and where it came from: the
  // manifest of the build - every source and the file it became - or the dev server, while it
  // runs. The clock is a Solid island started from the server's time.

  $dev      = padViteDev ();
  $manifest = $dev === '' ? ( padViteManifest () ?? [] ) : [];
  $written  = trim ( padViteTags ( [ 'src/main.js' ], FALSE, FALSE ) );

  $sources = [];

  foreach ( $manifest as $source => $one )
    if ( ! str_starts_with ( $source, '_' ) )
      $sources [] = [ 'source' => $source,
                      'file'   => $one ['file'],
                      'kind'   => ( $one ['isEntry'] ?? FALSE ) ? 'entry' : ( ( $one ['isDynamicEntry'] ?? FALSE ) ? 'loaded on demand' : 'chunk' ),
                      'css'    => implode ( ', ', $one ['css'] ?? [] ) ];

  $sourceCount = count ( $sources );
  $clock       = [ 'now' => time () ];

?>
 
<div class="cols-2">
  <div>
    <div class="panel-title">{pad:if $dev}The dev server runs: {$dev}{pad:else}The build: www/islands/build/{/pad:if}</div>
    <p class="muted small">{pad:if $dev}{pad:vite} asks the dev server - every save shows at once.{pad:else}Start <code>npm run dev</code> in <code>_frontend/</code> and reload: {pad:vite} then asks the dev server instead.{/pad:if}</p>
    <div class="panel-title" style="margin-top: 18px">What {pad:vite 'src/main.js'/} wrote into the head</div>
    {pad:echo $written | highlight('html')/}
  </div>
  <div>
    <div class="panel-title">The server's time, ticking in a Solid island</div>
    <div class="clock-box" data-island="Clock" data-props="{^clock}"><span class="clock">-</span></div>
  </div>
</div>

{pad:if $sourceCount}
<div class="panel-title" style="margin-top: 26px">The manifest - every source and what it became</div>
<table class="table">
  <thead><tr><th>Source</th><th>File</th><th>Kind</th><th>CSS</th></tr></thead>
  <tbody>
    {array:sources}<tr><td><code>{$source}</code></td><td><code>{$file}</code></td><td>{$kind}</td><td><code>{$css}</code></td></tr>{/array:sources}
  </tbody>
</table>
{/pad:if}
