App: charts - Page: heatmap/traffic
PHP
heatmap/traffic.php 		HTML
heatmap/traffic.pad 		Result 
<?php

  // Visits per weekday and hour: a busy lunch, a busier evening, a quiet weekend morning,
  // and a week that gets busier towards Thursday and quieter on Friday night.

  $traffic = [];
  $week    = [ 'Mon' => 0.85, 'Tue' => 0.95, 'Wed' => 1.0, 'Thu' => 1.1, 'Fri' => 0.9, 'Sat' => 1.2, 'Sun' => 1.3 ];

  foreach ( $week as $day => $busy )
    foreach ( [ 8, 10, 12, 14, 16, 18, 20, 22 ] as $hour ) {

      $visits = 40 + 60 * exp ( - ( ( $hour - 12.5 ) ** 2 ) / 3 )
                   + 90 * exp ( - ( ( $hour - 20 )   ** 2 ) / 4 );

      $visits = $visits * $busy - ( $busy > 1.15 and $hour < 12 ? 35 : 0 );

      $traffic [] = [ 'day' => $day, 'hour' => "{$hour}h", 'visits' => (int) round ( $visits ) ];

    }

?>
 
{chart 'heatmap',
       data='traffic',
       x='hour',
       y='day',
       value='visits',
       title='Visits per weekday and hour',
       width=640,
       height=360/}

App: regression/framework - Page: tags/chart_heatmap_takes_time_by_its_cells
PHP
tags/chart_heatmap_takes_time_by_its_cells.php 		HTML
tags/chart_heatmap_takes_time_by_its_cells.pad 		Result 
<?php

  // A heatmap's time goes by the cells it has, not by every pair of a row and a column:
  // 20,000 rows of a cell each - each its own row and column - took ten seconds.

  $cells = [];

  for ( $i = 0; $i < 20000; $i++ )
    $cells [] = [ 'r' => "r$i", 'c' => "c$i", 'v' => $i % 7 ];

  $start = hrtime ( TRUE );

?>
 
{chart 'heatmap', data='cells', x='c', y='r', value='v' | strlen/} {echo hrtime(TRUE) - $start lt 3e9 ? 'in time' : 'too slow'/}
