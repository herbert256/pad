App: regression/framework - Page: tags/a_disabled_segmented_and_treepicker_keep_their_value_after_a_failed_post
PHP
tags/a_disabled_segmented_and_treepicker_keep_their_value_after_a_failed_post.php 		HTML
tags/a_disabled_segmented_and_treepicker_keep_their_value_after_a_failed_post.pad 		Result 
<?php

  // The browser leaves a disabled radio group out of the post, and its padLists[], written
  // disabled with it, too: form f came back - the other field to be corrected - with
  // neither, so the locked choices show value= again, where a group the post left out was
  // always taken for one sent without a choice and showed none.

  $_SERVER ['REQUEST_METHOD'] = 'POST';
  $_POST = [ 'padForm' => 'f', 'other' => '' ];

?>

App: regression/framework - Page: tags/a_segmented_and_a_treepicker_in_a_get_form_keep_their_value_on_a_link_with_a_value
PHP
tags/a_segmented_and_a_treepicker_in_a_get_form_keep_their_value_on_a_link_with_a_value.php 		HTML
tags/a_segmented_and_a_treepicker_in_a_get_form_keep_their_value_on_a_link_with_a_value.pad 		Result 
<?php

  // The page of a form that asks by GET, asked with a value of its own - a pager link,
  // ?page&page=2 - and padLists[] naming two of its groups, as the form sends them without a
  // choice: view and cat, not named, show value=, where any value in the query emptied them;
  // emptied and spot, named and not in the query, have nothing chosen.

  $_GET ['page']     = '2';
  $_GET ['padLists'] = [ 'emptied', 'spot' ];

?>
 
{pad:form 'f', method='get'}{input:segmented 'view', options='List, Grid', value='Grid'/}{input:segmented 'emptied', options='S, M', value='M'/}{input:treepicker 'cat', options='A > B, D', value='D'/}{input:treepicker 'spot', options='X, Y', value='Y'/}{/pad:form}

App: regression/framework - Page: tags/a_segmented_refills_the_choice_its_form_posted
PHP
tags/a_segmented_refills_the_choice_its_form_posted.php 		HTML
tags/a_segmented_refills_the_choice_its_form_posted.pad 		Result 
<?php

  // A post of form f: its size shows the choice posted in place of value=, the size of form g,
  // which did not come back, shows value=; tone came back without a choice - a radio group
  // with none is left out of the post, the padLists[] beside it says it was there - so none
  // is chosen, not value= again; and view was posted a value that is no choice of it, which
  // is not shown.

  $_SERVER ['REQUEST_METHOD'] = 'POST';
  $_POST = [ 'padForm' => 'f', 'size' => 'L', 'view' => 'Map', 'padLists' => [ 'size', 'tone', 'view' ] ];

?>
 
{pad:form 'g'}{input:segmented 'size', options='S, M, L', value='S'/}{/pad:form}{pad:form 'f'}{input:segmented 'size', options='S, M, L', value='S'/}{input:segmented 'tone', options='Warm, Cool', value='Cool'/}{input:segmented 'view', options='List, Grid'/}{/pad:form}

App: regression/pages - Page: request/form_rules_segmented
PHP
request/form_rules_segmented.php 		HTML
request/form_rules_segmented.pad 		Result 
<?php

  // The fixture of request/form_rules_segmented_post: the rules of a segmented are those of
  // the one choice it posts, checked before the page's PHP runs, and this file - running after
  // the post pass - sees only whether the post kept them, and what it stores.

  $state = padPosted ( 'pick' )
         ? 'stored ' . ( $_POST ['size'] ?? '' ) . ' as ' . ( $_POST ['view'] ?? '' )
         : 'not stored';

?>
 
{pad:form 'pick', error}
{input:segmented 'size', label='Size', options='S, M, L', rules='required|in:S,M,L'/}
{input:segmented 'view', options='List, Grid', value='Grid'/}
{/pad:form}
{$state}
