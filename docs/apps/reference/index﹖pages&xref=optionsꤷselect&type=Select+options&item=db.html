select/demo/options.pad 		Result 
<?php

  // The options a declared select table takes on the tag itself: each section of the .pad
  // half names one, over the four-row staff table of the demo database.

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name' ];
  $padSelect ['elsewhere'] = [ 'key' => 'name' ];
  $padSelect ['staffJim'] = [ 'db' => 'staff', 'where' => "name = 'jim'" ];
  $padSelect ['staffRaw'] = [ 'db' => 'staff' ];

?>
 
where: [{select:staffSel where="salary >= 2000", order="salary"}{$name} {/select:staffSel}]
order: [{select:staffSel order="salary desc", fields="name"}{$name} {/select:staffSel}]
fields: [{select:staffSel fields="name, bonus", where="name = 'bob'"}{$name}/{$bonus}{/select:staffSel}]
group: [{select:staffSel fields="count(*) as heads", group="floor(salary / 2000)", order="heads"}{$heads} {/select:staffSel}]
having: [{select:staffSel fields="salary + bonus as total", group="total", having="total > 2500", order="total"}{$total} {/select:staffSel}]
rollup: [{select:staffRaw fields="sum(salary) as pay", group="name", rollup}{$pay} {/select:staffRaw}]
distinctrow: [{select:staffSel fields="floor(salary / 2000) as band", distinctrow, order="band"}{$band} {/select:staffSel}]
db: [{select:elsewhere db='staff', where="name = 'jim'"}{$name}/{$phone}{/select:elsewhere}]
union: [{select:staffSel where="name = 'bob'", union='staffJim', order="name"}{$name} {/select:staffSel}]
