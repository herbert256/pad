notations/abc_draws_a_long_line_of_slurs_and_tuplets_in_linear_time.pad 		Result 
<?php

  // A tune pasted by a visitor, a line of 16 KB: every slur looked for its notes among every
  // note of the line, every beamed tuplet counted every item of it, and a chord was named for
  // every character in it that is no note - 16 KB of slurs took four seconds, of tuplets as
  // long, and a chord of 16 000 unreadable characters half a second. The lenient walk: the
  // chord is a problem the strict check names.

  $padCheckSyntax = FALSE;

  $tune = "K:C\n" . str_repeat ( '(AB)', 4000 ) . "\n" . str_repeat ( '(3ABC', 3200 ) . "\n[C" . str_repeat ( '&', 16000 ) . "E]\n";

  $start = hrtime ( TRUE );

?>
 
{notation:abc $tune | strlen/} {pad:echo hrtime(TRUE) - $start lt 2e9 ? 'in time' : 'too slow'/}

App: regression/framework - Page: search/the_excerpts_are_made_for_the_rows_shown
PHP
search/the_excerpts_are_made_for_the_rows_shown.php 		HTML
search/the_excerpts_are_made_for_the_rows_shown.pad 		Result 
<?php

  // 3000 rows a query of short prefixes finds every one of: the excerpts are made for the
  // rows the page can show - the ten of first=10 - and padSearch's first answers the best
  // rows only, where every hit had its excerpt made before first= cut them, and a search
  // cost what its hits cost, not what the page showed: 0.53 seconds for the ten, 0.1 now.

  $syllables = [ 'ka', 'lo', 'mi', 'ter', 'san', 'dor', 'vel', 'pra', 'nu', 'qui', 'zen', 'bar', 'fo', 'lit', 'ron', 'ge' ];
  $seed      = 11;
  $next      = function ( $n ) use ( &$seed ) {
    $seed = ( $seed * 1103515245 + 12345 ) % 2147483648;
    return intdiv ( $seed, 65536 ) % $n;
  };

  $hitRows = [];
  for ( $r = 0; $r < 3000; $r++ ) {
    $body = [];
    for ( $w = 0; $w < 60; $w++ )
      $body [] = $syllables [ $next ( 16 ) ] . $syllables [ $next ( 16 ) ];
    $hitRows [] = [ 'id' => $r, 'body' => implode ( ' ', $body ) ];
  }

  unset ( $body );

  $prefixes = implode ( ' ', array_map ( fn ( $s ) => "$s*", $syllables ) );

  // The time of all hits with their excerpts is the measure: the ten best take a part of it.

  padSearch ( $hitRows, 'kalo', [ 'name' => 'hitRows' ] );

  $start  = hrtime ( TRUE );
  $all    = count ( padSearch ( $hitRows, $prefixes, [ 'name' => 'hitRows' ] ) );
  $allNs  = hrtime ( TRUE ) - $start;

  $start  = hrtime ( TRUE );
  $best   = padSearch ( $hitRows, $prefixes, [ 'name' => 'hitRows', 'first' => 3 ] );
  $shown  = "$all hits, " . count ( $best ) . ' best, ' . ( hrtime ( TRUE ) - $start < $allNs / 2 ? 'in part of the time' : 'in all of it' );

  $hitStart = hrtime ( TRUE );

?>
 
{$shown}|
{search:auto 'hitRows', q=$prefixes, first=10}{pad:if $excerpt ne ''}+{/pad:if}{/search:auto}|
{pad:echo php:hrtime(1) - $hitStart lt $allNs / 2 ? 'in part of the time' : 'in all of it'/}

App: regression/framework - Page: tags/chart_heatmap_takes_time_by_its_cells
PHP
tags/chart_heatmap_takes_time_by_its_cells.php 		HTML
tags/chart_heatmap_takes_time_by_its_cells.pad 		Result 
<?php

  // A heatmap's time goes by the cells it has, not by every pair of a row and a column:
  // 20,000 rows of a cell each - each its own row and column - took ten seconds.

  $cells = [];

  for ( $i = 0; $i < 20000; $i++ )
    $cells [] = [ 'r' => "r$i", 'c' => "c$i", 'v' => $i % 7 ];

  $start = hrtime ( TRUE );

?>
 
{chart:heatmap data='cells', x='c', y='r', value='v' | strlen/} {pad:echo hrtime(TRUE) - $start lt 3e9 ? 'in time' : 'too slow'/}

App: regression/framework - Page: tags/chart_sankey_takes_time_by_its_links

tags/chart_sankey_takes_time_by_its_links.php 		HTML

<?php

  // The flows into a sankey node are looked up by its name, not searched for in every link
  // per node: 10,000 nodes and 20,000 links took six seconds.

  $flows = [];

  for ( $i = 0; $i < 10000; $i++ ) {
    $flows [] = [ 's' => "a$i", 't' => "b$i", 'v' => 1 ];
    $flows [] = [ 's' => "b$i", 't' => 'c' . ( $i % 100 ), 'v' => 1 ];
  }

  $start = hrtime ( TRUE );

?>
 
{chart:sankey data='flows', source='s', target='t', value='v', height=2000 | strlen/} {pad:echo hrtime(TRUE) - $start lt 3e9 ? 'in time' : 'too slow'/}
