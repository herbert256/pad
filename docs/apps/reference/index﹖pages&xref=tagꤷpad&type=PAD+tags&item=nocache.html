<?php

  // Start the example from an empty entry, so it shows a miss and then a hit.

  padFragmentForget ( 'manual-nocache' );

  $built = 0;
  $visit = 1;

?>
 
{pad:cache 'manual-nocache', ttl=60}
  {pad:set $built = $built + 1/}
  <p>Cached: built {$built} time(s), {pad:nocache}live: visit {$visit}{/pad:nocache}</p>
{/pad:cache}

{pad:set $visit = 2/}

{pad:cache 'manual-nocache', ttl=60}
  {pad:set $built = $built + 1/}
  <p>Cached: built {$built} time(s), {pad:nocache}live: visit {$visit}{/pad:nocache}</p>
{/pad:cache}
