App: regression/framework - Page: printables/a_calendar_month_in_the_pdf_is_a_page_of_its_paper
PHP
printables/a_calendar_month_in_the_pdf_is_a_page_of_its_paper.php 		HTML
printables/a_calendar_month_in_the_pdf_is_a_page_of_its_paper.pad 		Result 
<?php

  // In a PDF a month of the calendar is a page of its paper - A4, 595.28 by 841.89 points, or
  // A5 turned - its texts set where the screen has them, in their colours: the month and the
  // red days in #c0392b (0.75 0.22 0.17), the weekdays grey, the other days black; and the
  // lines of the grid, drawn exactly by an SVG on the screen, are drawn by the PDF from the
  // markup, 8 down and 6 across, 0.5 pt.

  $html = padUnprotect ( padUnescape ( padCode ( "{printable:calendar month='2027-04', holidays='nl'/}{printable:calendar month='2027-02', paper='a5', landscape/}" ) ) );
  $pdf  = padPdf ( $html, [ 'compress' => FALSE ] );

  preg_match_all ( '#/MediaBox \[0 0 ([0-9.]+) ([0-9.]+)\]#', $pdf, $boxes, PREG_SET_ORDER );
  preg_match_all ( '#stream\n(.*?)\nendstream#s', $pdf, $streams );

  $first = $streams [1] [0] ?? '';

  preg_match_all ( '/BT ([0-9.]+ [0-9.]+ [0-9.]+) rg \/F\d+ ([0-9.]+) Tf [0-9.]+ [0-9.]+ Td \(((?:[^()\\\\]|\\\\.)*)\) Tj ET/', $first, $texts, PREG_SET_ORDER );

  $red = array_map ( fn ( $t ) => $t [3], array_filter ( $texts, fn ( $t ) => $t [1] == '0.75 0.22 0.17' ) );

  $answer = count ( $boxes ) . ' pages: ' . implode ( ', ', array_map ( fn ( $b ) => "{$b [1]} by {$b [2]}", $boxes ) )
          . ' // ' . count ( $texts ) . ' texts, red: ' . implode ( ' ', $red )
          . ' // ' . preg_match_all ( '/q [0-9. ]+ RG 0\.5 w [0-9. ]+ m [0-9. ]+ l S Q/', $first ) . ' lines of 0.5 pt';

?>
 
{!answer}

App: regression/framework - Page: printables/a_calendar_of_a_year_takes_its_items_in_turn_a_month_each
PHP
printables/a_calendar_of_a_year_takes_its_items_in_turn_a_month_each.php 		HTML
printables/a_calendar_of_a_year_takes_its_items_in_turn_a_month_each.pad 		Result 
<?php

  // year= makes the twelve months of the year, a sheet each: the items of the data are taken in
  // turn - three rows, Ann, Bob and Cy, over January, February, March, then again - and a
  // single tag gives every month the same empty top. The rows a month needs: February 2027
  // begins on a Monday and has four, May and August 2027 six (cal 2027). copies= makes every
  // month that many times, in a row.

  $rows = [ [ 'name' => 'Ann' ], [ 'name' => 'Bob' ], [ 'name' => 'Cy' ] ];

  function calendarYear ( $template ) {

    $out = padUnprotect ( padUnescape ( padCode ( $template ) ) );

    preg_match_all ( '/<section class="pp-sheet".*?<\/section>/s', $out, $sheets );

    return implode ( ', ', array_map ( function ( $sheet ) {
      preg_match ( '/size=12;align=center;valign=middle"><div class="pp-in">(.*?)<\/div>/s', $sheet, $top );
      preg_match ( '/size=22[^"]*"><div class="pp-in"><b>(.*?)<\/b>/', $sheet, $month );
      return trim ( $month [1] ?? '?' ) . ( isset ( $top [1] ) ? ' ' . trim ( $top [1] ) : '' ) . ' ' . ( preg_match_all ( '/data-pdf-line="[0-9.]+ ([0-9.]+) [0-9.]+ \1"/', $sheet ) - 1 ) . ' rows';
    }, $sheets [0] ) );

  }

  $answer = calendarYear ( "{printable:calendar year=2027, data='rows'}{\$name}{/printable:calendar}" )
          . ' // ' . calendarYear ( "{printable:calendar month='2027-05', months=2, copies=2/}" );

?>
 
{!answer}

App: regression/framework - Page: printables/a_calendar_of_this_year_outside_the_known_years_marks_no_day_of_them
PHP
printables/a_calendar_of_this_year_outside_the_known_years_marks_no_day_of_them.php 		HTML
printables/a_calendar_of_this_year_outside_the_known_years_marks_no_day_of_them.pad 		Result 
<?php

  // A wall calendar no year= or month= fixes shows this year - a year the template does not
  // choose, so outside the years its country's file knows it is no error, as a month the
  // visitor asks {pad:calendar} for is none: only the days inside those years are marked. New
  // Zealand's holidays are known for 2010 to 2052 (Matariki's schedule): with the clock at
  // 2053 no day of the year is marked - the rules' rows marked King's Birthday and Labour Day
  // and left Matariki out - and at 2052 they are all there. A year= or month= outside them is
  // the strict check's error (the errors suite).

  $names = fn ( $show ) => implode ( ', ', array_unique ( preg_match_all ( "/Waitangi Day|Anzac Day|King's Birthday|Matariki|Labour Day|Christmas Day/", $show, $m ) ? $m [0] : [] ) ) ?: 'none';

  padNowFreeze ( '2053-03-01 12:00:00' );

  $after = $names ( calendarShow ( "{printable:calendar holidays='nz'/}" ) );

  padNowFreeze ( '2052-03-01 12:00:00' );

  $last = $names ( calendarShow ( "{printable:calendar holidays='nz'/}" ) );

  padNowFreeze ();

  $answer = "2053: $after // 2052: $last";

?>
 
{!answer}
