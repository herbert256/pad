options/tobool_and_tocontent_on_one_tag_read_the_same_text.pad 		Result 
<?php

  // toBool and toContent - or toData - on one tag, in either order: each reads the tag's
  // text, where the first left nothing for the next - a flag of FALSE, an empty store.

  $xs = [ [ 'n' => 1 ], [ 'n' => 2 ] ];

?>
 
{xs toContent='c1', toBool='b1'}{$n}{/xs}{if bool='b1'}Y{else}N{/if}[{content:c1}] {xs toBool='b2', toContent='c2'}{$n}{/xs}{if bool='b2'}Y{else}N{/if}[{content:c2}] {xs toData='d3', toBool='b3'}{$n}{/xs}{if bool='b3'}Y{else}N{/if}
