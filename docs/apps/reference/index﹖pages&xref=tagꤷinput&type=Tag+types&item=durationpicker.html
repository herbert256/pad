App: regression/framework - Page: tags/a_durationpicker_refills_the_seconds_its_form_posted
PHP
tags/a_durationpicker_refills_the_seconds_its_form_posted.php 		HTML
tags/a_durationpicker_refills_the_seconds_its_form_posted.pad 		Result 
<?php

  // A post of form f: its length shows the seconds posted in place of value=, the length of
  // form g, which did not come back, shows value=, and a lap posted as text that is no number
  // of seconds is shown as it came - the number field marks it, and the page's rules say so.
  // The page's padValidate left a message for the length: it is tied to the field, and the
  // field's unit is said after it.

  $_SERVER ['REQUEST_METHOD'] = 'POST';
  $_POST = [ 'padForm' => 'f', 'length' => '5400', 'lap' => '1:30' ];

  padValidate ( [ 'length' => 'integer|max:3600' ] );

?>
 
{pad:form 'g'}{input:durationpicker 'length', value=60/}{/pad:form}{pad:form 'f'}{input:durationpicker 'length', value=60/}{input:durationpicker 'lap', parts='ms'/}{/pad:form}

App: regression/pages - Page: request/form_rules_durationpicker
PHP
request/form_rules_durationpicker.php 		HTML
request/form_rules_durationpicker.pad 		Result 
<?php

  // The fixture of request/form_rules_durationpicker_post: the rules of a duration picker are
  // those of the number of seconds it posts, checked before the page's PHP runs, and this
  // file - running after the post pass - sees only whether the post kept them, and what it
  // stores.

  $state = padPosted ( 'time' ) ? 'stored ' . ( $_POST ['length'] ?? '' ) . ' seconds' : 'not stored';

?>
 
{pad:form 'time', error}
{input:durationpicker 'length', label='Length', parts='hm', rules='required|integer|between:600,28800'/}
{/pad:form}
{$state}
