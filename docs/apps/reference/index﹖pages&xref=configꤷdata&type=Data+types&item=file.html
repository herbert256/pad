<main class="page home">

  <section class="hero">
    <div class="hero-text">
      <span class="kicker">PAD · {ignore}{chart}{/ignore}</span>

      <p class="lead">One tag turns rows of JSON, YAML, CSV or PHP into an accessible SVG chart -
      lines, areas, streams, bars, funnels, pyramids, waterfalls, pies, waffles, marimekkos,
      parliaments, venns, gauges, radars, roses, bullets, histograms, violins, scatters, heatmaps,
      candlesticks, spirals, sankeys, chords, networks, treemaps, packs, org charts and gantt
      charts. No JavaScript, so it prints, caches and shows up in an e-mail.</p>
      <ul class="stats">

        <li><b>{$exampleCount}</b> examples</li>
        <li><b>0</b> lines of JavaScript</li>

      {page 'sankey/energy'}
    </a>
  </section>

  {families}
  <section class="family" id="{$family}">
