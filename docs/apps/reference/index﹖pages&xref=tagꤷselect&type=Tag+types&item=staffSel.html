App: regression/pages - Page: db/a_negative_number_after_a_hyphen_stays_as_written_outside_sql_code
PHP
db/a_negative_number_after_a_hyphen_stays_as_written_outside_sql_code.php 		HTML
db/a_negative_number_after_a_hyphen_stays_as_written_outside_sql_code.pad 		Result 
<?php

  // A negative number spliced right after a minus gets a space before it in the SQL a tag
  // writes, outside its quoted literals, where 2000--1000 is a line comment on SQLite - and
  // nowhere else: the page's text, an attribute, an echo and a literal keep the value as it is,
  // a literal in the fields= or join= of a select too, where no other value may stand.

  $lo  = -12;
  $hi  = -4;
  $id  = -1;
  $adj = -1000;

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name' ];

?>
 
text:    [{$lo}-{$hi}] [row-{$id}] <i id="item-{$id}"></i>
echo:    [{echo 'UTC-{$id}'/}]
literal: [{field "concat('X-{$id}')"/}]
code:    [{field "2000-{$adj}"/}]
fields:  [{staffSel fields="name, concat('X-{$id}') as code", rows=1}{$code}{/staffSel}]
join:    [{staffSel join="staff as b on b.name = staff.name and concat('X-{$id}') = 'X--1'", fields="staff.name", rows=1}{$name}{/staffSel}]

App: regression/pages - Page: select/demo/a_line_comment_ends_its_own_clause
PHP
select/demo/a_line_comment_ends_its_own_clause.php 		HTML
select/demo/a_line_comment_ends_its_own_clause.pad 		Result 
<?php

  // A -- comment written in a clause - where=, group=, having=, order= - ends that clause:
  // the clauses of the statement stand on lines of their own. On one line the comment ran to
  // its end, so the having, the order and the limit after it were gone without a word, and
  // the closing parenthesis of the where with them. Over the staff table of the demo database.

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name' ];

?>
 
group: [{staffSel fields="floor(salary / 2000) as band, count(*) as n", group="band -- per band", having="n > 1", order="n desc", rows=1}{$band}={$n} {/staffSel}]
having: [{staffSel fields="floor(salary / 2000) as band, count(*) as n", group="band", having="n > 1 -- the shared ones", order="n desc"}{$band}={$n} {/staffSel}]
rollup: [{staffSel fields="floor(salary / 2000) as band, sum(salary) as pay", group="band -- per band", rollup}{$pay} {/staffSel}]
where: [{staffSel where="salary > 2000 -- the well paid", order="name"}{$name} {/staffSel}]
order: [{staffSel order="salary desc -- best paid first", rows=2}{$name} {/staffSel}]
pages: [{staffSel fields="floor(salary / 2000) as band, count(*) as n", group="band", having="n > 0 -- every band", page=1, rows=2}{$band} {/staffSel}] ({pager 'staffSel', window=0}{$kind}:{$page} {/pager})

App: regression/pages - Page: select/demo/a_subquery_or_a_window_in_fields_keeps_the_declared_order
PHP
select/demo/a_subquery_or_a_window_in_fields_keeps_the_declared_order.php 		HTML
select/demo/a_subquery_or_a_window_in_fields_keeps_the_declared_order.pad 		Result 
<?php

  // An aggregate in a subquery or over a window is a value of each row, no total: the select
  // keeps the order the table declares, and rows= the first page of it. Beside them a total
  // still drops that order - under ONLY_FULL_GROUP_BY, MySQL's default, it is an error there -
  // and so does one whose alias starts with over, which is no window.

  db ( "set session sql_mode = concat(@@sql_mode, ',ONLY_FULL_GROUP_BY')" );

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name' ];

?>
 
subquery: [{staffSel fields="name, (select max(s.salary) from staff as s where s.salary < staff.salary) as below", rows=2}{$name}:{$below} {/staffSel}]
window: [{staffSel fields="name, sum(bonus) over (order by salary desc) as running", rows=3}{$name}:{$running} {/staffSel}]
total: [{staffSel fields="count(*) as n, (select max(salary) from staff) as top"}{$n}:{$top}{/staffSel}]
alias: [{staffSel fields="count(*) overall"}{$overall}{/staffSel}]

App: regression/pages - Page: select/demo/a_total_is_no_row_the_declared_order_sorts
PHP
select/demo/a_total_is_no_row_the_declared_order_sorts.php 		HTML
select/demo/a_total_is_no_row_the_declared_order_sorts.pad 		Result 
<?php

  // The order a table declares - or its key - sorts the rows of the table, not the rows a
  // select makes of them: a rollup, a total over the whole table, one value. Under
  // ONLY_FULL_GROUP_BY, as MySQL has it by default, an order by a column the total does not
  // hold is an error; with a rollup it is one on MariaDB as well. A view that groups keeps an
  // order of its own. Over the four-row staff table of the demo database.

  db ( "set session sql_mode = concat(@@sql_mode, ',ONLY_FULL_GROUP_BY')" );

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name' ];
  $padSelect ['staffKey'] = [ 'db' => 'staff', 'key' => 'name' ];

  $padSelect ['perBand']  = [ 'base' => 'staffSel', 'fields' => 'floor(salary / 2000) as band, sum(bonus) as bonus',
                              'group' => 'band', 'rollup' => TRUE ];
  $padSelect ['bandDesc'] = [ 'base' => 'staffSel', 'fields' => 'floor(salary / 2000) as band, count(*) as n',
                              'group' => 'band', 'order' => 'band desc' ];

?>
 
rollup: [{staffSel fields="floor(salary / 2000) as band, sum(salary) as pay", group="band", rollup}{$pay} {/staffSel}]
key rollup: [{staffKey fields="floor(salary / 2000) as band, sum(salary) as pay", group="band", rollup}{$pay} {/staffKey}]
count: [{staffSel fields="count(*) as n"}{$n}{/staffSel}]
field: [{staffKey type='field', fields="max(salary)"/}]
view: [{perBand}{$bonus} {/perBand}]
view order: [{bandDesc}{$band}:{$n} {/bandDesc}]
tag order: [{staffSel fields="floor(salary / 2000) as band, count(*) as n", group="band", order="n desc, band"}{$band}:{$n} {/staffSel}]
rows: [{staffSel order="salary desc", rows=2}{$name} {/staffSel}]

App: regression/pages - Page: select/demo/a_union_is_no_row_the_declared_order_sorts
PHP
select/demo/a_union_is_no_row_the_declared_order_sorts.php 		HTML
select/demo/a_union_is_no_row_the_declared_order_sorts.pad 		Result 
<?php

  // A union makes rows of its own, as a group does: the declared order and the key are the
  // order of the table's rows, and after the union only its own columns can be named. A
  // select with union= - on the tag, declared, or a view of a table with an order - is
  // sorted by an order= written with it, or not at all. Over the staff and offices tables of
  // the demo database; sort= puts the answers in a fixed order.

  $padSelect ['staffSel']   = [ 'db' => 'staff',   'key' => 'name', 'order' => 'name' ];
  $padSelect ['cityPhones'] = [ 'db' => 'offices', 'fields' => 'phone' ];
  $padSelect ['phones']     = [ 'db' => 'staff',   'key' => 'name', 'fields' => 'phone', 'union' => 'cityPhones' ];
  $padSelect ['allPhones']  = [ 'base' => 'staffSel', 'fields' => 'phone', 'union' => 'cityPhones' ];

?>
 
tag: [{staffSel fields="phone", union="cityPhones", sort="phone"}{$phone} {/staffSel}]
key: [{phones sort="phone"}{$phone} {/phones}]
view: [{allPhones sort="phone"}{$phone} {/allPhones}]
order: [{allPhones order="phone desc", rows=3}{$phone} {/allPhones}]

App: regression/pages - Page: select/demo/paging
PHP
select/demo/paging.php 		HTML
select/demo/paging.pad 		Result 
<?php

  // Paging a declared select table: the page is cut once, by the SQL limit, so page 2 holds
  // the third and fourth rows; a row count in the declaration is a row count, not the
  // query's command word; and a joined table declared without fields joins, adding none.
  // Over the four-row staff table and the join tables of the demo database.

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name' ];
  $padSelect ['staffTwo'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name', 'rows' => 2 ];

  $padSelect ['bare']     = [ 'db' => 'table2', 'key' => 'key' ];
  $padSelect ['joined']   = [ 'db' => 'table1', 'fields' => 'table1.text', 'order' => 'table1.key',
                              'join' => [ 'inner' => 'bare', 'key' => 'table1.key' ] ];

?>
 
page 1: [{staffSel page=1, rows=2}{$name} {/staffSel}]
page 2: [{staffSel page=2, rows=2}{$name} {/staffSel}]
declared rows: [{staffTwo}{$name} {/staffTwo}]
join without fields: [{joined}{$text} {/joined}]

App: regression/pages - Page: select/demo/rows_0_on_a_select_keeps_no_row
PHP
select/demo/rows_0_on_a_select_keeps_no_row.php 		HTML
select/demo/rows_0_on_a_select_keeps_no_row.pad 		Result 
<?php

  // rows=0 on the tag of a select - or a value there that is no number - keeps no row, as
  // it does on every other data, and a declaration's 'rows' => 0 is no limit: since page=
  // and rows= make the limit when they are written, both were read as the 10 of a rows=
  // not written. A page= beside the declared 0 is a page of 10. Over the staff and the 122
  // customers of the demo database.

  $padSelect ['staffSel']     = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name' ];
  $padSelect ['allCustomers'] = [ 'db' => 'customers', 'key' => 'customerNumber', 'rows' => 0 ];

  $none  = '';
  $names = [ 'ann', 'bea' ];

?>
 
tag: [{staffSel rows=0}{$name} @else@none{/staffSel}]
value: [{staffSel rows=$none}{$name} @else@none{/staffSel}]
list: [{names rows=0}{$names} @else@none{/names}]
declared: [{allCustomers}{last@allCustomers}{count@allCustomers/}{/last@allCustomers}{/allCustomers}]
paged: [{allCustomers page=2}{last@allCustomers}{count@allCustomers/}{/last@allCustomers}{/allCustomers}]
