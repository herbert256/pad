App: regression/framework - Page: printables/a_folded_card_in_the_pdf_leaves_its_guides_out_and_turns_its_inside
PHP
printables/a_folded_card_in_the_pdf_leaves_its_guides_out_and_turns_its_inside.php 		HTML
printables/a_folded_card_in_the_pdf_leaves_its_guides_out_and_turns_its_inside.pad 		Result 
<?php

  // In the PDF a folded card is its content alone: the folds and the names of the panels are
  // the screen's (data-pdf-skip), and so not in the document - its four texts, no line - and
  // the two panels of the inside are turned half round their middles (-1 0 0 -1), the front
  // and the back not. With folds the folds are printed too, two light lines of 0.3 pt, which
  // end up on the edges of the card.

  function foldcardPdf ( $template ) {

    $pdf = padPdf ( padUnprotect ( padUnescape ( padCode ( $template ) ) ), [ 'compress' => FALSE ] );

    preg_match_all ( '/\((Front|Back|Left|Right|Inside left|Inside right)\) Tj/', $pdf, $texts );
    preg_match_all ( '/re W n -1 0 0 -1/', $pdf, $turns );
    preg_match_all ( '/q ([0-9. ]+) RG ([0-9.]+) w ([0-9. ]+) m ([0-9. ]+) l S Q/', $pdf, $lines, PREG_SET_ORDER );

    return implode ( ', ', $texts [1] ) . ' // ' . count ( $turns [0] ) . ' turned // '
         . ( $lines ? implode ( ', ', array_map ( fn ( $l ) => "line {$l [3]} to {$l [4]} of {$l [2]} pt", $lines ) ) : 'no lines' );

  }

  $card   = '<header>Front</header><aside>Left</aside><p>Right</p><footer>Back</footer>';
  $answer = foldcardPdf ( "{printable:foldcard}$card{/printable:foldcard}" ) . ' ## ' . foldcardPdf ( "{printable:foldcard folds}$card{/printable:foldcard}" );

?>
 
{!answer}

App: regression/framework - Page: printables/a_folded_card_puts_its_parts_on_the_panels_of_a_quarter_fold
PHP
printables/a_folded_card_puts_its_parts_on_the_panels_of_a_quarter_fold.php 		HTML
printables/a_folded_card_puts_its_parts_on_the_panels_of_a_quarter_fold.pad 		Result 
<?php

  // A folded card on A4, a quarter fold: the top half folded back along 148.5 mm, then the
  // sheet folded in two along 105 mm, a card of A6. Folded, the lower right quarter is the
  // front - the <header> - and the lower left the back - the <footer>; the upper half lies
  // inside, upside down on the sheet so that it reads the right way up in the card: the upper
  // right quarter - behind the front - is the inside left, the <aside>, and the upper left
  // the inside right, the rest of the item. The folds and the names of the panels are the
  // screen's alone. Letter turned: a card of 139.7 by 107.95 mm, the same quarters.

  $answer = printablesShow ( '{printable:foldcard}<header>Front</header><aside>Left</aside><p>Right</p><footer>Back</footer>{/printable:foldcard}', 10 )
          . ' ## ' . printablesShow ( "{printable:foldcard paper='letter', landscape}<header>Front</header><p>Right</p>{/printable:foldcard}", 8 );

?>
 
{!answer}

<?php

  // A printable, by the name in the address - ?labels, /printables/labels: one of the engine's
  // kinds, else a page not found. Its template - kinds/<kind>.pad, sample data from {fake:rows} with
  // fixed seeds - is shown beside what it makes, coloured, and run; its pdf option links the
  // same sheets as a PDF.

  $one = array_values ( array_filter ( $printables, fn ( $item ) => $item ['kind'] === $kind ) ) [0] ?? NULL;

  if ( ! $one )
    padAbort ( 404 );

  $title    = $one ['name'];
  $text     = $one ['text'];
  $template = printablesTemplate ( $kind );
  $source   = padHighlightTokens ( $template, 'pad' );

  // The icons at the top right of the template that take it to the WASM editor and the live
  // application (_common/_lib/playground.php).

  include_once COMMON . '_lib/playground.php';

  $play     = padPlayLink ( "$template\n" );

?>
 
<p class="pr-lead">{$text}</p>

  </section>
