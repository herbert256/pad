tags/a_field_inside_the_title_or_label_of_more_drawing_tags_is_escaped_once.pad 		Result 
<?php

  // A field written inside the quoted title, label, alt, button, past= or tab name of these
  // tags is escaped once, as the field alone is: it arrived escaped by its sanitize step and
  // the tag escaped it again - Q&A Å was shown and read out as Q&amp;A &Aring;. Each line
  // counts the texts written right; a tab named by the field is the one active= and open=
  // name with it.

  $t = 'Q&A Å';

?>
 
sudoku {sudoku '53..7....6..195....98....6.8...6...34..8.3..17...2...6.6....28....419..5....8..79', title='{$t}' | php:substr_count(@, 'Q&amp;A Å')};

crossword {crossword title='{$t}'}
  ECHO: Writes a value
  HELLO: The first page
{/crossword | php:substr_count(@, 'Q&amp;A Å')};

App: regression/framework - Page: tags/widget_styles_carry_the_csp_nonce
PHP
tags/widget_styles_carry_the_csp_nonce.php 		HTML
tags/widget_styles_carry_the_csp_nonce.pad 		Result 
<?php

  // Under a policy whose style-src asks for a nonce, the <style> of {tabs}, {accordion},
  // {modal} and {copy} carries it as {img} and {video} do: without their CSS a tab set showed
  // every panel and bare radio buttons, a modal its button and its no-script link both.

  $padCsp = "default-src 'self'; script-src 'nonce'; style-src 'nonce'";

?>
 
{copy 'x'}{modal 'm', title='T'}body{/modal}{tabs}{tab 'a'}x{/tab}{/tabs}{accordion}{tab 'q'}a{/tab}{/accordion}

<?php

  $plans = [
    [ 'name' => 'Starter', 'price' => '€ 9',  'about' => 'for one person and a side project',
      'features' => [ 'One site', '10 GB storage', 'E-mail support' ] ],
    [ 'name' => 'Team',    'price' => '€ 29', 'about' => 'for a team that ships every week',
      'features' => [ 'Ten sites', '100 GB storage', 'Chat support', 'Daily backups' ] ],
    [ 'name' => 'Studio',  'price' => '€ 79', 'about' => 'for an agency with many clients',

?>

  {tabs}
    {plans}
      {tab $name}
        <p><strong>{$price}</strong> a month - {$about}</p>
        <ul>{features}<li>{$features}</li>{/features}</ul>
      {/tab}
    {/plans}
  {/tabs}
</div>

App: tags - Page: examples/tab/1-conditional

 HTML
examples/tab/1-conditional.pad 		Result 
{meta title='A tab in an if', about='A tab only shows up when its condition holds - the hand-over skips the if around it.'}
{set $hasReviews = 0}
{tabs}
  {tab 'Overview'}<p>A kettle in brushed steel.</p>{/tab}
  {if $hasReviews gt 0}
    {tab 'Reviews'}<p>The reviews.</p>{/tab}
  {/if}
  {tab 'Delivery'}<p>Delivered the next working day.</p>{/tab}
{/tabs}
