examples/macro/2-loop.php 		HTML

<?php

  $macroProducts = [
    [ 'product' => 'Kettle',  'price' => 39.5 ],
    [ 'product' => 'Toaster', 'price' => 24   ],
    [ 'product' => 'Blender', 'price' => 79.9 ] ];

?>
 
{meta title='Used inside a loop', about='The macro\'s own parameters are its fields, and the row round the use stays visible behind them.'}
{macro 'macroPrice', amount}
  <li>{$product}: € {$amount | number_format(@, 2)}</li>
{/macro}

<ul>
{macroProducts}
  {macroPrice $price}
{/macroProducts}
</ul>
