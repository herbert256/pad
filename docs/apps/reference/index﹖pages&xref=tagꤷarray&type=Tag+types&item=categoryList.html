Reference - Tag types - categoryList

App: ledger - Page: transactions
PHP
transactions.php 		HTML
transactions.pad 		Result 
<?php

  // A month's transactions as one table - sorted and paged on the server by {datatable} -
  // narrowed to a category, an account or a word in the note. The same rows answer as CSV
  // (?transactions&padFormat=csv) and JSON: $padExpose names them.

  $only     = (int) padRequest ( 'category', 0 );
  $from     = (int) padRequest ( 'account', 0 );
  $word     = trim ( ledgerText ( 'q' ) );

  $rows     = array_map ( fn ( $r ) => array_diff_key ( $r, [ 'id' => 1, 'kind' => 1 ] ), ledgerRows ( $ledgerMonth, $only, $from, $word ) );
  $shown    = count ( $rows );
  $sum      = round ( array_sum ( array_column ( $rows, 'amount' ) ), 2 );

  $categoryList = array_map ( fn ( $c ) => $c + [ 'chosen' => $c ['id'] == $only ], ledgerCategories () );
  $accountList  = array_map ( fn ( $a ) => $a + [ 'chosen' => $a ['id'] == $from ], ledgerAccounts () );
  $monthName    = ledgerMonthName ( $ledgerMonth );
  $csvName      = "ledger-$ledgerMonth.csv";
  $csvQuery     = http_build_query ( [ 'month' => $ledgerMonth, 'category' => $only, 'account' => $from, 'q' => $word, 'padFormat' => 'csv' ] );

  $padExpose  = [ 'rows' ];
  $ledgerPage = "Transactions, $monthName";

?>
 
<div class="head">
  <h1>Transactions <span class="dim">{$monthName}</span></h1>
  <a class="button" href="{$padGo}transactions&{$csvQuery}" download="{$csvName}">⬇ CSV</a>
</div>

<form class="filters" method="get">
  <input type="hidden" name="transactions" value="">
  <input type="hidden" name="month" value="{$ledgerMonth}">
  <label>Category
    <select name="category"><option value="0">All</option>{categoryList}<option value="{$id}"{if $chosen} selected{/if}>{$name}</option>{/categoryList}</select>
  </label>
  <label>Account
    <select name="account"><option value="0">All</option>{accountList}<option value="{$id}"{if $chosen} selected{/if}>{$name}</option>{/accountList}</select>
  </label>
  <label>Note <input type="search" name="q" value="{$word}" placeholder="a word"></label>
  <button class="button">Show</button>
</form>

<section class="card table">
  {if $shown gt 0}
  {datatable data='rows', columns='day, category, account, note, amount', format="amount:money, day:date('D j M')", totals='amount', sortable, rows=15, striped}
  {else}
  <p class="dim">No transactions here.</p>
  {/if}
  <p class="dim sum">{$shown} transactions, together {$sum | money}.</p>
</section>
