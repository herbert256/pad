App: manual - Page: fragments/notify_3

fragments/notify_3.php 		HTML
fragments/notify_3.pad 		Result 
<?php

  // The webhook channel posts the notification as JSON, signed as Standard Webhooks has it. A
  // fake answers here and keeps what was sent; below it the receiver's side checks the
  // signature as any receiver does - the HMAC-SHA256 of id.timestamp.body under the secret.
  // In an application the address and the secret stand in .env: WEBHOOK_URL, WEBHOOK_SECRET.

  padCurlFake ( [ '*' => 204 ] );

  $signingKey = 'whsec_MfKQ9r8GKYqrTwjUPD8ILPZIo2LhLjHd';

  padNotify ( [ 'id' => 'manual-ann', 'email' => 'ann@example.com' ], 'order-shipped',
              [ 'order' => 42, 'name' => 'Ann', 'url' => 'https://shop.example/orders/42' ],
              [ 'webhook' ], [ 'webhook' => [ 'url' => 'https://hooks.example.com/orders', 'secret' => $signingKey ] ] );

  $sent = padCurlRecorded () [0];
  $head = $sent ['headers'];
  $body = json_decode ( $sent ['post'], TRUE );

  // The receiver: the signature made again from the headers and the body as they came.

  $signed = $head ['webhook-id'] . '.' . $head ['webhook-timestamp'] . '.' . $sent ['post'];
  $mac    = 'v1,' . base64_encode ( hash_hmac ( 'sha256', $signed, base64_decode ( substr ( $signingKey, 6 ) ), TRUE ) );
  $valid  = hash_equals ( $mac, $head ['webhook-signature'] ) ? 'yes' : 'no';

  $type  = $body ['type'];
  $title = $body ['data'] ['title'];

  padCurlFakeStop ();

?>
 
<p>The type: <code>{$type}</code>, the title: {$title}</p>
<p>The signature checks out: {$valid}</p>

<?php

  // The webhook channel's address is its credential - a catch hook of Zapier, Make or n8n
  // runs for anyone holding it - so the dumps and the error report (padRedact, $padCurlLast
  // in them) and the address _events/curl.php hears show it redacted, a service of the
  // application's own too: the address the channel posted to is redacted by value for the
  // rest of the request, from its path on - whole when its path is shorter than eight
  // characters. Zapier's, Make's and n8n's are known by their form in any text, and a
  // webhook's address under its usual name, WEBHOOK_URL, by the name. The fetch itself goes
  // to the address as it is.

  $GLOBALS ['heardCurlUrls'] = [];

  $padCurlLast = NULL;

  padCurlFake ( [ '*' => 204 ] );

  $_ENV ['WEBHOOK_SECRET'] = 'whsec_' . base64_encode ( str_repeat ( "\x01", 24 ) );

  $data = [ 'order' => 42, 'items' => 1, 'name' => 'Ann', 'url' => '' ];

  padNotify ( [ 'id' => 7 ], 'shipped', $data, [ 'webhook' ], [ 'webhook' => [ 'url' => 'https://automate.example.org/in/7f3c2a91e5/order?src=pad' ] ] );

  $last = $padCurlLast ['url'] ?? '';

  padNotify ( [ 'id' => 8 ], 'shipped', $data, [ 'webhook' ], [ 'webhook' => [ 'url' => 'https://short.example.net/x1' ] ] );

  $report = [
    'fetched'  => array_column ( padCurlRecorded (), 'url' ),
    'the hook' => $GLOBALS ['heardCurlUrls'],
    'a report' => padRedact ( [
      'last'        => $last,
      'text'        => 'The webhook did not answer (https://automate.example.org/in/7f3c2a91e5/order?src=pad)',
      'short'       => 'POST https://short.example.net/x1',
      'zapier'      => 'https://hooks.zapier.com/hooks/catch/1234567/abcdef1/',
      'make'        => 'https://hook.eu1.make.com/k3j4h5g6f7d8s9a0q1w2e3r4t5y6u7i8',
      'n8n'         => 'https://n8n.example.com/webhook/0f9c1d6e-5a7b-4c3d-9e8f-1a2b3c4d5e6f',
      'n8n test'    => 'https://n8n.example.com/webhook-test/orders/new?x=1',
      'WEBHOOK_URL' => 'https://hooks.example.net/pad',
      'plain'       => 'https://api.example.com/v1/items?page=2' ] ) ];

  $json = json_encode ( $report, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES );

?>
 
{$json}

App: regression/framework - Page: channels/webhook_signs_the_notification_as_standard_webhooks_and_keeps_its_id_for_a_retry
PHP
channels/webhook_signs_the_notification_as_standard_webhooks_and_keeps_its_id_for_a_retry.php 		HTML
channels/webhook_signs_the_notification_as_standard_webhooks_and_keeps_its_id_for_a_retry.pad 		Result 
<?php

  // The webhook channel, signed as Standard Webhooks has it. The signature is first checked on
  // the worked example of Svix's "verifying manually" page - its secret, id, timestamp and
  // payload give its v1 signature. Then a notification is posted to the configured address,
  // the clock frozen: the spec's body (type, timestamp, data), the webhook-id of the
  // notification, the timestamp of the attempt and a v1 signature per secret of a rotation,
  // each verified here as a receiver does it. Queued, the first attempt answered 500, the
  // retry carries the same webhook-id - a receiver drops a second delivery by it - and a 410
  // is said to be the spec's Gone.

  include_once PAD . 'subs/channels/_lib/webhook.php';

  $report ['svix example'] = padWebhookSignature ( 'msg_loFOjxBNrRLzqYUf', '1731705121', '{"event_type":"ping","data":{"success":true}}',
                                                   [ base64_decode ( 'plJ3nmyCDGBKInavdOK15jsl' ) ] );

  $old = 'whsec_' . base64_encode ( str_repeat ( "\x01", 24 ) );
  $new = 'whsec_' . base64_encode ( str_repeat ( "\x02", 32 ) );

  $_ENV ['WEBHOOK_URL']    = 'https://hooks.example.com/pad';
  $_ENV ['WEBHOOK_SECRET'] = "$new $old";

  padNowFreeze ( '2026-11-01 12:00:00 UTC' );

  $verify = function ( $call, $secret ) {
    $key  = base64_decode ( substr ( $secret, 6 ) );
    $want = 'v1,' . base64_encode ( hash_hmac ( 'sha256', $call ['headers'] ['webhook-id'] . '.' . $call ['headers'] ['webhook-timestamp'] . '.' . $call ['post'], $key, TRUE ) );
    return in_array ( $want, explode ( ' ', $call ['headers'] ['webhook-signature'] ), TRUE );
  };

  padCurlFake ( [ 'https://hooks.example.com/*' => 204 ] );

  padNotify ( [ 'id' => 7, 'email' => 'ann@example.com', 'phone' => '+31612345678' ], 'shipped',
              [ 'order' => 42, 'items' => 3, 'name' => 'Ann', 'url' => 'https://shop.example/orders/42' ], [ 'webhook' ] );

  $call = padCurlRecorded () [0];

  $report ['sent'] = [ 'call'       => $call ['method'] . ' ' . $call ['url'],
                       'type'       => $call ['headers'] ['Content-Type'],
                       'id'         => (bool) preg_match ( '/^msg_[0-9a-f]{24}$/D', $call ['headers'] ['webhook-id'] ),
                       'timestamp'  => $call ['headers'] ['webhook-timestamp'],
                       'signatures' => count ( explode ( ' ', $call ['headers'] ['webhook-signature'] ) ),
                       'new secret' => $verify ( $call, $new ),
                       'old secret' => $verify ( $call, $old ),
                       'body'       => json_decode ( $call ['post'], TRUE ),
                       'results'    => array_map ( fn ( $one ) => $one ['ok'] ? 'sent' : $one ['error'], padNotifyResults () ) ];

  // Queued: the first attempt fails, the retry is the same message.

  padQueueFlush ( 'notifyhook' );

  $tries = 0;

  padCurlFake ( [ 'https://hooks.example.com/*' => function () use ( &$tries ) {
    return ++$tries == 1 ? 500 : 200;
  } ] );

  padNotify ( [ 'id' => 8 ], 'shipped', [ 'order' => 43, 'items' => 1, 'name' => 'Bo', 'url' => '' ], [ 'webhook' ], [ 'queue' => 'notifyhook' ] );

  $first = padQueueWork ( 'notifyhook', 0, 60, 3 );

  foreach ( glob ( DATA . 'queue/regression/framework/notifyhook/*.job' ) as $file ) {
    $record = json_decode ( file_get_contents ( $file ), TRUE );
    $record ['available'] = time () - 1;
    file_put_contents ( $file, json_encode ( $record ) );
    rename ( $file, dirname ( $file ) . '/' . sprintf ( '%010d', time () - 1 ) . '-' . $record ['id'] . '.job' );
  }

  $second = padQueueWork ( 'notifyhook', 0, 60, 3 );
  $ids    = array_map ( fn ( $one ) => $one ['headers'] ['webhook-id'], padCurlRecorded () );

  padQueueFlush ( 'notifyhook' );

  $report ['queued'] = [ 'attempts' => count ( $ids ), 'first' => [ $first ['done'], $first ['retried'] ], 'second' => [ $second ['done'], $second ['retried'] ],
                         'one id' => count ( array_unique ( $ids ) ) == 1 ];

  padCurlFake ( [ 'https://hooks.example.com/*' => 410 ] );

  padNotify ( [ 'id' => 9 ], 'shipped', [ 'order' => 44, 'items' => 1, 'name' => 'Cy', 'url' => '' ], [ 'webhook' ] );

  $report ['gone'] = padNotifyResults () ['webhook'] ['error'];

  padNowFreeze ();

  $json = json_encode ( $report, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE );

?>
 
{$json}
