App: regression/framework - Page: formats/a_fetched_body_is_read_in_the_format_its_media_type_names

formats/a_fetched_body_is_read_in_the_format_its_media_type_names.php 		HTML
formats/a_fetched_body_is_read_in_the_format_its_media_type_names.pad 		Result 
<?php

  // A fetched body is read in the format of the formats subsystem its media type names - the
  // NDJSON of a stream API (application/x-ndjson), a calendar (text/calendar), TOML
  // (application/toml), TSV (text/tab-separated-values) - where application/x-ndjson was
  // taken for one JSON document by the json in its name and failed, and TOML read as CSV;
  // a feed (application/rss+xml) is the XML it is, as before, and JSON is JSON.

  padCurlFake ( [
    'https://api.example.com/ndjson' => [ 'data' => "{\"name\":\"rex\"}\n{\"name\":\"tom\"}", 'headers' => [ 'Content-Type' => 'application/x-ndjson; charset=utf-8' ] ],
    'https://api.example.com/ics'    => [ 'data' => "BEGIN:VCALENDAR\nBEGIN:VEVENT\nSUMMARY:walk\nDTSTART:20261101\nEND:VEVENT\nEND:VCALENDAR", 'headers' => [ 'Content-Type' => 'text/calendar' ] ],
    'https://api.example.com/toml'   => [ 'data' => "[[pets]]\nname = \"rex\"\n[[pets]]\nname = \"tom\"", 'headers' => [ 'Content-Type' => 'application/toml' ] ],
    'https://api.example.com/tsv'    => [ 'data' => "name\nrex\ntom", 'headers' => [ 'Content-Type' => 'text/tab-separated-values' ] ],
    'https://api.example.com/rss'    => [ 'data' => '<rss><channel><item><title>a</title></item></channel></rss>', 'headers' => [ 'Content-Type' => 'application/rss+xml' ] ],
    'https://api.example.com/json'   => [ 'data' => '[{"name":"rex"}]', 'headers' => [ 'Content-Type' => 'application/json' ] ]
  ] );

  $fetchShow = [];

  foreach ( [ 'ndjson', 'ics', 'toml', 'tsv', 'rss', 'json' ] as $fetchKind ) {
    $fetchCurl   = padCurl ( "https://api.example.com/$fetchKind" );
    $fetchShow [] = "$fetchKind: {$fetchCurl ['type']} " . json_encode ( padCurlData ( $fetchCurl, 'x' ) );
  }

  padCurlFakeStop ();

  $fetchShow = implode ( ' | ', $fetchShow );

?>
 
{pad:echo $fetchShow/}

App: regression/framework - Page: formats/a_tsv_cell_a_spreadsheet_would_run_gets_a_quote_as_in_csv

formats/a_tsv_cell_a_spreadsheet_would_run_gets_a_quote_as_in_csv.php 		HTML
formats/a_tsv_cell_a_spreadsheet_would_run_gets_a_quote_as_in_csv.pad 		Result 
<?php

  // A spreadsheet opens a .tsv as it opens a .csv and runs a cell that starts with =, +, -
  // or @: the TSV writer gives such a text cell - and a header name - the ' in front the CSV
  // writer gives it, where it wrote a visitor's =HYPERLINK(...) live, one padFormat= away
  // from the CSV answer that defused it; a number of either sign stays as it is.

  $tsvRows = [ [ 'name' => '=HYPERLINK("http://evil.example","click")', 'cmd' => "+cmd|' /C calc'!A0", 'at' => '@SUM(A1)', 'minus' => '-2+3', 'n' => -5, '=x' => 1.5 ] ];

  $tsvShow = str_replace ( [ "\t", "\n" ], [ '|', '~' ], padFormatWrite ( 'tsv', $tsvRows ) );
  $csvShow = str_replace ( "\n", '~', padFormatWrite ( 'csv', $tsvRows ) );

?>
 
tsv: {pad:echo $tsvShow/} csv: {pad:echo $csvShow/}

App: regression/framework - Page: formats/a_tsv_leading_tab_and_the_bytes_of_an_xlsx_reach_their_reader_untrimmed
PHP
formats/a_tsv_leading_tab_and_the_bytes_of_an_xlsx_reach_their_reader_untrimmed.php 		HTML
formats/a_tsv_leading_tab_and_the_bytes_of_an_xlsx_reach_their_reader_untrimmed.pad 		Result 
<?php

  // In TSV a leading tab is an empty cell, not white space: a crosstab whose first header
  // cell is empty reads as its rows, and so does the writer's own text of a column named ''
  // and a text where every line opens with a tab - each by the helper and as {format:auto} with
  // format='tsv', indented as a template writes it - where the trim of a data text took the
  // first line's tab ("a row of 3 cells under a header of 2 names") and the cut of the indent
  // the lines share took the first column of all of them. An xlsx handed to padData as its
  // bytes reads too, where the trim took the zip's closing nulls.

  $tsvCross = "\tQ1\tQ2\nNorth\t1\t2\nSouth\t3\t4";
  $tsvOwn   = padFormatWrite ( 'tsv', [ [ '' => 'North', 'Q1' => 1 ] ] );
  $tsvAll   = "\tname\n\tann\n\tbob";

  $tsvShow = fn ( $rows ) => json_encode ( $rows );

  $tsvRead = implode ( ' | ', [ $tsvShow ( padFormatRead ( 'tsv', $tsvCross ) ),
                                $tsvShow ( padFormatRead ( 'tsv', $tsvOwn ) ),
                                $tsvShow ( padFormatRead ( 'tsv', $tsvAll ) ),
                                $tsvShow ( padData ( $tsvCross, 'tsv' ) ) ] );

  $tsvBook = padFormatWrite ( 'xlsx', [ [ 'a' => 1, 'b' => 'x' ] ] );
  $tsvXlsx = $tsvShow ( padData ( $tsvBook, 'xlsx' ) );

?>
 
{format:auto 'cross', format='tsv'}
      Q1    Q2
  North    1    2
  South    3    4
{/format:auto}
[{data:cross}{$Q1}+{$Q2} {/data:cross}] {pad:echo $tsvRead/} | {pad:echo $tsvXlsx/}
{format:auto 'book', $tsvBook, format='xlsx'/}
{data:book}{$a}{$b}{/data:book}

App: regression/framework - Page: formats/a_tsv_text_is_known_on_sight_and_its_escapes_round_trip
PHP
formats/a_tsv_text_is_known_on_sight_and_its_escapes_round_trip.php 		HTML
formats/a_tsv_text_is_known_on_sight_and_its_escapes_round_trip.pad 		Result 
<?php

  // TSV has no quoting: a tab, a newline and a backslash in a cell are written \t, \n and
  // \\, and read back as what they were.

  $tsvRows = [ [ 'name' => "tab\there", 'note' => "two\nlines" ], [ 'name' => 'back\\slash', 'note' => 3 ] ];
  $tsvText = padFormatWrite ( 'tsv', $tsvRows );

  $tsvShow = json_encode ( $tsvText );
  $tsvBack = ( padFormatRead ( 'tsv', $tsvText ) == $tsvRows ) ? 'read back the same' : 'read back otherwise';

?>
 
{format:auto 'stock'}
item    count    place
pens    40    shelf, top
paper    500
{/format:auto}
[{data:stock}{$item}={$count} ({$place});{/data:stock}] {pad:echo $tsvShow/} {pad:echo $tsvBack/}
