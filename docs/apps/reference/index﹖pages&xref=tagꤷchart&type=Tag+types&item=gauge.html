gauge/servers.pad 		Result 
<?php

  // Three servers and how loaded they are, as a monitoring page would read them.

  $servers = [
    [ 'name' => 'web-1', 'cpu' => 34, 'memory' => 61 ],
    [ 'name' => 'web-2', 'cpu' => 71, 'memory' => 78 ],
    [ 'name' => 'db-1',  'cpu' => 93, 'memory' => 88 ],
  ];

?>
 
<div class="gauges">
  {array:servers}
  {chart:gauge value=$cpu,
         label=$name,
         unit='%',
         bands='70, 90',
         width=200,
         height=130/}
  {/array:servers}
</div>

App: regression/framework - Page: tags/chart_gauge_of_no_number_shows_its_else
PHP
tags/chart_gauge_of_no_number_shows_its_else.php 		HTML
tags/chart_gauge_of_no_number_shows_its_else.pad 		Result 
<?php

  // A gauge whose value could not be read - '' or 'n/a' - draws nothing and shows its
  // @else@, where it failed as a chart "without data", which a gauge never needs.

  $empty = '';
  $text  = 'n/a';

?>
 
[{chart:gauge value=$empty, unit='%'}@else@none{/chart:gauge}] [{chart:gauge value=$text, unit='%'}@else@none{/chart:gauge}]
