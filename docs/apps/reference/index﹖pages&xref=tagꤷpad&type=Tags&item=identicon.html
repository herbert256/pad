App: regression/framework - Page: tags/a_field_inside_the_quoted_text_of_a_tag_that_computes_from_it_counts_as_its_characters
PHP
tags/a_field_inside_the_quoted_text_of_a_tag_that_computes_from_it_counts_as_its_characters.php 		HTML
tags/a_field_inside_the_quoted_text_of_a_tag_that_computes_from_it_counts_as_its_characters.pad 		Result 
<?php

  // A field written inside the quoted text of {avatar:initials}, {avatar:gravatar}, {avatar:identicon} or {barcode:qr}
  // gives what the field alone gives: the characters it holds - not the stand-ins it
  // travels with, nor the HTML its sanitize step made of it, which went into the initials,
  // the hash and the code, and into a label escaped once more.

  $name  = 'Åsa Ørsted & Co';
  $email = "o'brien@example.com";
  $query = 'https://example.com/?a=1&b=2';

?>
 
{avatar:initials '{$name}'/}
{avatar:gravatar '{$email}'/}
{avatar:identicon '{$name}'/}
{barcode:qr '{$query}', size=21/}

App: images - Page: examples/identicon/accounts
PHP
examples/identicon/accounts.php 		HTML
examples/identicon/accounts.pad 		Result 
<?php

  $accounts = [
    [ 'email' => 'herbert@example.com', 'joined' => '2009-03-14' ],
    [ 'email' => 'ada@example.org',     'joined' => '2015-12-10' ],
    [ 'email' => 'grace@example.net',   'joined' => '2018-07-01' ],
    [ 'email' => 'alan@example.com',    'joined' => '2021-06-23' ],
    [ 'email' => 'Ada@Example.org ',    'joined' => 'the same address, other spelling' ] ];

?>
 
{pad:meta title='Accounts at a glance', about='An identicon per e-mail address: trimmed and lower-cased first, so every spelling of one address is one picture.'/}
<table style="border-collapse:collapse; width:100%">
  {array:accounts}
  <tr style="border-bottom:1px solid rgba(127,127,127,.25)">
    <td style="padding:8px 12px 8px 0; width:48px">{avatar:identicon $email, size=40/}</td>
    <td style="padding:8px 0"><strong>{$email}</strong><br><small style="opacity:.7">{$joined}</small></td>
  </tr>
  {/array:accounts}
</table>
