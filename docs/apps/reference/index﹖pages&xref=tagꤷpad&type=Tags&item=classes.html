<?php

  // A quote's stand-in - U+E022, which a request can carry in a value - is escaped as the
  // quote it will become at output (exits.php restores it after all escaping), so it cannot
  // close the attribute {attrs} or {classes} builds and add an event handler.

  $t = "hi\u{E022} onmouseover=1";
  $c = "p\u{E022} onx=1";

?>
 
<a {attrs title=$t}>x</a>|<i class="{classes p=1, $c}">y</i>

<?php

  $tasks = [
    [ 'title' => 'Write the release notes', 'done' => TRUE,  'days' => 0 ],
    [ 'title' => 'Update the screenshots',  'done' => FALSE, 'days' => 1 ],
    [ 'title' => 'Plan the next version',   'done' => FALSE, 'days' => 14 ] ];

?>
 
{meta title='Per row of a loop', about='Each row\'s classes from its own fields and from the loop\'s properties.'}
<ul>
  {tasks}
    <li class="{classes 'task', done=$done, first=first@tasks, urgent=$days lt 2}">{$title}: <code>{classes 'task', done=$done, first=first@tasks, urgent=$days lt 2}</code></li>
  {/tasks}
</ul>
