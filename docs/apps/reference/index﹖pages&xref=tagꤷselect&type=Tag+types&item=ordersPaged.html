Reference - Tag types - ordersPaged

App: regression/pages - Page: select/demo/a_pager_counts_a_select_with_a_join

select/demo/a_pager_counts_a_select_with_a_join.php 		HTML
select/demo/a_pager_counts_a_select_with_a_join.pad 		Result 
<?php

  // A pager after a select that joins two tables sharing a column name - orders and
  // customers, both with customerNumber - counts its rows: the count ran the select list
  // as a derived table, which MySQL refuses with two columns of one name ("1060: Duplicate
  // column name 'customerNumber'"), and the page ended at the pager.

  $padSelect ['ordersPaged'] = [ 'db' => 'orders', 'key' => 'orderNumber' ];

?>
 
[{ordersPaged join='customers on customers.customerNumber = orders.customerNumber', where="customers.country = 'Norway'", order='orderNumber', page=1, rows=3}{$orderNumber} {$customerName}; {/ordersPaged}]
{pager 'ordersPaged', window=0}{$kind}:{$page} {/pager}
