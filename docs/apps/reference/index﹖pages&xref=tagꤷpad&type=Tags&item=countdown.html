<?php

  // A value holding the stand-in of a quote - what %EE%80%A2 in a request is - written by
  // the tags that draw into an attribute: escaped as the quote it becomes when the page is

  // of the visitor's own, an event handler.

{progress 40, label=$hostile}
{placeholder '60x30', text=$hostile}

App: showcase - Page: countdown/events
PHP
countdown/events.php 		HTML
countdown/events.pad 		Result 
<?php

  // Events at moments counted from now - from half a minute ahead, so the words come out
  // round - and the page always has some to come.

  $events = [];
  $from   = padNow ()->modify ( '+30 seconds' );

  foreach ( [ 'Jazz in the park'   => '-2 days',
              'Pottery workshop'   => '+3 hours 20 minutes',
              'City half marathon' => '+4 days 6 hours',
              'Film night'         => '+26 days 2 hours',
              'New year concert'   => '+84 days 3 hours' ] as $name => $when )
    $events [] = [ 'name' => $name, 'starts' => $from->modify ( $when )->format ( 'Y-m-d H:i:s' ) ];

?>
 
{meta title='Upcoming events', about='The two largest units by default, a past= text for an event that has started, and the full moment in the title on hover.'}
<ul style="list-style:none;margin:0;padding:0;min-width:24em">
  {events}
  <li style="display:flex;justify-content:space-between;gap:24px;padding:10px 0;border-bottom:1px solid rgba(128,128,128,.25)">
    <strong>{$name}</strong>
    <span style="opacity:.8">{countdown $starts, past='Sold out'}</span>
  </li>
  {/events}
</ul>

<?php

  // The launch: twelve days, five hours and a half from now - from half a minute ahead, so
  // the minutes come out round while the page renders.

  $launch = padNow ()->modify ( '+12 days 5 hours 30 minutes 30 seconds' )->format ( 'Y-m-d H:i:s' );

?>
 
{meta title='A launch countdown', about='Three units of the time left, counting down every second in the browser with the live option - the server writes the same words first.'}
<section style="text-align:center;padding:28px 36px;border-radius:16px;background:linear-gradient(135deg,#2a78d6,#7c5cff);color:#fff;max-width:30em">
  <p style="margin:0;font-size:13px;letter-spacing:.12em;text-transform:uppercase;opacity:.85">Version 3.0 launches in</p>
  <p style="margin:10px 0 12px;font-size:30px;font-weight:700;font-variant-numeric:tabular-nums">{countdown $launch, units=3, live, past='We are live!'}</p>
  <p style="margin:0;font-size:14px;opacity:.85">{$launch | date('l j F, H:i')}</p>
</section>
