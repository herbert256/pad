App: regression/framework - Page: tags/a_cache_section_is_kept_per_locale
PHP
tags/a_cache_section_is_kept_per_locale.php 		HTML
tags/a_cache_section_is_kept_per_locale.pad 		Result 
<?php

  // With several locales a {cache} section is kept per locale, as a cached page is: the
  // section written for nl was served to en - "Je winkelwagen" for every language.

  $padLocales = [ 'en', 'nl' ];
  $padLocale  = 'nl';

  padFragmentForget ( 'cart-per-locale' );

  function cacheCaseLocale ( $locale ) { $GLOBALS ['padLocale'] = $locale; return ''; }

?>
 
[{cache 'cart-per-locale', ttl=60}{trans 'cart.title'}{/cache}]{echo php:cacheCaseLocale('en')}[{cache 'cart-per-locale', ttl=60}{trans 'cart.title'}{/cache}]
