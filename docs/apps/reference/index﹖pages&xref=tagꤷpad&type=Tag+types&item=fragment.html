<?php

  // The cart is the session's: product id => quantity. A button posts its product - op=add
  // or op=remove - and the answer is the {fragment 'item'} of that product: the card with its
  // new count, then the parts elsewhere on the page that changed too, each marked
  // hx-swap-oob - htmx puts the card back where the button was and the rest where their ids
  // are: the count in the header, the summary beside the shelf. Emptying the cart is an
  // ordinary {form} post that ends in a redirect, the page coming back whole.

  $counts   = padSession ( 'htmxCart', [] );
  $products = array_slice ( htmxData ( 'products' ), 0, 8 );
  $posted   = padRequestIs ( 'POST' );

  if ( $posted ) {

    $id    = (int) padRequest ( 'id', 0 );
    $op    = htmxText ( 'op' );
    $stock = array_column ( $products, 'stock', 'id' );

    if ( $op == 'clear' ) {
      padSessionPut ( 'htmxCart', [] );
      padRedirect ();
    }

    if ( $op == 'add' and ( $stock [$id] ?? 0 ) > ( $counts [$id] ?? 0 ) )
      $counts [$id] = ( $counts [$id] ?? 0 ) + 1;
    elseif ( $op == 'remove' and isset ( $counts [$id] ) and --$counts [$id] <= 0 )
      unset ( $counts [$id] );

    padSessionPut ( 'htmxCart', $counts );

    $products = array_values ( array_filter ( $products, fn ( $product ) => $product ['id'] == $id ) );

    // A product the shelf does not have - a stale or made-up request - has no card to
    // answer: a 404, which htmx swaps nothing for, where the empty loop rendered no
    // {fragment 'item'} and the answer was a 500, "there is no fragment named 'item'".

    if ( ! $products )
      padAbort ( 404, 'There is no such product' );

  }

  foreach ( $products as $at => $product )
    $products [$at] ['inCart'] = $counts [ $product ['id'] ] ?? 0;

  $cart      = htmxCart ( $counts );
  $cartCount = $cart ['count'];
  $lines     = $cart ['lines'];
  $total     = $cart ['total'];

?>
 
<div class="shop">
  <div id="shelf" class="grid shelf">
  {products}
  {fragment 'item'}
    <div class="card item" id="item-{$id}">
      <div class="item-top"><span class="item-emoji">{$emoji}</span>{if $inCart}<span class="badge is-accent">{$inCart} in cart</span>{/if}</div>
      <strong>{$name}</strong>
      <div class="muted small">{$price | currency('EUR')} &middot; {if $stock}{$stock} in stock{else}sold out{/if}</div>
      <div class="item-buttons">
        <button class="btn-sm" hx-post="?examples/cart&id={$id}&op=add&padFragment=item" hx-target="#item-{$id}" hx-swap="outerHTML">+ Add</button>
        {if $inCart}<button class="btn-sm btn-ghost" hx-post="?examples/cart&id={$id}&op=remove&padFragment=item" hx-target="#item-{$id}" hx-swap="outerHTML">−</button>{/if}
      </div>
    </div>
    {if $posted}
      <span id="cart-count" hx-swap-oob="true">{$cartCount}</span>
      {cartSummary}
    {/if}
  {/fragment}
  {/products}
  </div>
  {cartSummary}
</div>

<?php

  // The contacts are the session's copy of _data/contacts.json. htmx asks for one row at a
  // time - &id= with &padFragment=row - and this file narrows the loop to that row, so the
  // first {fragment 'row'} to render is the answer: the row to read, the row as a form
  // (mode=edit), or after a post the row as stored - or the form again with padValidate's
  // messages in it. The token of the post comes in the X-CSRF-Token header (_inits.pad).

  $contacts = padSession ( 'htmxContacts', htmxData ( 'contacts' ) );
  $id       = (int) padRequest ( 'id', 0 );
  $mode     = padRequest ( 'mode', '' ) === 'edit' ? 'edit' : 'view';
  $errors   = [];

  if ( padRequestIs ( 'POST' ) and padRequestHas ( 'reset' ) ) {
    $contacts = htmxData ( 'contacts' );
    padSessionPut ( 'htmxContacts', $contacts );
    $id = 0;
  }

  elseif ( padRequestIs ( 'POST' ) and $id ) {

    $errors = padValidate ( [ 'name' => 'required|max:40', 'email' => 'required|email', 'city' => 'required|max:30' ] );

    foreach ( $contacts as $at => $contact )
      if ( $contact ['id'] == $id and ! $errors ) {
        $contacts [$at] = [ 'id' => $id, 'name' => trim ( padRequest ( 'name' ) ), 'email' => trim ( padRequest ( 'email' ) ), 'city' => trim ( padRequest ( 'city' ) ) ];
        padSessionPut ( 'htmxContacts', $contacts );
      }

    $mode = $errors ? 'edit' : 'view';

  }

  // The rows the template walks: all of them, or the one asked for - a form of it showing
  // the posted values and the messages when the post had errors.

  $rows = [];

  foreach ( $contacts as $contact ) {

    if ( $id and $contact ['id'] != $id )
      continue;

    if ( $errors )
      $contact = array_replace ( $contact, [ 'name'  => htmxText ( 'name' ),
                                             'email' => htmxText ( 'email' ),
                                             'city'  => htmxText ( 'city' ) ] );

    $rows [] = $contact + [ 'editing'  => ( $id and $mode == 'edit' ),
                            'errName'  => $errors ['name']  ?? '',
                            'errEmail' => $errors ['email'] ?? '',
                            'errCity'  => $errors ['city']  ?? '' ];

  }

  // An id the session does not hold - a stale or made-up request - has no row to answer: a
  // 404, which htmx swaps nothing for, where the empty loop rendered no {fragment 'row'} and
  // the answer was a 500, "there is no fragment named 'row' on this page".

  if ( $id and ! $rows )
    padAbort ( 404, 'There is no such contact' );

?>
 
<table class="table">
  <thead><tr><th>Name</th><th>E-mail</th><th>City</th><th></th></tr></thead>
  <tbody hx-target="closest tr" hx-swap="outerHTML">
  {rows}
  {fragment 'row'}
    {if $editing}
    <tr class="is-editing">
      <td><input class="input{if $errName} is-invalid{/if}" name="name" value="{$name}" aria-label="Name">{if $errName}<span class="error">{$errName}</span>{/if}</td>
      <td><input class="input{if $errEmail} is-invalid{/if}" name="email" value="{$email}" aria-label="E-mail">{if $errEmail}<span class="error">{$errEmail}</span>{/if}</td>
      <td><input class="input{if $errCity} is-invalid{/if}" name="city" value="{$city}" aria-label="City">{if $errCity}<span class="error">{$errCity}</span>{/if}</td>
      <td class="actions">
        <button class="btn-sm" hx-post="?examples/edit&id={$id}&padFragment=row" hx-include="closest tr">Save</button>
        <button class="btn-sm btn-ghost" hx-get="?examples/edit&id={$id}&padFragment=row">Cancel</button>
      </td>
    </tr>
    {else}
    <tr>
      <td><strong>{$name}</strong></td>
      <td>{$email}</td>
      <td>{$city}</td>
      <td class="actions"><button class="btn-sm btn-ghost" hx-get="?examples/edit&id={$id}&mode=edit&padFragment=row">Edit</button></td>
    </tr>
    {/if}
  {/fragment}
  {/rows}
  </tbody>
</table>

<p class="small muted" style="margin-top: 14px">
  Your edits stay in your session.
  <button class="btn-sm btn-ghost" hx-post="?examples/edit&reset" hx-select="table" hx-target="previous table" hx-swap="outerHTML">Undo them all</button>
</p>

  // The slow part. The page itself renders a skeleton where the chart goes; once it has

<?php

  // Ninety orders, fifteen at a time. The page shows the first fifteen; the last row of every
  // batch carries hx-get for the next batch, sent when the row scrolls into view. That row
  // has no id, so the request names its fragment itself: &padFragment=rows - the answer is
  // the next rows, the last of them asking for the batch after it, and hx-swap="afterend"
  // puts them below the row that asked.

  $page  = max ( 1, (int) padRequest ( 'p', 1 ) );
  $per   = 15;
  $total = 90;
  $pages = (int) ceil ( $total / $per );
  $page  = min ( $page, $pages );

  $people = [ 'Ada', 'Bram', 'Chen', 'Dalia', 'Emeka', 'Freya', 'Goran', 'Hana' ];
  $states = [ 'paid', 'shipped', 'open', 'paid', 'returned' ];

  $orders = [];

  for ( $n = ( $page - 1 ) * $per + 1; $n <= min ( $total, $page * $per ); $n++ )
    $orders [] = [ 'number'   => 1000 + $n,
                   'customer' => $people [ ( $n * 3 ) % count ( $people ) ],
                   'amount'   => 20 + ( $n * 37 ) % 480,
                   'status'   => $states [ $n % count ( $states ) ],
                   'next'     => ( $n == $page * $per and $page < $pages ) ? $page + 1 : 0 ];

?>
 
<table class="table">
  <thead><tr><th>Order</th><th>Customer</th><th>Status</th><th class="num">Amount</th></tr></thead>
  <tbody>
  {fragment 'rows'}
    {orders}
    <tr{if $next} hx-get="?examples/scroll&p={$next}&padFragment=rows" hx-trigger="revealed" hx-swap="afterend"{/if}>
      <td>#{$number}</td>
      <td>{$customer}</td>
      <td><span class="badge status-{$status}">{$status}</span></td>
      <td class="num">{$amount | currency('EUR')}</td>
    </tr>
    {/orders}
  {/fragment}
  </tbody>
</table>
<p class="muted small">Scroll down - every fifteenth row asks for the next fifteen. Ninety in all.</p>

  // The chart the request asks for: its kind, how many months, which channels. The browser
  // gets the whole page; the component asks for &padFragment=chart with its choices and gets
  // the {fragment 'chart'} of chart.pad alone - the SVG PAD drew, nothing around it.

  $kind     = padRequest ( 'kind', 'bar' );

  $months   = max ( 3, min ( 12, (int) padRequest ( 'months', 12 ) ) );
  $channels = array_values ( array_intersect ( [ 'online', 'shop', 'wholesale' ], explode ( ',', reactText ( 'channels', 'online,shop,wholesale' ) ) ) );

  $value     = implode ( ', ', $channels );
  $stacked   = ( $kind == 'stacked' );
  $chartKind = $stacked ? 'bar' : $kind;
  $total     = array_sum ( array_map ( fn ( $row ) => array_sum ( array_intersect_key ( $row, array_flip ( $channels ) ) ), $sales ) );

  $choice = [ 'kind' => $kind, 'months' => $months, 'channels' => $channels ];

?>

{fragment 'chart'}
  <div class="chart-frame" data-total="{$total}">
    {if $stacked}
      {chart 'bar', data='sales', label='month', value=$value, stacked, title='Sales per month, in thousands'}
    {else}
      {chart $chartKind, data='sales', label='month', value=$value, title='Sales per month, in thousands'}
    {/if}
  </div>
{/fragment}
</div>
