<?php

  // A text with a byte that is no UTF-8, far before its end: it is cut in code points, as
  // the same text is when the byte stands near the end.

  $tail  = "ab\u{1F1F3}\u{1F1F1}e\u{301}";
  $long  = "\xFF" . str_repeat ( 'a', 5000 ) . $tail;
  $short = "\xFF" . str_repeat ( 'a', 10 ) . $tail;

?>
 
[{pad:echo $long | right(1) | php:bin2hex(@)/}][{pad:echo $long | right(2) | php:bin2hex(@)/}][{pad:echo $long | substr(-3, 1) | php:bin2hex(@)/}][{pad:echo $short | right(1) | php:bin2hex(@)/}]

App: regression/framework - Page: functions/the_cutting_functions_keep_a_character_whole_at_the_end_of_a_long_text
PHP
functions/the_cutting_functions_keep_a_character_whole_at_the_end_of_a_long_text.php 		HTML
functions/the_cutting_functions_keep_a_character_whole_at_the_end_of_a_long_text.pad 		Result 
<?php

  // Past 4096 bytes right() and substr() from the end read a window at the end of the text,
  // not the whole of it: a run of flags, whose pairs count from the run's start, and an
  // accent written apart must come out whole there too.

  $flags  = str_repeat ( "\u{1F1F3}\u{1F1F1}", 3001 );
  $accent = str_repeat ( "e\u{0301}", 3000 ) . "ab";

?>
 
[{pad:echo $flags | right(1) | php:bin2hex(@)/}][{pad:echo $flags | substr(-2, 1) | php:bin2hex(@)/}][{pad:echo $accent | right(3) | php:bin2hex(@)/}][{pad:echo $accent | mid(3000, 1) | php:bin2hex(@)/}][{pad:echo $accent | truncate(3, '') | php:bin2hex(@)/}]
