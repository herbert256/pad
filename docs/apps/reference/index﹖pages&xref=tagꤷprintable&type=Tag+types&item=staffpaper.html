App: regression/framework - Page: printables/grand_staff_paper_pairs_its_staves_joined_at_their_start
PHP
printables/grand_staff_paper_pairs_its_staves_joined_at_their_start.php 		HTML
printables/grand_staff_paper_pairs_its_staves_joined_at_their_start.pad 		Result 
<?php

  // With grand the staves come in pairs for the piano, 1.6 staff heights - 11.2 mm at rastral
  // 3 - between the two of a pair, which a line of 0.9 pt joins at their start, and at least
  // 2.2 staff heights between one pair and the next: six pairs on A4, the twelve staves of
  // the paper sold for the piano. With a line over them - the content between the tags - the
  // staves move down 12 mm, the box of the line in the top margin's place; in the PDF the
  // same lines, those of a staff 0.5 pt.

  $answer = staffShow ( '{printable:staffpaper grand/}' ) . ' // '
          . printablesShow ( '{printable:staffpaper grand}<b>Sonatina</b>{/printable:staffpaper}', 2 );

  $pdf = padPdf ( padUnprotect ( padUnescape ( padCode ( '{printable:staffpaper grand/}' ) ) ), [ 'compress' => FALSE ] );

  $answer .= ' // pdf: ' . implode ( ', ', array_map ( fn ( $w, $n ) => "$n lines of $w pt",
                                                       array_keys ( $c = array_count_values ( preg_match_all ( '/RG ([0-9.]+) w [0-9. ]+ m [0-9. ]+ l S Q/', $pdf, $m ) ? $m [1] : [] ) ), $c ) );

?>
 
{!answer}

App: regression/framework - Page: printables/staff_paper_spreads_staves_of_a_rastral_size_over_the_page
PHP
printables/staff_paper_spreads_staves_of_a_rastral_size_over_the_page.php 		HTML
printables/staff_paper_spreads_staves_of_a_rastral_size_over_the_page.pad 		Result 
<?php

  // Staff paper on A4: rastral 3 - a staff 7.0 mm high, its five lines 1.75 mm apart, the size
  // of a part as Gould's table in Behind Bars gives it - and as many staves as fit with 2.2
  // staff heights between them, twelve, spread from the top margin of 15 mm to the bottom one,
  // 15 to 282 mm, over the width inside the margins, 15 to 195. rastral=0 is 9.2 mm and
  // rastral=8 3.7 mm; lines=6 is tablature, six lines a space apart, 8.75 mm high at rastral
  // 3; staves= says how many, a staff alone at the top.

  $answer = staffShow ( '{printable:staffpaper/}' )
          . ' // ' . staffShow ( '{printable:staffpaper rastral=0, staves=4/}' )
          . ' // ' . staffShow ( '{printable:staffpaper rastral=8, staves=2/}' )
          . ' // ' . staffShow ( "{printable:staffpaper lines=6, staves=1, paper='letter'/}" );

?>
 
{!answer}

  // marks none; a folded card on a paper it does not fold is A4; staff paper of a rastral
  // that is none is rastral 3, and of more staves than fit has those that fit.

  $padCheckSyntax = FALSE;

  $answer = implode ( ' // ', array_map ( fn ( $template ) => printablesShow ( $template, 0 ) . ', '

<?php

  // A printable, by the name in the address - ?labels, /printables/labels: one of the engine's
  // kinds, else a page not found. Its template - kinds/<kind>.pad, sample data from {fake:rows} with
  // fixed seeds - is shown beside what it makes, coloured, and run; its pdf option links the
  // same sheets as a PDF.

  $one = array_values ( array_filter ( $printables, fn ( $item ) => $item ['kind'] === $kind ) ) [0] ?? NULL;

  if ( ! $one )
    padAbort ( 404 );

  $title    = $one ['name'];
  $text     = $one ['text'];
  $template = printablesTemplate ( $kind );
  $source   = padHighlightTokens ( $template, 'pad' );

  // The icons at the top right of the template that take it to the WASM editor and the live
  // application (_common/_lib/playground.php).

  include_once COMMON . '_lib/playground.php';

  $play     = padPlayLink ( "$template\n" );

?>
 
<p class="pr-lead">{$text}</p>

<div class="pr-stage">
  <section class="pr-template" aria-labelledby="pr-template">
    <h2 id="pr-template">The template</h2>
    <div class="pad-play-box">{!play}<pre class="pr-source"><code>{!source}</code></div>
  </section>
  <section class="pr-preview" aria-labelledby="pr-preview">
    <h2 id="pr-preview">What it prints</h2>
    {pad:echo $template | code/}
  </section>
</div>
