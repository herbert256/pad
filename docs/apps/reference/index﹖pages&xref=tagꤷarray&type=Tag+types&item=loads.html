Reference - Tag types - loads

App: islands - Page: examples/loading
PHP
examples/loading.php 		HTML
examples/loading.pad 		Result 
<?php

  // Six islands, three ways of loading: the first two at once, two when the browser is idle,
  // two only when they scroll into view - far down the page. The log is an island too, loaded
  // at once so it hears the others mount.

  $loads = [ [ 'island' => 'ReactCounter',  'load' => 'eager',   'framework' => 'React',  'start' => 1 ],
             [ 'island' => 'PreactCounter', 'load' => 'eager',   'framework' => 'Preact', 'start' => 2 ],
             [ 'island' => 'VueCounter',    'load' => 'idle',    'framework' => 'Vue',    'start' => 3 ],
             [ 'island' => 'SolidCounter',  'load' => 'idle',    'framework' => 'Solid',  'start' => 4 ],
             [ 'island' => 'SvelteCounter', 'load' => 'visible', 'framework' => 'Svelte', 'start' => 5 ] ];

  foreach ( $loads as $at => $one )
    $loads [$at] ['props'] = [ 'start' => $one ['start'], 'framework' => $one ['framework'] ];

?>
 
<div class="cols-2">
  <div>
    <div class="panel-title">When each island came</div>
    <div data-island="LoadLog"></div>
  </div>
  <div class="loads">
    {loads}
    <div class="card load-card">
      <div class="framework-head"><strong>{$framework}</strong><code class="badge">data-load="{$load}"</code></div>
      {if $load ne 'visible'}<div data-island="{$island}" data-load="{$load}" data-props="{^props}"><p class="muted small">Waiting to load ...</p></div>{/if}
    </div>
    {/loads}
  </div>
</div>

<div class="far-below">
  <p class="muted">Far down the page - the island below loads only when it scrolls into view. Scroll on.</p>
  {loads}{if $load eq 'visible'}
  <div class="card load-card">
    <div class="framework-head"><strong>{$framework}</strong><code class="badge">data-load="{$load}"</code></div>
    <div data-island="{$island}" data-load="{$load}" data-props="{^props}"><p class="muted small">Not loaded yet - it is not in view.</p></div>
  </div>
  {/if}{/loads}
</div>
