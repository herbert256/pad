App: regression/framework - Page: search/a_page_array_is_searched_and_its_index_made_again_when_it_changes
PHP
search/a_page_array_is_searched_and_its_index_made_again_when_it_changes.php 		HTML
search/a_page_array_is_searched_and_its_index_made_again_when_it_changes.pad 		Result 
<?php

  // A page's array - a db() result has this shape - searched by name from the template, and
  // the index kept for rows of one name made again when the rows change: the second search
  // of 'films' finds the film added, the third no longer the one removed.

  $films = [
    [ 'id' => 1, 'title' => 'The Running Man', 'plot' => 'A man runs for his life in a televised game.' ],
    [ 'id' => 2, 'title' => 'Chariots of Fire', 'plot' => 'Two athletes run in the 1924 Olympics.' ]
  ];

  $ids = function () use ( &$films ) {
    return implode ( ' ', array_column ( padSearch ( $films, 'run', [ 'name' => 'films', 'fields' => 'title^2, plot' ] ), 'id' ) );
  };

  $runners = [ $ids () ];

  $films [] = [ 'id' => 3, 'title' => 'Run Lola Run', 'plot' => 'Twenty minutes to find the money.' ];

  $runners [] = $ids ();

  array_shift ( $films );

  $runners [] = $ids ();

  $runners = implode ( ' / ', $runners );

  unset ( $ids );

?>
 
{$runners}|
{search:auto 'films', q='olympic runners', fields='title^2, plot'}{$id} {$title} {/search:auto}|
{search:auto 'films', q='olympic runners', fields='title^2, plot', first=1}{$id}{/search:auto}

App: regression/framework - Page: search/the_excerpts_are_made_for_the_rows_shown
PHP
search/the_excerpts_are_made_for_the_rows_shown.php 		HTML
search/the_excerpts_are_made_for_the_rows_shown.pad 		Result 
<?php

  // 3000 rows a query of short prefixes finds every one of: the excerpts are made for the
  // rows the page can show - the ten of first=10 - and padSearch's first answers the best
  // rows only, where every hit had its excerpt made before first= cut them, and a search
  // cost what its hits cost, not what the page showed: 0.53 seconds for the ten, 0.1 now.

  $syllables = [ 'ka', 'lo', 'mi', 'ter', 'san', 'dor', 'vel', 'pra', 'nu', 'qui', 'zen', 'bar', 'fo', 'lit', 'ron', 'ge' ];
  $seed      = 11;
  $next      = function ( $n ) use ( &$seed ) {
    $seed = ( $seed * 1103515245 + 12345 ) % 2147483648;
    return intdiv ( $seed, 65536 ) % $n;
  };

  $hitRows = [];
  for ( $r = 0; $r < 3000; $r++ ) {
    $body = [];
    for ( $w = 0; $w < 60; $w++ )
      $body [] = $syllables [ $next ( 16 ) ] . $syllables [ $next ( 16 ) ];
    $hitRows [] = [ 'id' => $r, 'body' => implode ( ' ', $body ) ];
  }

  unset ( $body );

  $prefixes = implode ( ' ', array_map ( fn ( $s ) => "$s*", $syllables ) );

  // The time of all hits with their excerpts is the measure: the ten best take a part of it.

  padSearch ( $hitRows, 'kalo', [ 'name' => 'hitRows' ] );

  $start  = hrtime ( TRUE );
  $all    = count ( padSearch ( $hitRows, $prefixes, [ 'name' => 'hitRows' ] ) );
  $allNs  = hrtime ( TRUE ) - $start;

  $start  = hrtime ( TRUE );
  $best   = padSearch ( $hitRows, $prefixes, [ 'name' => 'hitRows', 'first' => 3 ] );
  $shown  = "$all hits, " . count ( $best ) . ' best, ' . ( hrtime ( TRUE ) - $start < $allNs / 2 ? 'in part of the time' : 'in all of it' );

  $hitStart = hrtime ( TRUE );

?>
 
{$shown}|
{search:auto 'hitRows', q=$prefixes, first=10}{pad:if $excerpt ne ''}+{/pad:if}{/search:auto}|
{pad:echo php:hrtime(1) - $hitStart lt $allNs / 2 ? 'in part of the time' : 'in all of it'/}

App: regression/framework - Page: search/the_style_of_the_marks_comes_with_the_rows_that_are_shown
PHP
search/the_style_of_the_marks_comes_with_the_rows_that_are_shown.php 		HTML
search/the_style_of_the_marks_comes_with_the_rows_that_are_shown.pad 		Result 
<?php

  // The style of the excerpts' marks went with the excerpt of the best row found, and a page
  // that sorted the rows, cut them or printed no excerpt lost it - for every {widget:excerpt} after
  // it as well. Now padSearch's excerpts carry it each, a {search:auto} pair writes it before its
  // first row and the single tag before its list, and the page's own once is left to what
  // prints it: the {widget:excerpt} after a pair that printed none has it too.

  $found = padSearch ( 'kitchen', 'bread' );

?>
 
[{array:found sort='slug DESC', first=1}{$slug}: {!excerpt}{/array:found}]
[{search:auto 'kitchen', q='bread'}{$slug};{/search:auto}]
[{widget:excerpt 'some text with bread in it', words=3, highlight='bread'/}]
[{search:auto 'kitchen', q='bread', sort='slug DESC', first=1}{$slug}: {!excerpt}{/search:auto}]
[{search:auto 'kitchen', q='bread', link='slug'/}]
