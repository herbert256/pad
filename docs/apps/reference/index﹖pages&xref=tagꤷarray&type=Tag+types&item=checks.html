<?php

  // The autocontrast filter by PIL's ImageOps.autocontrast with preserve_tone, worked out by
  // hand on pictures made here with GD.
  //
  // A ramp of the greys 64 to 191, one pixel each, at cutoff 0: lo 64 and hi 191, a value v
  // becomes ( v - 64 ) x 255 / 127 cut to a whole number - 64 0, 100 72, 128 128, 191 255. At
  // cutoff 10: 12 of its 128 pixels let go at each end, lo 76 and hi 179 - 76 0, 128 128, 179
  // 254 (254.99999 in doubles, cut, as PIL's own table has it), and what lies beyond them kept
  // at 0 and 255. A colour goes through the same table on every channel: #806040 among those
  // greys is #804000 - each channel stretched as a grey of its value is, not each to a range
  // of its own. A picture of one grey: as it was. A ramp of 100 to 150 with a transparent
  // black and a transparent white pixel beside it: the stretch is of the 100 to 150 alone,
  // 125 becoming 127, the two keeping their opacity. Every value here is what PIL 13's
  // ImageOps.autocontrast ( preserve_tone=True ) gives for the same pixels, the photo's
  // thumbnails too.

  include_once PAD . 'subs/images/filters/_lib/filter.php';

  $spec = include PAD . 'subs/images/filters/types/autocontrast.php';

  $ramp = function ( $from, $to, $extra = [] ) {
    $image = imagecreatetruecolor ( $to - $from + 1 + count ( $extra ), 1 );
    imagealphablending ( $image, FALSE );
    for ( $v = $from; $v <= $to; $v++ )
      imagesetpixel ( $image, $v - $from, 0, 0x010101 * $v );
    foreach ( $extra as $k => $c )
      imagesetpixel ( $image, $to - $from + 1 + $k, 0, $c );
    return $image;
  };

  $at = function ( $image, $cutoff, $xs ) use ( $spec ) {
    $image = $spec ['apply'] ( $image, [ 'cutoff' => $cutoff ], 1 );
    $out   = [];
    foreach ( $xs as $label => $x ) {
      $c     = imagecolorat ( $image, $x, 0 );
      $out[] = "$label " . ( ( ( $c >> 16 ) & 255 ) == ( $c & 255 ) && ( ( $c >> 8 ) & 255 ) == ( $c & 255 ) ? $c & 255 : sprintf ( '#%06x', $c & 0xFFFFFF ) )
             . ( ( $c >> 24 ) & 127 ? ' alpha ' . ( ( $c >> 24 ) & 127 ) : '' );
    }
    return implode ( ', ', $out );
  };

  $flat = imagecreatetruecolor ( 4, 2 );
  imagefill ( $flat, 0, 0, 0x6a6a6a );

  $checks = [
    [ 'check' => 'greys 64 to 191 at 0',    'got' => $at ( $ramp ( 64, 191 ), 0, [ 64 => 0, 100 => 36, 128 => 64, 191 => 127 ] ) ],
    [ 'check' => 'at 10',                   'got' => $at ( $ramp ( 64, 191 ), 10, [ 64 => 0, 76 => 12, 128 => 64, 179 => 115, 191 => 127 ] ) ],
    [ 'check' => 'a colour among them',     'got' => $at ( $ramp ( 64, 191, [ 0x806040 ] ), 0, [ '#806040' => 128 ] ) ],
    [ 'check' => 'one grey',                'got' => $at ( $flat, 1, [ '#6a6a6a' => 0 ] ) ],
    [ 'check' => 'beside transparent ones', 'got' => $at ( $ramp ( 100, 150, [ 0x7f000000, 0x7fffffff ] ), 0, [ 100 => 0, 125 => 25, 150 => 50, 'black' => 51, 'white' => 52 ] ) ],
  ];

?>

<?php

  // BlurHash by what the reference algorithm gives, on pictures made here with GD.
  //
  // A solid #3366cc of 8 by 8 at 4,3: the size flag L (3 + 2 x 9 = 21), the DC term 5?~E -
  // 0x3366cc in four digits of base 83 - and AC terms that are not all zero: the reference
  // takes cos ( pi i x / width ) for x = 0 .. width - 1, whose sum over a row is 1 for an odd i,
  // so a solid picture of 8 columns has AC ( 1, 0 ) of a quarter of its linear colour - blue
  // 0.604 / 4 = 0.151, the largest, quantised to floor ( 0.151 x 166 - 0.5 ) = 24, O - and
  // p3 for the terms of odd i or j alone (red 11, green 13, blue 18 of base 19), j^ for those
  // of both (10, 10, 12), fQ (9, 9, 9: zero) for the even ones. Worked out by hand, this is the
  // hash below, character for character. One component is the DC term alone: 00 5?~E.
  //
  // Black to white across, 2,1: the one AC term is negative for every channel (00: 0, 0, 0,
  // the most negative step), and the decoded row goes from dark to light.
  //
  // The photo: its hash decoded, encoded again and decoded again is the same picture within a
  // few levels - the reference's decode and encode are not each other's exact inverse (the
  // largest AC term comes back quantised, the pixels clamped and rounded), so the second hash
  // is not the first, but what it draws is. And the photo against its hash's picture: a blur
  // of it. A text of a length no components give decodes to nothing.

  include_once PAD . 'subs/images/filters/_lib/blurhash.php';

  $solid = imagecreatetruecolor ( 8, 8 );
  imagefill ( $solid, 0, 0, 0x3366cc );

  $ramp = imagecreatetruecolor ( 16, 4 );

  for ( $x = 0; $x < 16; $x++ )
    imagefilledrectangle ( $ramp, $x, 0, $x, 3, 0x111111 * $x );

  $rampHash = padBlurhashEncode ( padBlurhashPixels ( $ramp ), 2, 1 );

  $photo = imagecreatefromjpeg ( dirname ( APPS ) . '/www/regression/framework/assets/photo.jpg' );
  $small = imagecreatetruecolor ( 32, 21 );
  imagecopyresampled ( $small, $photo, 0, 0, 0, 0, 32, 21, 320, 213 );

  $first   = padBlurhashEncode ( padBlurhashPixels ( $small ), 4, 3 );
  $drawn   = padBlurhashDecode ( $first, 32, 21 );
  $second  = padBlurhashEncode ( padBlurhashDecode ( $first, 320, 213 ), 4, 3 );
  $again   = padBlurhashDecode ( $second, 32, 21 );
  $apart   = function ( $one, $two ) {
    $sum = 0;
    foreach ( $one as $y => $row )
      foreach ( $row as $x => $pixel )
        foreach ( [ 0, 1, 2 ] as $c )
          $sum += abs ( $pixel [$c] - $two [$y] [$x] [$c] );
    return number_format ( $sum / ( count ( $one ) * count ( $one [0] ) * 3 ), 1 );
  };

  $checks = [
    [ 'check' => 'solid #3366cc',                'got' => padBlurhashEncode ( padBlurhashPixels ( $solid ), 4, 3 ) ],
    [ 'check' => 'one component',                'got' => padBlurhashEncode ( padBlurhashPixels ( $solid ), 1, 1 ) ],
    [ 'check' => 'black to white',               'got' => $rampHash ],
    [ 'check' => 'decoded across',               'got' => implode ( ' ', array_map ( fn ( $p ) => $p [0], padBlurhashDecode ( $rampHash, 4, 1 ) [0] ) ) ],
    [ 'check' => 'the photo',                    'got' => $first ],
    [ 'check' => 'decoded and encoded again',    'got' => $second ],
    [ 'check' => 'the two pictures apart',       'got' => $apart ( $drawn, $again ) . ' levels' ],
    [ 'check' => 'the photo and its picture',    'got' => $apart ( padBlurhashPixels ( $small ), $drawn ) . ' levels' ],
    [ 'check' => 'a hash of the wrong length',   'got' => count ( padBlurhashDecode ( 'L05?~E', 4, 3 ) ) . ' pixels' ],
  ];

?>

<?php

  // The edges filter by what Sobel's operator gives, worked out by hand, on pictures made here
  // with GD - each row of the result as its greys, a transparent pixel marked /t.
  //
  // One grey everywhere: nothing changes, no edge, the border neither (it reads its own edge
  // pixels again). A white pixel on black: its four sides 2 x 255 / 4 = 127.5, 128; its
  // corners the length of ( 255, 255 ) / 4 = 90.2, 90; itself 0. A step from black to white:
  // the full 4 x 255 / 4 on the last black and the first white pixel. A ramp of 32 a step: 64
  // inside, 32 and 31 at the ends, where the border is read twice. Red against green: their
  // luminances 76 and 150, 74 apart. A white square on a transparent ground: the inside 0, its
  // outline 255 from the opacity's own edge, the ground still transparent. amount=50: the
  // step at half, 128.

  include_once PAD . 'subs/images/filters/_lib/filter.php';

  $edges = include PAD . 'subs/images/filters/types/edges.php';

  $canvas = function ( $w, $h, $fill ) {
    $image = imagecreatetruecolor ( $w, $h );
    imagealphablending ( $image, FALSE );
    imagesavealpha ( $image, TRUE );
    imagefilledrectangle ( $image, 0, 0, $w - 1, $h - 1, $fill );
    return $image;
  };

  $rows = function ( $image, $amount = 100 ) use ( $edges ) {
    $image = $edges ['apply'] ( $image, [ 'colors' => [ '#000000', '#ffffff' ], 'amount' => $amount ], 1 );
    $lines = [];
    for ( $y = 0; $y < imagesy ( $image ); $y++ ) {
      $line = [];
      for ( $x = 0; $x < imagesx ( $image ); $x++ ) {
        $c      = imagecolorat ( $image, $x, $y );
        $line[] = ( $c & 255 ) . ( ( $c >> 24 ) & 127 ? '/t' : '' );
      }
      $lines [] = implode ( ' ', $line );
    }
    return implode ( ' | ', $lines );
  };

  $impulse = $canvas ( 5, 5, 0x000000 );
  imagesetpixel ( $impulse, 2, 2, 0xffffff );

  $step = $canvas ( 6, 2, 0x000000 );
  imagefilledrectangle ( $step, 3, 0, 5, 1, 0xffffff );

  $stepAgain = imagecrop ( $step, [ 'x' => 0, 'y' => 0, 'width' => 6, 'height' => 1 ] );

  $ramp = $canvas ( 9, 1, 0x000000 );
  for ( $x = 0; $x < 9; $x++ )
    imagesetpixel ( $ramp, $x, 0, 0x010101 * min ( 255, $x * 32 ) );

  $colours = $canvas ( 4, 1, 0xff0000 );
  imagefilledrectangle ( $colours, 2, 0, 3, 0, 0x00ff00 );

  $square = $canvas ( 6, 5, 0x7f000000 );
  imagefilledrectangle ( $square, 1, 1, 4, 3, 0xffffff );

  $checks = [
    [ 'check' => 'one grey',           'got' => $rows ( $canvas ( 5, 3, 0x808080 ) ) ],
    [ 'check' => 'a white pixel',      'got' => $rows ( $impulse ) ],
    [ 'check' => 'a step',             'got' => $rows ( $step ) ],
    [ 'check' => 'a ramp',             'got' => $rows ( $ramp ) ],
    [ 'check' => 'red against green',  'got' => $rows ( $colours ) ],
    [ 'check' => 'a square on nothing', 'got' => $rows ( $square ) ],
    [ 'check' => 'a step at amount=50', 'got' => $rows ( $stepAgain, 50 ) ],
  ];

?>

  // The edges filter on the 2x picture reads its neighbours a CSS pixel away - $scale pixels -
  // so a screen of any density shows the lines of the 1x picture: as strong, and as wide in
  // CSS pixels. Worked out by hand, white lines on black at amount=100 (a quarter of the
  // length of the gradient):
  //
  // A step from black to white: at 1x the full 255 on the pixel either side of it; at 2x on
  // the two either side - two CSS pixels in both, where the 2x picture had one CSS pixel.
  // A ramp of 32 a CSS pixel: 32 at 1x is 64 inside (4 x 2 x 32 / 4); 16 a pixel at 2x read
  // two pixels away is 64 too, where it was 32. At 1.5x, 20 a pixel: the gradients of the
  // strides 1 and 2 weighed half and half, 40 and 80, is 60 - 30 a CSS pixel, as 1x draws it.
  // The framework's photo, 160 wide at 1x and 320 at 2x: the mean of the lines about the
  // same, where the 2x picture's was about half.

  include_once PAD . 'subs/images/filters/_lib/filter.php';

  $edges = include PAD . 'subs/images/filters/types/edges.php';

  $row = function ( $values, $scale ) use ( $edges ) {
    $image = imagecreatetruecolor ( count ( $values ), 1 );
    foreach ( $values as $x => $v )
      imagesetpixel ( $image, $x, 0, 0x010101 * $v );
    $image = $edges ['apply'] ( $image, [ 'colors' => [ '#000000', '#ffffff' ], 'amount' => 100 ], $scale );
    $out = [];
    for ( $x = 0; $x < imagesx ( $image ); $x++ )
      $out [] = imagecolorat ( $image, $x, 0 ) & 255;
    return implode ( ' ', $out );
  };

  $mean = function ( $width, $scale ) use ( $edges ) {
    $photo = imagecreatefromjpeg ( dirname ( APPS ) . '/www/regression/framework/assets/photo.jpg' );

    $sum   = 0;
    for ( $y = 0; $y < imagesy ( $image ); $y++ )
      for ( $x = 0; $x < imagesx ( $image ); $x++ )
        $sum += imagecolorat ( $image, $x, $y ) & 255;
    return number_format ( $sum / ( imagesx ( $image ) * imagesy ( $image ) ), 1 );
  };

  $checks = [
    [ 'check' => 'a step at 1x',          'got' => $row ( [ 0, 0, 0, 255, 255, 255 ], 1 ) ],

    [ 'check' => 'the photo at 1x and 2x', 'got' => $mean ( 160, 1 ) . ' and ' . $mean ( 320, 2 ) ],
  ];

  // The solarize filter channel by channel, worked out by hand: a channel at the cut,
  // threshold x 255 / 100, or past it becomes 255 less itself, one below it stays. At 50
  // (127.5) the greys 0 64 127 128 192 255 become 0 64 127 127 63 0 - 128 is the first to turn,
  // as with PIL's ImageOps.solarize at its default 128 and ImageMagick's -solarize 50%. Each
  // channel on its own: #20c080 is #203f7f. At 0 everything turns, as invert; at 100 only 255.
  // A pixel keeps its opacity. And the photo as {image:thumbnail} makes it, the same bytes as PIL's
  // solarize gave for the same picture.

  include_once PAD . 'subs/images/filters/_lib/filter.php';

  $spec = include PAD . 'subs/images/filters/types/solarize.php';

  $pixels = function ( $colours, $threshold, $alpha = 0 ) use ( $spec ) {
    $image = imagecreatetruecolor ( count ( $colours ), 1 );
    imagealphablending ( $image, FALSE );
    foreach ( $colours as $x => $c )
      imagesetpixel ( $image, $x, 0, ( $alpha << 24 ) | $c );
    $image = $spec ['apply'] ( $image, [ 'threshold' => $threshold ], 1 );
    $out   = [];
    for ( $x = 0; $x < imagesx ( $image ); $x++ ) {
      $c     = imagecolorat ( $image, $x, 0 );
      $out[] = sprintf ( '#%06x', $c & 0xFFFFFF ) . ( ( $c >> 24 ) & 127 ? ' alpha ' . ( ( $c >> 24 ) & 127 ) : '' );
    }
    return implode ( ' ', $out );
  };

  $greys = [ 0x000000, 0x404040, 0x7f7f7f, 0x808080, 0xc0c0c0, 0xffffff ];

  $checks = [
    [ 'check' => 'greys at 50',        'got' => $pixels ( $greys, 50 ) ],
    [ 'check' => 'a colour at 50',     'got' => $pixels ( [ 0x20c080 ], 50 ) ],
    [ 'check' => 'greys at 0',         'got' => $pixels ( $greys, 0 ) ],

    [ 'check' => 'half transparent',   'got' => $pixels ( [ 0xc0c0c0 ], 50, 64 ) ],
  ];

App: regression/framework - Page: filters/thumbhash_encodes_and_decodes_as_the_reference_does

filters/thumbhash_encodes_and_decodes_as_the_reference_does.php 		HTML

<?php

  // ThumbHash by what the reference implementation gives - js/thumbhash.js of
  // github.com/evanw/thumbhash, rgbaToThumbHash and thumbHashToRGBA run by node on the same
  // RGBA pixels: every hash below is the reference's, byte for byte, and its picture decoded
  // the same bytes as the reference decodes.
  //
  // The photo and the picture with transparent parts, made 100 by 67 as the filter makes them:
  // the photo's hash has no alpha flag (bit 7 of its third byte), the other has, and decodes 32
  // by 19 - its shape - with a transparent corner. A ramp from black to white down a strip of
  // 20 by 60 decodes standing, 9 by 32. One pixel of #ff8000. A picture transparent
  // everywhere. A solid #3366cc: its terms beyond the average are no more than rounding noise,
  // whose nibbles depend on the last bit of a cosine - JavaScript's engine and the C library
  // can part there - so the check is what it draws, one colour, where the noise counts for
  // nothing (its scale quantises to 0). A hash shorter than its header decodes to nothing.

  include_once PAD . 'subs/images/filters/_lib/thumbhash.php';

  $fit = function ( $source ) {
    $out = imagecreatetruecolor ( 100, 67 );
    imagealphablending ( $out, FALSE );
    imagesavealpha ( $out, TRUE );
    imagefill ( $out, 0, 0, imagecolorallocatealpha ( $out, 0, 0, 0, 127 ) );
    imagecopyresampled ( $out, $source, 0, 0, 0, 0, 100, 67, imagesx ( $source ), imagesy ( $source ) );
    return $out;
  };

  $hashOf = fn ( $image ) => padThumbhashEncode ( imagesx ( $image ), imagesy ( $image ), padThumbhashPixels ( $image ) );

  $drawn = function ( $hash ) {
