<?php

  // An {pad:else} belongs to the innermost owner it stands in - {pad:if}, {pad:case}, {pad:auth}, {pad:guest},
  // {pad:can}, {pad:cannot}, {pad:feature} - whichever of them encloses which.

  padLogin ( [ 'id' => 1, 'name' => 'Bo' ] );

  padGate ( 'x', fn ( $user ) => FALSE );

  $padFeatures = [ 'off' => FALSE ];

?>
 
{pad:if 1 eq 2}no{pad:else}{pad:guest}no{pad:else}in{/pad:guest}{/pad:if} | {pad:auth}{pad:if 1 eq 2}no{pad:else}{$name}{/pad:if}{pad:else}no{/pad:auth} | {pad:case 'a'}{pad:when 'a'}{pad:feature 'off'}no{pad:else}off{/pad:feature}{pad:else}no{/pad:case} | {pad:can 'x'}{pad:if 1 eq 1}x{pad:else}no{/pad:if}{pad:else}not x{/pad:can}

<?php

  // A gate gets the user and the values {pad:can} names: $post a page array, $posts inside
  // {array:posts} the row of that occurrence.

  padGate ( 'edit-post', fn ( $user, $post ) => $user ['id'] == $post ['user_id'] );

  padLogin ( [ 'id' => 7, 'name' => 'Ann' ] );

  $posts = [ [ 'title' => 'mine', 'user_id' => 7 ], [ 'title' => 'theirs', 'user_id' => 8 ] ];
  $post  = $posts [1];

?>
 
{pad:can 'edit-post', $post}edit{pad:else}read{/pad:can} | {array:posts}{$title}: {pad:can 'edit-post', $posts}edit{pad:else}read{/pad:can}; {/array:posts}| {array:posts}{pad:cannot 'edit-post', $posts}{$title} is not mine{/pad:cannot}{/array:posts}

App: regression/framework - Page: tags/can_lets_a_guest_reach_only_a_gate_that_takes_null

tags/can_lets_a_guest_reach_only_a_gate_that_takes_null.php 		HTML
tags/can_lets_a_guest_reach_only_a_gate_that_takes_null.pad 		Result 
<?php

  // Nobody is logged in: a gate whose first parameter allows NULL is asked, any other is
  // not - and a before hook is skipped the same way.

  padGateBefore ( fn ( $user, $ability ) => TRUE );

  padGate ( 'view-post',   fn ( ?array $user, $post ) => $post ['public'] );
  padGate ( 'delete-post', fn ( $user, $post ) => TRUE );

  $post = [ 'public' => TRUE ];

?>
 
{pad:can 'view-post', $post}view{pad:else}no view{/pad:can} | {pad:can 'delete-post', $post}delete{pad:else}no delete{/pad:can} | {pad:cannot 'delete-post', $post}cannot delete@else@can delete{/pad:cannot}
