<?php

  // A <!DOCTYPE in a placemark's CDATA - the HTML page of its balloon - and in a comment of
  // the prolog is text, and the document is read: only a DOCTYPE the document holds is
  // refused.

  $kmlText = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<!-- written without a <!DOCTYPE kml> -->\n"
           . "<kml xmlns=\"http://www.opengis.net/kml/2.2\"><Placemark><name>Balloon</name>"

  $kmlRow  = padFormatRead ( 'kml', $kmlText ) [0] ?? [];
  $kmlShow = ( $kmlRow ['name'] ?? '' ) . ' at ' . ( $kmlRow ['lat'] ?? '' ) . ',' . ( $kmlRow ['lon'] ?? '' ) . ': ' . ( $kmlRow ['description'] ?? '' );

?>

App: regression/framework - Page: formats/a_kml_folder_whose_name_holds_a_slash_survives_a_read_and_a_write
PHP
formats/a_kml_folder_whose_name_holds_a_slash_survives_a_read_and_a_write.php 		HTML
formats/a_kml_folder_whose_name_holds_a_slash_survives_a_read_and_a_write.pad 		Result 
<?php

  // A folder named Rotterdam / Den Haag - or AC/DC - is one folder: the reader writes a / inside
  // a name as ∕ (U+2215), which the / between nested folders is not, and the writer writes it
  // as / again, so a document read and written keeps its folders - where it came back as a
  // folder Rotterdam holding a folder Den Haag.

  $kmlText = '<kml xmlns="http://www.opengis.net/kml/2.2"><Document>'
           . '<Folder><name>Rotterdam / Den Haag</name><Placemark><name>Station</name></Placemark>'
           . '<Folder><name>AC/DC</name><Placemark><name>Stage</name></Placemark></Folder></Folder>'
           . '</Document></kml>';

  $kmlRows  = padFormatRead ( 'kml', $kmlText );
  $kmlRead  = implode ( ' | ', array_column ( $kmlRows, 'folder' ) );
  $kmlAgain = padFormatWrite ( 'kml', $kmlRows );

  preg_match_all ( '#<(/?)Folder>(?:\s*<name>([^<]*))?#', $kmlAgain, $kmlMatch, PREG_SET_ORDER );

  $kmlTree = implode ( ' ', array_map ( fn ( $m ) => $m [1] === '/' ? ']' : '[' . $m [2], $kmlMatch ) );
  $kmlSame = ( padFormatRead ( 'kml', $kmlAgain ) == $kmlRows ) ? 'the same rows' : 'other rows';

?>
 
{$kmlRead} - {$kmlTree} - {$kmlSame}

App: regression/framework - Page: formats/a_kml_position_past_the_poles_is_not_written

<?php

  // A row whose latitude lies past the poles is written without its geometry - on the lenient
  // walk without a word - as the reader refuses such a position: written, it made the whole
  // document unreadable, the good rows lost with it. A longitude of 152 is one.

  $padCheckSyntax = FALSE;

  $kmlText = padFormatWrite ( 'kml', [ [ 'name' => 'Far east', 'lat' => 5.12, 'lon' => 152.09 ],
                                       [ 'name' => 'Bad', 'lat' => 152.09, 'lon' => 5.12 ],
                                       [ 'name' => 'Line', 'geometry' => 'LineString', 'coordinates' => [ [ 5.1, 52.0 ], [ 190.0, 52.1 ] ] ],
                                       [ 'name' => 'Good', 'lat' => 52.09, 'lon' => 5.12 ] ] );

  $kmlBack = implode ( ' ', array_map ( fn ( $row ) => "{$row['name']}={$row['geometry']}", padFormatRead ( 'kml', $kmlText ) ) );

?>

App: regression/framework - Page: formats/a_kml_year_or_month_survives_a_read_and_a_write
PHP
formats/a_kml_year_or_month_survives_a_read_and_a_write.php 		HTML
formats/a_kml_year_or_month_survives_a_read_and_a_write.pad 		Result 
<?php

  // KML's dateTime is a year, a year and a month, a day or a moment: a TimeStamp or a TimeSpan
  // of a year or a month read and written again is the year or the month it was - not a Data
  // field time, nor the first moment of the month.

  $kmlText = '<kml xmlns="http://www.opengis.net/kml/2.2"><Document>'
           . '<Placemark><name>Year</name><TimeStamp><when>1997</when></TimeStamp></Placemark>'
           . '<Placemark><name>Month</name><TimeStamp><when>1997-07</when></TimeStamp></Placemark>'
           . '<Placemark><name>Span</name><TimeSpan><begin>1941-09</begin><end>1943</end></TimeSpan></Placemark>'
           . '<Placemark><name>Day</name><TimeStamp><when>1997-07-16</when></TimeStamp></Placemark>'
           . '</Document></kml>';

  $kmlAgain = padFormatWrite ( 'kml', padFormatRead ( 'kml', $kmlText ) );

  preg_match_all ( '#<(when|begin|end)>([^<]*)|<Data name="([^"]*)"#', $kmlAgain, $kmlMatch, PREG_SET_ORDER );

  $kmlTimes = implode ( ' ', array_map ( fn ( $m ) => ( $m [3] ?? '' ) !== '' ? "data:{$m[3]}" : "{$m[1]}={$m[2]}", $kmlMatch ) );

?>
 
{$kmlTimes}
