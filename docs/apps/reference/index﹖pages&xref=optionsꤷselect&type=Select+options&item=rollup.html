App: regression/pages - Page: select/demo/a_line_comment_ends_its_own_clause
PHP
select/demo/a_line_comment_ends_its_own_clause.php 		HTML
select/demo/a_line_comment_ends_its_own_clause.pad 		Result 
<?php

  // A -- comment written in a clause - where=, group=, having=, order= - ends that clause:
  // the clauses of the statement stand on lines of their own. On one line the comment ran to
  // its end, so the having, the order and the limit after it were gone without a word, and
  // the closing parenthesis of the where with them. Over the staff table of the demo database.

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name' ];

?>
 
group: [{staffSel fields="floor(salary / 2000) as band, count(*) as n", group="band -- per band", having="n > 1", order="n desc", rows=1}{$band}={$n} {/staffSel}]
having: [{staffSel fields="floor(salary / 2000) as band, count(*) as n", group="band", having="n > 1 -- the shared ones", order="n desc"}{$band}={$n} {/staffSel}]
rollup: [{staffSel fields="floor(salary / 2000) as band, sum(salary) as pay", group="band -- per band", rollup}{$pay} {/staffSel}]
where: [{staffSel where="salary > 2000 -- the well paid", order="name"}{$name} {/staffSel}]
order: [{staffSel order="salary desc -- best paid first", rows=2}{$name} {/staffSel}]
pages: [{staffSel fields="floor(salary / 2000) as band, count(*) as n", group="band", having="n > 0 -- every band", page=1, rows=2}{$band} {/staffSel}] ({pager 'staffSel', window=0}{$kind}:{$page} {/pager})

App: regression/pages - Page: select/demo/a_rollup_takes_no_declared_order
PHP
select/demo/a_rollup_takes_no_declared_order.php 		HTML
select/demo/a_rollup_takes_no_declared_order.pad 		Result 
<?php

  // MariaDB, and MySQL before 8.0.12, take no ORDER BY beside WITH ROLLUP, whose rows come in
  // the order of the group: a rollup - on the tag or declared - writes only an order= given
  // on the tag, not the order of a declaration that groups, which ended the select on
  // "Incorrect usage of CUBE/ROLLUP and ORDER BY". Over the staff table of the demo database.

  $padSelect ['bands']       = [ 'db' => 'staff', 'fields' => 'floor(salary / 2000) as band, count(*) as n',
                                 'group' => 'band', 'order' => 'band desc' ];
  $padSelect ['bandsRollup'] = [ 'base' => 'bands', 'rollup' => TRUE ];

?>
 
grouped: [{bands}{$band}:{$n} {/bands}]
tag: [{bands rollup}{$n} {/bands}]
declared: [{bandsRollup}{$n} {/bandsRollup}]

App: regression/pages - Page: select/demo/a_total_is_no_row_the_declared_order_sorts
PHP
select/demo/a_total_is_no_row_the_declared_order_sorts.php 		HTML
select/demo/a_total_is_no_row_the_declared_order_sorts.pad 		Result 
<?php

  // The order a table declares - or its key - sorts the rows of the table, not the rows a
  // select makes of them: a rollup, a total over the whole table, one value. Under
  // ONLY_FULL_GROUP_BY, as MySQL has it by default, an order by a column the total does not
  // hold is an error; with a rollup it is one on MariaDB as well. A view that groups keeps an
  // order of its own. Over the four-row staff table of the demo database.

  db ( "set session sql_mode = concat(@@sql_mode, ',ONLY_FULL_GROUP_BY')" );

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name' ];
  $padSelect ['staffKey'] = [ 'db' => 'staff', 'key' => 'name' ];

  $padSelect ['perBand']  = [ 'base' => 'staffSel', 'fields' => 'floor(salary / 2000) as band, sum(bonus) as bonus',
                              'group' => 'band', 'rollup' => TRUE ];
  $padSelect ['bandDesc'] = [ 'base' => 'staffSel', 'fields' => 'floor(salary / 2000) as band, count(*) as n',
                              'group' => 'band', 'order' => 'band desc' ];

?>
 
rollup: [{staffSel fields="floor(salary / 2000) as band, sum(salary) as pay", group="band", rollup}{$pay} {/staffSel}]
key rollup: [{staffKey fields="floor(salary / 2000) as band, sum(salary) as pay", group="band", rollup}{$pay} {/staffKey}]
count: [{staffSel fields="count(*) as n"}{$n}{/staffSel}]
field: [{staffKey type='field', fields="max(salary)"/}]
view: [{perBand}{$bonus} {/perBand}]
view order: [{bandDesc}{$band}:{$n} {/bandDesc}]
tag order: [{staffSel fields="floor(salary / 2000) as band, count(*) as n", group="band", order="n desc, band"}{$band}:{$n} {/staffSel}]
rows: [{staffSel order="salary desc", rows=2}{$name} {/staffSel}]

select/demo/options.pad 		Result 
<?php

  // The options a declared select table takes on the tag itself: each section of the .pad
  // half names one, over the four-row staff table of the demo database.

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name' ];
  $padSelect ['elsewhere'] = [ 'key' => 'name' ];
  $padSelect ['staffJim'] = [ 'db' => 'staff', 'where' => "name = 'jim'" ];
  $padSelect ['staffRaw'] = [ 'db' => 'staff' ];

?>
 
where: [{staffSel where="salary >= 2000", order="salary"}{$name} {/staffSel}]
order: [{staffSel order="salary desc", fields="name"}{$name} {/staffSel}]
fields: [{staffSel fields="name, bonus", where="name = 'bob'"}{$name}/{$bonus}{/staffSel}]
group: [{staffSel fields="count(*) as heads", group="floor(salary / 2000)", order="heads"}{$heads} {/staffSel}]
having: [{staffSel fields="salary + bonus as total", group="total", having="total > 2500", order="total"}{$total} {/staffSel}]
rollup: [{staffRaw fields="sum(salary) as pay", group="name", rollup}{$pay} {/staffRaw}]
distinctrow: [{staffSel fields="floor(salary / 2000) as band", distinctrow, order="band"}{$band} {/staffSel}]
db: [{elsewhere db='staff', where="name = 'jim'"}{$name}/{$phone}{/elsewhere}]
union: [{staffSel where="name = 'bob'", union='staffJim', order="name"}{$name} {/staffSel}]
