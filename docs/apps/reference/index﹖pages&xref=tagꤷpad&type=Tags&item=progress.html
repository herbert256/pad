App: ledger - Page: index

index.pad 		Result 
<?php

  // The month at a glance: what came in, what went out, what is left; six months of it as a
  // chart, this month's spending by category, the budgets, the accounts, the latest entries
  // - and a quick entry that ledger.js posts as JSON, answered by quick.php.

  $totals    = ledgerTotals ( $ledgerMonth );
  $history   = ledgerHistory ( $ledgerMonth );
  $spending  = ledgerSpending ( $ledgerMonth );
  $budgets   = ledgerBudgets ( $ledgerMonth );
  $accounts  = ledgerAccounts ();
  $latest    = array_slice ( array_values ( array_filter ( ledgerRows ( $ledgerMonth ), fn ( $r ) => $r ['kind'] != 'transfer' ) ), 0, 8 );
  $spendable = array_values ( array_filter ( ledgerCategories (), fn ( $c ) => $c ['kind'] == 'expense' ) );
  $monthName = ledgerMonthName ( $ledgerMonth );
  $today     = date ( 'Y-m-d' );

  $ledgerPage = $monthName;

?>
 
<div class="head">
  <h1>{$monthName}</h1>
  <span class="dim">{$totals.entries} entries</span>
</div>

<section class="kpis" aria-label="This month">
  <div class="kpi in"><span>Income</span><b data-kpi="income">{$totals.income | money}</b></div>
  <div class="kpi out"><span>Spending</span><b data-kpi="spent">{$totals.spent | money}</b></div>
  <div class="kpi"><span>Left</span><b data-kpi="left">{$totals.left | money}</b></div>
  <div class="kpi"><span>Kept</span><b data-kpi="kept">{$totals.kept}%</b></div>
</section>

<div class="columns">

  <div class="main">

    <section class="card">
      <h2>Six months</h2>
      {chart 'bar', data='history', label='month', value='income, spending', title='Income and spending, six months'}
    </section>

    <section class="card">
      <h2>Where it went</h2>
      {if $totals.spent gt 0}
        {chart 'donut', data='spending', label='category', value='spent', title='Spending by category'}
      {else}
        <p class="dim">Nothing spent this month.</p>
      {/if}
    </section>

    <section class="card">
      <h2>Budgets <a class="more" href="{$padGo}budgets{?month}">Change</a></h2>
      <ul class="budgets">
        {budgets}
        <li class="{$state}">
          <div class="line"><span class="dot" style="background: {$color}"></span><b>{$name}</b><span>{$spent | money} of {$monthly | money}</span></div>
          {progress $share, max=100, label=$name, color=$tone}
        </li>
        @else@
        <li class="dim">No budgets yet.</li>
        {/budgets}
      </ul>
    </section>

  </div>

  <aside class="side">

    <section class="card quick">
      <h2>Quick entry</h2>
      {form 'quick', action='?quick', class='quick-form'}
        <input type="hidden" name="month" value="{$ledgerMonth}">
        <label for="q-amount">Amount</label>
        <input id="q-amount" name="amount" inputmode="decimal" placeholder="12.50" required pattern="[0-9]+([.,][0-9][0-9]?)?">
        <label for="q-category">Category</label>
        <select id="q-category" name="category">{spendable}<option value="{$id}">{$name}</option>{/spendable}</select>
        <label for="q-note">Note</label>
        <input id="q-note" name="note" placeholder="Coffee" maxlength="120">
        <button class="button primary">Spend it</button>
        <p class="said" role="status"></p>
      {/form}
    </section>

    <section class="card">
      <h2>Accounts</h2>
      <ul class="accounts">
        {accounts}<li><span>{$name}</span><b class="{if $balance lt 0}neg{/if}">{$balance | money}</b></li>{/accounts}
      </ul>
    </section>

    <section class="card">
      <h2>Latest <a class="more" href="{$padGo}transactions{?month}">All</a></h2>
      <ul class="latest">
        {latest}<li><span class="day">{$day | date('j M')}</span><span class="what">{$note}<small>{$category}</small></span><b class="{if $amount lt 0}neg{else}pos{/if}">{$amount | money}</b></li>
        @else@<li class="dim">No entries this month.</li>{/latest}
      </ul>
    </section>

  </aside>

</div>

<?php

  // {placeholder} is escaped once, as the field alone is: it arrived escaped by its

App: regression/framework - Page: tags/a_quote_stand_in_in_a_value_stays_inside_the_attribute_of_a_drawing_tag
PHP
tags/a_quote_stand_in_in_a_value_stays_inside_the_attribute_of_a_drawing_tag.php 		HTML
tags/a_quote_stand_in_in_a_value_stays_inside_the_attribute_of_a_drawing_tag.pad 		Result 
<?php

  // A value holding the stand-in of a quote - what %EE%80%A2 in a request is - written by
  // the tags that draw into an attribute: escaped as the quote it becomes when the page is
  // written out, where their escapers left it and it closed the attribute and opened one
  // of the visitor's own, an event handler.

  $hostile = "red\u{E022} onclick=\u{E022}alert(1)";

?>
 
{qr 'https://example.com', size=60, color=$hostile}
{countdown '2030-01-01', now='2026-01-01', past=$hostile, live}
{rating 3, label=$hostile}
{progress 40, label=$hostile}
{placeholder '60x30', text=$hostile}

App: regression/framework - Page: tags/a_tag_in_a_kept_cache_section_carries_its_own_style

tags/a_tag_in_a_kept_cache_section_carries_its_own_style.php 		HTML
tags/a_tag_in_a_kept_cache_section_carries_its_own_style.pad 		Result 
<?php

  // A {cache} section that is kept carries the style of the tags in it, also when the page
  // wrote that style before it: the section comes back on other pages, where nothing else
  // writes it - kept without it, its bars lost their colours and a video its play button.
  // The page's own style and the section's: two, on a miss and on a hit alike.

?>
 
{progress 10}{cache 'once-per-output', ttl=600}{progress 20}{/cache}

App: regression/pages - Page: mail/a_mail_rendered_first_leaves_the_page_its_style
PHP
mail/a_mail_rendered_first_leaves_the_page_its_style.php 		HTML
mail/a_mail_rendered_first_leaves_the_page_its_style.pad 		Result 
<?php

  // A mail rendered in the page's PHP is an output of its own: a {progress} in it writes
  // the style into the mail, and the page's {progress} still writes it into the page -
  // where the mail took the one style of the request, and the page's bar lost its colours.

  padMail ( 'ann@example.com', 'progress', 'Progress' );

  $raw = file_get_contents ( $padMailLast ['file'] );

  unlink ( $padMailLast ['file'] );

  $mailStyles = substr_count ( quoted_printable_decode ( $raw ), '--pad-progress-track:' ) ? 'with its style' : 'without a style';

?>
 
{progress 40}
the mail: {$mailStyles}
