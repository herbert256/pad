App: regression/framework - Page: tags/a_mail_an_offline_page_a_static_copy_and_the_console_have_the_assets_written_in
PHP
tags/a_mail_an_offline_page_a_static_copy_and_the_console_have_the_assets_written_in.php 		HTML
tags/a_mail_an_offline_page_a_static_copy_and_the_console_have_the_assets_written_in.pad 		Result 
<?php

  // Where a <link> cannot work the text of the files of www/assets/ is written in, as it was
  // everywhere before they were linked: a mail (a mail client fetches no style sheet - the
  // mail rendered during this request, whose page links), an offline application (pad pack,
  // PAD_WASM: no server behind the page), a static copy (pad export) and every output that
  // is not the web's (lib/widget.php padWidgetLinked).

  $padAssetLinks = TRUE;

  include_once PAD . 'tags/_lib/mail.php';

  $mailTpl  = padMailTemplate ( 'assets' );
  $mailHtml = padMailRender ( $mailTpl ['dir'], $mailTpl ['html'], 'pad' );

  $switches = [ 'web' => padWidgetLinked () ];

  $_SERVER ['PAD_WASM'] = '1';
  $switches ['offline'] = padWidgetLinked ();
  unset ( $_SERVER ['PAD_WASM'] );

  $GLOBALS ['padExportRun'] = TRUE;
  $switches ['export'] = padWidgetLinked ();
  $GLOBALS ['padExportRun'] = FALSE;

  $GLOBALS ['padOutputType'] = 'console';
  $switches ['console'] = padWidgetLinked ();
  $GLOBALS ['padOutputType'] = 'web';

  $GLOBALS ['padAssetLinks'] = FALSE;
  $switches ['off'] = padWidgetLinked ();
  $GLOBALS ['padAssetLinks'] = TRUE;

  $switchText = implode ( ' ', array_map ( fn ( $k, $v ) => "$k=" . ( $v ? 'linked' : 'written' ), array_keys ( $switches ), $switches ) );

?>
 
<pre id="mail">{$mailHtml}
<p id="switches">{$switchText}</p>
{widget:tabs}{widget:tab 'b'}y{/widget:tab}{/widget:tabs}

App: regression/framework - Page: tags/assets_of_a_subsystem_are_linked_with_a_hash_of_their_contents

tags/assets_of_a_subsystem_are_linked_with_a_hash_of_their_contents.pad 		Result 
<?php

  // On a web page the style sheets and the scripts of the subsystems are files of
  // www/assets/<sub>/ the page links - a <link> behind the colours of the widget, a <script
  // src> - each address with ?v= and ten hex digits of a hash of the file's contents, so a
  // browser keeps the file and a changed file is a new address. Their text was written into
  // every page, the style of a tab set and the whole runtime of the games with each.

  $padAssetLinks = TRUE;

  $tabsHash = substr ( hash ( 'xxh128', file_get_contents ( dirname ( PAD ) . '/www/assets/widgets/tabs.css' ) ), 0, 10 );
  $gameHash = substr ( hash ( 'xxh128', file_get_contents ( dirname ( PAD ) . '/www/assets/games/game.js'    ) ), 0, 10 );

?>
