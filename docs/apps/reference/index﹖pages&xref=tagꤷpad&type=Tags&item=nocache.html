App: manual - Page: fragments/cache_2
PHP
fragments/cache_2.php 		HTML
fragments/cache_2.pad 		Result 
<?php

  // Start the example from an empty entry, so it shows a miss and then a hit.

  padFragmentForget ( 'manual-nocache' );

  $built = 0;
  $visit = 1;

?>
 
{cache 'manual-nocache', ttl=60}
  {set $built = $built + 1}
  <p>Cached: built {$built} time(s), {nocache}live: visit {$visit}{/nocache}</p>
{/cache}

{set $visit = 2}

{cache 'manual-nocache', ttl=60}
  {set $built = $built + 1}
  <p>Cached: built {$built} time(s), {nocache}live: visit {$visit}{/nocache}</p>
{/cache}

App: regression/cache_file - Page: nocacheprobe
PHP
nocacheprobe.php 		HTML
nocacheprobe.pad 		Result 
<?php

  // The fixture of nocache: its build moment in nanoseconds, and rows for a loop. A hit
  // of the page cache runs none of this - the {nocache} parts render without it.

  $stamp = hrtime ( TRUE );
  $rows  = [ [ 'v' => 1 ], [ 'v' => 2 ] ];

?>
 
built {$stamp} live {nocache}{php:microtime}{/nocache} rows {rows}{$v}{nocache}={$v}{/nocache};{/rows}

App: regression/framework - Page: tags/a_poll_inside_a_nocache_part_leaves_its_cache_section_kept

tags/a_poll_inside_a_nocache_part_leaves_its_cache_section_kept.php 		HTML
tags/a_poll_inside_a_nocache_part_leaves_its_cache_section_kept.pad 		Result 
<?php

  // A {poll} inside a {nocache} part renders on every request, the hit included, so the
  // section round it is kept: the second section of the name is a hit - the stored text,
  // the poll made again from its source.

  padFragmentForget ( 'fwPollNocache' );

?>
 
{cache 'fwPollNocache', ttl=60}kept {nocache}{poll 'fw-poll-nocache', options='Yes, No' | php:substr_count(@, 'padPollAnswer')}{/nocache}{/cache}|{cache 'fwPollNocache', ttl=60}its own {nocache}{poll 'fw-poll-nocache', options='Yes, No' | php:substr_count(@, 'padPollAnswer')}{/nocache}{/cache}

<?php

  $nocacheVisitor  = padRequest ( 'visitor', 'guest' );

  $nocacheProducts = [
    [ 'name' => 'Kettle'  ],
    [ 'name' => 'Toaster' ],
    [ 'name' => 'Blender' ] ];

?>
 
{meta title='A live part in a cached section', about='The list is kept five minutes; the greeting renders on every request - try ?examples/nocache/1-section&visitor=Ann.'}
{cache 'nocacheDemo', ttl=300}
  <p>{nocache}Hello, {$nocacheVisitor}!{/nocache}</p>
  <ul>
  {nocacheProducts}
    <li>{$name}</li>
  {/nocacheProducts}
  </ul>
{/cache}
