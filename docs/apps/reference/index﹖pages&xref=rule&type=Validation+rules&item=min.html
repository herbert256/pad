  {pad:if $done}
    <div class="callout is-good rise">
      <p><strong>Welcome aboard, {$name}!</strong> PAD checked the post against the rules of the form and the page's PHP ran for a good one only.</p>
      <p><button class="btn-sm btn-ghost" hx-get="?examples/form" hx-target="#signup">Sign up someone else</button></p>
    </div>
  {pad:else}
    {input:form 'signup', client, error, hx-post='?examples/form', hx-target='#signup', class='stack'}

      {input:field 'age', label='Age', rules='integer|min:18', class='input'/}

App: regression/framework - Page: rules/different_refuses_the_text_of_the_other_field
PHP
rules/different_refuses_the_text_of_the_other_field.php 		HTML
rules/different_refuses_the_text_of_the_other_field.pad 		Result 
<?php

  // different:field - not the text of the other field, the mirror of same: - the other found
  // where PHP files it and trimmed, case counting; one absent, empty or a list holds no text
  // the value could be. The rows are the shared table of the Pages suite's forms/_rules.php,
  // which its client_parity holds the browser's checker to.

  $ruleRows     = ( include APPS . 'regression/pages/forms/_rules.php' ) ['different'];
  $ruleVerdicts = [];

  foreach ( $ruleRows as [ $ruleField, $ruleList, $ruleData ] )
    $ruleVerdicts [] = "$ruleField $ruleList " . json_encode ( $ruleData, JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE ) . ' : '
                     . ( padValidate ( [ $ruleField => $ruleList ], $ruleData ) [$ruleField] ?? 'passes' );

  $ruleVerdicts = implode ( ' ; ', $ruleVerdicts );

?>
 
{!ruleVerdicts}

App: regression/framework - Page: rules/required_if_requires_the_field_when_the_other_holds_one_of_the_values

rules/required_if_requires_the_field_when_the_other_holds_one_of_the_values.php 		HTML
rules/required_if_requires_the_field_when_the_other_holds_one_of_the_values.pad 		Result 
<?php

  // required_if:field,value - the field required, as required judges it, when the other field
  // holds one of the values as written: an absent or empty other field holds none, blanks
  // around it are trimmed, case counts, a list holds a value when one of its items does,
  // kinds finds kinds[] and kinds[] kinds, user[kind] is the field the form sends, an array
  // with keys of its own is no value, and a list field is required as required requires it.
  // The rows are the shared table of the Pages suite's forms/_rules.php, which its
  // client_parity holds the browser's checker to.

  $ruleRows     = ( include APPS . 'regression/pages/forms/_rules.php' ) ['required_if'];
  $ruleVerdicts = [];

  foreach ( $ruleRows as [ $ruleField, $ruleList, $ruleData ] )
    $ruleVerdicts [] = "$ruleField $ruleList " . json_encode ( $ruleData, JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE ) . ' : '
                     . ( padValidate ( [ $ruleField => $ruleList ], $ruleData ) [$ruleField] ?? 'passes' );

  $ruleVerdicts = implode ( ' ; ', $ruleVerdicts );

?>
 
{!ruleVerdicts}

App: regression/framework - Page: tags/padValidate_reads_a_field_that_is_an_object
PHP
tags/padValidate_reads_a_field_that_is_an_object.php 		HTML
tags/padValidate_reads_a_field_that_is_an_object.pad 		Result 
<?php

  // A field of the data that is an object: one read like an array - an ArrayObject - is a
  // list, kept by tags[] and refused for a field of one value; one that is text - a
  // Stringable - is its text. Any object but a Stringable was cast to a text, and the
  // request ended on "could not be converted to string".

  $validated = padValidate (
    [ 'user' => 'required', 'tags[]' => 'required|in:a,b', 'name' => 'required|min:3' ],
    new ArrayObject ( [ 'user' => new ArrayObject ( [ 'age' => 1 ] ),
                        'tags' => new ArrayObject ( [ 'a', 'b' ] ),
                        'name' => new class { function __toString () { return 'Ann'; } } ] ) );

  $result = '';

  foreach ( $validated as $field => $message )
    $result .= "$field: $message. ";

?>
 
{$result}

<?php

  // A rule name is read in any case - Integer is integer - and so is the one that makes
  // min: and max: compare the value rather than its length: with Integer|max:10, 99999 was

  $validated = padValidate (
    [ 'qty'   => 'Integer|max:10',
      'price' => 'NUMERIC|min:5',
      'code'  => 'max:3' ],
    [ 'qty' => '99999', 'price' => '4.5', 'code' => '1234' ] );

  $result = '';

  foreach ( $validated as $field => $message )
    $result .= "$field: $message. ";

?>
