App: live - Page: index
PHP
index.php 		HTML
index.pad 		Result 
<?php

  // The live application's editor: a template and its data on the left, the page they make on
  // the right. The data is a text in any format PAD reads - JSON, YAML,
  // XML, CSV ... - or, switched with the button beside its label, PHP code as a page's .php.
  // Run posts them to ?save, which saves them as a page of this server and sends the output
  // frame there: the page is a real request, answered as any page is - its PHP, request
  // values, sessions, forms that post back, scripts. www/live/live.js keeps the boxes in the
  // URL hash, so a link carries the example, and runs on load and on Ctrl+Enter.
  //
  // The editor stands in the frame every PAD application shares, which it fills - the brand
  // and the menu on the side, the title, the buttons, the theme switcher in the bar. This page opens the frame itself, not an _inits.pad: that would wrap
  // the pages saved here too, ?page/<id>, which are the visitor's own and answer as they are.

  $title     = 'Live: Runs on the PAD http server';
  $frameMenu = [ [ 'link' => '?index', 'label' => 'Live', 'icon' => 'zap', 'current' => TRUE ] ];

  include_once COMMON . '_lib/playground.php';

  [ $sampleSource, $sampleData, $samplePhp ] = padPlaySamples ();

  // The boxes become editors with colours when Monaco loads from $liveMonaco, in the PAD
  // colouring of the edit application (_common/_lib/playground.php, padPlayEditor).

  $editorMonaco = $liveMonaco ?? '';

  [ $editorPadMode, $editorNames ] = $editorMonaco ? padPlayEditor () : [ '', [] ];

?>
 
{common:padFrame fill}

  {pad:slot 'head'}
    {pad:asset 'live.css', tag/}
    <noscript><style>.live #previewOut[hidden] { display: block; }</style></noscript>
  {/pad:slot}

  {pad:slot 'tools'}
    <span class="liveTools">
      <button type="submit" form="liveForm" class="liveRun" id="runPage">Run page</button>
      <button type="submit" form="liveForm" formaction="?render" formtarget="previewOut" id="renderTemplate">Render template</button>
      <button type="submit" form="liveForm" formaction="?render&amp;view=source" formtarget="previewOut" id="showHtml">Show HTML</button>
      <a id="toOther" class="toOther" href="{$padRoot}wasm/" title="The same template, data and PHP running in your browser">{icon:play size=14/}Open in WASM</a>
    </span>
  {/pad:slot}

{/common:padFrame}

<form method="post" action="?save" target="liveOut" id="liveForm" class="live" data-monaco="{$editorMonaco}" data-names="{^editorNames}">
  <div class="pane">
    <label for="source">Template</label>
    <textarea id="source" name="source" spellcheck="false" rows="14">{$sampleSource}</textarea>
    <div class="dataHead">
      <label for="data" data-mode="data">Data - JSON, YAML, XML or CSV</label>
      <label for="php" data-mode="php" hidden>Data - PHP code</label>
      <button type="button" id="dataSwitch" hidden>Switch to PHP</button>
    </div>
    <input type="hidden" id="mode" name="mode" value="data">
    <textarea id="data" name="data" data-mode="data" spellcheck="false" rows="10" placeholder="No data - the template has only what it sets itself">{$sampleData}</textarea>
    <textarea id="php" name="php" data-mode="php" spellcheck="false" rows="10" placeholder="No PHP - the template has only what it sets itself" hidden>{$samplePhp}</textarea>
    <p class="hint">Ctrl+Enter runs; the address bar always links to what is typed here.</p>
  </div>
  <div class="pane">
    <div class="outHead"><span class="caption">Output</span><a id="liveOpen" href="#" target="_blank" hidden>open alone</a></div>
    <iframe id="liveOut" name="liveOut" title="The page" sandbox="allow-scripts allow-same-origin allow-forms allow-popups allow-modals allow-downloads"></iframe>
    <iframe id="previewOut" name="previewOut" title="Template preview" sandbox="allow-scripts" hidden></iframe>
  </div>
</form>

{pad:if $editorPadMode}<script src="{$editorPadMode}"></script>{/pad:if}
<script src="{pad:asset 'live.js'/}"></script>

{common:padFrameEnd/}
