App: regression/pages - Page: flush/csrf
PHP
flush/csrf.php 		HTML
flush/csrf.pad 		Result 
<?php

  // With $padCsrf on, a flushed page carries the session's token in its forms on both sides
  // of the {flush}: the part sent early goes through the same form check as the rest, and
  // the session starts with the first flush, while its cookie can still go out - a form
  // below the flush asked for it after the headers had gone, and got an empty token.

  $padCsrf = TRUE;

?>
 
<form method="post"><input name="a"></form>{flush}<form method="post"><input name="b"></form>

<?php

  // A <base href> sent early by {flush} still decides where the forms below it post: the

flush/fragment.pad 		Result 
<?php

  // A request for one response fragment is answered with that fragment alone, a {flush}
  // above it on the page or in its wrapper notwithstanding: the flush sent the part of the
  // page above it first, and the fragment an HTMX swap or an {ajax fragment=} put in its

App: regression/pages - Page: flush/login
PHP
flush/login.php 		HTML
flush/login.pad 		Result 
<?php

  // A login below a {flush}, on a session already open: the id cannot be renewed once the
  // headers have gone out, and the login was made on the id from before it - a fixation,
  // without a word. padLogin refuses it now.

  padSessionStart ();

?>
 
<p>top</p>{flush}{page "flush/login_inner"}

App: regression/pages - Page: flush/logout
PHP
flush/logout.php 		HTML
flush/logout.pad 		Result 
<?php

  // A logout below a {flush} while the browser holds a remember-me cookie: the cookie
  // cannot be dropped once the headers have gone out, and the logout answered TRUE - the
  // next request was logged in again by it. padLogout refuses it now.

  $_COOKIE [ padAuthCookieName () ] = 'kept by the browser';

?>
 
<p>top</p>{flush}{page "flush/logout_inner"}
