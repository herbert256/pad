App: regression/error_ignore - Page: index
PHP
index.php 		HTML
index.pad 		Result 
<?php

  // Test fetches the boom page - an undefined variable its .php reads - and asserts what
  // the 'ignore' error action is supposed to do with it: nothing reaches the visitor and
  // the page finishes. A plain load only offers the link, so a crawl raises nothing.

  $tested = isset ( $test ) ? 1 : 0;

  if ( $tested ) {

    $r    = padCurl ( $padHost . 'regression/error_ignore/?boom&padInclude' );
    $code = $r ['result'];
    $body = $r ['data'];

    $verdict = ( $code == 200 and str_contains ( $body, "before" )
               and str_contains ( $body, "after" )
               and ! str_contains ( $body, "Warning" ) ) ? "yes" : "NO";

  }

  $action = $padErrorAction;

?>
 
<p>action: {$action}</p>

{if $tested eq 1}
  <p>the boom page behaved as this action promises: <b>{$verdict}</b></p>
{else}
  <p><a href="?index&test">Test</a> the boom page</p>
{/if}

App: regression/error_ignore - Page: verdict
PHP
verdict.php 		HTML
verdict.pad 		Result 
<?php

  // The check the index runs when asked with &test, run every time. The crawl fetches this
  // page and compares it with its stored answer, so every suite run holds the action to
  // its promise - the index alone only offers a Test link, which no run follows.

  $test = TRUE;

  include APP . 'index.php';

?>
 
{$verdict}

App: regression/events - Page: curl
PHP
curl.php 		HTML
curl.pad 		Result 
<?php

  // Two remote fetches - a page of this application, and a port nothing listens on - and
  // the curl hook hears both, the failure too.

  padCurl ( $padHost . 'regression/events/?output&padInclude' );
  padCurl ( 'http://127.0.0.1:1/' );

  $heard = implode ( ', ', $GLOBALS ['heardCurl'] ?? [] );

?>
 
<p>{$heard}</p>
