App: regression/framework - Page: variables/a_collection_page_variable_stays_the_object_after_a_list_lookup

<?php

  // An object a page leaves that can be read as a list again - an IteratorAggregate, an
  // ArrayObject - stays that object after a tag read it as a list; only an iterator that runs
  // once is replaced by its rows. Replaced, a later {$cart.items.1.name} was not found and a
  // function calling $cart->total() met an array.

  class VariablesKeptCart implements IteratorAggregate {

    public $items = [ [ 'name' => 'Pen', 'price' => 2 ], [ 'name' => 'Ink', 'price' => 5 ] ];

    function getIterator (): Iterator { return new ArrayIterator ( $this -> items ); }

    function total () { return array_sum ( array_column ( $this -> items, 'price' ) ); }

  }

  function variablesKeptCartTotal () { return $GLOBALS ['cart'] -> total (); }

  function variablesKeptListKind () { return get_class ( $GLOBALS ['list'] ); }

  $cart = new VariablesKeptCart;
  $list = new ArrayObject ( [ [ 'n' => 1 ], [ 'n' => 2 ] ] );

?>
 
before[{$cart.items.1.name}] {cart}{$name} {/cart}after[{$cart.items.1.name}] total[{php:variablesKeptCartTotal}] list[{list}{$n}{/list}] kind[{php:variablesKeptListKind}]

<?php

  // A JsonSerializable whose jsonSerialize answers the object itself - return $this - is its
  // public properties, as json_encode writes it: it stayed an object, so {me}..{/me} was no
  // tag, {^me} was not found and a list of them printed the tag's name per row.

  class padCaseJsonSelf implements JsonSerializable {

    public    $name   = 'Ann';
    protected $secret= '*REDACTED*';

    function jsonSerialize (): mixed { return $this; }

  }

  $me   = new padCaseJsonSelf;
  $list = [ new padCaseJsonSelf, new padCaseJsonSelf ];
  $enc  = json_encode ( $me );

?>
 
enc[{$enc}] tag[{me}[{$me}]{/me}] json[{^me}] list[{list}{$name},{/list}]
