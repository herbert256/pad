index.pad 		Result 
<?php

  // The page shows PAD in big letters on top, a heading of its own: the bar shows no title.

  $title     = 'PAD Home';
  $skipTitle = TRUE;

?>
 
<h1 class="home-pad">PAD</h1>

{block}

  <blockquote>

    <h3>Concept</h3>

    <blockquote>

      <p>PAD is an <a href="https://en.wikipedia.org/wiki/Inversion_of_control">inversion of control</a> PHP application driver.
      
      <p>PAD: <br>
        - first executes the application PHP code, <br> 
        - then reads the application HTML markup code,<br>
        - merges both and send the result to the browser.
      
      <p>There is no PAD code in the application PHP code at all !
        
    </blockquote>

    <h3>The Hello World example</h3>
 
     <blockquote> 
      <p>{example 'hello', skipNames/}
    </blockquote>

    <h3>Don't call us</h3>

    <blockquote>
  
      <div style="max-width:560px">
        {embed 'https://www.youtube.com/watch?v=i4njPe2_rho&t=45s', title="Sugarloaf - Don't call us, we'll call you."/}
      </div>

    </blockquote>

  </blockquote>

{/block}
