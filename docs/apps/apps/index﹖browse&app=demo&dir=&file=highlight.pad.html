<p>The <code>{ignore}{highlight}{/ignore}</code> tag colours source code on the server - PAD, PHP, HTML,
CSS, JavaScript, JSON, YAML, SQL and Bash. No JavaScript library, and the braces inside need no
<code>&#123;ignore&#125;</code>.</p>

<h2>PHP, with line numbers and a marked line</h2>

{highlight 'php', lines, mark='4'}
  <?php

    $orders = db ( "ARRAY * FROM orders WHERE status = 'open'" );
    $total  = padArrSum ( $orders, 'amount' );

    if ( ! $orders )
      padRedirect ( 'empty' );

  ?>
{/highlight}

<h2>PAD itself</h2>

{highlight 'pad'}
  {orders sort='date', first=10}
    <li class="{even@orders ? 'even' : 'odd'}">
      {$number}: {$total | money}
    </li>
  @else@
    <li>No orders yet</li>
  {/orders}
{/highlight}
