App: regression/framework - Page: tags/excerpt_html_strips_scripts_in_linear_time
PHP
tags/excerpt_html_strips_scripts_in_linear_time.php 		HTML
tags/excerpt_html_strips_scripts_in_linear_time.pad 		Result 
<?php

  // The scripts and styles of an {excerpt ..., html} text are left out in one pass: an
  // unclosed <script> before a megabyte of text emptied the excerpt (or ended the request),
  // and openers without a closer took seconds - every one scanned on to the end.

  include_once PAD . 'tags/_lib/excerpt.php';

  $big   = '<p>Hello</p><script>' . str_repeat ( 'x', 1000000 );
  $start = hrtime ( TRUE );
  $many  = strlen ( padExcerptCode ( str_repeat ( '<style>x ', 60000 ) ) );
  $time  = ( hrtime ( TRUE ) - $start < 1e9 ) ? 'in time' : 'too slow';
  $kept  = padExcerptCode ( 'a<script>x</script>b<STYLE>y</style >c<script src=1>d' );

?>
 
{excerpt $big, words=1, html} {$many} {$time} {$kept}

App: regression/framework - Page: tags/excerpt_with_thousands_of_search_words_or_a_huge_one_still_marks
PHP
tags/excerpt_with_thousands_of_search_words_or_a_huge_one_still_marks.php 		HTML
tags/excerpt_with_thousands_of_search_words_or_a_huge_one_still_marks.pad 		Result 
<?php

  // A search form that posts: four thousand words, and one word of seventy thousand
  // letters - each made a pattern PCRE refuses to compile.

  $many = [];
  for ( $i = 0; $i < 4000; $i++ )
    $many [] = 'w' . dechex ( $i * 7919 );

  $many = 'excerpting ' . implode ( ' ', $many );
  $huge = 'excerpting ' . str_repeat ( 'templates', 7800 );

?>
 
{excerpt "Templates drive the excerpting of long texts", words=3, highlight=$many}
{excerpt "Templates drive the excerpting of long texts", words=3, highlight=$huge}

App: showcase - Page: excerpt/search
PHP
excerpt/search.php 		HTML
excerpt/search.pad 		Result 
<?php

  $q = 'cache';

  $results = [
    [ 'title' => 'Fragment cache',
      'url'   => '?manual/cache',
      'body'  => 'Some parts of a page cost more to render than others: a list of the best selling products, a menu built from the database, a report over a year of orders. The fragment cache keeps the rendered text of such a section for a number of seconds and serves it again on the next request without running the tags inside it. Name the section, give it a ttl, and forget it when what it shows has changed.' ],
    [ 'title' => 'Remote data',
      'url'   => '?manual/curl',
      'body'  => 'A page that shows exchange rates or the weather fetches them from another server. Every request would wait for that server, so the answer is kept for a while: the ttl option says how many seconds. When the source fails afterwards, the last good copy in the cache is served and the failure is logged, so a slow or broken service never takes the page down with it.' ],
    [ 'title' => 'Page cache',
      'url'   => '?manual/config',
      'body'  => 'With the page cache switched on in the configuration, a whole page is kept after its first rendering, with an ETag, and a browser that asks again with that tag gets a 304 Not Modified. Pages with a guard, a flash message or a form are never cached. The cache can live in files, APCu, Redis, Memcached or the database.' ] ];

?>
 
{meta title='Search results', about='Each result cut to 24 words around the first match of the query, every match marked.'}
<div>
  <p>Results for <strong>{$q}</strong></p>
  {results}
    <article>
      <h3><a href="{$url}">{$title}</a></h3>
      <p>{excerpt $body, words=24, highlight=$q}</p>
    </article>
  {/results}
</div>
