fragments/cache_1.php 		HTML
fragments/cache_1.pad 		Result 
<?php

  // Start the example from an empty entry, so it shows a miss and then a hit.

  padFragmentForget ( 'manual-cache' );

  $built = 0;

?>
 
{cache 'manual-cache', ttl=60}
  {set $built = $built + 1}
  <p>Rendered while building: built {$built} time(s)</p>
{/cache}

{cache 'manual-cache', ttl=60}
  {set $built = $built + 1}
  <p>Rendered while building: built {$built} time(s)</p>
{/cache}

<p>The section was built {$built} time: the second one came from the cache.</p>
