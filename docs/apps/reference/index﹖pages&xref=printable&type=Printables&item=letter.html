App: regression/framework - Page: printables/the_text_options_of_a_printable_pair_are_text
PHP
printables/the_text_options_of_a_printable_pair_are_text.php 		HTML
printables/the_text_options_of_a_printable_pair_are_text.pad 		Result 
<?php

  // A text option of a printable - sender=, title=, signed= - is the characters it holds in the
  // pair form as in the single tag: a value never becomes template code. The pair writes its
  // sheets on the end walk, into content the level walks again, and a field's {pad:echo 6*7/} in
  // sender= ran there - X42Y in the return address of the envelope and the letter, the
  // invoice's heading, the band of the badges, the certificate's title and its signature.

  $v = 'X{pad:echo 6*7/}Y';

  $answer = [];

  foreach ( [ 'envelope' => 'sender=$v', 'letter' => 'sender=$v', 'invoice' => 'title=$v', 'badges' => 'title=$v',
              'certificate' => 'title=$v, signed=$v' ] as $kind => $options ) {
    $out       = padUnprotect ( padUnescape ( padCode ( "{printable:$kind $options}Ann{/printable:$kind}{printable:$kind $options/}" ) ) );
    $answer [] = "$kind: " . substr_count ( $out, 'X{pad:echo 6*7/}Y' ) . ' as text, ' . substr_count ( $out, 'X42Y' ) . ' run';
  }

  $answer = implode ( ' // ', $answer );

?>
 
{!answer}

App: printables - Page: letter
PHP
[kind].php 		HTML
[kind].pad 		Result 
<?php

  // A printable, by the name in the address - ?labels, /printables/labels: one of the engine's
  // kinds, else a page not found. Its template - kinds/<kind>.pad, sample data from {fake:rows} with
  // fixed seeds - is shown beside what it makes, coloured, and run; its pdf option links the
  // same sheets as a PDF.

  $one = array_values ( array_filter ( $printables, fn ( $item ) => $item ['kind'] === $kind ) ) [0] ?? NULL;

  if ( ! $one )
    padAbort ( 404 );

  $title    = $one ['name'];
  $text     = $one ['text'];
  $template = printablesTemplate ( $kind );
  $source   = padHighlightTokens ( $template, 'pad' );

  // The icons at the top right of the template that take it to the WASM editor and the live
  // application (_common/_lib/playground.php).

  include_once COMMON . '_lib/playground.php';

  $play     = padPlayLink ( "$template\n" );

?>
 
<p class="pr-lead">{$text}</p>

<div class="pr-stage">
  <section class="pr-template" aria-labelledby="pr-template">
    <h2 id="pr-template">The template</h2>
    <div class="pad-play-box">{!play}<pre class="pr-source"><code>{!source}</code></div>
  </section>
  <section class="pr-preview" aria-labelledby="pr-preview">
    <h2 id="pr-preview">What it prints</h2>
    {pad:echo $template | code/}
  </section>
</div>
