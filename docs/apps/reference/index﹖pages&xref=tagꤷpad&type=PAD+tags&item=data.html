<?php

  $customers = [
    [ 'id' => 1, 'name' => 'Ann',  'country' => 'NL' ],
    [ 'id' => 2, 'name' => 'Bob',  'country' => 'BE' ]
  ];

  $orders = [
    [ 'number' => 101, 'customer_id' => 2 ],
    [ 'number' => 102, 'customer_id' => 1 ],
    [ 'number' => 103, 'customer_id' => 7 ]
  ];

?>
 
{data 'countries'}
  [ {"code": "NL", "label": "Netherlands"}, {"code": "BE", "label": "Belgium"} ]
{/data}

{customers}
  <p>{$name} lives in {$country | lookup('countries', 'code', 'label')}</p>
{/customers}
