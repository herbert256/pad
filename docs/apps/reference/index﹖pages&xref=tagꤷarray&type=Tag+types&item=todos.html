App: alpine - Page: examples/todos
PHP
examples/todos.php 		HTML
examples/todos.pad 		Result 
<?php

  // The list is the session's, started from _data/todos.json. Every change is a post - op=add
  // with a text, toggle or remove with an id, clear for the done ones - and the answer is
  // the list as it now stands ($padExpose): the component shows a change at once and then
  // takes the server's list as the truth.

  $todos = padSession ( 'alpineTodos', alpineData ( 'todos' ) );

  if ( padRequestIs ( 'POST' ) ) {

    $op   = alpineText ( 'op' );
    $id   = (int) padRequest ( 'id', 0 );
    $text = trim ( alpineText ( 'text' ) );

    if ( $op == 'add' and $text !== '' )
      $todos [] = [ 'id' => max ( array_column ( $todos, 'id' ) ?: [ 0 ] ) + 1, 'text' => mb_substr ( $text, 0, 80 ), 'done' => FALSE ];

    foreach ( $todos as $at => $todo ) {
      if ( $op == 'toggle' and $todo ['id'] == $id ) $todos [$at] ['done'] = ! $todo ['done'];
      if ( $op == 'remove' and $todo ['id'] == $id ) unset ( $todos [$at] );
    }

    if ( $op == 'clear' )
      $todos = array_filter ( $todos, fn ( $todo ) => ! $todo ['done'] );

    $todos = array_values ( $todos );

    padSessionPut ( 'alpineTodos', $todos );

  }

  $padExpose = [ 'todos' ];

?>
 
<div class="todos" x-data="todos({^todos})">

  <form class="todo-add" @submit.prevent="add()">
    <input class="input" x-model="text" placeholder="Something to do ..." aria-label="A new todo" maxlength="80">
    <button>Add</button>
  </form>

  <div class="bar"><span :style="'width:' + progress() + '%'"></span></div>
  <p class="muted small"><span x-text="done()"></span> of <span x-text="list.length"></span> done
     <span x-show="syncing" x-cloak> &middot; saving ...</span></p>

  <ul class="todo-list">
    <template x-for="todo in list" :key="todo.id">
      <li :class="todo.done && 'is-done'">
        <label><input type="checkbox" :checked="todo.done" :disabled="!!todo.pending" @change="toggle(todo)"> <span x-text="todo.text"></span></label>
        <button class="btn-icon btn-ghost" @click="remove(todo)" :disabled="!!todo.pending" :aria-label="'Remove ' + todo.text">×</button>
      </li>
    </template>
  </ul>

  <button class="btn-sm btn-ghost" @click="clear()" x-show="done() > 0" x-cloak>Clear the done ones</button>

  <ul class="todo-list" x-show="false">
    {array:todos}<li>{$text}</li>{/array:todos}
  </ul>
</div>

<script src="{$padRoot}alpine/examples/todos.js"></script>

todo.php 		HTML
todo.pad 		Result 
<?php

  $title    = 'Todo List';
  $dataFile = 'demo/todos.json';   // under DATA/, for padFileGet and padFilePut
  $message  = $message ?? '';      // todoPost sets it, then restarts this page

  // A notice from the request - ?todo&message[]=x - is no text: {pad:if $message} ended the
  // page on a 500 over the list.

  if ( ! is_string ( $message ) )
    $message = '';

  $todos = json_decode ( padFileGet ( $dataFile ), TRUE ) ?: [];

  $hasTodos     = count ( $todos ) > 0;
  $doneCount    = count ( padArrWhere ( $todos, 'done' ) );
  $pendingCount = count ( $todos ) - $doneCount;

?>
 
<p>Manage your tasks with this simple todo list.</p>

{pad:if $message}
  <div class="message">{$message}</div>
{/pad:if}

<h2>Add New Task</h2>

{include:todo 'add', 'Add Task'}
  <input type="text" name="task" aria-label="New task" placeholder="Enter a new task..." required style="flex: 1;">
{/include:todo}

<h2>Tasks ({$pendingCount} pending, {$doneCount} done)</h2>

{array:todos}

  <table>

    <tr>
      <th>Task</th>
      <th style="width: 120px;">Added</th>
      <th style="width: 5%; text-align: center;" colspan="2">Actions</th>
    </tr>

  @start@

    <tr{pad:if $done} class="todo-item done"{/pad:if}>
      <td>{$task}</td>
      <td style="font-size: 12px; color: #999;">{$date | substr (0, 10)}</td>
      <td>
        {pad:if ! $done}
          {include:todo 'done', 'Done'}
            <input type="hidden" name="id" value="{$id}">
          {/include:todo}
        {/pad:if}
      </td>
      <td>
        {include:todo 'delete', 'Delete'}
          <input type="hidden" name="id" value="{$id}">
        {/include:todo}
      </td>
    </tr>

  @end@

  </table>

  {pad:if $doneCount > 0}
    {include:todo 'clear', 'Clear Completed Tasks'}{/include:todo}
  {/pad:if}
    
@else@

  <p>No tasks yet. Add one above!</p>

{/array:todos}
