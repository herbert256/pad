App: regression/pages - Page: db/a_database_tag_binds_a_value_put_in_as_name
PHP
db/a_database_tag_binds_a_value_put_in_as_name.php 		HTML
db/a_database_tag_binds_a_value_put_in_as_name.pad 		Result 
<?php

  // The SQL of a database tag is a quoted string, and a value goes into it as $name, bound
  // as an escaped literal of that variable - an array as a list, a row's field inside a
  // loop too. A value concatenated in was SQL, and one put in as {$name} reached the
  // database as PAD's inert stand-ins, so no tag had a way to take a value that worked.

  $who   = 'jim';
  $names = [ 'bob', 'joe' ];
  $odd   = "O'Br\\ien=a{b}";

?>
 
field:  [{field "phone from staff where name = $who"}]
array:  [{array "name from staff where name in ($names) order by name"}{$name} {/array}]
check:  [{check "staff where name = $who"}yes@else@no{/check}]
record: [{record "name, phone from staff where name = $who"}{$name} {$phone}{/record}]
odd:    [{field "concat($odd, '|')"}]
quoted: [{field "count(*) from staff where name = '$who'"}]
loop:   [{array "name from staff where salary > 2000 order by name"}{$name} {field "phone from staff where name = $name"} {/array}]

App: regression/pages - Page: db/sqlite_binds_a_name_after_a_literal_ending_in_a_backslash
PHP
db/sqlite_binds_a_name_after_a_literal_ending_in_a_backslash.php 		HTML
db/sqlite_binds_a_name_after_a_literal_ending_in_a_backslash.pad 		Result 
<?php

  // On SQLite a backslash in a quoted literal is a character like any other, so 'C:\' ends
  // where its quote does and a $name after it is bound. The bind took the backslash for an
  // escape on every driver: the quote that ends the literal counted as escaped, the rest of
  // the statement as quoted, and $n went to SQLite as written - a parameter of its own, NULL.

  $GLOBALS ['padSqlConnect'] = padDbSqlite ( ':memory:' );

  db ( "create table staff (name text, folder text)" );
  db ( "insert into staff values ('bob', 'C:\\'), ('jim', 'D:\\')" );

  $n = 'bob';

?>
 
A[{field "count(*) from staff where name = $n"}] B[{field "count(*) from staff where folder <> 'D:\\' and name = $n"}] C[{record "folder from staff where folder = 'C:\\' and name = $n"}{$folder}{/record}]
