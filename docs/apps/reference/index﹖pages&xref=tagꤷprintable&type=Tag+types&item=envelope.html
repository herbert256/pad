App: regression/framework - Page: printables/a_printable_in_a_pdf_is_a_page_of_its_own_size
PHP
printables/a_printable_in_a_pdf_is_a_page_of_its_own_size.php 		HTML
printables/a_printable_in_a_pdf_is_a_page_of_its_own_size.pad 		Result 
<?php

  // In a PDF a printable's sheet is a page of its own size - an envelope DL, 623.62 by 311.81
  // points, a sheet of labels A4 - the first one the document's first page while nothing
  // stands on it, and what the flow sets after the sheets goes on a page of the document's own
  // size, Letter here. The sheets carry no page number, and the one page of the flow none
  // either: a number stands under the pages of a flow of more than one.

  $html = padUnprotect ( padUnescape ( padCode ( <<<'PAD'
{printable:envelope sender='PAD, Utrecht'}Ann Smith{/printable:envelope}
{printable:labels sheet='a4-2x7', copies=3}Fragile{/printable:labels}
<p>After the sheets.</p>
PAD ) ) );

  $pdf = padPdf ( $html, [ 'compress' => FALSE, 'size' => 'Letter' ] );

  preg_match_all ( '#/MediaBox \[0 0 ([0-9.]+) ([0-9.]+)\]#', $pdf, $boxes, PREG_SET_ORDER );
  preg_match_all ( '#stream\n(.*?)\nendstream#s', $pdf, $streams );

  $pages = [];

  foreach ( $boxes as $i => $box ) {
    preg_match_all ( '/\(((?:[^()\\\\]|\\\\.)*)\) Tj/', $streams [1] [$i] ?? '', $texts );
    $pages [] = $box [1] . ' by ' . $box [2] . ': ' . implode ( ', ', $texts [1] );
  }

  $answer = count ( $boxes ) . ' pages - ' . implode ( ' // ', $pages );

?>
 
{!answer}

App: regression/framework - Page: printables/a_printable_inside_an_item_of_a_printable_stays_in_its_box_in_the_pdf
PHP
printables/a_printable_inside_an_item_of_a_printable_stays_in_its_box_in_the_pdf.php 		HTML
printables/a_printable_inside_an_item_of_a_printable_stays_in_its_box_in_the_pdf.pad 		Result 
<?php

  // A printable inside an item of another - an envelope on a business card - stays in the
  // box of that item in the PDF, its text set there as the screen shows it in the box: one
  // A4 page with both cards. Its sheet started a page of its own inside the outer box, and
  // the boxes after it went to a third page: the card's tail and the whole second card.

  $html = padUnprotect ( padUnescape ( padCode ( <<<'PAD'
{format:auto 'cards'}
[ { "n": "one", "inner": 1 }, { "n": "two", "inner": 0 } ]
{/format:auto}
{printable:cards data='cards'}card {$n}{pad:if $inner}{printable:envelope sender='PAD'}Ann{/printable:envelope}tail{/pad:if}{/printable:cards}
PAD ) ) );

  $pdf = padPdf ( $html, [ 'compress' => FALSE ] );

  preg_match_all ( '#/MediaBox \[0 0 ([0-9.]+) ([0-9.]+)\]#', $pdf, $boxes, PREG_SET_ORDER );
  preg_match_all ( '/stream\n(.*?)\nendstream/s', $pdf, $streams );

  $pages = [];

  foreach ( $boxes as $i => $box ) {
    preg_match_all ( '/\(([^)]*)\) Tj/', $streams [1] [$i] ?? '', $words );
    $pages [] = "$box[1] by $box[2]: " . implode ( ' ', $words [1] );
  }

  $answer = count ( $pages ) . ( count ( $pages ) == 1 ? ' page - ' : ' pages - ' ) . implode ( ' // ', $pages );

?>
 
{!answer}

App: regression/framework - Page: printables/the_style_prints_the_sheets_alone_on_pages_of_their_size
PHP
printables/the_style_prints_the_sheets_alone_on_pages_of_their_size.php 		HTML
printables/the_style_prints_the_sheets_alone_on_pages_of_their_size.pad 		Result 
<?php

  // The style of a printable: every sheet the size of its paper in millimetres and on a named
  // page of that size without a margin - @page pp-<id> - its places classes of the style, no
  // inline style anywhere, scale= a zoom of the screen alone; and the style every printable
  // shares prints the sheets alone, everything round them out of the way.

  $out = padUnprotect ( padUnescape ( padCode ( "{printable:envelope scale=0.5}Ann{/printable:envelope}" ) ) );

  preg_match ( '/\.(pad-printable-[0-9a-f]{8}) \.pp-sheet\{width:([^;]+);height:([^;]+);page:(pp-[0-9a-f]{8})\}@page \4\{size:([^}]+)\}/', $out, $m );

  $answer = ( $m ? "sheet {$m [2]} by {$m [3]} on page " . ( $m [4] ? 'pp-id' : '?' ) . ", @page size:{$m [5]}" : 'no sheet rule' )
          . ( str_contains ( $out, ' style="' ) ? ', inline styles' : ', no inline style' )
          . ( preg_match ( '/@media screen\{\.pad-printable-[0-9a-f]{8} \.pp-sheets\{zoom:0\.5\}\}/', $out ) ? ', zoom 0.5 on the screen' : '' )
          . ( str_contains ( $out, 'body :not(:has(.pad-printable)):not(.pad-printable):not(.pad-printable *){display:none!important}' ) ? ', printed alone' : '' );

?>
 
{!answer}

App: regression/framework - Page: printables/the_text_options_of_a_printable_pair_are_text
PHP
printables/the_text_options_of_a_printable_pair_are_text.php 		HTML
printables/the_text_options_of_a_printable_pair_are_text.pad 		Result 
<?php

  // A text option of a printable - sender=, title=, signed= - is the characters it holds in the
  // pair form as in the single tag: a value never becomes template code. The pair writes its
  // sheets on the end walk, into content the level walks again, and a field's {pad:echo 6*7/} in
  // sender= ran there - X42Y in the return address of the envelope and the letter, the
  // invoice's heading, the band of the badges, the certificate's title and its signature.

  $v = 'X{pad:echo 6*7/}Y';

  $answer = [];

  foreach ( [ 'envelope' => 'sender=$v', 'letter' => 'sender=$v', 'invoice' => 'title=$v', 'badges' => 'title=$v',
              'certificate' => 'title=$v, signed=$v' ] as $kind => $options ) {
    $out       = padUnprotect ( padUnescape ( padCode ( "{printable:$kind $options}Ann{/printable:$kind}{printable:$kind $options/}" ) ) );
    $answer [] = "$kind: " . substr_count ( $out, 'X{pad:echo 6*7/}Y' ) . ' as text, ' . substr_count ( $out, 'X42Y' ) . ' run';
  }

  $answer = implode ( ' // ', $answer );

?>
 
{!answer}

  // A printable, by the name in the address - ?labels, /printables/labels: one of the engine's
  // kinds, else a page not found. Its template - kinds/<kind>.pad, sample data from {fake:rows} with
