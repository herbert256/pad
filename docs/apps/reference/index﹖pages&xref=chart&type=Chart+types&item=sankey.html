App: charts - Page: sankey/salary
PHP
sankey/salary.php 		HTML
sankey/salary.pad 		Result 
<?php

  // A net salary of 3,200 a month: the posts, and what they are made of.

  $spending = [
    'Housing'   => [ 'Rent' => 1150, 'Insurance' => 120 ],
    'Living'    => [ 'Groceries' => 450, 'Energy' => 160, 'Phone' => 40 ],
    'Transport' => [ 'Car' => 260, 'Train' => 90 ],
    'Saving'    => [ 'Pension' => 250, 'Holiday' => 180 ],
    'Fun'       => [ 'Dining out' => 220, 'Sport' => 80, 'Streaming' => 30 ]
  ];

  $salary = [];

  foreach ( $spending as $post => $parts )
    foreach ( $parts as $part => $amount ) {
      $salary [] = [ 'from' => 'Salary', 'to' => $post, 'amount' => $amount ];
      $salary [] = [ 'from' => $post,    'to' => $part, 'amount' => $amount ];
    }

?>
 
{chart 'sankey',
       data='salary',
       source='from',
       target='to',
       value='amount',
       title='A salary of 3,200, spent',
       width=640,
       height=420/}

App: regression/framework - Page: tags/chart_sankey_takes_time_by_its_links
PHP
tags/chart_sankey_takes_time_by_its_links.php 		HTML
tags/chart_sankey_takes_time_by_its_links.pad 		Result 
<?php

  // The flows into a sankey node are looked up by its name, not searched for in every link
  // per node: 10,000 nodes and 20,000 links took six seconds.

  $flows = [];

  for ( $i = 0; $i < 10000; $i++ ) {
    $flows [] = [ 's' => "a$i", 't' => "b$i", 'v' => 1 ];
    $flows [] = [ 's' => "b$i", 't' => 'c' . ( $i % 100 ), 'v' => 1 ];
  }

  $start = hrtime ( TRUE );

?>
 
{chart 'sankey', data='flows', source='s', target='t', value='v', height=2000 | strlen/} {echo hrtime(TRUE) - $start lt 3e9 ? 'in time' : 'too slow'/}
