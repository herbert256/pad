App: regression/framework - Page: tags/a_field_inside_the_title_or_label_of_more_drawing_tags_is_escaped_once

tags/a_field_inside_the_title_or_label_of_more_drawing_tags_is_escaped_once.php 		HTML
tags/a_field_inside_the_title_or_label_of_more_drawing_tags_is_escaped_once.pad 		Result 
<?php

  // A field written inside the quoted title, label, alt, button, past= or tab name of these
  // tags is escaped once, as the field alone is: it arrived escaped by its sanitize step and
  // the tag escaped it again - Q&A Å was shown and read out as Q&amp;A &Aring;. Each line
  // counts the texts written right; a tab named by the field is the one active= and open=
  // name with it.

  $t = 'Q&A Å';

?>
 
sudoku {sudoku '53..7....6..195....98....6.8...6...34..8.3..17...2...6.6....28....419..5....8..79', title='{$t}' | php:substr_count(@, 'Q&amp;A Å')};
chess {chess 'rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1', title='{$t}' | php:substr_count(@, 'Q&amp;A Å')};
crossword {crossword title='{$t}'}
  ECHO: Writes a value
  HELLO: The first page
{/crossword | php:substr_count(@, 'Q&amp;A Å')};
emoji {emoji 'rocket', label='{$t}' | php:substr_count(@, 'Q&amp;A Å')};
copy {copy 'abc', label='{$t}' | php:substr_count(@, 'Q&amp;A Å')};
modal {modal 'fw-esc-once', title='{$t}', button='{$t}'}x{/modal | php:substr_count(@, 'Q&amp;A Å')};
video {video 'https://youtu.be/dQw4w9WgXcQ', title='{$t}' | php:substr_count(@, 'Q&amp;A Å')};
countdown {countdown '2000-01-01', past='{$t}' | php:substr_count(@, 'Q&amp;A Å')};
tabs {tabs active='{$t}'}{tab 'B'}b{/tab}{tab '{$t}'}a{/tab}{/tabs | php:substr_count(@, 'Q&amp;A Å')} {tabs active='{$t}'}{tab 'B'}b{/tab}{tab '{$t}'}a{/tab}{/tabs | php:preg_match('/-2-panel. checked/u', @)};
accordion {accordion open='{$t}'}{tab 'B'}b{/tab}{tab '{$t}'}a{/tab}{/accordion | php:preg_match('/open><summary class..pad-accordion-summary.>Q&amp;A Å/u', @)};
carousel {carousel label='{$t}'}{tab '{$t}'}a{/tab}{/carousel | php:substr_count(@, 'Q&amp;A Å')};
poll {poll 'fw-esc-once', title='{$t}', button='{$t}', options='{$t}, B' | php:substr_count(@, 'Q&amp;A Å')};
img {img 'assets/photo.jpg', width=40, alt='{$t}' | php:substr_count(@, 'Q&amp;A Å')}

App: showcase - Page: emoji/reactions
PHP
emoji/reactions.php 		HTML
emoji/reactions.pad 		Result 
<?php

  // The reactions a post got, by shortcode - as GitHub and Slack write them.

  $reactions = [
    [ 'code' => '+1',      'count' => 24 ],
    [ 'code' => 'tada',    'count' => 17 ],
    [ 'code' => 'heart',   'count' => 9  ],
    [ 'code' => 'rocket',  'count' => 12 ],
    [ 'code' => 'eyes',    'count' => 3  ],
    [ 'code' => 'joy',     'count' => 2  ],
  ];

?>
 
{meta title='A reaction bar', about='Reactions under a post, each emoji from its shortcode in a span with role img and its name as the label a screen reader says.'}
<div style="max-width:30em">
  <p style="margin:0 0 14px;line-height:1.5"><strong>Release 3.0 is out</strong> - charts, tables and a faster first render.</p>
  <div style="display:flex;flex-wrap:wrap;gap:8px">
    {reactions}
    <button type="button" style="display:inline-flex;align-items:center;gap:6px;padding:5px 12px;border-radius:999px;border:1px solid rgba(128,128,128,.35);background:transparent;color:inherit;font:inherit;font-size:15px">
      {emoji $code} <span style="font-size:13px;font-variant-numeric:tabular-nums">{$count}</span>
    </button>
    {/reactions}
  </div>
</div>
