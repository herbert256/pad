App: regression/framework - Page: formats/a_fetched_json_text_sequence_or_webvtt_is_read_by_its_media_type
PHP
formats/a_fetched_json_text_sequence_or_webvtt_is_read_by_its_media_type.php 		HTML
formats/a_fetched_json_text_sequence_or_webvtt_is_read_by_its_media_type.pad 		Result 
<?php

  // A fetched body of application/json-seq is read as the JSON text sequence it is - where it
  // was CSV, one column named after its first record - and one of text/vtt as the cues of
  // its captions, by the formats whose media type it is (padFormatMedia).

  padCurlFake ( [
    'https://api.example.com/seq' => [ 'data' => "\x1E{\"name\":\"rex\"}\n\x1E{\"name\":\"tom\"}\n", 'headers' => [ 'Content-Type' => 'application/json-seq' ] ],
    'https://api.example.com/vtt' => [ 'data' => "WEBVTT\n\n00:01.000 --> 00:02.000\nHello", 'headers' => [ 'Content-Type' => 'text/vtt; charset=utf-8' ] ]
  ] );

  $fetchShow = [];

  foreach ( [ 'seq', 'vtt' ] as $fetchKind ) {
    $fetchCurl    = padCurl ( "https://api.example.com/$fetchKind" );
    $fetchRows    = padCurlData ( $fetchCurl, 'x' );
    $fetchShow [] = "$fetchKind: {$fetchCurl ['type']} " . json_encode ( $fetchKind == 'vtt' ? array_column ( $fetchRows, 'text', 'start' ) : $fetchRows );
  }

  padCurlFakeStop ();

  $fetchShow = implode ( ' | ', $fetchShow );

?>

App: regression/framework - Page: formats/a_malformed_cue_is_passed_over_on_the_lenient_walk
PHP
formats/a_malformed_cue_is_passed_over_on_the_lenient_walk.php 		HTML
formats/a_malformed_cue_is_passed_over_on_the_lenient_walk.pad 		Result 
<?php

  // With the strict check off, a cue a browser drops is passed over and the others are read: a
  // timing with minutes past 59, a block that is no cue, a cue that ends before it starts - in
  // WebVTT and in SRT, where the reading goes on at the next subtitle behind a blank line. A
  // text without its WEBVTT line is no WebVTT and no rows; a row without a start is left out
  // of what is written, without a word.

  $padCheckSyntax = FALSE;

  $lenientVtt = padFormatRead ( 'vtt', "WEBVTT\n\n00:60.000 --> 00:61.000\nminutes past 59\n\njust text\n\n"
                                     . "00:05.000 --> 00:04.000\nends before it starts\n\nkept\n00:01.000 --> 00:02.000\nkept\n" );

  $lenientSrt = padFormatRead ( 'srt', "1\n00:00:01,000 -> 00:00:02,000\nno arrow\n\n2\n00:00:03,000 --> 00:00:04,000\nkept\n\n"
                                     . "3\n00:00:09,000 --> 00:00:08,000\nends before it starts\n\n4\n00:00:10,000 --> 00:00:11,000\nkept too\n" );

  $lenientNone = count ( padFormatRead ( 'vtt', "00:01.000 --> 00:02.000\nno signature" ) );
  $lenientOut  = substr_count ( padFormatWrite ( 'vtt', [ [ 'text' => 'no start' ], [ 'start' => 1, 'end' => 2, 'text' => 'timed' ] ] ), '-->' );

  $lenientShow = implode ( ' ', array_map ( fn ( $row ) => "{$row['id']}={$row['text']}", $lenientVtt ) ) . ' / '
               . implode ( ' ', array_map ( fn ( $row ) => "{$row['id']}={$row['text']}", $lenientSrt ) )
               . " / no signature: $lenientNone rows / written: $lenientOut cue";

?>
 
{pad:echo $lenientShow/}

App: regression/framework - Page: formats/an_srt_file_reads_its_subtitles_and_a_legacy_encoding
PHP
formats/an_srt_file_reads_its_subtitles_and_a_legacy_encoding.php 		HTML
formats/an_srt_file_reads_its_subtitles_and_a_legacy_encoding.pad 		Result 
<?php

  // SRT read as players read it: the Library of Congress's example with its CRLF line ends, a
  // blank line inside a text with no number and timing behind it going on with the text, a
  // subtitle without its number, a dot before the milliseconds, SRT's coordinates as its
  // settings, its <font> and {\an8} out of plain; a file of Windows-1252 bytes - no UTF-8 -
  // read as that encoding. The rows written as WebVTT are its cues, the numbers their
  // identifiers. Each subtitle: id|start|end|text|plain|settings.

  $srtText = "1\r\n00:02:16,612 --> 00:02:19,376\r\nSenator, we're making\r\nour final approach into Coruscant.\r\n\r\n"
           . "2\r\n00:02:19,482 --> 00:02:21,609\r\nVery good, Lieutenant.\r\n\r\nstill the lieutenant\r\n\r\n"
           . "00:02:22.000 --> 00:02:23.500 X1:100 X2:200 Y1:10 Y2:20\r\n{\\an8}<font color=\"#ffff00\">Top</font>\r\n\r\n"
           . "4\r\n00:02:24,000 --> 00:02:25,000\r\nCaf\xe9 \x93quoted\x94";

  $srtOut = [];

  foreach ( padFormatRead ( 'srt', $srtText ) as $srtRow )
    $srtOut [] = "{$srtRow['id']}|{$srtRow['start']}|{$srtRow['end']}|" . str_replace ( "\n", '/', $srtRow ['text'] ) . '|'
               . str_replace ( "\n", '/', $srtRow ['plain'] ) . "|{$srtRow['settings']}";

  $srtShow = implode ( ' ## ', $srtOut );
  $srtVtt  = str_replace ( "\n", '~', padFormatWrite ( 'vtt', padFormatRead ( 'srt', $srtText ) ) );

?>

App: regression/framework - Page: formats/cues_are_written_in_the_order_of_their_start_escaped_and_read_back
PHP
formats/cues_are_written_in_the_order_of_their_start_escaped_and_read_back.php 		HTML
formats/cues_are_written_in_the_order_of_their_start_escaped_and_read_back.pad 		Result 
<?php

  // Rows written as WebVTT and as SRT (line ends shown as ~): the cues in the order of their
  // start, a start read from a timestamp of any length, a number of seconds or startms, an end
  // left out the start of the next cue - chapters; WebVTT's & < > of a plain title escaped, a
  // tag it has kept and SRT's <font> and {\an8} taken out, a blank line and an arrow of a text
  // never ending the cue, an identifier one line, only the settings of each format kept; SRT
  // numbered from 1 with a comma before the milliseconds, WebVTT's character references the
  // characters, its own tags out, an arrow ->. Read back, both give the times written and the
  // plain text - and the SRT written again from what it read is the same bytes.

  $cueRows = [
    [ 'start' => '1:30', 'title' => 'Formats & <cues>' ],
    [ 'start' => 0, 'title' => 'Intro', 'id' => "chapter\n1 --> a" ],
    [ 'startms' => 125500, 'endms' => 180000, 'text' => "{\\an8}<c.loud>Cues</c> &amp; <font color=\"red\">rows</font> <00:02:06.000>a < b\n\n--> next", 'settings' => 'align:start X1:5 line:0 color:red' ]
  ];

  $cueVtt = padFormatWrite ( 'vtt', $cueRows );
  $cueSrt = padFormatWrite ( 'srt', $cueRows );

  $cueShow = str_replace ( [ "\r\n", "\n" ], [ '~', '~' ], $cueVtt ) . ' ## ' . str_replace ( "\r\n", '~', $cueSrt );

  $cueTimes = fn ( $rows ) => implode ( ' ', array_map ( fn ( $row ) => "{$row['startms']}-{$row['endms']}=" . str_replace ( "\n", '/', $row ['plain'] ), $rows ) );

  $cueBack = 'read back: ' . $cueTimes ( padFormatRead ( 'vtt', $cueVtt ) ) . ' ## ' . $cueTimes ( padFormatRead ( 'srt', $cueSrt ) );

  $cueSame = ( $cueSrt === padFormatWrite ( 'srt', padFormatRead ( 'srt', $cueSrt ) ) ) ? 'the same bytes again' : 'other bytes';

?>
 
{pad:echo $cueShow/} - {pad:echo $cueBack/} - {pad:echo $cueSame/}

App: regression/framework - Page: formats/webvtt_cues_are_read_as_the_spec_examples_have_them
PHP
formats/webvtt_cues_are_read_as_the_spec_examples_have_them.php 		HTML
formats/webvtt_cues_are_read_as_the_spec_examples_have_them.pad 		Result 
<?php

  // The examples of the W3C WebVTT specification read as a browser's parser reads them - the
  // cues and their identifiers, the times with and without hours, the settings after the end
  // time, the voice of a <v> span with its classes, the markup kept in text and taken out of
  // plain with its character references decoded - and a STYLE block, the NOTE comments and
  // the header passed over. Each cue: id|start|end|startms-endms|voice|plain|settings.

  $vttExamples = [

    // 3.1, a cue with settings and the voices of two speakers
    "WEBVTT\n\n00:30.000 --> 00:31.500 align:right size:50%\n<v Roger Bingham>When we e-mailed—\n\n"
    . "00:32.500 --> 00:33.500 align:left size:50%\n<v Neil deGrasse Tyson><i>Laughs</i>\n",

    // 4.1, a STYLE block with its braces and a NOTE between, a cue with its identifier
    "WEBVTT\n\nSTYLE\n::cue {\n  color: papayawhip;\n}\n\nNOTE comment blocks can be used between style blocks.\n\n"
    . "hello\n00:00:00.000 --> 00:00:10.000\nHello <b>world</b>.\n\nNOTE style blocks cannot appear after the first cue.\n",

    // 4.2, voices with classes, a closed voice span before an italic one
    "WEBVTT\n\n00:00.000 --> 00:02.000\n<v.first.loud Esme>It's a blue apple tree!\n\n00:04.000 --> 00:06.000\n<v Esme>Hee!</v> <i>laughter</i>\n",

    // a header of an older file, character references, ruby and a timestamp tag
    "WEBVTT - Kind: captions\nLanguage: en\n\n1\n01:02:03.004 --> 01:02:05.000 line:0 position:10%,line-left\n"
    . "Tom &amp; Jerry &lt;3\n<ruby>漢<rt>kan</rt></ruby> <00:00:12.000>later\n"
  ];

  $vttOut = [];

  foreach ( $vttExamples as $vttText )
    foreach ( padFormatRead ( 'vtt', $vttText ) as $vttRow )
      $vttOut [] = "{$vttRow['id']}|{$vttRow['start']}|{$vttRow['end']}|{$vttRow['startms']}-{$vttRow['endms']}|{$vttRow['voice']}|"
                 . str_replace ( "\n", '/', $vttRow ['plain'] ) . "|{$vttRow['settings']}";

  $vttShow = implode ( ' ## ', $vttOut );
  $vttKept = json_encode ( padFormatRead ( 'vtt', $vttExamples [2] ) [1] ['text'], JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES );

?>
 
{pad:echo $vttShow/} - text {pad:echo $vttKept/}
