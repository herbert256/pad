App: regression/pages - Page: db/named
PHP
db/named.php 		HTML
db/named.pad 		Result 
<?php

  // A _data/*.sql file is a named query: {staffByPhone} runs it, and each {$field} in it is
  // bound as a placeholder - this hostile pattern compares as a value, it never joins the
  // statement.

  $pattern = '%4%';
  $max     = 10;
  $hostile = "%' or '1'='1";

?>
 
<p>{staffByPhone}{$name} {/staffByPhone}</p>
<p>{staffByPhone pattern='%3%', max=2}{$name} {/staffByPhone}</p>
<p>[{staffByPhone pattern=$hostile}{$name} @else@none{/staffByPhone}]</p>
