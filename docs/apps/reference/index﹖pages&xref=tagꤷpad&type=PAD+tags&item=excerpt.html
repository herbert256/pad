App: regression/framework - Page: search/the_style_of_the_marks_comes_with_the_rows_that_are_shown
PHP
search/the_style_of_the_marks_comes_with_the_rows_that_are_shown.php 		HTML
search/the_style_of_the_marks_comes_with_the_rows_that_are_shown.pad 		Result 
<?php

  // The style of the excerpts' marks went with the excerpt of the best row found, and a page
  // that sorted the rows, cut them or printed no excerpt lost it - for every {pad:excerpt} after
  // it as well. Now padSearch's excerpts carry it each, a {pad:search} pair writes it before its
  // first row and the single tag before its list, and the page's own once is left to what
  // prints it: the {pad:excerpt} after a pair that printed none has it too.

  $found = padSearch ( 'kitchen', 'bread' );

?>
 
[{array:found sort='slug DESC', first=1}{$slug}: {!excerpt}{/array:found}]
[{pad:search 'kitchen', q='bread'}{$slug};{/pad:search}]
[{pad:excerpt 'some text with bread in it', words=3, highlight='bread'/}]
[{pad:search 'kitchen', q='bread', sort='slug DESC', first=1}{$slug}: {!excerpt}{/pad:search}]
[{pad:search 'kitchen', q='bread', link='slug'/}]

App: regression/framework - Page: tags/excerpt_html_strips_scripts_in_linear_time
PHP
tags/excerpt_html_strips_scripts_in_linear_time.php 		HTML
tags/excerpt_html_strips_scripts_in_linear_time.pad 		Result 
<?php

  // The scripts and styles of an {pad:excerpt ..., html} text are left out in one pass: an
  // unclosed <script> before a megabyte of text emptied the excerpt (or ended the request),
  // and openers without a closer took seconds - every one scanned on to the end.

  include_once PAD . 'tags/_lib/excerpt.php';

  $big   = '<p>Hello</p><script>' . str_repeat ( 'x', 1000000 );
  $start = hrtime ( TRUE );
  $many  = strlen ( padExcerptCode ( str_repeat ( '<style>x ', 60000 ) ) );
  $time  = ( hrtime ( TRUE ) - $start < 1e9 ) ? 'in time' : 'too slow';
  $kept  = padExcerptCode ( 'a<script>x</script>b<STYLE>y</style >c<script src=1>d' );

?>
 
{pad:excerpt $big, words=1, html/} {$many} {$time} {$kept}

App: regression/framework - Page: tags/excerpt_with_thousands_of_search_words_or_a_huge_one_still_marks

tags/excerpt_with_thousands_of_search_words_or_a_huge_one_still_marks.php 		HTML
tags/excerpt_with_thousands_of_search_words_or_a_huge_one_still_marks.pad 		Result 
<?php

  // A search form that posts: four thousand words, and one word of seventy thousand
  // letters - each made a pattern PCRE refuses to compile.

  $many = [];
  for ( $i = 0; $i < 4000; $i++ )
    $many [] = 'w' . dechex ( $i * 7919 );

  $many = 'excerpting ' . implode ( ' ', $many );
  $huge = 'excerpting ' . str_repeat ( 'templates', 7800 );

?>
 
{pad:excerpt "Templates drive the excerpting of long texts", words=3, highlight=$many/}
{pad:excerpt "Templates drive the excerpting of long texts", words=3, highlight=$huge/}
