App: demo - Page: contact
PHP
contact.php 		HTML
contact.pad 		Result 
<?php

  $title    = 'Contact Form';
  $dataFile = 'demo/messages.json';   // under DATA/, for padFileGet and padFilePut

  // The rules of a valid message stand on the fields in contact.pad - rules='required|email' -
  // and are checked before this file runs: padPosted ( 'contact' ) is TRUE only for a post
  // that keeps them. A post that breaks one never gets here; the form comes back refilled
  // with what was typed, each message beside its field.

  // The rules let a field posted as a list - name[]=x - through, so a message whose fields
  // are not all text is not stored: htmlspecialchars() was handed the array and the post
  // ended on a 500. The form comes back as it was sent.

  $sent = padRequestOnly ( [ 'name', 'email', 'subject', 'message' ] );

  if ( padPosted ( 'contact' ) and count ( array_filter ( $sent, 'is_string' ) ) == count ( $sent ) ) {

    $messages = json_decode ( padFileGet ( $dataFile ), TRUE ) ?: [];

    $entry          = array_map ( 'htmlspecialchars', $sent );
    $entry ['date'] = padNow ( 'Y-m-d H:i:s' );

    $messages [] = $entry;

    padFilePut ( $dataFile, json_encode ( $messages, JSON_PRETTY_PRINT ) );

    // Post, redirect, get: the browser is sent on to a plain GET of this page, so a refresh
    // shows the empty form instead of sending the message a second time - the thanks go
    // along as a flash message.

    padFlash ( 'Thank you for your message! We will get back to you soon.' );

    padRedirect ( 'contact' );

  }

?>
 
<p>Have a question or feedback? Send us a message!</p>

{pad:flash}
  <div class="message">{$message}</div>
{/pad:flash}

{input:form 'contact', error='Please correct the errors below.'}

  <p>{input:field 'name', label='Your Name: *', rules='required|max:100'/}</p>

  <p>{input:field 'email', type='email', label='Your Email: *', rules='required|email'/}</p>

  <p>{input:field 'subject', label='Subject: *', rules='required|max:200'/}</p>

  <p>{input:textarea 'message', label='Message: *', rows=6, rules='required|max:2000'/}</p>

  <p>
    <input type="submit" value="Send Message">
  </p>

{/input:form}

<p style="color: #999; font-size: 12px;">* Required fields</p>

qr.php 		HTML
qr.pad 		Result 
<?php

  $title = 'QR Codes';

  // The text to encode: what the form posted - its rules, on the field in qr.pad, are kept
  // before this runs - else the address of PAD itself.

  if ( ! padPosted ( 'qr' ) or ! is_string ( $text ?? NULL ) )
    $text = 'https://github.com/herbert256/pad';

  $levels = [
    [ 'level' => 'L', 'restores' => '7%'  ],
    [ 'level' => 'M', 'restores' => '15%' ],
    [ 'level' => 'Q', 'restores' => '25%' ],
    [ 'level' => 'H', 'restores' => '30%' ],
  ];

  $wifi = 'WIFI:T:WPA;S:PAD Demo;P:inversion-of-control;;';

?>
 
<p>The <code>{pad:ignore}{barcode:qr}{/pad:ignore}</code> tag encodes a text as a QR code and writes it as inline SVG -
no image library, no remote service, no JavaScript. Type a text and scan the result with your phone.</p>

{input:form 'qr', error='Please correct the errors below.'}
  <p>{input:field 'text', label='Text or address:', value=$text, rules='required|max:1000'/}</p>
  <p><input type="submit" value="Make the QR code"></p>
{/input:form}

<div class="qr-big">{barcode:qr $text, size=220/}</div>

<p><code>{pad:ignore}{barcode:qr $text, size=220/}{/pad:ignore}</code></p>

<h2>Error correction</h2>

<p>The higher the level, the more of the symbol may be damaged or covered - a logo in the
middle - and the more modules it takes.</p>

<table>
  <tr>
    {array:levels}<th>level='{$level}' - restores {$restores}</th>{/array:levels}
  </tr>
  <tr>
    {array:levels}<td>{barcode:qr $text, level=$level, size=150/}</td>{/array:levels}
  </tr>
</table>

App: htmx - Page: examples/form
PHP
examples/form.php 		HTML
examples/form.pad 		Result 
<?php

  // The rules= of form.pad checked the post before this file runs: padPosted ( 'signup' ) is
  // TRUE for a post that kept them alone. htmx posts the form (hx-post) and swaps the answer
  // into its place - hx-target="#signup" sends HX-Target: signup, so the answer is the
  // {pad:fragment 'signup'}: the form again, refilled, with PAD's messages beside its fields, or
  // the welcome. The browser checks the same rules first (client); switch that off to see
  // PAD's own answer.

  $done = padPosted ( 'signup' );
  $name = htmxText ( 'name' );

?>
 
<label class="small switch-line"><input type="checkbox" checked data-toggle-rules="#signup form"> Check in the browser first - turn it off to see PAD's answer to a bad post</label>

<div id="signup">
{pad:fragment 'signup'}
  {pad:if $done}
    <div class="callout is-good rise">
      <p><strong>Welcome aboard, {$name}!</strong> PAD checked the post against the rules of the form and the page's PHP ran for a good one only.</p>
      <p><button class="btn-sm btn-ghost" hx-get="?examples/form" hx-target="#signup">Sign up someone else</button></p>
    </div>
  {pad:else}
    {input:form 'signup', client, error, hx-post='?examples/form', hx-target='#signup', class='stack'}
      {input:field 'name', label='Name', rules='required|max:40', class='input'/}
      {input:field 'email', type='email', label='E-mail', rules='required|email', class='input'/}
      {input:field 'age', label='Age', rules='integer|min:18', class='input'/}
      <button>Sign up</button>
    {/input:form}
  {/pad:if}
{/pad:fragment}
</div>
