App: widgets - Page: examples/sortable/tasks
PHP
examples/sortable/tasks.php 		HTML
examples/sortable/tasks.pad 		Result 
<?php

  $tasks = [
    [ 'id' => 't1', 'title' => 'Write the quarterly report' ],
    [ 'id' => 't2', 'title' => 'Call the bank about the loan' ],
    [ 'id' => 't3', 'title' => 'Order the new kettle boxes' ],
    [ 'id' => 't4', 'title' => 'Answer the mail of the week' ]
  ];

  // The form posts priority[key] = place for every task: sorted by place, the keys are the
  // order - only the keys this page wrote count.

  $saved = '';

  if ( padPosted ( 'ranking' ) ) {
    $titles = array_column ( $tasks, 'title', 'id' );
    $places = array_intersect_key ( (array) padRequest ( 'priority', [] ), $titles );
    asort ( $places );
    $saved  = implode ( ', ', array_map ( fn ( $key ) => $titles [$key], array_keys ( $places ) ) );
  }

?>
 
{pad:meta title='Tasks in order', about='Drag a task by its grip, click its arrows, or press Space on the grip and move it with the arrow keys; the form posts every task with its place - number fields without scripting.'/}
<div style="width:100%; max-width:520px">
  {input:form 'ranking'}
    {widget:sortable 'priority', label='The tasks, most urgent first'}
      {array:tasks}{widget:tab $id}{$title}{/widget:tab}{/array:tasks}
    {/widget:sortable}
    <button>Save the order</button>
  {/input:form}
  {pad:if $saved}<p>Saved: {$saved}</p>{/pad:if}
</div>
