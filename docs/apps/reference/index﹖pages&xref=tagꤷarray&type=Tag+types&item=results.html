App: examples - Page: index
PHP
index.php 		HTML
index.pad 		Result 
<?php

  // The search. $q arrives as a global from the form; the walk is over the .html files,
  // which every example has, and a query matches on the example's name first, then on the
  // contents of its three files.

  $q        = padMakeSafe ( $q ?? '', 200 );
  $searched = ( strlen ( $q ) >= 2 ) ? 1 : 0;
  $short    = ( $q != '' and ! $searched ) ? 1 : 0;
  $found    = [];
  $total    = 0;
  $capped   = 0;

  if ( $searched ) {

    $dir = DATA . 'examples/';

    // More than one word searches for each and keeps only the examples that hold them
    // all. Every word is placed in its best class - the example's name, inside a {...}
    // tag of the .pad, inside an html header, or plainly in the text - and the example
    // ranks by its weakest word: it is only as much about the query as its least-covered
    // word. All words in the name ranks above every content hit; within a content class
    // a small example ranks above a big one - a few lines that match are about the
    // query, a big page merely mentions it. The bucket is keyed class;size;name so one
    // ksort orders it all.

    $words = array_filter ( padExplode ( $q, ' ' ), 'strlen' );

    $foundName    = [];
    $foundContent = [];

    $directory = new RecursiveDirectoryIterator ( $dir );
    $iterator  = new RecursiveIteratorIterator  ( $directory );

    foreach ( $iterator as $one ) {

      $path = padCorrectPath ( $one->getPathname () );

      if ( ! str_ends_with ( $path, '.html' ) )
        continue;

      $item = substr ( $path, strlen ( $dir ), -5 );

      $size  = 0;
      $texts = [];

      foreach ( [ 'html', 'pad', 'php' ] as $ext ) {
        $texts [$ext] = padFileGet ( $dir . "$item.$ext" );
        $size        += strlen ( $texts [$ext] );
      }

      preg_match_all ( '/\{[^\s{}][^{}]*\}/', $texts ['pad'], $padTags );
      preg_match_all ( '/<h[1-6][^>]*>(.*?)<\/h[1-6]>/is', $texts ['pad'] . $texts ['html'], $padHeads );

      $inTags  = implode ( ' ', $padTags  [0] );
      $inHeads = implode ( ' ', $padHeads [1] );
      $inAll   = implode ( ' ', $texts );

      $worst = 0;

      foreach ( $words as $word ) {

        if     ( stripos ( $item,    $word ) !== FALSE ) $class = 0;
        elseif ( stripos ( $inTags,  $word ) !== FALSE ) $class = 1;
        elseif ( stripos ( $inHeads, $word ) !== FALSE ) $class = 2;
        elseif ( stripos ( $inAll,   $word ) !== FALSE ) $class = 3;
        else   { $worst = 9; break; }

        $worst = max ( $worst, $class );

      }

      if ( $worst == 9 )
        continue;

      if ( $worst == 0 )
        $foundName [$item] = [ 'item' => $item ];
      else
        $foundContent [ sprintf ( '%d;%012d;%s', $worst, $size, $item ) ] = [ 'item' => $item ];

    }

    ksort ( $foundName );
    ksort ( $foundContent );

    $found = $foundName + $foundContent;

    $total = count ( $found );

    if ( $total > 200 ) {
      $found  = array_slice ( $found, 0, 200 );
      $capped = 200;
    }

  }

  $results = array_values ( $found );

  $title = 'Examples';

?>
 
<form method="post" class="example-search">
  <input type="text" name="q" value="{$q}" aria-label="Search the examples" placeholder="search the examples by name or content">
  <input type="submit" value="search">
</form>

{pad:if $short eq 1}
  <p>type at least two characters</p>
{/pad:if}

{pad:if $searched eq 1 and $total eq 0}
  <p>nothing found for '{$q}'</p>
{/pad:if}

{pad:if $capped gt 0}
  <p>showing the first {$capped} of {$total} hits</p>
{/pad:if}

{array:results}
  {property:first@results}<table class="example-list">{/property:first@results}
  <tr><td><a href="?show{?item}{?q}">{$item}</a></td></tr>
  {property:last@results}</table>{/property:last@results}
{/array:results}

App: regression/framework - Page: oauth/google_checks_the_id_token_issuer_audience_expiry_and_nonce
PHP
oauth/google_checks_the_id_token_issuer_audience_expiry_and_nonce.php 		HTML
oauth/google_checks_the_id_token_issuer_audience_expiry_and_nonce.pad 		Result 
<?php

  // Sign in with Google, OpenID Connect: the authorize address asks for a nonce, and the
  // id_token that comes with the tokens must name Google as its issuer, this client as its
  // audience, an expiry still ahead and that nonce - a login is refused for each one that is
  // wrong. Its signature is not checked: it comes straight from the token endpoint over TLS
  // (OpenID Connect Core 3.1.3.7, item 6). The userinfo's sub must be the token's.

  $options = [ 'client_id' => 'web.apps.googleusercontent.com', 'client_secret' => 'cs', 'redirect' => 'https://shop.example/oauth/google' ];

  function googleToken ( $claims ) {
    return padCryptB64 ( '{"alg":"RS256","kid":"k1"}' ) . '.' . padCryptB64 ( json_encode ( $claims ) ) . '.c2lnbmF0dXJl';
  }

  $cases = [
    'right'          => [],
    'other issuer'   => [ 'iss' => 'https://accounts.example.com' ],
    'other client'   => [ 'aud' => 'another.apps.googleusercontent.com' ],
    'two audiences'  => [ 'aud' => [ 'web.apps.googleusercontent.com', 'other' ] ],
    'expired'        => [ 'exp' => time () - 3600 ],
    'other nonce'    => [ 'nonce' => 'replayed' ],
    'other userinfo' => [ 'sub' => '999' ],
  ];

  $results = [];

  foreach ( $cases as $case => $change ) {

    padCurlFakeStop ();

    parse_str ( parse_url ( padOAuthUrl ( 'google', $options ), PHP_URL_QUERY ), $asked );

    $claims = $change + [ 'iss' => 'https://accounts.google.com', 'aud' => 'web.apps.googleusercontent.com', 'sub' => '1093', 'exp' => time () + 3600,
                          'iat' => time (), 'nonce' => $asked ['nonce'], 'email' => 'ann@example.com' ];

    padCurlFake ( [
      'https://oauth2.googleapis.com/token' => [ 'data' => [ 'access_token' => 'ya29', 'id_token' => googleToken ( $claims ), 'expires_in' => 3599 ] ],
      'https://openidconnect.googleapis.com/v1/userinfo' => [ 'data' => [ 'sub' => '1093', 'name' => 'Ann Smith', 'email' => 'ann@example.com', 'picture' => 'https://lh3.example/a.png' ] ],
    ] );

    $_GET = [ 'code' => 'c', 'state' => $asked ['state'] ];

    $user = padOAuthUser ( 'google', $options );

    $results [] = [ 'case' => $case, 'nonce' => isset ( $asked ['nonce'] ) ? 'asked' : 'not asked', 'scope' => $asked ['scope'],
                    'user' => $user ? $user ['id'] . ' ' . $user ['name'] . ' <' . $user ['email'] . '>' : '', 'error' => padOAuthError () ];

  }

?>
 
{array:results}
{$case} ({$scope}, nonce {$nonce}): {$user}{$error} |
{/array:results}

App: regression/framework - Page: oauth/microsoft_checks_the_issuer_by_the_tenant_of_the_account_and_a_tenant_given_by_its_id
PHP
oauth/microsoft_checks_the_issuer_by_the_tenant_of_the_account_and_a_tenant_given_by_its_id.php 		HTML
oauth/microsoft_checks_the_issuer_by_the_tenant_of_the_account_and_a_tenant_given_by_its_id.pad 		Result 
<?php

  // Microsoft's id_token names the tenant of the account in its issuer -
  // login.microsoftonline.com/<tid>/v2.0 - so with the tenant common any tenant's issuer
  // passes when it matches the token's own tid, and a tenant given by its id lets only that
  // tenant's accounts in. The username is the id_token's preferred_username, under Graph's
  // userinfo.

  $tid   = '9188040d-6c67-4c5b-b112-36a304b66dad';
  $nonce = '';
  $iss   = '';

  padCurlFake ( [
    'https://login.microsoftonline.com/*/oauth2/v2.0/token' => function () use ( &$nonce, &$iss, $tid ) {
      $claims = [ 'iss' => $iss, 'tid' => $tid, 'aud' => 'app-id', 'exp' => time () + 300, 'sub' => 'AAAAbbbb', 'nonce' => $nonce,
                  'preferred_username' => 'ann@contoso.com' ];
      return [ 'data' => [ 'access_token' => 'at', 'id_token' => 'e30.' . padCryptB64 ( json_encode ( $claims ) ) . '.c2ln' ] ];
    },
    'https://graph.microsoft.com/oidc/userinfo' => [ 'data' => [ 'sub' => 'AAAAbbbb', 'name' => 'Ann Smith', 'email' => 'ann@contoso.com' ] ],
  ] );

  $cases = [ 'common, its own tid'          => [ 'common', "https://login.microsoftonline.com/$tid/v2.0" ],
             'common, another tid in iss'   => [ 'common', 'https://login.microsoftonline.com/72f988bf-86f1-41af-91ab-2d7cd011db47/v2.0' ],
             'the tenant given by its id'   => [ $tid, "https://login.microsoftonline.com/$tid/v2.0" ],
             'another tenant given'         => [ '72f988bf-86f1-41af-91ab-2d7cd011db47', "https://login.microsoftonline.com/$tid/v2.0" ] ];

  $results = [];

  foreach ( $cases as $case => [ $tenant, $iss ] ) {

    $options = [ 'tenant' => $tenant, 'client_id' => 'app-id', 'client_secret' => 'cs', 'redirect' => 'https://shop.example/oauth/microsoft' ];

    parse_str ( parse_url ( padOAuthUrl ( 'microsoft', $options ), PHP_URL_QUERY ), $asked );

    $nonce = $asked ['nonce'];
    $_GET  = [ 'code' => 'c', 'state' => $asked ['state'] ];
    $user  = padOAuthUser ( 'microsoft', $options );

    $results [] = [ 'case' => $case, 'got' => $user ? $user ['id'] . ' ' . $user ['username'] : padOAuthError () ];

  }

?>
 
{array:results}
{$case}: {$got} |
{/array:results}

      'body'  => 'Some parts of a page cost more to render than others: a list of the best selling products, a menu built from the database, a report over a year of orders. The fragment cache keeps the rendered text of such a section for a number of seconds and serves it again on the next request without running the tags inside it. Name the section, give it a ttl, and forget it when what it shows has changed.' ],
    [ 'title' => 'Remote data',
      'url'   => '../manual/?pages/remote_data',
      'body'  => 'A page that shows exchange rates or the weather fetches them from another server. Every request would wait for that server, so the answer is kept for a while: the ttl option says how many seconds. When the source fails afterwards, the last good copy in the cache is served and the failure is logged, so a slow or broken service never takes the page down with it.' ],
    [ 'title' => 'Page cache',
      'url'   => '../manual/?pages/fragment_cache',
      'body'  => 'With the page cache switched on in the configuration, a whole page is kept after its first rendering, with an ETag, and a browser that asks again with that tag gets a 304 Not Modified. Pages with a guard, a flash message or a form are never cached. The cache can live in files, APCu, Redis, Memcached or the database.' ] ];
