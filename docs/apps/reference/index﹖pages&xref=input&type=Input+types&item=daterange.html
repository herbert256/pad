App: regression/framework - Page: tags/a_daterange_refills_both_days_and_shows_the_message_of_each
PHP
tags/a_daterange_refills_both_days_and_shows_the_message_of_each.php 		HTML
tags/a_daterange_refills_both_days_and_shows_the_message_of_each.pad 		Result 
<?php

  // A post of form f: its two days show what was posted for each - stay_from and stay_to - in
  // place of value=, the range of form g, which did not come back, shows value=. The page's
  // padValidate holds the last day after the first: the message stands after the frame, tied
  // to the last day's field alone.

  $_SERVER ['REQUEST_METHOD'] = 'POST';
  $_POST = [ 'padForm' => 'f', 'stay_from' => '2026-08-10', 'stay_to' => '2026-08-03' ];

  padValidate ( [ 'stay_from' => 'required|date', 'stay_to' => 'required|date|after:stay_from' ] );

?>
 
{input:form 'g'}{input:daterange 'stay', value='2026-07-06, 2026-07-17'/}{/input:form}{input:form 'f'}{input:daterange 'stay', label='Stay', value='2026-07-06, 2026-07-17', nights/}{/input:form}

App: regression/pages - Page: request/form_daterange
PHP
request/form_daterange.php 		HTML
request/form_daterange.pad 		Result 
<?php

  // The fixture of request/form_daterange_post: a date range posts two fields, stay_from and
  // stay_to, which the page's PHP checks with padValidate - its rules= are refused - the
  // departure after the arrival, and stores the nights between them.

  $state = 'not stored';

  if ( padFormCameBack ( 'book' ) and ! padValidate ( [ 'stay_from' => 'required|date', 'stay_to' => 'required|date|after:stay_from' ] ) )
    $state = 'stored ' . ( ( strtotime ( $_POST ['stay_to'] . ' UTC' ) - strtotime ( $_POST ['stay_from'] . ' UTC' ) ) / 86400 ) . ' nights';

?>
 
{input:form 'book', error}
{input:daterange 'stay', label='Stay', labels='Arrival, Departure', nights/}
{/input:form}
{$state}
