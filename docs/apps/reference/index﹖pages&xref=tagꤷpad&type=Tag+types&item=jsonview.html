App: regression/framework - Page: tags/jsonview_cuts_an_array_that_holds_itself_and_a_tree_too_deep
PHP
tags/jsonview_cuts_an_array_that_holds_itself_and_a_tree_too_deep.php 		HTML
tags/jsonview_cuts_an_array_that_holds_itself_and_a_tree_too_deep.pad 		Result 
<?php

  // An array holding a reference to itself, a list of an object holding itself, and a list
  // nested deeper than json_decode would ever give.

  $self = [ 'x' => 1 ];
  $self ['me'] = &$self;

  $node = new stdClass;
  $node->name = 'root';
  $node->parent = $node;
  $nodes = [ $node ];

  $deep = 1;
  for ( $i = 0; $i < 600; $i++ )
    $deep = [ $deep ];

?>
 
{jsonview $self, open=0}
{jsonview $nodes, open=0}
{jsonview $deep, open=0}
