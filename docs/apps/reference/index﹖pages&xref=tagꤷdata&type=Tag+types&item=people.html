  // A CSV row with fewer cells than the header - a trailing empty cell left out, as a
  // spreadsheet export does - has every header field, the missing ones empty: Bob's row had
  // no city, so {$city} on it found the page's own $city, and with no such variable the
  // page ended "Field '$c' not found".

  $city = 'the page city';

?>

Bob
{/pad:data}{pad:data 'c2'}

<?php

  // The rows of {fake:rows ..., rows=, seed=} are those padFakeRows gives for the seed, and an
  // email of a row with a name is that name's.

  $fromPhp = padFakeRows ( [ 'customer' => 'name', 'mail' => 'email', 'total' => [ 'price', [ 'from' => 10, 'to' => 99 ] ] ], 3, 11 );

?>
 
{fake:rows 'customer:name, mail:email, total:price', from=10, to=99, rows=3, seed=11, toData='people'/}{data:people}[{$customer}|{$mail}|{$total}]{/data:people} = {array:fromPhp}[{$customer}|{$mail}|{$total}]{/array:fromPhp}
