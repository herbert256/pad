  // A field written inside the quoted title, label, alt, button, past= or tab name of these
  // tags is escaped once, as the field alone is: it arrived escaped by its sanitize step and
  // the tag escaped it again - Q&A Å was shown and read out as Q&amp;A &Aring;. Each line
  // counts the texts written right; a tab named by the field is the one active= and open=
  // name with it.

  $t = 'Q&A Å';

?>

App: regression/framework - Page: tags/sudoku_reads_a_drawn_row_without_givens_and_windows_line_ends
PHP
tags/sudoku_reads_a_drawn_row_without_givens_and_windows_line_ends.php 		HTML
tags/sudoku_reads_a_drawn_row_without_givens_and_windows_line_ends.pad 		Result 
<?php

  // A row with no given written as dashes with spaces or bars between them is nine empty
  // cells, not a line between bands, and a grid with the \r\n line ends of a posted
  // textarea passes over its band lines as well: the first puzzle had 72 cells, the
  // second 99.

  $spaced = "- - - | - - - | - - -\n"
          . "6 - - | 1 9 5 | - - -\n"
          . "- 9 8 | - - - | - 6 -\n"
          . "8 - - | - 6 - | - - 3\n"
          . "4 - - | 8 - 3 | - - 1\n"
          . "7 - - | - 2 - | - - 6\n"
          . "- 6 - | - - - | 2 8 -\n"
          . "---|---|---\n"
          . "- - - - 8 - - 7 9\n";

  $posted = str_replace ( "\n", "\r\n",
              "5 3 . | . 7 . | . . .\n6 . . | 1 9 5 | . . .\n. 9 8 | . . . | . 6 .\n------+-------+------\n"
            . "8 . . | . 6 . | . . 3\n4 . . | 8 . 3 | . . 1\n7 . . | . 2 . | . . 6\n------+-------+------\n"
            . ". 6 . | . . . | 2 8 .\n. . . | 4 1 9 | . . 5\n. . . | . 8 . | . 7 9\n" );

?>
 
{sudoku $spaced}
{sudoku $posted, solve}

App: regression/framework - Page: tags/sudoku_reads_band_lines_of_equals_and_of_dashes_as_wide_as_a_row
PHP
tags/sudoku_reads_band_lines_of_equals_and_of_dashes_as_wide_as_a_row.php 		HTML
tags/sudoku_reads_band_lines_of_equals_and_of_dashes_as_wide_as_a_row.pad 		Result 
<?php

  // A drawn grid's band lines without a + are band lines too: = is no cell, so a line of =
  // with spaces and bars is one, and a line of nine dashes is one when the grid has 81
  // cells without such lines and more with them - a compact grid's ---|---|---, as wide as
  // its rows. The first was refused with "it has a '='", the second and third with "it has
  // 99 cells, not 81".

  $rows = [ [ '5 3 . | . 7 . | . . .', '6 . . | 1 9 5 | . . .', '. 9 8 | . . . | . 6 .' ],
            [ '8 . . | . 6 . | . . 3', '4 . . | 8 . 3 | . . 1', '7 . . | . 2 . | . . 6' ],
            [ '. 6 . | . . . | 2 8 .', '. . . | 4 1 9 | . . 5', '. . . | . 8 . | . 7 9' ] ];

  $grid = fn ( $band, $compact ) => implode ( "\n$band\n", array_map (
            fn ( $band ) => implode ( "\n", array_map ( fn ( $row ) => $compact ? str_replace ( ' ', '', $row ) : $row, $band ) ),
            $rows ) );

  $spaced  = $grid ( '= = = | = = = | = = =', FALSE );
  $packed  = $grid ( '===|===|===', TRUE );
  $dashed  = $grid ( '---|---|---', TRUE );

?>
 
{sudoku $spaced}
{sudoku $packed}
{sudoku $dashed, solve}
