tags/a_field_inside_the_quoted_text_of_a_tag_that_computes_from_it_counts_as_its_characters.pad 		Result 
<?php

  // A field written inside the quoted text of {pad:avatar}, {pad:gravatar}, {pad:identicon} or {barcode:qr}
  // gives what the field alone gives: the characters it holds - not the stand-ins it
  // travels with, nor the HTML its sanitize step made of it, which went into the initials,
  // the hash and the code, and into a label escaped once more.

  $name  = 'Åsa Ørsted & Co';
  $email = "o'brien@example.com";
  $query = 'https://example.com/?a=1&b=2';

?>
 
{pad:avatar '{$name}'/}
{pad:gravatar '{$email}'/}
{pad:identicon '{$name}'/}
{barcode:qr '{$query}', size=21/}

App: showcase - Page: gravatar/comments
PHP
gravatar/comments.php 		HTML
gravatar/comments.pad 		Result 
<?php

  $comments = [
    [ 'name' => 'Herbert Jebbink', 'email' => 'herbert@example.com', 'text' => 'The page cache halves the time of the home page.' ],
    [ 'name' => 'Ada Lovelace',    'email' => 'ada@example.org',     'text' => 'Could the sequence tags take a step of their own?' ],
    [ 'name' => 'Grace Hopper',    'email' => 'grace@example.net',   'text' => 'A bug report with a template line number - lovely.' ] ];

?>
 
{pad:meta title='Comments with Gravatars', about='The picture each address registered at gravatar.com - only its SHA-256 hash leaves the page, lazily, without a referrer.'/}
<div style="display:grid; gap:16px">
  {array:comments}
  <div style="display:flex; gap:12px; align-items:flex-start">
    {pad:gravatar $email, size=40, default='identicon', alt=$name/}
    <div><strong>{$name}</strong><br>{$text}</div>
  </div>
  {/array:comments}
</div>
