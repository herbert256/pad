index.php 		HTML
index.pad 		Result 
<?php

  $groups  = typesByGroup ();
  $indexes = [
    [ 'page' => 'names',      'label' => 'By name',     'text' => 'Every type from A to Z, with its one line.' ],
    [ 'page' => 'order',      'label' => 'Type prefixes', 'text' => 'Choose the namespace explicitly for every named tag.' ],
    [ 'page' => 'groups',     'label' => 'By group',    'text' => 'The types by what they reach - tags, stores, values, sequences, subsystems.' ],
    [ 'page' => 'cheatsheet', 'label' => 'Cheat sheet', 'text' => 'The syntax of every type on one page.' ] ];

  $memberCount = $exampleCount = 0;

  foreach ( typesCatalog () as $name => $type ) {
    $memberCount  += count ( typesMembers ( $name ) );
    $exampleCount += count ( typesExamples ( $name ) );
  }

?>
 
{pad:meta title='Every PAD type'/}
<section class="hero">
  <h1>Every PAD type</h1>
  <p class="lead">A tag has a type, the prefix before the colon: <code>&#123;pad:if&#125;</code> is a
  built-in tag, <code>&#123;chart:pie&#125;</code> a chart type, <code>&#123;field:name&#125;</code>
  a field of the row, <code>&#123;sequence:fibonacci&#125;</code> a sequence. Every named tag
  must include its type prefix, on both halves of a pair. Variable and expression shorthand,
  such as <code>&#123;$name&#125;</code>, stays available. Each type has its syntax, available
  names and live examples. The options and helpers group lists subsystem kinds selected
  through parameters or PHP helpers.</p>
  <ul class="stats">
    <li><b>{$typesCount}</b> types and kind families</li>
    <li><b>{$memberCount}</b> available names</li>
    <li><b>{$exampleCount}</b> examples</li>
  </ul>
</section>

<div class="cards">
  {array:indexes}
  <a class="card" href="?{$page}">
    <h2>{$label}</h2>
    <p>{$text}</p>
  </a>
  {/array:indexes}
</div>
