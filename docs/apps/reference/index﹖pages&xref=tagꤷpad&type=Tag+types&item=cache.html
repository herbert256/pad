  padFragmentForget ( 'manual-cache' );

  $built = 0;

  <p>Rendered while building: built {$built} time(s)</p>
{/cache}

App: regression/framework - Page: tags/a_chart_from_the_fragment_cache_and_a_chart_drawn_now_have_their_own_ids
PHP
tags/a_chart_from_the_fragment_cache_and_a_chart_drawn_now_have_their_own_ids.php 		HTML
tags/a_chart_from_the_fragment_cache_and_a_chart_drawn_now_have_their_own_ids.pad 		Result 
<?php

  // An earlier request stores the named section with the bar chart, under the ids that
  // request gave it; this request serves the bar from the cache and draws the line itself.

  if ( ! isset ( $warm ) ) {
    padFragmentForget ( 'fwChartIds' );
    padCurl ( [ 'url' => $padHost . "regression/framework/?tags/a_chart_from_the_fragment_cache_and_a_chart_drawn_now_have_their_own_ids&padInclude&warm=1" ] );
  }

?>
 
{data 'sales'}[{"month":"Jan","amount":120},{"month":"Feb","amount":90}]{/data}
{cache 'fwChartIds'}{chart 'bar', data='sales', value='amount', width=300, height=150}{/cache}
{chart 'line', data='sales', value='amount', width=300, height=150}
