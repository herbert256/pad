App: regression/pages - Page: select/demo/a_huge_page_number_is_a_page_past_the_last

select/demo/a_huge_page_number_is_a_page_past_the_last.php 		HTML
select/demo/a_huge_page_number_is_a_page_past_the_last.pad 		Result 
<?php

  // A page number from the request too high for an integer offset is a page past the last,
  // as page 1000 is: the page became PHP_INT_MAX, times the rows a float, and the statement
  // ended in "limit 1.844674407371E+19, 2" - an SQL error a visitor could raise at will.

  $padSelect ['officesPaged'] = [ 'db' => 'offices', 'key' => 'officeCode' ];

  $pg   = '99999999999999999999';
  $high = 1000;

?>
 
huge: [{officesPaged page=$pg, rows=2}{$officeCode} @else@empty{/officesPaged}]
high: [{officesPaged page=$high, rows=2}{$officeCode} @else@empty{/officesPaged}]
