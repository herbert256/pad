App: manual - Page: fragments/notify_1

fragments/notify_1.php 		HTML
fragments/notify_1.pad 		Result 
<?php

  // Slack and ntfy are not asked here: a fake answers every request, as a test answers them,
  // and keeps what was sent. The mail goes to DATA/mail/ - the file transport. In an
  // application the addresses stand in .env: SLACK_WEBHOOK_URL, NTFY_TOPIC.

  padCurlFake ( [ '*' => 'ok' ] );

  $ann = [ 'id' => 'manual-ann', 'email' => 'ann@example.com', 'ntfy' => 'ann-orders' ];

  padNotify ( $ann, 'order-shipped', [ 'order' => 42, 'name' => 'Ann', 'url' => 'https://shop.example/orders/42' ],
              [ 'mail', 'slack', 'ntfy' ], [ 'slack' => [ 'url' => 'https://hooks.slack.com/services/T0/B0/example' ] ] );

  $results = [];

  foreach ( padNotifyResults () as $channel => $result )
    $results [] = [ 'channel' => $channel, 'ok' => $result ['ok'] ? 'sent' : $result ['error'] ];

  $subject = $padMailLast ['subject'];
  $slack   = json_decode ( padCurlRecorded () [0] ['post'], TRUE ) ['text'];
  $ntfy    = padCurlRecorded () [1] ['post'];

  padCurlFakeStop ();

?>
 
<ul>
  {array:results}<li>{$channel}: {$ok}</li>{/array:results}
</ul>
<p>The mail's subject: {$subject}</p>
<p>Slack got: <code>{$slack}</code></p>
<p>ntfy got: <code>{$ntfy}</code></p>

channels/a_queued_notification_keeps_no_secret_of_its_recipient.pad 		Result 
<?php

  // A queued notification is a job file under DATA/queue/ - kept with the failed jobs after
  // its tries, listed by the admin console - and its recipient goes along for the worker. Of
  // a user row it keeps what padLogin keeps: the password hash, the remember token, the
  // second factor's secret are left out; the fields the channels and the template read - the
  // id, the email, a push subscription with its keys - stay.

  padQueueFlush ( 'notifysecret' );

  padNotify ( [ 'id' => 'r26f', 'name' => 'Ann', 'email' => 'ann@example.com', 'password_hash' => '$2y$12$abcdefghijklmnopqrstuv',
                'remember_token' => 'rt_9f8e7d', 'totp_secret' => 'JBSWY3DPEHPK3PXP', 'api_key'=> '*REDACTED*',
                'webpush' => [ 'endpoint' => 'https://fcm.googleapis.com/fcm/send/x', 'keys' => [ 'p256dh' => 'BP', 'auth'=> '*REDACTED*' ] ] ],
              'shipped', [ 'order' => 42, 'items' => 1, 'name' => 'Ann', 'url' => '' ], [ 'mail' ], [ 'queue' => 'notifysecret' ] );

  $kept = [];

  foreach ( glob ( DATA . 'queue/regression/framework/notifysecret/*.job' ) as $file )
    $kept = json_decode ( file_get_contents ( $file ), TRUE ) ['data'] ['to'] ?? [];

  padQueueFlush ( 'notifysecret' );

  $json = json_encode ( $kept, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES );

?>
 
{$json}

<?php

  // The mail channel hands the @mail@ section to padMail as the HTML part, the text part made
  // from it, the @title@ section as the subject - the file transport keeps it under
  // DATA/mail/, nothing is sent. A recipient without an email gets no mail: FALSE, and the
  // results say why; that is the data's, not the author's, so it is no error.

  padNotify ( [ 'id' => 3, 'email' => 'ann@example.com', 'name' => 'Ann' ], 'shipped',
              [ 'order' => 42, 'items' => 3, 'name' => 'Ann & Bob', 'url' => 'https://shop.example/orders/42' ], [ 'mail' ] );

  $mail = $padMailLast;

  $none   = padNotify ( [ 'id' => 4 ], 'shipped', [ 'order' => 43, 'items' => 1, 'name' => 'Cy', 'url' => '' ], [ 'mail' ] ) ? 'sent' : 'not sent';
  $reason = padNotifyResults () ['mail'] ['error'];

?>
 
to: {$mail.to}
subject: {$mail.subject}

App: regression/framework - Page: channels/queued_every_channel_is_a_job_of_its_own_and_a_failing_one_is_retried_alone
PHP
channels/queued_every_channel_is_a_job_of_its_own_and_a_failing_one_is_retried_alone.php 		HTML
channels/queued_every_channel_is_a_job_of_its_own_and_a_failing_one_is_retried_alone.pad 		Result 
<?php

  // Queued, a notification is a job per channel ('@notify'): the request answers at once, the
  // worker renders and sends - here padQueueWork in the same request, the fakes still on. Slack
  // answers 500 the first time: its job comes back for a retry, the mail that went is not
  // sent again; the retry delivers it.

  padQueueFlush ( 'notifytest' );

  $slack = 0;

  padCurlFake ( [ 'https://hooks.slack.com/*' => function () use ( &$slack ) {
    return ++$slack == 1 ? [ 'status' => 500, 'data' => 'rollup_error' ] : 'ok';
  } ] );

  $queued = padNotify ( [ 'id' => 9, 'email' => 'cy@example.com' ], 'shipped', [ 'order' => 77, 'items' => 1, 'name' => 'Cy', 'url' => '' ],
                        [ 'mail', 'slack' ], [ 'queue' => 'notifytest', 'slack' => [ 'url' => 'https://hooks.slack.com/services/T/B/q' ] ] );

  $jobs   = array_map ( fn ( $one ) => is_string ( $one ['job'] ) ? 'a job' : 'no job', padNotifyResults () );
  $size   = padQueueSize ( 'notifytest' );
  $first  = padQueueWork ( 'notifytest', 0, 60, 3 );
  $mailed = $padMailLast ['subject'] ?? '';

  $padMailLast = NULL;

  // The retry is due after the backoff: the job file is made due now.

  foreach ( glob ( DATA . 'queue/regression/framework/notifytest/*.job' ) as $file ) {
    $record = json_decode ( file_get_contents ( $file ), TRUE );
    $record ['available'] = time () - 1;
    file_put_contents ( $file, json_encode ( $record ) );
    rename ( $file, dirname ( $file ) . '/' . sprintf ( '%010d', time () - 1 ) . '-' . $record ['id'] . '.job' );
  }

  $second = padQueueWork ( 'notifytest', 0, 60, 3 );

  $report = json_encode ( [ 'queued' => $queued, 'jobs' => $jobs, 'waiting' => $size,
                            'first'  => [ $first ['done'], $first ['retried'], $first ['failed'] ], 'mailed' => $mailed,
                            'second' => [ $second ['done'], $second ['retried'], $second ['failed'] ], 'mailed again' => $padMailLast !== NULL,
                            'slack asked' => $slack ], JSON_UNESCAPED_SLASHES );

?>
 
{$report}
