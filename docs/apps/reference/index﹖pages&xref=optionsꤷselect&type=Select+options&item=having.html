App: regression/pages - Page: select/demo/a_line_comment_ends_its_own_clause
PHP
select/demo/a_line_comment_ends_its_own_clause.php 		HTML
select/demo/a_line_comment_ends_its_own_clause.pad 		Result 
<?php

  // A -- comment written in a clause - where=, group=, having=, order= - ends that clause:
  // the clauses of the statement stand on lines of their own. On one line the comment ran to
  // its end, so the having, the order and the limit after it were gone without a word, and
  // the closing parenthesis of the where with them. Over the staff table of the demo database.

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name', 'order' => 'name' ];

?>
 
group: [{select:staffSel fields="floor(salary / 2000) as band, count(*) as n", group="band -- per band", having="n > 1", order="n desc", rows=1}{$band}={$n} {/select:staffSel}]
having: [{select:staffSel fields="floor(salary / 2000) as band, count(*) as n", group="band", having="n > 1 -- the shared ones", order="n desc"}{$band}={$n} {/select:staffSel}]
rollup: [{select:staffSel fields="floor(salary / 2000) as band, sum(salary) as pay", group="band -- per band", rollup}{$pay} {/select:staffSel}]
where: [{select:staffSel where="salary > 2000 -- the well paid", order="name"}{$name} {/select:staffSel}]
order: [{select:staffSel order="salary desc -- best paid first", rows=2}{$name} {/select:staffSel}]
pages: [{select:staffSel fields="floor(salary / 2000) as band, count(*) as n", group="band", having="n > 0 -- every band", page=1, rows=2}{$band} {/select:staffSel}] ({pad:pager 'staffSel', window=0}{$kind}:{$page} {/pad:pager})

App: regression/pages - Page: select/demo/options
PHP
select/demo/options.php 		HTML
select/demo/options.pad 		Result 
<?php

  // The options a declared select table takes on the tag itself: each section of the .pad
  // half names one, over the four-row staff table of the demo database.

  $padSelect ['staffSel'] = [ 'db' => 'staff', 'key' => 'name' ];
  $padSelect ['elsewhere'] = [ 'key' => 'name' ];
  $padSelect ['staffJim'] = [ 'db' => 'staff', 'where' => "name = 'jim'" ];
  $padSelect ['staffRaw'] = [ 'db' => 'staff' ];

?>
 
where: [{select:staffSel where="salary >= 2000", order="salary"}{$name} {/select:staffSel}]
order: [{select:staffSel order="salary desc", fields="name"}{$name} {/select:staffSel}]
fields: [{select:staffSel fields="name, bonus", where="name = 'bob'"}{$name}/{$bonus}{/select:staffSel}]
group: [{select:staffSel fields="count(*) as heads", group="floor(salary / 2000)", order="heads"}{$heads} {/select:staffSel}]
having: [{select:staffSel fields="salary + bonus as total", group="total", having="total > 2500", order="total"}{$total} {/select:staffSel}]
rollup: [{select:staffRaw fields="sum(salary) as pay", group="name", rollup}{$pay} {/select:staffRaw}]
distinctrow: [{select:staffSel fields="floor(salary / 2000) as band", distinctrow, order="band"}{$band} {/select:staffSel}]
db: [{select:elsewhere db='staff', where="name = 'jim'"}{$name}/{$phone}{/select:elsewhere}]
union: [{select:staffSel where="name = 'bob'", union='staffJim', order="name"}{$name} {/select:staffSel}]
